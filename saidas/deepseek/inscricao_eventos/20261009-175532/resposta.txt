const express = require('express');
const session = require('express-session');
const cookieParser = require('cookie-parser');
const bcrypt = require('bcryptjs');
const sqlite3 = require('sqlite3').verbose();
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

// ---------- Banco de dados ----------
const db = new sqlite3.Database(path.join(__dirname, 'portal.db'));

db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT NOT NULL CHECK(tipo IN ('participante','organizador'))
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS eventos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    descricao TEXT,
    local TEXT,
    data TEXT,
    vagas INTEGER DEFAULT 0,
    organizador_id INTEGER NOT NULL,
    cancelado INTEGER DEFAULT 0,
    criado_em TEXT DEFAULT CURRENT_TIMESTAMP
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS inscricoes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    evento_id INTEGER NOT NULL,
    usuario_id INTEGER NOT NULL,
    status TEXT DEFAULT 'pendente',
    criado_em TEXT DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(evento_id, usuario_id)
  )`);
});

function requireAuth(req, res, next) {
  if (!req.session.usuario) return res.status(401).json({ erro: 'Não autenticado' });
  next();
}
function requireOrganizador(req, res, next) {
  if (!req.session.usuario) return res.status(401).json({ erro: 'Não autenticado' });
  if (req.session.usuario.tipo !== 'organizador') return res.status(403).json({ erro: 'Apenas organizador' });
  next();
}

app.post('/api/registrar', async (req, res) => {
  const { nome, email, senha, tipo } = req.body;
  if (!nome || !email || !senha || !tipo) return res.status(400).json({ erro: 'Preencha todos os campos' });
  if (!['participante', 'organizador'].includes(tipo)) return res.status(400).json({ erro: 'Tipo inválido' });

  const hash = await bcrypt.hash(senha, 10);
  db.run(
    `INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)`,
    [nome, email.toLowerCase(), hash, tipo],
    function (err) {
      if (err) {
        if (String(err.message).includes('UNIQUE')) return res.status(400).json({ erro: 'Este e-mail já está cadastrado' });
        return res.status(500).json({ erro: 'Erro ao cadastrar' });
      }
      req.session.usuario = { id: this.lastID, nome, email: email.toLowerCase(), tipo };
      res.json({ ok: true, usuario: req.session.usuario });
    }
  );
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;
  if (!email || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });

  db.get(`SELECT * FROM usuarios WHERE email = ?`, [email.toLowerCase()], async (err, user) => {
    if (err || !user) return res.status(400).json({ erro: 'E-mail ou senha inválidos' });
    const ok = await bcrypt.compare(senha, user.senha);
    if (!ok) return res.status(400).json({ erro: 'E-mail ou senha inválidos' });
    req.session.usuario = { id: user.id, nome: user.nome, email: user.email, tipo: user.tipo };
    res.json({ ok: true, usuario: req.session.usuario });
  });
});

app.get('/api/eventos', (req, res) => {
  const u = req.session.usuario;
  const sql = `
    SELECT e.*, u.nome AS organizador_nome,
      (SELECT COUNT(*) FROM inscricoes i WHERE i.evento_id = e.id AND i.status != 'cancelada') AS total_inscritos
    FROM eventos e
    JOIN usuarios u ON u.id = e.organizador_id
    WHERE e.cancelado = 0
    ORDER BY e.data ASC, e.id DESC
  `;
  db.all(sql, [], (err, eventos) => {
    if (err) return res.status(500).json({ erro: 'Erro ao buscar eventos' });
    if (!u) return res.json({ eventos: eventos.map(e => ({ ...e, minha_inscricao: null })) });
    db.all(`SELECT evento_id, status FROM inscricoes WHERE usuario_id = ?`, [u.id], (err2, insc) => {
      const map = {};
      (insc || []).forEach(i => { map[i.evento_id] = i.status; });
      res.json({ eventos: eventos.map(e => ({ ...e, minha_inscricao: map[e.id] || null })) });
    });
  });
});

async function api(url, opts = {}) {
  const res = await fetch(url, {
    headers: { 'Content-Type': 'application/json' },
    credentials: 'same-origin',
    ...opts
  });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.erro || 'Erro');
  return data;
}

function toast(msg, tipo = 'ok') {
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.className = 'toast show ' + (tipo === 'erro' ? 'erro' : '');
  clearTimeout(el._t);
  el._t = setTimeout(() => { el.className = 'toast'; }, 3000);
}

function esc(s) {
  if (s == null) return '';
  return String(s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
}

function dataFmt(d) {
  if (!d) return 'Data a definir';
  const dt = new Date(d);
  if (isNaN(dt)) return d;
  return dt.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit', year: 'numeric' });
}

function badgeStatus(s) {
  if (s === 'confirmada') return '<span class="badge badge-confirmada">Confirmada</span>';
  if (s === 'cancelada') return '<span class="badge badge-cancelada">Cancelada</span>';
  if (s === 'pendente') return '<span class="badge badge-pendente">Pendente</span>';
  return '';
}

function renderNav() {
  const nav = document.getElementById('nav');
  if (!usuario) {
    nav.innerHTML = '';
    return;
  }
  let html = `<span class="usuario-info">👤 ${esc(usuario.nome)} <small>(${usuario.tipo})</small></span>`;
  if (usuario.tipo === 'participante') {

    html += `<button onclick="irPara('minhas')">Minhas Inscrições</button>`;

    html += `<button onclick="irPara('criar')">Novo Evento</button>`;
  }
  html += `<button onclick="logout()">Sair</button>`;
  nav.innerHTML = html;
}

  const d = await api('/api/minhas-inscricoes');
  minhasInscricoes = d.inscricoes || [];
}

async function carregarMeusEventos() {
  // organizador vê eventos próprios; reutilizamos a lista de eventos gerais e filtramos
  // mas o backend só devolve eventos ativos de todos; então fazemos uma query específica:
  const all = await api('/api/eventos');
  eventos = all.eventos || [];
  meusEventos = eventos.filter(e => usuario && e.organizador_id === usuario.id);
}

async function carregarTudo() {
  if (!usuario) return;
  if (usuario.tipo === 'participante') {
    await Promise.all([carregarEventos(), carregarMinhas()]);
  } else {
    await carregarMeusEventos();
  }
}

async function salvarEvento(ev) {
  ev.preventDefault();
  const f = ev.target;
  const dados = {
    titulo: f.titulo.value,
    descricao: f.descricao.value,
    local: f.local.value,
    data: f.data.value,
    vagas: Number(f.vagas.value) || 0
  };
  try {
    if (eventoEditando) {
      await api('/api/eventos/' + eventoEditando.id, { method: 'PUT', body: JSON.stringify(dados) });
      toast('Evento atualizado!');
    } else {
      await api('/api/eventos', { method: 'POST', body: JSON.stringify(dados) });
      toast('Evento publicado!');
    }
    eventoEditando = null;
    irPara('painel');
  } catch (e) {
    document.getElementById('msg').innerHTML = `<div class="erro-msg">${esc(e.message)}</div>`;
  }
}

# Portal de Inscrição para Eventos

Portal web completo para inscrição em eventos, com contas de **participante** e **organizador**, tudo salvo no próprio servidor.

## O que faz

**Participante:**
- Vê todos os eventos disponíveis
- Se inscreve em um evento
- Vê suas inscrições e cancela quando quiser

**Organizador:**
- Cria, edita e cancela eventos
- Vê todas as inscrições dos eventos que criou
- Confirma a presença dos inscritos

Todas as informações ficam salvas no servidor (banco de dados SQLite), então funcionam de qualquer computador ou celular, e persistem mesmo fechando o site.

## Como instalar e rodar

Você precisa ter o **Node.js** instalado (versão 16 ou superior). Baixe em https://nodejs.org.

No terminal, dentro da pasta do projeto, execute:

```bash
npm install
npm start
```

Depois abra no navegador:

```
http://localhost:3000
```

O site já está funcionando. Crie uma conta escolhendo "Participante" ou "Organizador" na tela de cadastro.

## Colocar na internet (opcional)

Para que outras pessoas acessem de qualquer lugar, hospede em um serviço que rode Node.js, por exemplo:

- **Render** (render.com) — plano gratuito disponível
- **Railway** (railway.app)
- **Fly.io**

Passos gerais:
1. Crie uma conta no serviço escolhido
2. Suba esta pasta do projeto (via GitHub ou upload direto)
3. Configure o comando de início como `npm start`
4. O serviço dará um endereço público (ex: `https://seusite.onrender.com`)

Não é necessário criar contas em outros sites nem conectar com serviços externos — o banco de dados fica dentro do próprio projeto (`portal.db`).

## Estrutura

- `server.js` — servidor e API (backend)
- `public/index.html` — página principal
- `public/style.css` — aparência
- `public/app.js` — comportamento das telas
- `portal.db` — banco de dados (criado automaticamente na primeira execução)
