const express = require('express');
const path = require('path');
const crypto = require('crypto');
const Database = require('better-sqlite3');

const app = express();
const PORT = process.env.PORT || 3000;
const DB_FILE = process.env.DB_FILE || path.join(__dirname, 'agendamentos.db');

// ===== Banco de dados =====
const db = new Database(DB_FILE);
db.pragma('journal_mode = WAL');
db.pragma('foreign_keys = ON');

db.exec(`
  CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    salt TEXT NOT NULL,
    name TEXT NOT NULL,
    phone TEXT NOT NULL,
    role TEXT NOT NULL CHECK (role IN ('client','owner')),
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
  );

  CREATE TABLE IF NOT EXISTS services (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    description TEXT,
    price REAL NOT NULL DEFAULT 0,
    duration_min INTEGER NOT NULL DEFAULT 30,
    active INTEGER NOT NULL DEFAULT 1
  );

  CREATE TABLE IF NOT EXISTS time_slots (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    weekday INTEGER NOT NULL,
    time TEXT NOT NULL,
    UNIQUE (weekday, time)
  );

  CREATE TABLE IF NOT EXISTS bookings (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    client_id INTEGER NOT NULL,
    service_id INTEGER NOT NULL,
    date TEXT NOT NULL,
    time TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('pending','confirmed','rejected','cancelled')) DEFAULT 'pending',
    note TEXT,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (client_id) REFERENCES users(id),
    FOREIGN KEY (service_id) REFERENCES services(id)
  );

  CREATE UNIQUE INDEX IF NOT EXISTS idx_booking_unique
    ON bookings (service_id, date, time)
    WHERE status IN ('pending','confirmed');
`);

// ===== Senhas (hash) =====
function hashPassword(password, salt) {
  return crypto.scryptSync(password, salt, 64).toString('hex');
}
function makeSalt() {
  return crypto.randomBytes(16).toString('hex');
}

function checkAuth(req, res, next) {
  const user = req.session && req.session.user;
  if (!user) return res.status(401).json({ error: 'Faça login.' });
  next();
}
function requireOwner(req, res, next) {
  checkAuth(req, res, () => {
    if (req.session.user.role !== 'owner') return res.status(403).json({ error: 'Acesso negado.' });
    next();
  });
}

// ===== Sessões simples (JWT-like com expiração) =====
const sessions = new Map(); // token -> { user, expires }
const SESSION_MS = 7 * 24 * 60 * 60 * 1000;

function createSession(user) {
  const token = crypto.randomBytes(32).toString('hex');
  sessions.set(token, { user, expires: Date.now() + SESSION_MS });
  return token;
}
function readSession(req) {
  const token = (req.get('Authorization') || '').replace(/^Bearer /, '');
  const s = sessions.get(token);
  if (s && s.expires > Date.now()) {
    req.session = s;
    return true;
  }
  sessions.delete(token);
  return false;
}

// ===== Helpers de data =====
function todayISO() {
  const d = new Date();
  return d.toISOString().slice(0, 10);
}
function addDaysISO(days) {
  const d = new Date();
  d.setDate(d.getDate() + days);
  return d.toISOString().slice(0, 10);
}
function weekdayOf(dateStr) {
  // 1970-01-01 era quarta-feira (4). Usamos 0=Seg..6=Dom.
  const dt = new Date(dateStr + 'T12:00:00Z');
  const js = dt.getUTCDay(); // 0=Dom
  return (js + 6) % 7; // 0=Seg ... 6=Dom
}

app.post('/api/register', (req, res) => {
  const { username, password, name, phone } = req.body || {};
  if (!username || !password || !name || !phone) {
    return res.status(400).json({ error: 'Preencha nome, telefone, usuário e senha.' });
  }
  if (String(username).trim().length < 3) {
    return res.status(400).json({ error: 'Usuário deve ter pelo menos 3 caracteres.' });
  }
  if (String(password).length < 4) {
    return res.status(400).json({ error: 'Senha deve ter pelo menos 4 caracteres.' });
  }
  const u = db.prepare('SELECT id FROM users WHERE username = ?').get(String(username).trim());
  if (u) return res.status(409).json({ error: 'Este usuário já existe.' });

  const salt = makeSalt();
  const hash = hashPassword(String(password), salt);
  const info = db.prepare(
    'INSERT INTO users (username, password_hash, salt, name, phone, role) VALUES (?, ?, ?, ?, ?, ?)'
  ).run(String(username).trim(), hash, salt, String(name).trim(), String(phone).trim(), 'client');

  const user = db.prepare('SELECT id, username, name, phone, role FROM users WHERE id = ?').get(info.lastInsertRowid);
  const token = createSession(user);
  res.json({ token, user });
});

app.post('/api/login', (req, res) => {
  const { username, password } = req.body || {};
  if (!username || !password) return res.status(400).json({ error: 'Usuário e senha são obrigatórios.' });
  const row = db.prepare('SELECT * FROM users WHERE username = ?').get(String(username).trim());
  if (!row) return res.status(401).json({ error: 'Usuário ou senha incorretos.' });
  const hash = hashPassword(String(password), row.salt);
  if (hash !== row.password_hash) return res.status(401).json({ error: 'Usuário ou senha incorretos.' });
  const user = { id: row.id, username: row.username, name: row.name, phone: row.phone, role: row.role };
  const token = createSession(user);
  res.json({ token, user });
});

// Retorna, para um serviço e um intervalo de dias, quais horários estão livres.
app.get('/api/availability', checkAuth, (req, res) => {
  const { serviceId, days } = req.query;
  const sid = parseInt(serviceId, 10);
  const n = Math.min(parseInt(days, 10) || 14, 60);
  if (!sid) return res.status(400).json({ error: 'Informe o serviço.' });

  const svc = db.prepare('SELECT * FROM services WHERE id = ?').get(sid);
  if (!svc) return res.status(404).json({ error: 'Serviço não encontrado.' });

  const slots = db.prepare('SELECT weekday, time FROM time_slots ORDER BY weekday, time').all();
  const taken = db.prepare(
    `SELECT date, time FROM bookings
     WHERE service_id = ? AND status IN ('pending','confirmed') AND date >= ? AND date <= ?`
  ).all(sid, addDaysISO(0), addDaysISO(n - 1));
  const takenSet = new Set(taken.map(t => t.date + '|' + t.time));

  const result = [];
  for (let i = 0; i < n; i++) {
    const date = addDaysISO(i);
    const wd = weekdayOf(date);
    const free = slots.filter(s => s.weekday === wd && !takenSet.has(date + '|' + s.time)).map(s => s.time);
    result.push({ date, weekday: wd, freeTimes: free });
  }
  res.json({ serviceId: sid, durationMin: svc.duration_min, days: result });
});

app.post('/api/bookings', checkAuth, (req, res) => {
  const { serviceId, date, time, note } = req.body || {};
  const clientId = req.session.user.id;
  const sid = parseInt(serviceId, 10);
  if (!sid || !date || !time) return res.status(400).json({ error: 'Preencha serviço, data e horário.' });

  const svc = db.prepare('SELECT * FROM services WHERE id = ?').get(sid);
  if (!svc) return res.status(404).json({ error: 'Serviço não encontrado.' });

  // Valida se existe slot para esse dia da semana
  const wd = weekdayOf(date);
  const slotOk = db.prepare('SELECT 1 FROM time_slots WHERE weekday = ? AND time = ?').get(wd, time);
  if (!slotOk) return res.status(400).json({ error: 'Este horário não está disponível para este dia.' });

  // Verifica conflito (mesmo serviço, mesma data/hora)
  const conflict = db.prepare(
    "SELECT id FROM bookings WHERE service_id = ? AND date = ? AND time = ? AND status IN ('pending','confirmed')"
  ).get(sid, date, time);
  if (conflict) return res.status(409).json({ error: 'Este horário acabou de ser ocupado. Escolha outro.' });

  const info = db.prepare(
    "INSERT INTO bookings (client_id, service_id, date, time, status, note) VALUES (?, ?, ?, ?, 'pending', ?)"
  ).run(clientId, sid, date, time, note ? String(note).slice(0, 500) : null);

  const row = db.prepare('SELECT * FROM bookings WHERE id = ?').get(info.lastInsertRowid);
  res.status(201).json(row);
});

app.get('/api/bookings/mine', checkAuth, (req, res) => {
  const clientId = req.session.user.id;
  const rows = db.prepare(
    `SELECT b.id, b.date, b.time, b.status, b.created_at, b.note,
            s.name AS service_name, s.price
     FROM bookings b JOIN services s ON s.id = b.service_id
     WHERE b.client_id = ?
     ORDER BY b.date DESC, b.time DESC`
  ).all(clientId);
  res.json(rows);
});

app.post('/api/bookings/:id/cancel', checkAuth, (req, res) => {
  const id = parseInt(req.params.id, 10);
  const clientId = req.session.user.id;
  const row = db.prepare('SELECT * FROM bookings WHERE id = ?').get(id);
  if (!row) return res.status(404).json({ error: 'Agendamento não encontrado.' });
  if (row.client_id !== clientId && req.session.user.role !== 'owner') {
    return res.status(403).json({ error: 'Você não pode cancelar este agendamento.' });
  }
  if (!['pending', 'confirmed'].includes(row.status)) {
    return res.status(400).json({ error: 'Este agendamento já não pode ser cancelado.' });
  }
  db.prepare("UPDATE bookings SET status = 'cancelled' WHERE id = ?").run(id);
  res.json(db.prepare('SELECT * FROM bookings WHERE id = ?').get(id));
});

app.get('/api/owner/bookings', requireOwner, (req, res) => {
  const { status } = req.query;
  let sql = `SELECT b.id, b.date, b.time, b.status, b.created_at, b.note,
            s.name AS service_name, s.price,
            u.name AS client_name, u.phone AS client_phone, u.username AS client_username
     FROM bookings b
     JOIN services s ON s.id = b.service_id
     JOIN users u ON u.id = b.client_id`;
  const params = [];
  if (status) {
    sql += ' WHERE b.status = ?';
    params.push(status);
  }
  sql += ' ORDER BY b.date DESC, b.time DESC';
  res.json(db.prepare(sql).all(...params));
});

app.put('/api/owner/services/:id', requireOwner, (req, res) => {
  const id = parseInt(req.params.id, 10);
  const existing = db.prepare('SELECT * FROM services WHERE id = ?').get(id);
  if (!existing) return res.status(404).json({ error: 'Serviço não encontrado.' });
  const { name, description, price, duration_min, active } = req.body || {};
  const nameVal = name !== undefined ? String(name).trim() : existing.name;
  const descVal = description !== undefined ? String(description) : existing.description;
  const priceNum = price !== undefined ? (isNaN(parseFloat(price)) ? 0 : parseFloat(price)) : existing.price;
  const dur = duration_min !== undefined ? (parseInt(duration_min, 10) || existing.duration_min) : existing.duration_min;
  const act = active !== undefined ? (active ? 1 : 0) : existing.active;
  db.prepare('UPDATE services SET name=?, description=?, price=?, duration_min=?, active=? WHERE id=?')
    .run(nameVal, descVal, priceNum, dur, act, id);
  res.json(db.prepare('SELECT * FROM services WHERE id = ?').get(id));
});

const state = {
  token: localStorage.getItem('token') || null,

const STATUS_LABEL = { pending: 'Pendente', confirmed: 'Confirmado', rejected: 'Recusado', cancelled: 'Cancelado' };

// ===== API =====
async function api(path, opts = {}) {
  opts.headers = opts.headers || {};
  if (state.token) opts.headers['Authorization'] = 'Bearer ' + state.token;
  if (opts.body && typeof opts.body === 'object' && !(opts.body instanceof FormData)) {
    opts.headers['Content-Type'] = 'application/json';
    opts.body = JSON.stringify(opts.body);
  }
  const res = await fetch('/api' + path, opts);
  let data = null;
  try { data = await res.json(); } catch (e) {}
  if (!res.ok) {
    const msg = (data && data.error) ? data.error : 'Erro inesperado. Tente novamente.';
    throw new Error(msg);
  }
  return data;
}

function fmtDate(dateStr) {
  // dateStr = AAAA-MM-DD -> DD/MM/AAAA
  const [y, m, d] = dateStr.split('-');
  return d + '/' + m + '/' + y;
}
function fmtPrice(p) {
  const n = parseFloat(p || 0);
  if (!isFinite(n)) return 'Gratuito';
  return 'R$ ' + n.toFixed(2).replace('.', ',');
}
function esc(s) {
  return String(s == null ? '' : s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

function toast(msg, type = 'success') {
  const el = document.createElement('div');
  el.className = 'toast ' + type;
  el.textContent = msg;

  setTimeout(() => { el.style.opacity = '0'; el.style.transition = 'opacity .3s'; setTimeout(() => el.remove(), 300); }, 3000);
}

}

async function doLogin() {
  const username = document.getElementById('lUser').value.trim();
  const password = document.getElementById('lPass').value;
  if (!username || !password) return toast('Preencha usuário e senha.', 'error');
  try {
    const d = await api('/login', { method: 'POST', body: { username, password } });
    state.token = d.token; state.user = d.user;
    localStorage.setItem('token', d.token); localStorage.setItem('user', JSON.stringify(d.user));
    state.view = d.user.role === 'owner' ? 'owner' : 'services';
    renderAuthArea(); render();
    toast('Olá, ' + d.user.name + '!');
  } catch (e) { toast(e.message, 'error'); }
}

async function doRegister() {
  const name = document.getElementById('rName').value.trim();
  const phone = document.getElementById('rPhone').value.trim();
  const username = document.getElementById('rUser').value.trim();
  const password = document.getElementById('rPass').value;
  if (!name || !phone || !username || !password) return toast('Preencha todos os campos.', 'error');
  try {
    const d = await api('/register', { method: 'POST', body: { name, phone, username, password } });
    state.token = d.token; state.user = d.user;
    localStorage.setItem('token', d.token); localStorage.setItem('user', JSON.stringify(d.user));
    state.view = 'services';
    renderAuthArea(); render();
    toast('Conta criada! Agora escolha um serviço.');
  } catch (e) { toast(e.message, 'error'); }
}

}

async function renderServices(app) {
  app.innerHTML += `<div class="card"><h2>Escolha um serviço</h2><p class="muted" style="margin-top:0">Toque em um serviço para ver os horários livres.</p><div class="grid" id="svcGrid"><p class="muted">Carregando...</p></div></div><div class="avail-wrap" id="availWrap"></div>`;
  let services;
  try { services = await api('/services'); }
  catch (e) { document.getElementById('svcGrid').innerHTML = '<p class="muted">Erro ao carregar serviços: ' + esc(e.message) + '</p>'; return; }
  if (!services.length) {
    document.getElementById('svcGrid').innerHTML = '<p class="empty">Nenhum serviço cadastrado ainda.</p>';
    return;
  }
  const grid = document.getElementById('svcGrid');
  grid.innerHTML = services.map(s => `
    <div class="card service-card ${state.selectedService && state.selectedService.id === s.id ? 'selected' : ''}" onclick="pickService(${s.id})">
      <div class="service-name">${esc(s.name)}</div>
      <div class="service-desc">${esc(s.description || '')}</div>
      <div class="service-meta">
        <span class="price">${fmtPrice(s.price)}</span>
        <span class="duration">⏱ ${esc(s.duration_min)} min</span>
      </div>
    </div>`).join('');
}

async function pickService(id) {
  const s = await api('/services').then(a => a.find(x => x.id === id));
  state.selectedService = s;
  state.selectedDay = null; state.selectedTime = null;
  renderServices(document.getElementById('app'));
  loadAvailability();
}

async function loadAvailability() {
  const wrap = document.getElementById('availWrap');
  wrap.innerHTML = '<div class="card"><p class="muted">Carregando horários...</p></div>';
  let data;
  try { data = await api('/availability?serviceId=' + state.selectedService.id + '&days=21'); }
  catch (e) { wrap.innerHTML = '<div class="card"><p class="muted">Erro: ' + esc(e.message) + '</p></div>'; return; }
  wrap.innerHTML = `<div class="card">
    <h2>Horários para: ${esc(state.selectedService.name)}</h2>
    <p class="muted" style="margin-top:0">Escolha um dia e um horário livre.</p>
    <div class="avail-days" id="dayChips"></div>
    <div id="slotsArea"><p class="muted" style="margin-top:12px">Selecione um dia acima.</p></div>
    <div id="summaryArea"></div>
  </div>`;
  const chips = document.getElementById('dayChips');
  chips.innerHTML = data.days.map(d => {
    const hasFree = d.freeTimes.length > 0;
    return `<div class="day-chip ${!hasFree ? 'nofree' : ''}" data-date="${d.date}" onclick="${hasFree ? 'pickDay(\'' + d.date + '\')' : ''}">
      <div class="dw">${WEEKDAYS[d.weekday]}</div>
      <div class="dd">${d.date.slice(8,10)}</div>
    </div>`;
  }).join('');
  state._days = data.days;
}

function pickDay(date) {
  state.selectedDay = date;
  state.selectedTime = null;
  const day = state._days.find(d => d.date === date);
  document.querySelectorAll('.day-chip').forEach(c => c.classList.toggle('active', c.dataset.date === date));
  const area = document.getElementById('slotsArea');
  if (!day.freeTimes.length) { area.innerHTML = '<p class="muted" style="margin-top:12px">Nenhum horário livre neste dia.</p>'; }
  else {
    area.innerHTML = '<div class="slots">' + day.freeTimes.map(t =>
      `<div class="slot" onclick="pickTime('${t}')">${t}</div>`).join('') + '</div>';
  }
  document.getElementById('summaryArea').innerHTML = '';
}

function pickTime(time) {
  state.selectedTime = time;
  document.querySelectorAll('.slot').forEach(s => s.classList.toggle('selected', s.textContent === time));
  const s = state.selectedService;
  document.getElementById('summaryArea').innerHTML = `
    <div class="summary-bar">
      <div>
        <div style="font-weight:700">${esc(s.name)}</div>
        <div class="muted">${fmtDate(state.selectedDay)} às ${time} · ${fmtPrice(s.price)}</div>
      </div>
      <button class="btn" onclick="confirmBooking()">Confirmar agendamento</button>
    </div>`;
}

async function confirmBooking() {
  if (!state.selectedDay || !state.selectedTime) return toast('Escolha dia e horário.', 'error');

    const row = await api('/bookings', {

async function ownerLoadBookings(c) {
  let rows;
  try { rows = await api('/owner/bookings'); }
  catch (e) { c.innerHTML = '<p class="muted">Erro: ' + esc(e.message) + '</p>'; return; }
  let filter = 'all';
  if (!rows.length) { c.innerHTML = '<p class="empty">Nenhum agendamento ainda.</p>'; return; }
  function draw() {
    const shown = rows.filter(r => filter === 'all' ? true : r.status === filter);
    if (!shown.length) { c.innerHTML = '<p class="empty">Nenhum agendamento nesta situação.</p>'; return; }
    c.innerHTML = `<div class="filter-row" style="margin-top:14px">
        <select onchange="ownerFilterStatus(this.value)">
          <option value="all">Todos</option>
          <option value="pending">Pendentes</option>
          <option value="confirmed">Confirmados</option>
          <option value="rejected">Recusados</option>
          <option value="cancelled">Cancelados</option>
        </select>
      </div>
      <div class="section-title">Lista</div>
      ${shown.map(b => `
        <div class="booking">
          <div class="info">
            <div class="sname">${esc(b.service_name)}</div>
            <div class="stime">${fmtDate(b.date)} às ${esc(b.time)}</div>
            <div class="client">👤 ${esc(b.client_name)} · 📞 ${esc(b.client_phone)} · <span class="badge ${b.status}" style="margin-left:4px">${STATUS_LABEL[b.status]}</span></div>
          </div>
          <div class="actions">
            ${b.status === 'pending' ? `
              <button class="btn small" onclick="ownerAction(${b.id},'confirm')">Confirmar</button>
              <button class="btn small danger" onclick="ownerAction(${b.id},'reject')">Recusar</button>` : ''}
          </div>
        </div>`).join('')}
    `;
  }
  draw();
  window._ownerRows = rows;
}
function ownerFilterStatus(v) {
  const c = document.getElementById('ownerContent');
  const rows = window._ownerRows || [];
  const shown = rows.filter(r => v === 'all' ? true : r.status === v);
  if (!shown.length) { c.innerHTML = '<p class="empty">Nenhum agendamento nesta situação.</p>'; return; }
  c.innerHTML = `<div class="filter-row"><select onchange="ownerFilterStatus(this.value)">
      <option value="all">Todos</option>
      <option value="pending" ${v==='pending'?'selected':''}>Pendentes</option>
      <option value="confirmed" ${v==='
