=== README.md ===
# Biblioteca Digital

Sistema completo de biblioteca digital com empréstimo de itens.

## Requisitos
- Node.js 18+ instalado

## Instalação e Execução

```bash
# Instalar dependências
npm install

# Executar o projeto
npm start
```

O site estará disponível em: http://localhost:3000

## Contas padrão

### Bibliotecário
- Email: admin@biblioteca.com
- Senha: admin123

### Leitor
- Email: leitor@exemplo.com
- Senha: leitor123

## Como usar

### Para Leitores:
1. Faça login com sua conta
2. Busque itens no acervo
3. Clique em "Emprestar" para pegar um item
4. Veja seus empréstimos na aba "Meus Empréstimos"
5. Devolva itens quando terminar

### Para Bibliotecário:
1. Faça login com a conta de administrador
2. Cadastre novos itens na aba "Gerenciar Acervo"
3. Veja todos os empréstimos na aba "Todos os Empréstimos"
4. Bloqueie leitores com atraso se necessário

Todos os dados ficam salvos no servidor e sincronizam automaticamente entre dispositivos.

// Criar tabelas
db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT NOT NULL CHECK(tipo IN ('leitor', 'bibliotecario')),
    bloqueado INTEGER DEFAULT 0,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS itens (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    autor TEXT NOT NULL,
    tipo TEXT NOT NULL,
    ano INTEGER,
    isbn TEXT,
    descricao TEXT,
    quantidade_total INTEGER DEFAULT 1,
    quantidade_disponivel INTEGER DEFAULT 1,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS emprestimos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    item_id INTEGER NOT NULL,
    data_emprestimo DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_devolucao_prevista DATETIME NOT NULL,
    data_devolucao_real DATETIME,
    status TEXT DEFAULT 'ativo' CHECK(status IN ('ativo', 'devolvido', 'atrasado')),
    FOREIGN KEY (usuario_id) REFERENCES usuarios (id),
    FOREIGN KEY (item_id) REFERENCES itens (id)
  );

  CREATE INDEX IF NOT EXISTS idx_emprestimos_usuario ON emprestimos(usuario_id);
  CREATE INDEX IF NOT EXISTS idx_emprestimos_item ON emprestimos(item_id);
  CREATE INDEX IF NOT EXISTS idx_emprestimos_status ON emprestimos(status);
`);

// Middleware de autenticação
function requireAuth(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({ erro: 'Não autenticado' });
  }
  next();
}

function requireBibliotecario(req, res, next) {
  if (!req.session.userId || req.session.userTipo !== 'bibliotecario') {
    return res.status(403).json({ erro: 'Acesso negado' });
  }
  next();
}

// Rotas de autenticação
app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;

  const usuario = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email);

  if (!usuario || !bcrypt.compareSync(senha, usuario.senha)) {
    return res.status(401).json({ erro: 'Email ou senha inválidos' });
  }

  req.session.userId = usuario.id;
  req.session.userNome = usuario.nome;
  req.session.userTipo = usuario.tipo;
  req.session.userBloqueado = usuario.bloqueado;

  res.json({
    id: usuario.id,
    nome: usuario.nome,
    email: usuario.email,
    tipo: usuario.tipo,
    bloqueado: usuario.bloqueado
  });
});

app.post('/api/registrar', (req, res) => {
  const { nome, email, senha } = req.body;

  if (!nome || !email || !senha) {
    return res.status(400).json({ erro: 'Todos os campos são obrigatórios' });
  }

  const usuarioExiste = db.prepare('SELECT id FROM usuarios WHERE email = ?').get(email);
  if (usuarioExiste) {
    return res.status(400).json({ erro: 'Email já cadastrado' });
  }

  const senhaHash = bcrypt.hashSync(senha, 10);

  try {
    const result = db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)').run(
      nome,
      email,
      senhaHash,
      'leitor'
    );

    req.session.userId = result.lastInsertRowid;
    req.session.userNome = nome;
    req.session.userTipo = 'leitor';
    req.session.userBloqueado = 0;

    res.json({
      id: result.lastInsertRowid,
      nome,
      email,
      tipo: 'leitor',
      bloqueado: 0
    });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao registrar usuário' });
  }
});

app.post('/api/logout', (req, res) => {

app.get('/api/itens', requireAuth, (req, res) => {
  const { busca } = req.query;
  let query = 'SELECT * FROM itens';
  let params = [];

  if (busca) {
    query += ' WHERE titulo LIKE ? OR autor LIKE ? OR tipo LIKE ?';
    const buscaParam = `%${busca}%`;
    params = [buscaParam, buscaParam, buscaParam];
  }

  query += ' ORDER BY titulo';

  const itens = db.prepare(query).all(...params);
  res.json(itens);
});

app.post('/api/itens', requireBibliotecario, (req, res) => {
  const { titulo, autor, tipo, ano, isbn, descricao, quantidade_total } = req.body;

  if (!titulo || !autor || !tipo || !quantidade_total) {
    return res.status(400).json({ erro: 'Campos obrigatórios faltando' });
  }

  try {
    const result = db.prepare(`
      INSERT INTO itens (titulo, autor, tipo, ano, isbn, descricao, quantidade_total, quantidade_disponivel)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `).run(titulo, autor, tipo, ano || null, isbn || null, descricao || null, quantidade_total, quantidade_total);

    const item = db.prepare('SELECT * FROM itens WHERE id = ?').get(result.lastInsertRowid);
    res.json(item);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao cadastrar item' });
  }
});

app.put('/api/itens/:id', requireBibliotecario, (req, res) => {
  const { titulo, autor, tipo, ano, isbn, descricao, quantidade_total } = req.body;
  const itemId = req.params.id;

  const itemAtual = db.prepare('SELECT * FROM itens WHERE id = ?').get(itemId);
  if (!itemAtual) {
    return res.status(404).json({ erro: 'Item não encontrado' });
  }

  const emprestimosAtivos = db.prepare('SELECT COUNT(*) as count FROM emprestimos WHERE item_id = ? AND status = ?')
    .get(itemId, 'ativo').count;

  const novaQuantidadeDisponivel = quantidade_total - emprestimosAtivos;

  if (novaQuantidadeDisponivel < 0) {
    return res.status(400).json({ erro: 'Quantidade total não pode ser menor que empréstimos ativos' });
  }

  try {
    db.prepare(`
      UPDATE itens
      SET titulo = ?, autor = ?, tipo = ?, ano = ?, isbn = ?, descricao = ?,
          quantidade_total = ?, quantidade_disponivel = ?
      WHERE id = ?
    `).run(titulo, autor, tipo, ano || null, isbn || null, descricao || null, quantidade_total, novaQuantidadeDisponivel, itemId);

    const item = db.prepare('SELECT * FROM itens WHERE id = ?').get(itemId);
    res.json(item);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao atualizar item' });
  }
});

app.delete('/api/itens/:id', requireBibliotecario, (req, res) => {
  const itemId = req.params.id;

  const emprestimosAtivos = db.prepare('SELECT COUNT(*) as count FROM emprestimos WHERE item_id = ? AND status = ?')
    .get(itemId, 'ativo').count;

  if (emprestimosAtivos > 0) {
    return res.status(400).json({ erro: 'Não é possível remover item com empréstimos ativos' });
  }

  try {
    db.prepare('DELETE FROM itens WHERE id = ?').run(itemId);
    res.json({ sucesso: true });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao remover item' });
  }
});

// Rotas de empréstimos
app.get('/api/emprestimos/meus', requireAuth, (req, res) => {
  const emprestimos = db.prepare(`
    SELECT e.*, i.titulo, i.autor, i.tipo, u.nome as usuario_nome
    FROM emprestimos e
    JOIN itens i ON e.item_id = i.id
    JOIN usuarios u ON e.usuario_id = u.id
    WHERE e.usuario_id = ?
    ORDER BY e.data_emprestimo DESC
  `).all(req.session.userId);

  // Atualizar status de atrasados
  const hoje = new Date().toISOString();
  emprestimos.forEach(emp => {
    if (emp.status === 'ativo' && emp.data_devolucao_prevista < hoje) {
      db.prepare('UPDATE emprestimos SET status = ? WHERE id = ?').run('atrasado', emp.id);
      emp.status = 'atrasado';
    }
  });

  res.json(emprestimos);
});

app.post('/api/emprestimos', requireAuth, (req, res) => {
  const { item_id } = req.body;
  const usuario_id = req.session.userId;

  // Verificar se usuário está bloqueado
  const usuario = db.prepare('SELECT bloqueado FROM usuarios WHERE id = ?').get(usuario_id);
  if (usuario.bloqueado) {
    return res.status(403).json({ erro: 'Usuário bloqueado. Entre em contato com a biblioteca.' });
  }

  // Verificar disponibilidade
  const item = db.prepare('SELECT * FROM itens WHERE id = ?').get(item_id);
  if (!item || item.quantidade_disponivel <= 0) {
    return res.status(400).json({ erro: 'Item não disponível para empréstimo' });
  }

  // Verificar se já tem empréstimo ativo deste item
  const emprestimoExistente = db.prepare(
    'SELECT id FROM emprestimos WHERE usuario_id = ? AND item_id = ? AND status = ?'
  ).get(usuario_id, item_id, 'ativo');

  if (emprestimoExistente) {
    return res.status(400).json({ erro: 'Você já possui este item emprestado' });
  }

  // Data de devolução: 14 dias
  const dataDevolucao = new Date();
  dataDevolucao.setDate(dataDevolucao.getDate() + 14);

  try {
    const result = db.prepare(`
      INSERT INTO emprestimos (usuario_id, item_id, data_devolucao_prevista)
      VALUES (?, ?, ?)
    `).run(usuario_id, item_id, dataDevolucao.toISOString());

    db.prepare('UPDATE itens SET quantidade_disponivel = quantidade_disponivel - 1 WHERE id = ?').run(item_id);

    const emprestimo = db.prepare(`
      SELECT e.*, i.titulo, i.autor, i.tipo
      FROM emprestimos e
      JOIN itens i ON e.item_id = i.id
      WHERE e.id = ?
    `).get(result.lastInsertRowid);

    res.json(emprestimo);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao realizar empréstimo' });
  }
});

app.post('/api/emprestimos/:id/devolver', requireAuth, (req, res) => {
  const emprestimoId = req.params.id;
  const usuario_id = req.session.userId;

  const emprestimo = db.prepare('SELECT * FROM emprestimos WHERE id = ?').get(emprestimoId);

  if (!emprestimo) {
    return res.status(404).json({ erro: 'Empréstimo não encontrado' });
  }

  if (emprestimo.usuario_id !== usuario_id && req.session.userTipo !== 'bibliotecario') {
    return res.status(403).json({ erro: 'Sem permissão para devolver este empréstimo' });
  }

  if (emprestimo.status === 'devolvido') {
    return res.status(400).json({ erro: 'Item já foi devolvido' });
  }

  try {
    const dataAtual = new Date().toISOString();

    db.prepare(`
      UPDATE emprestimos
      SET status = 'devolvido', data_devolucao_real = ?
      WHERE id = ?
    `).run(dataAtual, emprestimoId);

    db.prepare('UPDATE itens SET quantidade_disponivel = quantidade_disponivel + 1 WHERE id = ?')
      .run(emprestimo.item_id);

    const emprestimoAtualizado = db.prepare(`
      SELECT e.*, i.titulo, i.autor, i.tipo
      FROM emprestimos e
      JOIN itens i ON e.item_id = i.id
      WHERE e.id = ?
    `).get(emprestimoId);

    res.json(emprestimoAtualizado);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao devolver item' });
  }
});

// Rotas de gerenciamento de usuários (bibliotecário)
app.get('/api/usuarios', requireBibliotecario, (req, res) => {
  const usuarios = db.prepare(`
    SELECT u.id, u.nome, u.email, u.tipo, u.bloqueado, u.criado_em,
           COUNT(CASE WHEN e.status = 'atrasado' THEN 1 END) as emprestimos_atrasados
    FROM usuarios u
    LEFT JOIN emprestimos e ON u.id = e.usuario_id
    WHERE u.tipo = 'leitor'
    GROUP BY u.id
    ORDER BY u.nome
  `).all();

  res.json(usuarios);
});

app.post('/api/usuarios/:id/bloquear', requireBibliotecario, (req, res) => {
  const usuarioId = req.params.id;
  const { bloqueado } = req.body;

  try {
    db.prepare('UPDATE usuarios SET bloqueado = ? WHERE id = ?').run(bloqueado ? 1 : 0, usuarioId);
    const usuario = db.prepare('SELECT id, nome, email, tipo, bloqueado FROM usuarios WHERE id = ?').get(usuarioId);
    res.json(usuario);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao atualizar usuário' });
  }
});

let usuarioAtual = null;

// Inicialização
document.addEventListener('DOMContentLoaded', () => {
  verificarAutenticacao();
  setupEventListeners();
});

function setupEventListeners() {
  // Login
  document.getElementById('login-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    await fazerLogin();
  });

  // Registro
  document.getElementById('registrar-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    await fazerRegistro();
  });

  // Busca no acervo
  document.getElementById('busca-acervo').addEventListener('keyup', (e) => {
    if (e.key === 'Enter') {
      buscarAcervo();
    }
  });

  // Form de item
  document.getElementById('item-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    await salvarItem();
  });
}

async function verificarAutenticacao() {
  try {
    const response = await fetch('/api/usuario-atual');
    if (response.ok) {
      usuarioAtual = await response.json();
      mostrarTelaPrincipal();
    } else {
      mostrarTelaLogin();
    }
  } catch (erro) {
    mostrarTelaLogin();
  }
}

function mostrarTelaLogin() {
  document.getElementById('login-screen').classList.add('active');
  document.getElementById('main-screen').classList.remove('active');
}

function mostrarTelaPrincipal() {
  document.getElementById('login-screen').classList.remove('active');
  document.getElementById('main-screen').classList.add('active');
  
  document.getElementById('usuario-nome').textContent = `Olá, ${usuarioAtual.nome}`;

  // Mostrar/ocultar tabs baseado no tipo de usuário
  if (usuarioAtual.tipo === 'bibliotecario') {
    document.getElementById('tab-gerenciar').style.display = 'block';
    document.getElementById('tab-todos-emprestimos').style.display = 'block';
    document.getElementById('tab-usuarios').style.display = 'block';
  } else {
    document.getElementById('tab-gerenciar').style.display = 'none';
    document.getElementById('tab-todos-emprestimos').style.display = 'none';
    document.getElementById('tab-usuarios').style.display = 'none';
  }

  // Mostrar alerta se bloqueado
  if (usuarioAtual.bloqueado) {
    document.getElementById('alerta-bloqueado').style.display = 'block';
  } else {
    document.getElementById('alerta-bloqueado').style.display = 'none';
  }

  carregarAcervo();
}

// Autenticação
async function fazerLogin() {
  const email = document.getElementById('login-email').value;
  const senha = document.getElementById('login-senha').value;

  try {
    const response = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, senha })
    });

    if (response.ok) {
      usuarioAtual = await response.json();
      mostrarTelaPrincipal();
    } else {
      const erro = await response.json();
      mostrarErro('login-erro', erro.erro || 'Erro ao fazer login');
    }
  } catch (erro) {
    mostrarErro('login-erro', 'Erro ao conectar com o servidor');
  }
}

async function fazerRegistro() {
  const nome = document.getElementById('registrar-nome').value;
  const email = document.getElementById('registrar-email').value;
  const senha = document.getElementById('registrar-senha').value;

  try {
    const response = await fetch('/api/registrar', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ nome, email, senha })
    });

    if (response.ok) {
      usuarioAtual = await response.json();
      mostrarTelaPrincipal();
    } else {
      const erro = await response.json();
      mostrarErro('registrar-erro', erro.erro || 'Erro ao registrar');
    }
  } catch (erro) {
    mostrarErro('registrar-erro', 'Erro ao conectar com o servidor');
  }
}

async function sair() {
  try {
    await fetch('/api/logout', { method: 'POST' });
    usuarioAtual = null;
    mostrarTelaLogin();
    limparFormularios();
  } catch (erro) {
    console.error('Erro ao sair:', erro);
  }
}

function mostrarSecao(secao) {
  // Tabs principais
  document.querySelectorAll('.tab-main').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.secao').forEach(s => s.classList.remove('active'));
  
  event.target.classList.add('active');
  document.getElementById(`secao-${secao}`).classList.add('active');
  
  // Carregar dados da seção
  if (secao === 'acervo') {
    carregarAcervo();
  } else if (secao === 'meus-emprestimos') {
    carregarMeusEmprestimos();
  } else if (secao === 'gerenciar-acervo') {
    carregarGerenciarAcervo();
  } else if (secao === 'todos-emprestimos') {
    carregarTodosEmprestimos();
  } else if (secao === 'usuarios') {
    carregarUsuarios();
  }
}

// Acervo
async function carregarAcervo() {
  const container = document.getElementById('lista-acervo');
  container.innerHTML = '<div class="loading">Carregando acervo...</div>';

  try {
    const busca = document.getElementById('busca-acervo').value;
    const url = busca ? `/api/itens?busca=${encodeURIComponent(busca)}` : '/api/itens';
    const response = await fetch(url);
    const itens = await response.json();

    if (itens.length === 0) {
      container.innerHTML = `
        <div class="empty-state">
          <div class="empty-state-icon">📚</div>
          <div class="empty-state-text">Nenhum item encontrado</div>
        </div>
      `;
      return;
    }

    container.innerHTML = itens.map(item => `
      <div class="card">
        <div class="card-title">${item.titulo}</div>
        <div class="card-info">👤 ${item.autor}</div>
        <div class="card-info">📖 ${item.tipo}</div>
        ${item.ano ? `<div class="card-info">📅 ${item.ano}</div>` : ''}
        ${item.isbn ? `<div class="card-info">🔢 ${item.isbn}</div>` : ''}
        ${item.descricao ? `<div class="card-info" style="margin-top: 10px;">${item.descricao}</div>` : ''}
        <div class="card-info" style="margin-top: 10px;">
          <strong>Disponíveis:</strong> ${item.quantidade_disponivel} de ${item.quantidade_total}
        </div>
        <span class="card-badge ${item.quantidade_disponivel > 0 ? 'badge-disponivel' : 'badge-indisponivel'}">
          ${item.quantidade_disponivel > 0 ? 'Disponível' : 'Indisponível'}
        </span>
        ${item.quantidade_disponivel > 0 && !usuarioAtual.bloqueado ? `
          <div class="card-actions">
            <button class="btn btn-primary" onclick="emprestarItem(${item.id})">
              Emprestar
            </button>
          </div>
        ` : ''}
      </div>
    `).join('');
  } catch (erro) {
    container.innerHTML = `
      <div class="empty-state">
        <div class="empty-state-icon">⚠️</div>
        <div class="empty-state-text">Erro ao carregar acervo</div>
      </div>
    `;
  }
}

function buscarAcervo() {
  carregarAcervo();
}

async function emprestarItem(itemId) {
  if (!confirm('Deseja emprestar este item?')) return;

  try {
    const response = await fetch('/api/emprestimos', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ item_id: itemId })
    });

    if (response.ok) {
      alert('Item emprestado com sucesso!');
      carregarAcervo();
    } else {
      const erro = await response.json();
      alert(erro.erro || 'Erro ao emprestar item');
    }
  } catch (erro) {
    alert('Erro ao conectar com o servidor');
  }
}

    container.innerHTML = `

async function salvarItem() {
  const itemId = document.getElementById('item-id').value;
  const dados = {
    titulo: document.getElementById('item-titulo').value,
    autor: document.getElementById('item-autor').value,
    tipo: document.getElementById('item-tipo').value,
    ano: document.getElementById('item-ano').value || null,
    isbn: document.getElementById('item-isbn').value || null,
    descricao: document.getElementById('item-descricao').value || null,
    quantidade_total: parseInt(document.getElementById('item-quantidade').value)
  };

  try {
    const url = itemId ? `/api/itens/${itemId}` : '/api/itens';
    const method = itemId ? 'PUT' : 'POST';

    const response = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(dados)
    });

    if (response.ok) {
      alert('Item salvo com sucesso!');
      cancelarFormItem();
      carregarGerenciarAcervo();
    } else {
      const erro = await response.json();
      mostrarErro('item-erro', erro.erro || 'Erro ao salvar item');
    }
  } catch (erro) {
    mostrarErro('item-erro', 'Erro ao conectar com o servidor');
  }
}

  try {
    const response = await fetch('/api/usuarios');
    const usuarios = await response.json();

      container.innerHTML = `

          <div class="empty-state-icon">👥</div>

                  Desbloquear
                </button>
              ` : `
                <button class="btn btn-warning" onclick="bloquearUsuario(${usuario.id})">
                  Bloquear
                </button>

          </div>
        `).join('')}

// Utilitários
function formatarData(dataISO) {
  const data = new Date(dataISO);
  return data.toLocaleDateString('pt-BR', {
    day: '2-digit',
    month: '2-digit',
    year: 'numeric'
  });
}

function mostrarErro(elementoId, mensagem) {
  const elemento = document.getElementById(elementoId);
  elemento.textContent = mensagem;
  elemento.classList.add('show');
}

function limparErros() {
  document.querySelectorAll('.erro-msg').forEach(el => {
    el.classList.remove('show');
    el.textContent = '';
  });
}

function limparFormularios() {
  document.getElementById('login-form').reset();
  document.getElementById('registrar-form').reset();
  limparErros();
}
