=== README.md ===
# Sistema de Gestão de Tarefas de Equipe

## Requisitos
- Node.js (versão 14 ou superior)
- npm (geralmente vem com Node.js)

## Instalação

1. Instale o Node.js do site oficial: https://nodejs.org/

2. Abra o terminal/prompt de comando na pasta do projeto

3. Instale as dependências:
```bash
npm install
```

## Executar o projeto

1. Inicie o servidor:
```bash
npm start
```

2. Abra o navegador e acesse:
```
http://localhost:3000
```

## Como usar

### Para o Gerente:
- Criar tarefas e atribuir para membros da equipe
- Editar e remover qualquer tarefa
- Ver todas as tarefas de todos os membros
- Gerenciar usuários (criar novos membros)

### Para Membros da Equipe:
- Ver tarefas atribuídas a você
- Mudar o status das suas tarefas (Pendente → Em Progresso → Concluída)
- Criar tarefas para você mesmo

## Notas importantes

- Todos os dados ficam salvos no arquivo `database.json`
- Os dados persistem mesmo após fechar o navegador
- Funciona em qualquer dispositivo (computador, celular, tablet)
- Múltiplos usuários podem usar ao mesmo tempo

const express = require('express');
const session = require('express-session');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 3000;
const DB_FILE = path.join(__dirname, 'database.json');

// Middleware
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static('public'));
app.use(session({
  secret: 'sistema-tarefas-secreto-2024',
  resave: false,
  saveUninitialized: false,
  cookie: { maxAge: 24 * 60 * 60 * 1000 } // 24 horas
}));

// Inicializar banco de dados
function initDatabase() {
  if (!fs.existsSync(DB_FILE)) {
    const initialData = {
      users: [
        { id: 1, username: 'gerente', password: 'gerente123', name: 'Gerente', role: 'manager' },

        { id: 3, username: 'maria', password: 'maria123', name: 'Maria Santos', role: 'member' },

      ],
      tasks: [
        { id: 1, title: 'Revisar relatório mensal', description: 'Revisar e aprovar o relatório de vendas', assignedTo: 2, status: 'pending', createdBy: 1, createdAt: new Date().toISOString() },

      ],

      nextTaskId: 4

    fs.writeFileSync(DB_FILE, JSON.stringify(initialData, null, 2));
  }
}

// Funções de leitura e escrita do banco de dados
function readDatabase() {
  const data = fs.readFileSync(DB_FILE, 'utf8');
  return JSON.parse(data);
}

function writeDatabase(data) {
  fs.writeFileSync(DB_FILE, JSON.stringify(data, null, 2));
}

// Middleware de autenticação
function requireAuth(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({ error: 'Não autorizado' });
  }
  next();
}

function requireManager(req, res, next) {
  const db = readDatabase();
  const user = db.users.find(u => u.id === req.session.userId);
  if (!user || user.role !== 'manager') {
    return res.status(403).json({ error: 'Acesso negado. Apenas gerentes podem realizar esta ação.' });
  }
  next();
}

// Rotas de autenticação
app.post('/api/login', (req, res) => {
  const { username, password } = req.body;
  const db = readDatabase();
  
  const user = db.users.find(u => u.username === username && u.password === password);
  
  if (user) {
    req.session.userId = user.id;
    res.json({ 
      success: true, 
      user: { 
        id: user.id, 
        name: user.name, 
        username: user.username, 
        role: user.role 
      } 
    });
  } else {
    res.status(401).json({ error: 'Usuário ou senha incorretos' });
  }
});

app.post('/api/logout', (req, res) => {
  req.session.destroy();
  res.json({ success: true });
});

app.get('/api/me', requireAuth, (req, res) => {
  const db = readDatabase();
  const user = db.users.find(u => u.id === req.session.userId);
  
  if (user) {
    res.json({ 
      id: user.id, 
      name: user.name, 
      username: user.username, 
      role: user.role 
    });
  } else {
    res.status(404).json({ error: 'Usuário não encontrado' });
  }
});

// Rotas de usuários
app.get('/api/users', requireAuth, (req, res) => {
  const db = readDatabase();
  const users = db.users.map(u => ({ 
    id: u.id, 
    name: u.name, 
    username: u.username, 
    role: u.role 
  }));
  res.json(users);
});

app.post('/api/users', requireAuth, requireManager, (req, res) => {
  const { username, password, name } = req.body;
  
  if (!username || !password || !name) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios' });
  }
  
  const db = readDatabase();
  
  if (db.users.find(u => u.username === username)) {
    return res.status(400).json({ error: 'Nome de usuário já existe' });
  }
  
  const newUser = {
    id: db.nextUserId,
    username,
    password,
    name,
    role: 'member'
  };
  
  db.users.push(newUser);
  db.nextUserId++;
  writeDatabase(db);
  
  res.json({ 
    id: newUser.id, 
    name: newUser.name, 
    username: newUser.username, 
    role: newUser.role 
  });
});

// Rotas de tarefas
app.get('/api/tasks', requireAuth, (req, res) => {
  const db = readDatabase();
  const currentUser = db.users.find(u => u.id === req.session.userId);
  
  let tasks = db.tasks;
  
  // Membros só veem suas próprias tarefas
  if (currentUser.role === 'member') {
    tasks = tasks.filter(t => t.assignedTo === currentUser.id);
  }
  
  // Adicionar informações do usuário atribuído
  tasks = tasks.map(task => {
    const assignedUser = db.users.find(u => u.id === task.assignedTo);
    return {
      ...task,
      assignedToName: assignedUser ? assignedUser.name : 'Não atribuído'
    };
  });
  
  res.json(tasks);
});

app.post('/api/tasks', requireAuth, (req, res) => {
  const { title, description, assignedTo } = req.body;
  
  if (!title) {
    return res.status(400).json({ error: 'Título é obrigatório' });
  }
  
  const db = readDatabase();
  const currentUser = db.users.find(u => u.id === req.session.userId);
  
  // Membros só podem criar tarefas para si mesmos
  let finalAssignedTo = assignedTo;
  if (currentUser.role === 'member') {
    finalAssignedTo = currentUser.id;
  }
  
  const newTask = {
    id: db.nextTaskId,
    title,
    description: description || '',
    assignedTo: finalAssignedTo || currentUser.id,
    status: 'pending',
    createdBy: currentUser.id,
    createdAt: new Date().toISOString()
  };
  
  db.tasks.push(newTask);
  db.nextTaskId++;
  writeDatabase(db);
  
  const assignedUser = db.users.find(u => u.id === newTask.assignedTo);
  res.json({
    ...newTask,
    assignedToName: assignedUser ? assignedUser.name : 'Não atribuído'
  });
});

app.put('/api/tasks/:id', requireAuth, (req, res) => {
  const taskId = parseInt(req.params.id);
  const { title, description, assignedTo, status } = req.body;
  
  const db = readDatabase();
  const currentUser = db.users.find(u => u.id === req.session.userId);
  const taskIndex = db.tasks.findIndex(t => t.id === taskId);
  
  if (taskIndex === -1) {
    return res.status(404).json({ error: 'Tarefa não encontrada' });
  }
  
  const task = db.tasks[taskIndex];
  
  // Membros só podem editar suas próprias tarefas e apenas o status
  if (currentUser.role === 'member') {
    if (task.assignedTo !== currentUser.id) {
      return res.status(403).json({ error: 'Você só pode editar suas próprias tarefas' });
    }
    // Membros só podem alterar o status
    if (status) {
      task.status = status;
    }
  } else {
    // Gerentes podem editar tudo
    if (title) task.title = title;
    if (description !== undefined) task.description = description;
    if (assignedTo) task.assignedTo = assignedTo;
    if (status) task.status = status;
  }
  
  db.tasks[taskIndex] = task;
  writeDatabase(db);
  
  const assignedUser = db.users.find(u => u.id === task.assignedTo);
  res.json({
    ...task,
    assignedToName: assignedUser ? assignedUser.name : 'Não atribuído'
  });
});

app.delete('/api/tasks/:id', requireAuth, requireManager, (req, res) => {
  const taskId = parseInt(req.params.id);
  
  const db = readDatabase();
  const taskIndex = db.tasks.findIndex(t => t.id === taskId);
  
  if (taskIndex === -1) {
    return res.status(404).json({ error: 'Tarefa não encontrada' });
  }
  
  db.tasks.splice(taskIndex, 1);
  writeDatabase(db);
  
  res.json({ success: true });
});

// Inicializar e iniciar servidor
initDatabase();

app.listen(PORT, () => {

// Estado da aplicação
let currentUser = null;
let allTasks = [];
let allUsers = [];

// Elementos do DOM
const loginScreen = document.getElementById('login-screen');
const mainScreen = document.getElementById('main-screen');
const loginForm = document.getElementById('login-form');
const loginError = document.getElementById('login-error');
const logoutBtn = document.getElementById('logout-btn');
const userNameSpan = document.getElementById('user-name');
const newTaskBtn = document.getElementById('new-task-btn');
const newUserBtn = document.getElementById('new-user-btn');
const tasksContainer = document.getElementById('tasks-container');
const emptyState = document.getElementById('empty-state');
const statusFilter = document.getElementById('status-filter');
const taskModal = document.getElementById('task-modal');
const taskModalTitle = document.getElementById('task-modal-title');
const taskForm = document.getElementById('task-form');
const userModal = document.getElementById('user-modal');
const userForm = document.getElementById('user-form');
const userError = document.getElementById('user-error');

// Inicializar aplicação
document.addEventListener('DOMContentLoaded', () => {
  checkAuth();
  setupEventListeners();
});

// Verificar autenticação
async function checkAuth() {
  try {
    const response = await fetch('/api/me');
    if (response.ok) {
      currentUser = await response.json();
      showMainScreen();
    } else {
      showLoginScreen();
    }
  } catch (error) {
    showLoginScreen();
  }
}

// Configurar event listeners
function setupEventListeners() {
  // Login
  loginForm.addEventListener('submit', handleLogin);
  
  // Logout
  logoutBtn.addEventListener('click', handleLogout);
  
  // Nova tarefa
  newTaskBtn.addEventListener('click', () => openTaskModal());
  
  // Novo usuário
  newUserBtn.addEventListener('click', () => openUserModal());
  
  // Filtro de status
  statusFilter.addEventListener('change', renderTasks);
  
  // Formulário de tarefa
  taskForm.addEventListener('submit', handleTaskSubmit);
  
  // Formulário de usuário
  userForm.addEventListener('submit', handleUserSubmit);
  
  // Fechar modais
  document.querySelectorAll('.close-modal').forEach(btn => {
    btn.addEventListener('click', closeModals);
  });
  
  // Fechar modal clicando fora
  window.addEventListener('click', (e) => {
    if (e.target.classList.contains('modal')) {
      closeModals();
    }
  });
}

// Login
async function handleLogin(e) {
  e.preventDefault();
  
  const username = document.getElementById('username').value;
  const password = document.getElementById('password').value;
  
  try {
    const response = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username, password })
    });
    
    const data = await response.json();
    
    if (response.ok) {
      currentUser = data.user;
      showMainScreen();
    } else {
      showError(loginError, data.error);
    }
  } catch (error) {
    showError(loginError, 'Erro ao fazer login. Tente novamente.');
  }
}

// Logout
async function handleLogout() {
  try {
    await fetch('/api/logout', { method: 'POST' });
    currentUser = null;
    showLoginScreen();
  } catch (error) {
    console.error('Erro ao fazer logout:', error);
  }
}

// Mostrar tela de login
function showLoginScreen() {
  loginScreen.classList.add('active');
  mainScreen.classList.remove('active');
  loginForm.reset();
  loginError.classList.remove('show');
}

// Mostrar tela principal
function showMainScreen() {
  loginScreen.classList.remove('active');
  mainScreen.classList.add('active');
  userNameSpan.textContent = currentUser.name;
  
  // Mostrar botão de novo usuário apenas para gerentes
  if (currentUser.role === 'manager') {
    newUserBtn.style.display = 'block';
  } else {
    newUserBtn.style.display = 'none';
  }
  
  loadData();
}

// Carregar dados
async function loadData() {
  await Promise.all([loadTasks(), loadUsers()]);
  renderTasks();
}

// Carregar tarefas
async function loadTasks() {
  try {
    const response = await fetch('/api/tasks');
    if (response.ok) {
      allTasks = await response.json();
    }
  } catch (error) {
    console.error('Erro ao carregar tarefas:', error);
  }
}

// Carregar usuários
async function loadUsers() {
  try {
    const response = await fetch('/api/users');
    if (response.ok) {
      allUsers = await response.json();
      updateUserSelects();
    }
  } catch (error) {
    console.error('Erro ao carregar usuários:', error);
  }
}

// Atualizar selects de usuários
function updateUserSelects() {
  const taskAssignedSelect = document.getElementById('task-assigned');
  taskAssignedSelect.innerHTML = '<option value="">Selecione um membro</option>';
  
  const members = allUsers.filter(u => u.role === 'member');
  members.forEach(user => {
    const option = document.createElement('option');
    option.value = user.id;
    option.textContent = user.name;
    taskAssignedSelect.appendChild(option);
  });
}

// Renderizar tarefas
function renderTasks() {
  const filter = statusFilter.value;
  let filteredTasks = allTasks;
  
  if (filter !== 'all') {
    filteredTasks = allTasks.filter(task => task.status === filter);
  }
  
  if (filteredTasks.length === 0) {
    tasksContainer.style.display = 'none';
    emptyState.style.display = 'block';
    return;
  }
  
  tasksContainer.style.display = 'grid';
  emptyState.style.display = 'none';
  
  tasksContainer.innerHTML = '';
  
  filteredTasks.forEach(task => {
    const taskCard = createTaskCard(task);
    tasksContainer.appendChild(taskCard);
  });
}

// Criar card de tarefa
function createTaskCard(task) {
  const card = document.createElement('div');
  card.className = 'task-card';
  
  const statusText = {
    'pending': 'Pendente',
    'in-progress': 'Em Progresso',
    'completed': 'Concluída'
  };
  
  let actionsHTML = '';
  
  if (currentUser.role === 'manager') {
    actionsHTML = `
      <button class="btn btn-primary btn-sm" onclick="openTaskModal(${task.id})">Editar</button>
      <button class="btn btn-danger btn-sm" onclick="deleteTask(${task.id})">Excluir</button>
    `;
  } else if (task.assignedTo === currentUser.id) {
    actionsHTML = `
      <button class="btn btn-primary btn-sm" onclick="openTaskModal(${task.id})">Alterar Status</button>
    `;
  }
  
  card.innerHTML = `
    <div class="task-header">
      <h3 class="task-title">${escapeHtml(task.title)}</h3>
      <div class="task-actions">
        ${actionsHTML}
      </div>
    </div>
    ${task.description ? `<p class="task-description">${escapeHtml(task.description)}</p>` : ''}
    <div class="task-meta">
      <div class="task-assigned">
        <strong>👤 ${escapeHtml(task.assignedToName)}</strong>
      </div>
      <span class="task-status ${task.status}">${statusText[task.status]}</span>
    </div>
  `;
  
  return card;
}

// Abrir modal de tarefa
function openTaskModal(taskId = null) {
  const taskIdInput = document.getElementById('task-id');
  const taskTitle = document.getElementById('task-title');
  const taskDescription = document.getElementById('task-description');
  const taskAssigned = document.getElementById('task-assigned');
  const taskStatus = document.getElementById('task-status');
  const assignGroup = document.getElementById('assign-group');
  const statusGroup = document.getElementById('status-group');
  
  if (taskId) {
    const task = allTasks.find(t => t.id === taskId);
    if (!task) return;
    
    taskModalTitle.textContent = currentUser.role === 'manager' ? 'Editar Tarefa' : 'Alterar Status da Tarefa';
    taskIdInput.value = task.id;
    taskTitle.value = task.title;
    taskDescription.value = task.description;
    taskAssigned.value = task.assignedTo;
    taskStatus.value = task.status;
    
    if (currentUser.role === 'manager') {
      taskTitle.disabled = false;
      taskDescription.disabled = false;
      taskAssigned.disabled = false;
      assignGroup.style.display = 'block';
      statusGroup.style.display = 'block';
    } else {
      taskTitle.disabled = true;
      taskDescription.disabled = true;
      taskAssigned.disabled = true;
      assignGroup.style.display = 'none';
      statusGroup.style.display = 'block';
    }
  } else {
    taskModalTitle.textContent = 'Nova Tarefa';
    taskForm.reset();
    taskIdInput.value = '';
    taskTitle.disabled = false;
    taskDescription.disabled = false;
    
    if (currentUser.role === 'manager') {
      taskAssigned.disabled = false;
      assignGroup.style.display = 'block';
    } else {
      taskAssigned.disabled = true;
      assignGroup.style.display = 'none';
    }
    statusGroup.style.display = 'none';
  }
  
  taskModal.classList.add('active');
}

// Enviar formulário de tarefa
async function handleTaskSubmit(e) {
  e.preventDefault();
  
  const taskId = document.getElementById('task-id').value;
  const title = document.getElementById('task-title').value;
  const description = document.getElementById('task-description').value;
  const assignedTo = document.getElementById('task-assigned').value;
  const status = document.getElementById('task-status').value;
  
  const data = { title, description, assignedTo: parseInt(assignedTo) || undefined, status };
  
  try {
    let response;
    if (taskId) {
      response = await fetch(`/api/tasks/${taskId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      });
    } else {
      response = await fetch('/api/tasks', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      });
    }
    
    if (response.ok) {
      closeModals();
      await loadTasks();
      renderTasks();
    } else {
      const error = await response.json();
      alert(error.error || 'Erro ao salvar tarefa');
    }
  } catch (error) {
    alert('Erro ao salvar tarefa. Tente novamente.');
  }
}

// Excluir tarefa
async function deleteTask(taskId) {
  if (!confirm('Tem certeza que deseja excluir esta tarefa?')) {
    return;
  }
  
  try {
    const response = await fetch(`/api/tasks/${taskId}`, {
      method: 'DELETE'
    });
    
    if (response.ok) {
      await loadTasks();
      renderTasks();
    } else {
      const error = await response.json();
      alert(error.error || 'Erro ao excluir tarefa');
    }
  } catch (error) {
    alert('Erro ao excluir tarefa. Tente novamente.');
  }
}

// Abrir modal de usuário
function openUserModal() {
  userForm.reset();
  userError.classList.remove('show');
  userModal.classList.add('active');
}

// Enviar formulário de usuário
async function handleUserSubmit(e) {
  e.preventDefault();
  
  const name = document.getElementById('user-name').value;
  const username = document.getElementById('user-username').value;
  const password = document.getElementById('user-password').value;
  
  try {
    const response = await fetch('/api/users', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, username, password })
    });
    
    const data = await response.json();
    
    if (response.ok) {
      closeModals();
      await loadUsers();
      alert(`Membro ${data.name} criado com sucesso!`);
    } else {
      showError(userError, data.error);
    }
  } catch (error) {
    showError(userError, 'Erro ao criar membro. Tente novamente.');
  }
}

// Fechar modais
function closeModals() {
  taskModal.classList.remove('active');
  userModal.classList.remove('active');
}

// Mostrar erro
function showError(element, message) {
  element.textContent = message;
  element.classList.add('show');
  setTimeout(() => {
    element.classList.remove('show');
  }, 5000);
}

// Escapar HTML
function escapeHtml(text) {
  const div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;
}
