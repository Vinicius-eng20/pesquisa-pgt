# Plataforma de Cursos Online

Vou criar uma plataforma completa de cursos online com autenticação, matrículas e painel administrativo.

=== server.js ===
import express from 'express';
import Database from 'better-sqlite3';
import bcrypt from 'bcryptjs';
import cookieParser from 'cookie-parser';
import { fileURLToPath } from 'url';
import { dirname, join } from 'path';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const app = express();
const db = new Database('plataforma.db');

// Middleware
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(cookieParser());
app.use(express.static('public'));

// Criar tabelas
db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT NOT NULL CHECK(tipo IN ('aluno', 'admin')),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS cursos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    descricao TEXT NOT NULL,
    instrutor TEXT NOT NULL,
    duracao TEXT NOT NULL,
    nivel TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS matriculas (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    curso_id INTEGER NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (curso_id) REFERENCES cursos(id),
    UNIQUE(usuario_id, curso_id)
  );
`);

// Criar usuário admin padrão se não existir
const adminExiste = db.prepare('SELECT * FROM usuarios WHERE email = ?').get('admin@plataforma.com');
if (!adminExiste) {
  const senhaHash = bcrypt.hashSync('admin123', 10);
  db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)').run(
    'Administrador',
    'admin@plataforma.com',
    senhaHash,
    'admin'
  );
}

// Sessões em memória
const sessoes = new Map();

// Middleware de autenticação
function verificarAuth(req, res, next) {
  const token = req.cookies.sessao;
  if (!token || !sessoes.has(token)) {
    return res.status(401).json({ erro: 'Não autenticado' });
  }
  req.usuario = sessoes.get(token);
  next();
}

function verificarAdmin(req, res, next) {
  if (req.usuario.tipo !== 'admin') {
    return res.status(403).json({ erro: 'Acesso negado' });
  }
  next();
}

// Rotas de autenticação
app.post('/api/registrar', (req, res) => {
  const { nome, email, senha } = req.body;
  
  if (!nome || !email || !senha) {
    return res.status(400).json({ erro: 'Todos os campos são obrigatórios' });
  }

  try {
    const senhaHash = bcrypt.hashSync(senha, 10);
    const resultado = db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)').run(
      nome,
      email,
      senhaHash,
      'aluno'
    );

    res.json({ mensagem: 'Usuário registrado com sucesso', id: resultado.lastInsertRowid });
  } catch (erro) {
    if (erro.message.includes('UNIQUE constraint failed')) {
      return res.status(400).json({ erro: 'Email já cadastrado' });
    }
    res.status(500).json({ erro: 'Erro ao registrar usuário' });
  }
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;

  const usuario = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email);
  
  if (!usuario || !bcrypt.compareSync(senha, usuario.senha)) {
    return res.status(401).json({ erro: 'Email ou senha inválidos' });
  }

  const token = Math.random().toString(36).substring(2) + Date.now().toString(36);
  sessoes.set(token, {
    id: usuario.id,
    nome: usuario.nome,
    email: usuario.email,
    tipo: usuario.tipo
  });

  res.cookie('sessao', token, { httpOnly: true, maxAge: 24 * 60 * 60 * 1000 });
  res.json({
    usuario: {
      id: usuario.id,
      nome: usuario.nome,
      email: usuario.email,
      tipo: usuario.tipo
    }
  });
});

app.post('/api/logout', (req, res) => {
  const token = req.cookies.sessao;
  if (token) {
    sessoes.delete(token);
  }
  res.clearCookie('sessao');
  res.json({ mensagem: 'Logout realizado' });
});

app.get('/api/usuario-atual', verificarAuth, (req, res) => {
  res.json({ usuario: req.usuario });
});

// Rotas de cursos
app.get('/api/cursos', verificarAuth, (req, res) => {
  const cursos = db.prepare('SELECT * FROM cursos ORDER BY created_at DESC').all();
  
  if (req.usuario.tipo === 'aluno') {
    const matriculas = db.prepare('SELECT curso_id FROM matriculas WHERE usuario_id = ?').all(req.usuario.id);
    const cursosMatriculadosIds = matriculas.map(m => m.curso_id);
    
    const cursosComStatus = cursos.map(curso => ({
      ...curso,
      matriculado: cursosMatriculadosIds.includes(curso.id)
    }));
    
    return res.json(cursosComStatus);
  }
  
  res.json(cursos);
});

app.post('/api/cursos', verificarAuth, verificarAdmin, (req, res) => {
  const { titulo, descricao, instrutor, duracao, nivel } = req.body;
  
  if (!titulo || !descricao || !instrutor || !duracao || !nivel) {
    return res.status(400).json({ erro: 'Todos os campos são obrigatórios' });
  }

  try {
    const resultado = db.prepare('INSERT INTO cursos (titulo, descricao, instrutor, duracao, nivel) VALUES (?, ?, ?, ?, ?)').run(
      titulo,
      descricao,
      instrutor,
      duracao,
      nivel
    );

    res.json({ mensagem: 'Curso criado com sucesso', id: resultado.lastInsertRowid });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao criar curso' });
  }
});

app.put('/api/cursos/:id', verificarAuth, verificarAdmin, (req, res) => {
  const { id } = req.params;
  const { titulo, descricao, instrutor, duracao, nivel } = req.body;

  if (!titulo || !descricao || !instrutor || !duracao || !nivel) {
    return res.status(400).json({ erro: 'Todos os campos são obrigatórios' });
  }

  try {
    db.prepare('UPDATE cursos SET titulo = ?, descricao = ?, instrutor = ?, duracao = ?, nivel = ? WHERE id = ?').run(
      titulo,
      descricao,
      instrutor,
      duracao,
      nivel,
      id
    );

    res.json({ mensagem: 'Curso atualizado com sucesso' });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao atualizar curso' });
  }
});

app.delete('/api/cursos/:id', verificarAuth, verificarAdmin, (req, res) => {
  const { id } = req.params;

  try {
    db.prepare('DELETE FROM matriculas WHERE curso_id = ?').run(id);
    db.prepare('DELETE FROM cursos WHERE id = ?').run(id);
    res.json({ mensagem: 'Curso removido com sucesso' });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao remover curso' });
  }
});

// Rotas de matrículas
app.get('/api/minhas-matriculas', verificarAuth, (req, res) => {
  const matriculas = db.prepare(`
    SELECT m.id as matricula_id, m.created_at as data_matricula, c.*
    FROM matriculas m
    JOIN cursos c ON m.curso_id = c.id
    WHERE m.usuario_id = ?
    ORDER BY m.created_at DESC
  `).all(req.usuario.id);

  res.json(matriculas);
});

app.post('/api/matriculas', verificarAuth, (req, res) => {
  const { curso_id } = req.body;

  if (!curso_id) {
    return res.status(400).json({ erro: 'ID do curso é obrigatório' });
  }

  const curso = db.prepare('SELECT * FROM cursos WHERE id = ?').get(curso_id);
  if (!curso) {
    return res.status(404).json({ erro: 'Curso não encontrado' });
  }

  try {
    db.prepare('INSERT INTO matriculas (usuario_id, curso_id) VALUES (?, ?)').run(
      req.usuario.id,
      curso_id
    );

    res.json({ mensagem: 'Matrícula realizada com sucesso' });
  } catch (erro) {
    if (erro.message.includes('UNIQUE constraint failed')) {
      return res.status(400).json({ erro: 'Você já está matriculado neste curso' });
    }
    res.status(500).json({ erro: 'Erro ao realizar matrícula' });
  }
});

app.delete('/api/matriculas/:id', verificarAuth, (req, res) => {
  const { id } = req.params;

  if (req.usuario.tipo === 'admin') {
    db.prepare('DELETE FROM matriculas WHERE id = ?').run(id);
  } else {
    db.prepare('DELETE FROM matriculas WHERE id = ? AND usuario_id = ?').run(id, req.usuario.id);
  }

  res.json({ mensagem: 'Matrícula cancelada com sucesso' });
});

// Rotas administrativas
app.get('/api/admin/matriculas', verificarAuth, verificarAdmin, (req, res) => {
  const matriculas = db.prepare(`
    SELECT 
      m.id as matricula_id,
      m.created_at as data_matricula,
      u.id as usuario_id,
      u.nome as usuario_nome,
      u.email as usuario_email,
      c.id as curso_id,
      c.titulo as curso_titulo
    FROM matriculas m
    JOIN usuarios u ON m.usuario_id = u.id
    JOIN cursos c ON m.curso_id = c.id
    ORDER BY m.created_at DESC
  `).all();

  res.json(matriculas);
});

      mensagem.className = 'mensagem erro';

    }

    mensagem.className = 'mensagem erro';

});

  try {
    const response = await fetch('/api/usuario-atual');

      window.location.href = '/login';
      return;

    const data = await response.json();

async function carregarCursos() {
  try {
    const response = await fetch('/api/cursos');
    if (!response.ok) throw new Error('Erro ao carregar cursos');
    
    cursosData = await response.json();
    renderizarCursos();
  } catch (erro) {
    mostrarMensagem('Erro ao carregar cursos', 'erro');
  }

async function carregarMatriculas() {
  try {
    const response = await fetch('/api/minhas-matriculas');
    if (!response.ok) throw new Error('Erro ao carregar matrículas');
    
    matriculasData = await response.json();
    renderizarMatriculas();
  } catch (erro) {
    mostrarMensagem('Erro ao carregar suas matrículas', 'erro');
  }

function renderizarCursos() {
  const lista = document.getElementById('listaCursos');
  
  if (cursosData.length === 0) {
    lista.innerHTML = '<p class="text-muted">Nenhum curso disponível no momento.</p>';
    return;
  }

  lista.innerHTML = cursosData.map(curso => `
    <div class="curso-card">
      ${curso.matriculado ? '<span class="badge-matriculado">✓ Matriculado</span>' : ''}
      <h3>${curso.titulo}</h3>
      <p>${curso.descricao}</p>
      <div class="curso-info">
        <span class="badge">👨‍🏫 ${curso.instrutor}</span>
        <span class="badge">⏱️ ${curso.duracao}</span>
        <span class="badge">📊 ${curso.nivel}</span>
      </div>
      <div class="curso-actions">
        ${!curso.matriculado ? 
          `<button class="btn btn-success" onclick="matricular(${curso.id})">Matricular-se</button>` :
          `<button class="btn btn-secondary" disabled>Já Matriculado</button>`
        }
      </div>
    </div>
  `).join('');
}

function renderizarMatriculas() {
  const lista = document.getElementById('listaMatriculas');
  
  if (matriculasData.length === 0) {
    lista.innerHTML = '<p class="text-muted">Você ainda não está matriculado em nenhum curso.</p>';
    return;
  }

  lista.innerHTML = matriculasData.map(m => `
    <div class="curso-card">
      <h3>${m.titulo}</h3>
      <p>${m.descricao}</p>
      <div class="curso-info">

}

  }
}

  }
}

# Plataforma de Cursos Online

Uma plataforma completa de cursos online com sistema de autenticação, matrículas e painel administrativo.

## Funcionalidades

### Para Alunos:
- Criar conta e fazer login
- Ver todos os cursos disponíveis
- Matricular-se em cursos
- Ver cursos em que está matriculado
- Cancelar matrículas

### Para Administradores:
- Cadastrar novos cursos
- Editar cursos existentes
- Remover cursos
- Ver todas as matrículas de todos os alunos
- Cancelar matrículas

## Requisitos

- Node.js (versão 14 ou superior)

## Instalação

1. Instale as dependências:
```bash
npm install
```

## Como Executar

1. Inicie o servidor:
```bash
npm run dev
```

2. Abra seu navegador e acesse:
```
http://localhost:3000
```

## Credenciais Padrão

**Administrador:**
- Email: admin@plataforma.com
- Senha: admin123

## Como Usar

1. **Primeira vez:**
   - Acesse a página inicial
   - Clique em "Criar Conta" para criar uma conta de aluno
   - Ou faça login como administrador usando as credenciais acima

2. **Como Aluno:**
   - Navegue pelos cursos disponíveis
   - Clique em "Matricular-se" para se inscrever em um curso
   - Acesse a aba "Meus Cursos" para ver seus cursos
   - Cancele matrículas se necessário

3. **Como Administrador:**
   - Clique em "Novo Curso" para cadastrar um curso
   - Clique em "Editar" para modificar um curso existente
   - Clique em "Remover" para excluir um curso
   - Acesse a aba "Todas as Matrículas" para ver e gerenciar todas as matrículas

## Armazenamento de Dados

Todos os dados são armazenados em um banco de dados SQLite local (arquivo `plataforma.db`). Os dados persistem mesmo após reiniciar o servidor.

## Porta

O servidor roda na porta 3000 por padrão. Se precisar mudar, defina a variável de ambiente PORT:
```bash
PORT=8080 npm run dev
```

## Observações

- Todos os dados ficam salvos no arquivo de banco de dados
- As sessões de login funcionam em qualquer dispositivo
- O sistema funciona completamente offline, sem necessidade de serviços externos
- Senha do admin pode ser alterada diretamente no banco de dados se necessário
