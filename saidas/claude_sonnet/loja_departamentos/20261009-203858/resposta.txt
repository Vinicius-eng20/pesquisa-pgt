// Criar tabelas
db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT DEFAULT 'cliente',
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS produtos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    descricao TEXT,
    preco REAL NOT NULL,
    categoria TEXT NOT NULL,
    imagem TEXT,
    estoque INTEGER DEFAULT 0,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS pedidos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    total REAL NOT NULL,
    status TEXT DEFAULT 'pendente',
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
  );

  CREATE TABLE IF NOT EXISTS itens_pedido (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pedido_id INTEGER NOT NULL,
    produto_id INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    preco_unitario REAL NOT NULL,
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
  );

  CREATE TABLE IF NOT EXISTS sessoes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    token TEXT UNIQUE NOT NULL,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
  );
`);

// Middleware de autenticação
function autenticar(req, res, next) {
  const token = req.cookies.token;
  if (!token) {
    return res.status(401).json({ erro: 'Não autenticado' });
  }

  const sessao = db.prepare('SELECT usuario_id FROM sessoes WHERE token = ?').get(token);
  if (!sessao) {
    return res.status(401).json({ erro: 'Sessão inválida' });
  }

  const usuario = db.prepare('SELECT id, nome, email, tipo FROM usuarios WHERE id = ?').get(sessao.usuario_id);
  if (!usuario) {
    return res.status(401).json({ erro: 'Usuário não encontrado' });
  }

  req.usuario = usuario;
  next();
}

function verificarAdmin(req, res, next) {
  if (req.usuario.tipo !== 'admin') {
    return res.status(403).json({ erro: 'Acesso negado' });
  }
  next();
}

// Rotas de autenticação
app.post('/api/cadastro', (req, res) => {
  try {
    const { nome, email, senha } = req.body;

    if (!nome || !email || !senha) {
      return res.status(400).json({ erro: 'Todos os campos são obrigatórios' });
    }

    const usuarioExiste = db.prepare('SELECT id FROM usuarios WHERE email = ?').get(email);
    if (usuarioExiste) {
      return res.status(400).json({ erro: 'Email já cadastrado' });
    }

    const senhaHash = bcrypt.hashSync(senha, 10);
    const resultado = db.prepare('INSERT INTO usuarios (nome, email, senha) VALUES (?, ?, ?)').run(nome, email, senhaHash);

    const token = Math.random().toString(36).substring(2) + Date.now().toString(36);
    db.prepare('INSERT INTO sessoes (usuario_id, token) VALUES (?, ?)').run(resultado.lastInsertRowid, token);

    res.cookie('token', token, { httpOnly: true, maxAge: 30 * 24 * 60 * 60 * 1000 });
    res.json({ sucesso: true, usuario: { id: resultado.lastInsertRowid, nome, email, tipo: 'cliente' } });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao cadastrar usuário' });
  }
});

app.post('/api/login', (req, res) => {
  try {
    const { email, senha } = req.body;

    const usuario = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email);
    if (!usuario || !bcrypt.compareSync(senha, usuario.senha)) {
      return res.status(401).json({ erro: 'Email ou senha incorretos' });
    }

    const token = Math.random().toString(36).substring(2) + Date.now().toString(36);
    db.prepare('INSERT INTO sessoes (usuario_id, token) VALUES (?, ?)').run(usuario.id, token);

    res.cookie('token', token, { httpOnly: true, maxAge: 30 * 24 * 60 * 60 * 1000 });
    res.json({ 
      sucesso: true, 
      usuario: { 
        id: usuario.id, 
        nome: usuario.nome, 
        email: usuario.email, 
        tipo: usuario.tipo 
      } 
    });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao fazer login' });
  }
});

app.post('/api/logout', autenticar, (req, res) => {
  const token = req.cookies.token;
  db.prepare('DELETE FROM sessoes WHERE token = ?').run(token);
  res.clearCookie('token');
  res.json({ sucesso: true });
});

// Rotas de pedidos
app.post('/api/pedidos', autenticar, (req, res) => {
  try {
    const { itens } = req.body;

    if (!itens || itens.length === 0) {
      return res.status(400).json({ erro: 'Carrinho vazio' });
    }

    let total = 0;
    const itensValidados = [];

    for (const item of itens) {
      const produto = db.prepare('SELECT * FROM produtos WHERE id = ?').get(item.produtoId);
      if (!produto) {
        return res.status(400).json({ erro: `Produto ${item.produtoId} não encontrado` });
      }
      if (produto.estoque < item.quantidade) {
        return res.status(400).json({ erro: `Estoque insuficiente para ${produto.nome}` });
      }
      total += produto.preco * item.quantidade;
      itensValidados.push({
        produtoId: produto.id,
        quantidade: item.quantidade,
        precoUnitario: produto.preco
      });
    }

    const pedido = db.prepare('INSERT INTO pedidos (usuario_id, total) VALUES (?, ?)').run(req.usuario.id, total);

    const stmtItem = db.prepare('INSERT INTO itens_pedido (pedido_id, produto_id, quantidade, preco_unitario) VALUES (?, ?, ?, ?)');
    const stmtEstoque = db.prepare('UPDATE produtos SET estoque = estoque - ? WHERE id = ?');

    for (const item of itensValidados) {
      stmtItem.run(pedido.lastInsertRowid, item.produtoId, item.quantidade, item.precoUnitario);
      stmtEstoque.run(item.quantidade, item.produtoId);
    }

    res.json({ sucesso: true, pedidoId: pedido.lastInsertRowid });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao criar pedido' });
  }
});

app.get('/api/pedidos', autenticar, (req, res) => {
  try {
    const pedidos = db.prepare(`
      SELECT p.*, u.nome as usuario_nome, u.email as usuario_email
      FROM pedidos p
      JOIN usuarios u ON p.usuario_id = u.id
      WHERE p.usuario_id = ?
      ORDER BY p.criado_em DESC
    `).all(req.usuario.id);

    for (const pedido of pedidos) {
      pedido.itens = db.prepare(`
        SELECT ip.*, prod.nome as produto_nome, prod.imagem as produto_imagem
        FROM itens_pedido ip
        JOIN produtos prod ON ip.produto_id = prod.id
        WHERE ip.pedido_id = ?
      `).all(pedido.id);
    }

    res.json({ pedidos });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao buscar pedidos' });
  }
});

// Estado da aplicação
let estado = {
  usuario: null,
  carrinho: [],
  produtos: [],
  categorias: [],
  categoriaAtual: 'todas'
};

// Carregar carrinho do localStorage
function carregarCarrinho() {
  const carrinhoSalvo = localStorage.getItem('carrinho');
  if (carrinhoSalvo) {
    estado.carrinho = JSON.parse(carrinhoSalvo);
  }
}

// Salvar carrinho no localStorage
function salvarCarrinho() {
  localStorage.setItem('carrinho', JSON.stringify(estado.carrinho));
  atualizarBadgeCarrinho();
}

// Atualizar badge do carrinho
function atualizarBadgeCarrinho() {
  const totalItens = estado.carrinho.reduce((total, item) => total + item.quantidade, 0);
  const badge = document.querySelector('.carrinho-badge .badge');
  if (badge) {
    badge.textContent = totalItens;
    badge.style.display = totalItens > 0 ? 'flex' : 'none';
  }
}

// API helpers
async function api(url, options = {}) {
  try {
    const response = await fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options.headers
      }
    });
    const data = await response.json();
    if (!response.ok) {
      throw new Error(data.erro || 'Erro na requisição');
    }
    return data;
  } catch (erro) {
    mostrarNotificacao(erro.message, 'error');
    throw erro;
  }
}

// Notificações

}

async function paginaProdutos() {
  try {
    const [dataProdutos, dataCategorias] = await Promise.all([
      api(`/api/produtos?categoria=${estado.categoriaAtual}`),
      api('/api/categorias')
    ]);
    
    estado.produtos = dataProdutos.produtos;
    estado.categorias = dataCategorias.categorias;

    const opcoesCategorias = ['todas', ...estado.categorias].map(cat => 
      `<option value="${cat}" ${cat === estado.categoriaAtual ? 'selected' : ''}>${cat === 'todas' ? 'Todas as Categorias' : cat}</option>`
    ).join('');

    const produtosHtml = estado.produtos.map(produto => `
      <div class="card product-card" onclick="navegar('/produto/${produto.id}')">
        <img src="${produto.imagem || 'https://via.placeholder.com/300x300/CCCCCC/FFFFFF?text=Sem+Imagem'}" alt="${produto.nome}" class="product-image">
        <div class="card-body">
          <span class="product-category">${produto.categoria}</span>
          <h3 class="card-title">${produto.nome}</h3>
          <p class="card-text">${produto.descricao}</p>
          <p class="product-price">R$ ${produto.preco.toFixed(2)}</p>
          <p class="product-stock">Estoque: ${produto.estoque} unidades</p>
          ${estado.usuario && estado.usuario.tipo !== 'admin' ? `
            <button class="btn btn-primary btn-block" onclick="event.stopPropagation(); adicionarAoCarrinho(${produto.id})">
              Adicionar ao Carrinho
            </button>
          ` : ''}
        </div>
      </div>
    `).join('');

    return `
      <div class="products-header">
        <h2>Nossos Produtos</h2>
        <div class="filter-group">
          <select class="form-control" onchange="filtrarPorCategoria(this.value)">
            ${opcoesCategorias}
          </select>
        </div>
      </div>
      <div class="products-grid">
        ${produtosHtml || '<div class="empty-state"><h3>Nenhum produto encontrado</h3></div>'}
      </div>
    `;
  } catch (erro) {
    return '<div class="error-message">Erro ao carregar produtos</div>';
  }
}

function filtrarPorCategoria(categoria) {
  estado.categoriaAtual = categoria;
  renderizar();
}

function adicionarAoCarrinho(produtoId) {
  const produto = estado.produtos.find(p => p.id === produtoId);
  if (!produto) return;

  const itemCarrinho = estado.carrinho.find(item => item.produtoId === produtoId);
  
  if (itemCarrinho) {
    if (itemCarrinho.quantidade < produto.estoque) {
      itemCarrinho.quantidade++;
      mostrarNotificacao('Quantidade atualizada no carrinho', 'success');
    } else {
      mostrarNotificacao('Estoque insuficiente', 'error');
      return;
    }
  } else {
    estado.carrinho.push({
      produtoId: produto.id,
      nome: produto.nome,
      preco: produto.preco,
      imagem: produto.imagem,
      quantidade: 1,
      estoque: produto.estoque
    });
    mostrarNotificacao('Produto adicionado ao carrinho', 'success');
  }
  
  salvarCarrinho();
}

            <button class="btn btn-primary btn-block" onclick="adicionarAoCarrinhoComQuantidade(${produto.id})">
              Adicionar ao Carrinho
            </button>
          ` : produto.estoque === 0 ? '<p style="color: var(--danger); font-weight: bold;">Produto indisponível</p>' : ''}
        </div>
      </div>
    `;
  } catch (erro) {
    return '<div class="error-message">Produto não encontrado</div>';
  }
}

function alterarQuantidade(delta) {
  const input = document.getElementById('quantidade');
  const novoValor = parseInt(input.value) + delta;
  if (novoValor >= parseInt(input.min) && novoValor <= parseInt(input.max)) {
    input.value = novoValor;
  }
}

function adicionarAoCarrinhoComQuantidade(produtoId) {
  const quantidade = parseInt(document.getElementById('quantidade').value);
  const produto = estado.produtos.find(p => p.id === produtoId);
  
  if (!produto) {
    api(`/api/produtos/${produtoId}`).then(data => {
      const prod = data.produto;
      const itemCarrinho = estado.carrinho.find(item => item.produtoId === produtoId);
      
      if (itemCarrinho) {
        const novaQuantidade = itemCarrinho.quantidade + quantidade;
        if (novaQuantidade <= prod.estoque) {
          itemCarrinho.quantidade = novaQuantidade;
          mostrarNotificacao('Carrinho atualizado', 'success');
        } else {
          mostrarNotificacao('Estoque insuficiente', 'error');
          return;
        }
      } else {
        estado.carrinho.push({
          produtoId: prod.id,
          nome: prod.nome,
          preco: prod.preco,
          imagem: prod.imagem,
          quantidade: quantidade,
          estoque: prod.estoque
        });
        mostrarNotificacao(`${quantidade} ${quantidade > 1 ? 'itens adicionados' : 'item adicionado'} ao carrinho`, 'success');
      }
      
      salvarCarrinho();
    });
  }
}

  if (!estado.usuario || estado.usuario.tipo === 'admin') {

function alterarQuantidadeCarrinho(index, delta) {
  const item = estado.carrinho[index];
  const novaQuantidade = item.quantidade + delta;
  
  if (novaQuantidade <= 0) {
    removerDoCarrinho(index);
  } else if (novaQuantidade <= item.estoque) {
    item.quantidade = novaQuantidade;
    salvarCarrinho();
    renderizar();
  } else {
    mostrarNotificacao('Estoque insuficiente', 'error');
  }
}

function removerDoCarrinho(index) {
  estado.carrinho.splice(index, 1);
  salvarCarrinho();
  renderizar();
  mostrarNotificacao('Item removido do carrinho', 'info');
}

async function finalizarPedido() {

    // Já tratado
  }
}

      return;

    if (estado.usuario) {
      navegar('/');
      return;

    html = await paginaCadastro();
  } else if (caminho === '/carrinho') {
    html = await paginaCarrinho();
  } else if (caminho === '/pedidos') {
    html = await paginaPedidos();
  } else if (caminho === '/admin') {
    html = await paginaAdmin();
  } else if (caminho === '/admin/pedidos') {
    html = await paginaAdminPedidos();
  } else if (caminho.startsWith('/produto/')) {
    const id = caminho.split('/')[2];
    html = await paginaProdutoDetalhe(id);
  } else {
    html = await paginaProdutos();

  conteudo.innerHTML = html;

# Loja Virtual - Shopping Online

Uma loja virtual completa com carrinho de compras, sistema de pedidos e painel administrativo.

## Funcionalidades

### Para Clientes:
- Navegação por produtos e categorias
- Visualização detalhada de produtos
- Carrinho de compras persistente
- Finalização de pedidos
- Histórico de pedidos
- Sistema de autenticação (cadastro e login)

### Para Administradores:
- Gerenciamento completo de produtos (criar, editar, excluir)
- Visualização de todos os pedidos
- Alteração de status dos pedidos
- Dashboard com estatísticas

## Instalação

### Pré-requisitos
- Node.js (versão 14 ou superior)

### Passos para instalar e executar:

1. Clone ou extraia os arquivos do projeto em uma pasta

2. Abra o terminal/prompt de comando na pasta do projeto

3. Instale as dependências:
```bash
npm install
```

4. Inicie o servidor:
```bash
npm start
```

5. Acesse o site no navegador:
```
http://localhost:3000
```
