function readData() {
  initDb();
  const raw = fs.readFileSync(DB_FILE, 'utf-8');
  return JSON.parse(raw);
}

function writeData(data) {
  const tempPath = DB_FILE + '.tmp';
  fs.writeFileSync(tempPath, JSON.stringify(data, null, 2), 'utf-8');
  fs.renameSync(tempPath, DB_FILE);
}

// Gerenciamento de sessões em memória mapeadas aos usuários
const sessions = new Map();

function authMiddleware(req, res, next) {
  const token = req.cookies.session_token;
  if (!token || !sessions.has(token)) {
    req.user = null;
    return next();
  }
  const session = sessions.get(token);
  const db = readData();
  const user = db.users.find((u) => u.id === session.userId);
  if (!user) {
    sessions.delete(token);
    req.user = null;
    return next();
  }
  req.user = {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role,
    phone: user.phone
  };
  next();
}

app.use(authMiddleware);

function requireAuth(req, res, next) {
  if (!req.user) {
    return res.status(401).json({ error: 'Você precisa estar conectado para realizar esta ação.' });
  }
  next();
}

function requireManager(req, res, next) {
  if (!req.user || req.user.role !== 'manager') {
    return res.status(403).json({ error: 'Acesso restrito apenas ao gerente do hotel.' });
  }
  next();
}

app.post('/api/auth/register', (req, res) => {
  const { name, email, password, phone } = req.body;

  if (!name || !email || !password) {
    return res.status(400).json({ error: 'Preencha nome, e-mail e senha.' });
  }

  const db = readData();
  const existing = db.users.find((u) => u.email.toLowerCase() === email.trim().toLowerCase());
  if (existing) {
    return res.status(400).json({ error: 'Este e-mail já está cadastrado no sistema.' });
  }

  const salt = bcrypt.genSaltSync(10);
  const newUser = {
    id: 'usr_' + crypto.randomUUID().slice(0, 8),
    name: name.trim(),
    email: email.trim().toLowerCase(),
    passwordHash: bcrypt.hashSync(password, salt),
    role: 'guest',
    phone: phone ? phone.trim() : '',
    createdAt: new Date().toISOString()
  };

  db.users.push(newUser);
  writeData(db);

  const token = crypto.randomUUID();
  sessions.set(token, { userId: newUser.id, createdAt: Date.now() });

  res.cookie('session_token', token, {
    httpOnly: true,
    maxAge: 30 * 24 * 60 * 60 * 1000,
    sameSite: 'lax'
  });

  res.status(201).json({
    message: 'Cadastro realizado com sucesso!',
    user: {
      id: newUser.id,
      name: newUser.name,
      email: newUser.email,
      role: newUser.role,
      phone: newUser.phone
    }
  });
});

app.post('/api/auth/login', (req, res) => {
  const { email, password } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: 'Informe e-mail e senha.' });
  }

  const db = readData();

  res.cookie('session_token', token, {
    httpOnly: true,
    maxAge: 30 * 24 * 60 * 60 * 1000,
    sameSite: 'lax'
  });

  res.json({
    message: 'Login realizado com sucesso!',
    user: {
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role,
      phone: user.phone
    }
  });
});

function checkRoomConflict(booking, checkIn, checkOut) {
  if (booking.status === 'recusada' || booking.status === 'cancelada') {
    return false;
  }
  const bIn = new Date(booking.checkIn);
  const bOut = new Date(booking.checkOut);
  const qIn = new Date(checkIn);
  const qOut = new Date(checkOut);

  return qIn < bOut && qOut > bIn;
}

app.get('/api/rooms', (req, res) => {
  const { checkIn, checkOut, guests } = req.query;
  const db = readData();

  if (!checkIn || !checkOut) {
    return res.json(db.rooms);
  }

  const inDate = new Date(checkIn);
  const outDate = new Date(checkOut);

  if (isNaN(inDate.getTime()) || isNaN(outDate.getTime()) || inDate >= outDate) {
    return res.status(400).json({ error: 'Datas de check-in e check-out inválidas.' });
  }

  const guestsCount = guests ? parseInt(guests, 10) : 1;

  const availableRooms = db.rooms.filter((room) => {
    if (room.capacity < guestsCount) {
      return false;
    }
    const hasConflict = db.bookings.some(
      (b) => b.roomId === room.id && checkRoomConflict(b, checkIn, checkOut)
    );
    return !hasConflict;
  });

  res.json(availableRooms);
});

app.post('/api/bookings', requireAuth, (req, res) => {
  const { roomId, checkIn, checkOut, guestsCount } = req.body;

  if (!roomId || !checkIn || !checkOut) {
    return res.status(400).json({ error: 'Informe quarto, check-in e check-out.' });
  }

  const inDate = new Date(checkIn);
  const outDate = new Date(checkOut);
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  if (isNaN(inDate.getTime()) || isNaN(outDate.getTime())) {
    return res.status(400).json({ error: 'Datas inválidas.' });
  }

  if (inDate < today) {
    return res.status(400).json({ error: 'A data de check-in não pode ser no passado.' });
  }

  if (inDate >= outDate) {
    return res.status(400).json({ error: 'A data de check-out deve ser posterior ao check-in.' });
  }

  const db = readData();
  const room = db.rooms.find((r) => r.id === roomId);
  if (!room) {
    return res.status(404).json({ error: 'Quarto selecionado não existe.' });
  }

  const guestsNum = parseInt(guestsCount, 10) || 1;
  if (guestsNum > room.capacity) {
    return res.status(400).json({ error: `Este quarto acomoda no máximo ${room.capacity} pessoas.` });
  }

  const hasConflict = db.bookings.some(
    (b) => b.roomId === room.id && checkRoomConflict(b, checkIn, checkOut)
  );

  if (hasConflict) {
    return res.status(400).json({ error: 'Este quarto já está reservado no período selecionado.' });
  }

  const diffTime = Math.abs(outDate - inDate);
  const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
  const totalPrice = diffDays * room.pricePerNight;

  const newBooking = {
    id: 'res_' + crypto.randomUUID().slice(0, 8),
    userId: req.user.id,
    userName: req.user.name,
    userEmail: req.user.email,
    userPhone: req.user.phone,
    roomId: room.id,
    roomName: room.name,
    roomCategory: room.category,
    checkIn,
    checkOut,
    nights: diffDays,
    guestsCount: guestsNum,
    pricePerNight: room.pricePerNight,
    totalPrice,
    status: 'pendente', // pendente, confirmada, recusada, cancelada
    createdAt: new Date().toISOString()
  };

  db.bookings.unshift(newBooking);
  writeData(db);

  res.status(201).json({
    message: 'Reserva solicitada com sucesso! Aguarde a confirmação do hotel.',
    booking: newBooking
  });
});

app.get('/api/my-bookings', requireAuth, (req, res) => {

app.post('/api/bookings/:id/cancel', requireAuth, (req, res) => {
  const { id } = req.params;
  const db = readData();
  const booking = db.bookings.find((b) => b.id === id);

  if (!booking) {
    return res.status(404).json({ error: 'Reserva não encontrada.' });
  }

  if (booking.userId !== req.user.id && req.user.role !== 'manager') {
    return res.status(403).json({ error: 'Você não tem permissão para cancelar esta reserva.' });
  }

  if (booking.status === 'cancelada') {
    return res.status(400).json({ error: 'Esta reserva já foi cancelada.' });
  }

  if (booking.status === 'recusada') {
    return res.status(400).json({ error: 'Esta reserva já foi recusada pelo hotel.' });
  }

  booking.status = 'cancelada';
  booking.cancelledAt = new Date().toISOString();
  writeData(db);

  res.json({ message: 'Reserva cancelada com sucesso.', booking });
});

app.post('/api/admin/rooms', requireManager, (req, res) => {

    pricePerNight: Number(pricePerNight),

    amenities: Array.isArray(amenities)
      ? amenities
      : (amenities || '').split(',').map((a) => a.trim()).filter(Boolean)

  writeData(db);

  res.status(201).json({ message: 'Quarto cadastrado com sucesso!', room: newRoom });

app.put('/api/admin/rooms/:id', requireManager, (req, res) => {
  const { id } = req.params;
  const { name, category, description, pricePerNight, capacity, imageUrl, amenities } = req.body;

  const db = readData();
  const room = db.rooms.find((r) => r.id === id);

  if (!room) {
    return res.status(404).json({ error: 'Quarto não encontrado.' });
  }

  if (name) room.name = name.trim();
  if (category) room.category = category.trim();
  if (description !== undefined) room.description = description.trim();
  if (pricePerNight) room.pricePerNight = Number(pricePerNight);
  if (capacity) room.capacity = Number(capacity);
  if (imageUrl) room.imageUrl = imageUrl.trim();
  if (amenities !== undefined) {
    room.amenities = Array.isArray(amenities)
      ? amenities
      : amenities.split(',').map((a) => a.trim()).filter(Boolean);
  }

  writeData(db);
  res.json({ message: 'Quarto atualizado com sucesso!', room });
});

app.delete('/api/admin/rooms/:id', requireManager, (req, res) => {
  const { id } = req.params;
  const db = readData();
  const roomIndex = db.rooms.findIndex((r) => r.id === id);

  if (roomIndex === -1) {
    return res.status(404).json({ error: 'Quarto não encontrado.' });
  }

  // Verifica se há reservas ativas para o quarto
  const hasActiveBookings = db.bookings.some(
    (b) => b.roomId === id && (b.status === 'pendente' || b.status === 'confirmada')
  );

  if (hasActiveBookings) {
    return res.status(400).json({
      error: 'Não é possível excluir este quarto pois ele possui reservas pendentes ou confirmadas.'
    });
  }

  db.rooms.splice(roomIndex, 1);
  writeData(db);

  res.json({ message: 'Quarto removido com sucesso!' });
});

function setupDateDefaults() {
  const today = new Date();
  const tomorrow = new Date();
  tomorrow.setDate(today.getDate() + 1);

  const format = (d) => d.toISOString().split('T')[0];

  const inInput = document.getElementById('searchCheckIn');
  const outInput = document.getElementById('searchCheckOut');
  
  inInput.min = format(today);
  inInput.value = format(today);

  outInput.min = format(tomorrow);
  outInput.value = format(tomorrow);

  currentSearch.checkIn = format(today);
  currentSearch.checkOut = format(tomorrow);
}

async function checkAuth() {
  try {
    const res = await fetch('/api/auth/me');
    const data = await res.json();
    currentUser = data.user;
    updateAuthUI();

    console.error('Erro ao verificar sessão:', err);

function updateAuthUI() {
  const authBox = document.getElementById('authBox');
  const btnMyBookings = document.getElementById('btnNavMyBookings');
  const btnManager = document.getElementById('btnNavManager');

  if (currentUser) {
    const isManager = currentUser.role === 'manager';
    btnMyBookings.style.display = isManager ? 'none' : 'inline-block';
    btnManager.style.display = isManager ? 'inline-block' : 'none';

    authBox.innerHTML = `
      <div class="user-tag">
        <div>
          <span class="user-name">${escapeHtml(currentUser.name)}</span>
          <span class="user-role-badge ${isManager ? 'manager' : ''}">${isManager ? 'Gerente' : 'Hóspede'}</span>
        </div>
        <button class="btn-secondary" onclick="handleLogout()">Sair</button>
      </div>
    `;
  } else {
    btnMyBookings.style.display = 'none';
    btnManager.style.display = 'none';
    authBox.innerHTML = `<button class="btn-primary" onclick="openModal('authModal')">Entrar / Cadastrar</button>`;
  }
}

async function handleLogin(e) {
  e.preventDefault();
  const email = document.getElementById('loginEmail').value;
  const password = document.getElementById('loginPassword').value;

    const res = await fetch('/api/auth/login', {
      method: 'POST',

  }
}

    document.getElementById('btnNavMyBookings').classList.add('active');

    document.getElementById('btnNavManager').classList.add('active');

async function loadRooms(params = null) {
  try {
    let url = '/api/rooms';
    if (params) {
      const q = new URLSearchParams(params);
      url += `?${q.toString()}`;
    }

    const res = await fetch(url);
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao buscar quartos.', 'error');
      return;
    }

    allRooms = data;
    renderRoomsGrid(allRooms);
  } catch (err) {
    showToast('Erro ao carregar quartos.', 'error');
  }
}

function startBooking(roomId) {
  if (!currentUser) {
    showToast('Faça login ou cadastre-se para reservar um quarto.', 'error');
    openModal('authModal');
    return;
  }

  const room = allRooms.find((r) => r.id === roomId);
  if (!room) return;

  selectedRoomForBooking = room;

  document.getElementById('bookRoomId').value = room.id;
  document.getElementById('bookCheckIn').value = currentSearch.checkIn || '';
  document.getElementById('bookCheckOut').value = currentSearch.checkOut || '';
  document.getElementById('bookGuestsCount').value = currentSearch.guests || 2;
  document.getElementById('bookGuestsCount').max = room.capacity;

  const todayStr = new Date().toISOString().split('T')[0];
  document.getElementById('bookCheckIn').min = todayStr;

  document.getElementById('bookingSummary').innerHTML = `
    <strong>${escapeHtml(room.name)}</strong>
    <p style="font-size:13px; color:#64748b; margin-top:4px;">Capacidade: até ${room.capacity} pessoas | R$ ${Number(room.pricePerNight).toFixed(2)}/noite</p>
  `;

  recalcBookingModal();
  openModal('bookingModal');
}

function recalcBookingModal() {
  if (!selectedRoomForBooking) return;

  const cIn = document.getElementById('bookCheckIn').value;
  const cOut = document.getElementById('bookCheckOut').value;
  const btnSubmit = document.getElementById('btnSubmitBooking');

  if (!cIn || !cOut || cIn >= cOut) {
    document.getElementById('calcNights').innerText = '0 noites';
    document.getElementById('calcPerNight').innerText = `R$ ${Number(selectedRoomForBooking.pricePerNight).toFixed(2)}`;
    document.getElementById('calcTotalPrice').innerText = 'R$ 0,00';
    btnSubmit.disabled = true;
    return;
  }

  const dIn = new Date(cIn);
  const dOut = new Date(cOut);
  const diffDays = Math.ceil((dOut - dIn) / (1000 * 60 * 60 * 24));
  const total = diffDays * selectedRoomForBooking.pricePerNight;

  document.getElementById('calcNights').innerText = `${diffDays} noite(s)`;
  document.getElementById('calcPerNight').innerText = `R$ ${Number(selectedRoomForBooking.pricePerNight).toFixed(2)}`;
  document.getElementById('calcTotalPrice').innerText = `R$ ${total.toFixed(2)}`;
  btnSubmit.disabled = false;
}

  const payload = { name, category, capacity, pricePerNight, imageUrl, amenities, description };

  try {
    const url = id ? `/api/admin/rooms/${id}` : '/api/admin/rooms';
    const method = id ? 'PUT' : 'POST';

    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao salvar quarto.', 'error');
      return;
    }

    closeModal('roomModal');
    showToast(data.message, 'success');
    await loadRooms();
    await loadManagerRooms();
  } catch (err) {
    showToast('Erro ao salvar quarto.', 'error');
  }
}

async function deleteRoom(roomId) {
  if (!confirm('Deseja realmente excluir este quarto?')) return;

  try {
    const res = await fetch(`/api/admin/rooms/${roomId}`, { method: 'DELETE' });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao excluir quarto.', 'error');
      return;
    }

    showToast('Quarto excluído com sucesso!', 'success');
    await loadRooms();
    await loadManagerRooms();
  } catch (err) {
    showToast('Erro ao excluir quarto.', 'error');
  }
}

function showToast(message, type = 'info') {
  const toast = document.getElementById('toast');
  toast.innerText = message;
  toast.className = `toast ${type}`;
  toast.style.display = 'block';

  setTimeout(() => {
    toast.style.display = 'none';
  }, 4000);
}

function formatDate(dateStr) {
  if (!dateStr) return '';
  const [year, month, day] = dateStr.split('-');
  return `${day}/${month}/${year}`;
}

function formatDateTime(isoStr) {
  if (!isoStr) return '';
  const d = new Date(isoStr);
  return d.toLocaleDateString('pt-BR') + ' às ' + d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

# Sistema de Reservas de Hotel - Hotel Bellagio

Sistema web completo de reservas para hotel, pronto para uso tanto por **hóspedes** quanto pelo **gerente do hotel**. Todos os dados ficam salvos em arquivo permanente no servidor (`data/hotel.json`), permitindo que as informações persistam mesmo se o computador ou servidor for reiniciado, e que hóspedes e gerente acessem de qualquer aparelho (computador ou celular) com sincronização em tempo real.

---

## 🚀 Como Instalar e Rodar (Passo a Passo Simples)

### 1. Pré-requisito
Você só precisa ter o **Node.js** instalado no seu computador.
Se não tiver, baixe a versão recomendada no site oficial: [https://nodejs.org/](https://nodejs.org/) (a instalação é apenas avançar).

### 2. Baixar e Entrar na Pasta do Projeto
Abra o **Terminal** (no Mac/Linux) ou **Prompt de Comando / PowerShell** (no Windows) e entre na pasta do projeto:

```bash
cd caminho/para/a/pasta/do/projeto
```

### 3. Instalar as Dependências
Execute o comando abaixo:

```bash
npm install
```

### 4. Iniciar o Sistema
Execute o comando:

```bash
npm start
```

Pronto! O terminal mostrará:
```text
===============================================
 Hotel Bellagio - Sistema de Reservas Online
 Servidor rodando com sucesso na porta: 3000
 Acesse pelo computador: http://localhost:3000
===============================================
```

Abra seu navegador (Google Chrome, Firefox, Edge ou Safari) e acesse:
👉 **`http://localhost:3000`**

## 📱 Como Usar no Celular (Mesma Rede Wi-Fi)

Para testar o site funcionando no seu celular ao mesmo tempo que no computador:
1. Conecte o celular na mesma rede Wi-Fi do computador.
2. Descubra o IP local do seu computador:
   * **Windows:** Abra o Prompt de Comando e digite `ipconfig` (procure por *Endereço IPv4*, ex: `192.168.1.15`).
   * **Mac/Linux:** Abra o Terminal e digite `ifconfig` ou `ip a`.
3. No navegador do seu celular, digite:
   `http://SEU_IP_AQUI:3000` (exemplo: `http://192.168.1.15:3000`).
4. Tudo o que você fizer no celular aparecerá imediatamente no computador e vice-versa!
