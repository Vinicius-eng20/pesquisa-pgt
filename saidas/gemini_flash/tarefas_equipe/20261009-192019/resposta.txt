const sqlite3 = require('sqlite3').verbose();
const bcrypt = require('bcryptjs');
const path = require('path');

const dbPath = path.join(__dirname, 'tarefas.db');
const db = new sqlite3.Database(dbPath);

function initDB() {
  db.serialize(() => {
    // Tabela de usuários
    db.run(`
      CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        email TEXT UNIQUE NOT NULL,
        password TEXT NOT NULL,
        role TEXT NOT NULL CHECK(role IN ('gerente', 'membro')),
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `);

    // Tabela de tarefas
    db.run(`
      CREATE TABLE IF NOT EXISTS tasks (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        description TEXT,
        status TEXT NOT NULL DEFAULT 'pendente' CHECK(status IN ('pendente', 'em_andamento', 'concluida')),
        priority TEXT NOT NULL DEFAULT 'media' CHECK(priority IN ('baixa', 'media', 'alta')),
        assigned_to INTEGER NOT NULL,
        created_by INTEGER NOT NULL,
        due_date TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (assigned_to) REFERENCES users(id),
        FOREIGN KEY (created_by) REFERENCES users(id)
      )
    `);

    // Inserir usuários padrão se não existirem
    db.get('SELECT COUNT(*) as count FROM users', (err, row) => {
      if (err) {
        console.error('Erro ao verificar usuários:', err);
        return;
      }

      if (row.count === 0) {
        const salt = bcrypt.genSaltSync(10);
        const passGerente = bcrypt.hashSync('123456', salt);
        const passAna = bcrypt.hashSync('123456', salt);
        const passBruno = bcrypt.hashSync('123456', salt);

        const insertUser = db.prepare(`
          INSERT INTO users (name, email, password, role) VALUES (?, ?, ?, ?)
        `);

        insertUser.run('Carlos (Gerente)', 'gerente@equipe.com', passGerente, 'gerente');
        insertUser.run('Ana Silva', 'ana@equipe.com', passAna, 'membro');
        insertUser.run('Bruno Souza', 'bruno@equipe.com', passBruno, 'membro', function (err) {
          if (!err) {
            console.log('Usuários de demonstração criados com sucesso!');
            // Criar tarefas de exemplo
            criarTarefasIniciais();
          }
        });
        insertUser.finalize();
      }
    });
  });
}

function criarTarefasIniciais() {
  const insertTask = db.prepare(`
    INSERT INTO tasks (title, description, status, priority, assigned_to, created_by, due_date)
    VALUES (?, ?, ?, ?, ?, ?, ?)
  `);

  insertTask.run(
    'Revisar relatório mensal',
    'Conferir os números do mês anterior e preparar resumo para diretoria.',
    'em_andamento',
    'alta',
    2, // Ana
    1, // Carlos
    '2025-05-10'
  );

  insertTask.run(
    'Organizar arquivos do projeto',
    'Padronizar pastas e documentos no servidor compartilhado.',
    'pendente',
    'media',
    3, // Bruno
    1, // Carlos
    '2025-05-15'
  );

  insertTask.run(
    'Atualizar cadastro de fornecedores',
    'Telefonar para confirmação dos dados bancários atuais.',
    'concluida',
    'baixa',
    2, // Ana
    2, // Ana (criada por ela mesma)
    '2025-05-02'
  );

  insertTask.finalize();
}

const express = require('express');
const session = require('express-session');
const bcrypt = require('bcryptjs');
const path = require('path');
const db = require('./database');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static(path.join(__dirname, 'public')));

app.use(
  session({
    secret: 'chave-secreta-gestao-tarefas-2025',
    resave: false,
    saveUninitialized: false,
    cookie: {
      maxAge: 1000 * 60 * 60 * 24 * 7, // 7 dias de sessão salva
      httpOnly: true
    }
  })
);

// Middleware para verificar se está logado
function checkAuth(req, res, next) {
  if (!req.session.user) {
    return res.status(401).json({ error: 'Não autorizado. Faça login.' });
  }
  next();
}

// Middleware para verificar se é gerente
function checkManager(req, res, next) {
  if (!req.session.user || req.session.user.role !== 'gerente') {
    return res.status(403).json({ error: 'Acesso restrito para gerentes.' });
  }
  next();
}

// --- ROTAS DE AUTENTICAÇÃO ---

// Obter usuário logado atual
app.get('/api/me', (req, res) => {
  if (!req.session.user) {
    return res.json({ loggedIn: false });
  }
  res.json({ loggedIn: true, user: req.session.user });
});

// Login
app.post('/api/login', (req, res) => {
  const { email, password } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: 'Preencha email e senha.' });
  }

  db.get('SELECT * FROM users WHERE LOWER(email) = LOWER(?)', [email.trim()], (err, user) => {
    if (err) return res.status(500).json({ error: 'Erro no servidor.' });
    if (!user) return res.status(400).json({ error: 'E-mail ou senha incorretos.' });

    const passwordMatch = bcrypt.compareSync(password, user.password);
    if (!passwordMatch) {
      return res.status(400).json({ error: 'E-mail ou senha incorretos.' });
    }

    req.session.user = {
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role
    };

    res.json({ message: 'Login realizado com sucesso!', user: req.session.user });
  });
});

// Cadastro de nova conta
app.post('/api/register', (req, res) => {
  const { name, email, password, role } = req.body;

  if (!name || !email || !password) {
    return res.status(400).json({ error: 'Preencha todos os campos obrigatórios.' });
  }

  const userRole = role === 'gerente' ? 'gerente' : 'membro';

  db.get('SELECT id FROM users WHERE LOWER(email) = LOWER(?)', [email.trim()], (err, row) => {
    if (err) return res.status(500).json({ error: 'Erro no servidor.' });
    if (row) return res.status(400).json({ error: 'Este e-mail já está em uso.' });

    const hashedPassword = bcrypt.hashSync(password, 10);
    db.run(
      'INSERT INTO users (name, email, password, role) VALUES (?, ?, ?, ?)',
      [name.trim(), email.trim().toLowerCase(), hashedPassword, userRole],
      function (insertErr) {
        if (insertErr) return res.status(500).json({ error: 'Erro ao cadastrar usuário.' });

        const newUser = {
          id: this.lastID,
          name: name.trim(),
          email: email.trim().toLowerCase(),
          role: userRole
        };

        req.session.user = newUser;
        res.json({ message: 'Conta criada com sucesso!', user: newUser });
      }
    );
  });
});

// Listar tarefas
app.get('/api/tasks', checkAuth, (req, res) => {
  const currentUser = req.session.user;

  let query = `
    SELECT 
      t.id, 
      t.title, 
      t.description, 
      t.status, 
      t.priority, 
      t.due_date, 
      t.created_at, 
      t.assigned_to, 
      t.created_by,
      u_assignee.name AS assigned_to_name,
      u_creator.name AS created_by_name
    FROM tasks t
    LEFT JOIN users u_assignee ON t.assigned_to = u_assignee.id
    LEFT JOIN users u_creator ON t.created_by = u_creator.id
  `;

  const params = [];

  // Se for membro, vê apenas as suas tarefas atribuídas
  if (currentUser.role === 'membro') {
    query += ' WHERE t.assigned_to = ?';
    params.push(currentUser.id);
  }

  query += ' ORDER BY CASE t.status WHEN "em_andamento" THEN 1 WHEN "pendente" THEN 2 WHEN "concluida" THEN 3 END, t.due_date ASC, t.id DESC';

  db.all(query, params, (err, rows) => {
    if (err) return res.status(500).json({ error: 'Erro ao carregar tarefas.' });
    res.json(rows);
  });
});

// Criar tarefa
app.post('/api/tasks', checkAuth, (req, res) => {
  const currentUser = req.session.user;
  const { title, description, priority, assigned_to, due_date } = req.body;

  if (!title || !title.trim()) {
    return res.status(400).json({ error: 'O título da tarefa é obrigatório.' });
  }

  let finalAssignedTo = currentUser.id;

  // Se for gerente, pode atribuir para qualquer membro. Se for membro, a tarefa é sempre para si mesmo.
  if (currentUser.role === 'gerente') {
    finalAssignedTo = assigned_to ? parseInt(assigned_to, 10) : currentUser.id;
  }

  const finalPriority = ['baixa', 'media', 'alta'].includes(priority) ? priority : 'media';

  db.run(
    `INSERT INTO tasks (title, description, priority, assigned_to, created_by, due_date)
     VALUES (?, ?, ?, ?, ?, ?)`,
    [title.trim(), description ? description.trim() : '', finalPriority, finalAssignedTo, currentUser.id, due_date || null],
    function (err) {
      if (err) return res.status(500).json({ error: 'Erro ao criar a tarefa.' });
      res.json({ message: 'Tarefa criada com sucesso!', taskId: this.lastID });
    }
  );
});

// Atualizar status da tarefa (Tanto gerente quanto o membro dono podem mudar)
app.patch('/api/tasks/:id/status', checkAuth, (req, res) => {
  const taskId = req.params.id;
  const { status } = req.body;
  const currentUser = req.session.user;

  if (!['pendente', 'em_andamento', 'concluida'].includes(status)) {
    return res.status(400).json({ error: 'Situação inválida.' });
  }

  db.get('SELECT * FROM tasks WHERE id = ?', [taskId], (err, task) => {
    if (err) return res.status(500).json({ error: 'Erro ao buscar tarefa.' });
    if (!task) return res.status(404).json({ error: 'Tarefa não encontrada.' });

    // Permissão: Gerente pode mudar qualquer uma; Membro só pode mudar se a tarefa for dele
    if (currentUser.role !== 'gerente' && task.assigned_to !== currentUser.id) {
      return res.status(403).json({ error: 'Você só pode alterar o status das suas próprias tarefas.' });
    }

    db.run(
      'UPDATE tasks SET status = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?',
      [status, taskId],
      function (updateErr) {
        if (updateErr) return res.status(500).json({ error: 'Erro ao atualizar situação da tarefa.' });
        res.json({ message: 'Situação atualizada com sucesso!' });
      }
    );
  });
});

// Editar dados da tarefa (Apenas Gerente)
app.put('/api/tasks/:id', checkManager, (req, res) => {
  const taskId = req.params.id;
  const { title, description, status, priority, assigned_to, due_date } = req.body;

  if (!title || !title.trim()) {
    return res.status(400).json({ error: 'O título da tarefa é obrigatório.' });
  }

  db.run(
    `UPDATE tasks 
     SET title = ?, description = ?, status = ?, priority = ?, assigned_to = ?, due_date = ?, updated_at = CURRENT_TIMESTAMP
     WHERE id = ?`,
    [title.trim(), description || '', status, priority, assigned_to, due_date || null, taskId],
    function (err) {
      if (err) return res.status(500).json({ error: 'Erro ao atualizar tarefa.' });
      if (this.changes === 0) return res.status(404).json({ error: 'Tarefa não encontrada.' });
      res.json({ message: 'Tarefa alterada com sucesso!' });
    }
  );
});

// Excluir tarefa (Apenas Gerente)
app.delete('/api/tasks/:id', checkManager, (req, res) => {
  const taskId = req.params.id;

  db.run('DELETE FROM tasks WHERE id = ?', [taskId], function (err) {
    if (err) return res.status(500).json({ error: 'Erro ao excluir tarefa.' });
    if (this.changes === 0) return res.status(404).json({ error: 'Tarefa não encontrada.' });
    res.json({ message: 'Tarefa excluída com sucesso!' });
  });
});

async function verificarSessao() {
  try {
    const res = await fetch('/api/me');
    const data = await res.json();
    if (data.loggedIn && data.user) {
      currentUser = data.user;
      iniciarAplicacao();
    } else {
      mostrarTelaAuth();
    }
  } catch (err) {
    mostrarTelaAuth();
  }
}

async function tratarLogin(e) {
  e.preventDefault();
  authError.classList.add('hidden');

  const email = document.getElementById('login-email').value;
  const password = document.getElementById('login-password').value;

  try {
    const res = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });
    const data = await res.json();

    if (!res.ok) {
      mostrarErroAuth(data.error || 'Erro ao realizar login.');
      return;
    }

    currentUser = data.user;
    iniciarAplicacao();
  } catch (err) {
    mostrarErroAuth('Não foi possível conectar ao servidor.');
  }
}

async function tratarRegistro(e) {
  e.preventDefault();
  authError.classList.add('hidden');

  const name = document.getElementById('reg-name').value;
  const email = document.getElementById('reg-email').value;
  const password = document.getElementById('reg-password').value;
  const role = document.getElementById('reg-role').value;

  try {
    const res = await fetch('/api/register', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, email, password, role })
    });
    const data = await res.json();

    if (!res.ok) {
      mostrarErroAuth(data.error || 'Erro ao criar conta.');
      return;
    }

    currentUser = data.user;
    iniciarAplicacao();
  } catch (err) {
    mostrarErroAuth('Erro ao conectar ao servidor.');
  }
}

async function tratarLogout() {
  try {
    await fetch('/api/logout', { method: 'POST' });
  } finally {
    currentUser = null;
    mostrarTelaAuth();
  }
}

// Atualizar cards de estatísticas
function atualizarEstatisticas() {
  const total = allTasks.length;
  const pending = allTasks.filter(t => t.status === 'pendente').length;
  const progress = allTasks.filter(t => t.status === 'em_andamento').length;
  const completed = allTasks.filter(t => t.status === 'concluida').length;

  document.getElementById('stat-progress').textContent = progress;

// Renderizar lista filtrada
function renderizarTarefas() {
  const busca = filterSearch.value.toLowerCase().trim();
  const filtroStatus = filterStatus.value;
  const filtroPrioridade = filterPriority.value;
  const filtroMembro = filterMember.value;

  const tarefasFiltradas = allTasks.filter(t => {
    // Filtro texto
    const matchTexto = (t.title && t.title.toLowerCase().includes(busca)) ||
                       (t.description && t.description.toLowerCase().includes(busca));
    if (!matchTexto) return false;

    // Filtro status
    if (filtroStatus !== 'todos' && t.status !== filtroStatus) return false;

    // Filtro prioridade
    if (filtroPrioridade !== 'todas' && t.priority !== filtroPrioridade) return false;

    // Filtro membro (gerente)
    if (currentUser.role === 'gerente' && filtroMembro !== 'todos') {
      if (t.assigned_to != filtroMembro) return false;
    }

    return true;
  });

  tasksContainer.innerHTML = '';

  if (tarefasFiltradas.length === 0) {
    emptyState.classList.remove('hidden');
    return;
  }

  emptyState.classList.add('hidden');

  tarefasFiltradas.forEach(tarefa => {
    const card = criarCardTarefa(tarefa);
    tasksContainer.appendChild(card);
  });
}

          <span>Situação:</span>
        </span>

        <select onchange="alterarStatusTarefa(${t.id}, this.value)" class="text-xs font-semibold px-2 py-1 rounded-lg border bg-white focus:outline-none focus:ring-1 focus:ring-indigo-500 cursor-pointer ${statusInfo.border} ${statusInfo.text}">
          <option value="pendente" ${t.status === 'pendente' ? 'selected' : ''}>Pendente</option>
          <option value="em_andamento" ${t.status === 'em_andamento' ? 'selected' : ''}>Em Andamento</option>
          <option value="concluida" ${t.status === 'concluida' ? 'selected' : ''}>Concluída</option>
        </select>
      </div>
    </div>
  `;

  return card;
}

// --- CONTROLE DO MODAL DE CRIAÇÃO E EDIÇÃO ---

function abrirModalCriacao() {
  isEditing = false;
  taskForm.reset();
  taskIdInput.value = '';
  modalTitle.textContent = currentUser.role === 'gerente' ? 'Criar Nova Tarefa' : 'Criar Tarefa para Mim';
  statusEditContainer.classList.add('hidden');

  // Se for membro, o responsável padrão é ele mesmo
  if (currentUser.role === 'membro') {
    taskAssigneeInput.value = currentUser.id;
  }

  taskModal.classList.remove('hidden');
}

function abrirModalEdicao(id) {
  const task = allTasks.find(t => t.id === id);
  if (!task) return;

  isEditing = true;
  taskIdInput.value = task.id;
  taskTitleInput.value = task.title;
  taskDescInput.value = task.description || '';
  taskPriorityInput.value = task.priority;
  taskDueDateInput.value = task.due_date || '';
  taskAssigneeInput.value = task.assigned_to;
  taskStatusInput.value = task.status;

  modalTitle.textContent = 'Editar Tarefa';
  statusEditContainer.classList.remove('hidden');
  taskModal.classList.remove('hidden');
}

function fecharModalTarefa() {
  taskModal.classList.add('hidden');
  taskForm.reset();
}

// Salvar Tarefa (Criar ou Atualizar)
async function salvarTarefa(e) {
  e.preventDefault();

  const title = taskTitleInput.value.trim();
  const description = taskDescInput.value.trim();
  const priority = taskPriorityInput.value;
  const due_date = taskDueDateInput.value;
  const assigned_to = taskAssigneeInput.value;
  const status = taskStatusInput.value;

  if (!title) return;

  try {
    if (isEditing) {
      // Atualização via PUT (Gerente)
      const id = taskIdInput.value;
      const res = await fetch(`/api/tasks/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ title, description, priority, due_date, assigned_to, status })
      });
      if (!res.ok) throw new Error();
    } else {
      // Criação via POST
      const res = await fetch('/api/tasks', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ title, description, priority, due_date, assigned_to })
      });
      if (!res.ok) throw new Error();
    }

    fecharModalTarefa();
    await carregarTarefas();
  } catch (err) {
    alert('Erro ao salvar a tarefa. Tente novamente.');
  }
}

// Alterar Status
async function alterarStatusTarefa(id, novoStatus) {
  try {
    const res = await fetch(`/api/tasks/${id}/status`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ status: novoStatus })
    });

    if (!res.ok) {
      const data = await res.json();
      alert(data.error || 'Erro ao atualizar situação.');
      await carregarTarefas();
      return;
    }

    await carregarTarefas();
  } catch (err) {
    alert('Não foi possível atualizar a tarefa.');
    await carregarTarefas();
  }
}

// Excluir Tarefa
async function excluirTarefa(id) {
  if (!confirm('Tem certeza de que deseja remover esta tarefa definitivamente?')) {
    return;
  }

  try {
    const res = await fetch(`/api/tasks/${id}`, { method: 'DELETE' });
    if (!res.ok) {
      const data = await res.json();
      alert(data.error || 'Erro ao excluir tarefa.');
      return;
    }
    await carregarTarefas();
  } catch (err) {
    alert('Erro ao excluir tarefa.');
  }
}

// Função utilitária para evitar injeção de HTML
function escapeHtml(text) {
  if (!text) return '';
  return text
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

# Gestão de Tarefas da Equipe

Sistema completo de gestão e distribuição de tarefas para equipes e gerentes. Funciona em tempo real pelo navegador, tanto no computador quanto no celular. Os dados ficam salvos em um banco de dados próprio e nunca somem ao fechar a janela.

---

## 🚀 Como Colocar para Funcionar (Passo a Passo Fácil)

Você só precisa ter o **Node.js** instalado no seu computador. (Se ainda não tiver, baixe a versão recomendada em [nodejs.org](https://nodejs.org/)).

### 1. Abra o Terminal ou Prompt de Comando
Abra a pasta do projeto no Terminal (ou no Prompt de Comando do Windows).

### 2. Instale os componentes do sistema
Digite o comando abaixo e aperte **Enter**:
```bash
npm install
```
*Espere alguns segundos até que ele termine de baixar tudo o que precisa.*

### 3. Inicie o site
Digite o comando abaixo e aperte **Enter**:
```bash
npm start
```

Pronto! Você verá uma mensagem confirmando que o sistema está no ar:
```text
===============================================
 Sistema de Gestão de Tarefas Rodando!
 Acesso local: http://localhost:3000
===============================================
```

### 4. Abra no Navegador
Abra o navegador de sua preferência (Chrome, Edge, Firefox, Safari) e acesse:
👉 **`http://localhost:3000`**
