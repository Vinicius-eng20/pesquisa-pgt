# SuperShop - Shopping Virtual de Departamentos

Loja virtual de departamentos completa, persistente e pronta para uso. Funciona em computadores, tablets e celulares, sincronizando os dados em tempo real no servidor SQLite local sem depender de nenhum serviço externo ou criar contas em outros sites.

---

## 🚀 Como Instalar e Rodar o Projeto

Você só precisa ter o **Node.js** instalado no seu computador (se não tiver, baixe a versão recomendada LTS no site oficial: [https://nodejs.org](https://nodejs.org)).

### Passo 1: Abrir a pasta do projeto no Terminal ou Prompt de Comando
Abra o Prompt de Comando (Windows) ou Terminal (Mac/Linux) na pasta onde estão os arquivos deste projeto.

### Passo 2: Instalar as dependências
Digite o comando abaixo e aperte Enter:
```bash
npm install
```

### Passo 3: Iniciar o site
Após o término da instalação, execute o comando:
```bash
npm start
```

### Passo 4: Acessar a loja
Abra seu navegador de internet (Chrome, Edge, Firefox, Safari) e acesse:
```
http://localhost:3000
```

> **Para acessar pelo celular ou outro computador na mesma rede Wi-Fi:**  
> Descubra o endereço de IP do seu computador (ex: `192.168.1.15`) e acesse no celular:  
> `http://192.168.1.15:3000`

  fs.mkdirSync(uploadsDir, { recursive: true });

const dbPath = path.join(dataDir, 'loja.db');
const db = new sqlite3.Database(dbPath);

function run(sql, params = []) {
  return new Promise((resolve, reject) => {
    db.run(sql, params, function (err) {
      if (err) reject(err);
      else resolve(this);
    });
  });
}

function get(sql, params = []) {
  return new Promise((resolve, reject) => {
    db.get(sql, params, (err, row) => {
      if (err) reject(err);
      else resolve(row);
    });
  });
}

function all(sql, params = []) {
  return new Promise((resolve, reject) => {
    db.all(sql, params, (err, rows) => {
      if (err) reject(err);
      else resolve(rows);
    });
  });
}

async function initDatabase() {
  await run(`
    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      email TEXT UNIQUE NOT NULL,
      password TEXT NOT NULL,
      role TEXT DEFAULT 'customer',
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS categories (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT UNIQUE NOT NULL,
      slug TEXT UNIQUE NOT NULL,
      icon TEXT DEFAULT 'bi-tag'
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS products (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      category_id INTEGER NOT NULL,
      price REAL NOT NULL,
      stock INTEGER NOT NULL DEFAULT 10,
      description TEXT,
      image TEXT,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (category_id) REFERENCES categories (id)
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS cart_items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      user_id INTEGER NOT NULL,
      product_id INTEGER NOT NULL,
      quantity INTEGER NOT NULL DEFAULT 1,
      FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE,
      FOREIGN KEY (product_id) REFERENCES products (id) ON DELETE CASCADE,
      UNIQUE(user_id, product_id)
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS orders (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      user_id INTEGER NOT NULL,
      total REAL NOT NULL,
      status TEXT DEFAULT 'Pendente',
      shipping_address TEXT NOT NULL,
      payment_method TEXT NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (user_id) REFERENCES users (id)
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS order_items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      order_id INTEGER NOT NULL,
      product_id INTEGER,
      product_name TEXT NOT NULL,
      price REAL NOT NULL,
      quantity INTEGER NOT NULL,
      FOREIGN KEY (order_id) REFERENCES orders (id) ON DELETE CASCADE
    )
  `);

  await seedInitialData();
}

const express = require('express');
const router = express.Router();
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');
const { run, get } = require('../database');

const JWT_SECRET = 'supershop_chave_secreta_local_12345';

router.post('/register', async (req, res) => {
  try {
    const { name, email, password } = req.body;
    if (!name || !email || !password) {
      return res.status(400).json({ error: 'Preencha todos os campos obrigatórios.' });
    }

    const cleanEmail = email.trim().toLowerCase();
    const existing = await get(`SELECT id FROM users WHERE email = ?`, [cleanEmail]);
    if (existing) {
      return res.status(400).json({ error: 'Este e-mail já está cadastrado.' });
    }

    const hashedPassword = bcrypt.hashSync(password, 10);
    const result = await run(
      `INSERT INTO users (name, email, password, role) VALUES (?, ?, ?, 'customer')`,
      [name.trim(), cleanEmail, hashedPassword]
    );

    const user = {
      id: result.lastID,
      name: name.trim(),
      email: cleanEmail,
      role: 'customer'
    };

    const token = jwt.sign(user, JWT_SECRET, { expiresIn: '30d' });
    res.cookie('token', token, { httpOnly: true, maxAge: 30 * 24 * 60 * 60 * 1000, sameSite: 'lax' });

    res.json({ message: 'Conta criada com sucesso!', user, token });
  } catch (err) {
    res.status(500).json({ error: 'Erro ao registrar usuário: ' + err.message });
  }
});

router.post('/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    if (!email || !password) {
      return res.status(400).json({ error: 'Informe e-mail e senha.' });
    }

    const cleanEmail = email.trim().toLowerCase();
    const user = await get(`SELECT * FROM users WHERE email = ?`, [cleanEmail]);
    if (!user) {
      return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
    }

    const valid = bcrypt.compareSync(password, user.password);
    if (!valid) {
      return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
    }

    const userPayload = {
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role
    };

    const token = jwt.sign(userPayload, JWT_SECRET, { expiresIn: '30d' });
    res.cookie('token', token, { httpOnly: true, maxAge: 30 * 24 * 60 * 60 * 1000, sameSite: 'lax' });

    res.json({ message: 'Login realizado com sucesso!', user: userPayload, token });
  } catch (err) {
    res.status(500).json({ error: 'Erro ao fazer login: ' + err.message });
  }
});

router.post('/logout', (req, res) => {
  res.clearCookie('token');
  res.json({ message: 'Sessão encerrada com sucesso.' });
});

router.get('/me', async (req, res) => {
  try {
    const token = req.cookies.token || (req.headers.authorization && req.headers.authorization.split(' ')[1]);
    if (!token) {
      return res.json({ user: null });
    }

    const decoded = jwt.verify(token, JWT_SECRET);
    const user = await get(`SELECT id, name, email, role FROM users WHERE id = ?`, [decoded.id]);
    if (!user) {
      return res.json({ user: null });
    }

    res.json({ user });
  } catch (err) {
    res.json({ user: null });
  }
});

  destination: (req, file, cb) => {

function requireAdmin(req, res, next) {

  }
  next();
}

router.get('/', async (req, res) => {
  try {
    const { category, search, sort } = req.query;
    let query = `
      SELECT p.*, c.name as category_name, c.slug as category_slug
      FROM products p
      JOIN categories c ON p.category_id = c.id
      WHERE 1=1
    `;
    const params = [];

    if (category && category !== 'todos') {
      query += ` AND (c.slug = ? OR c.id = ?)`;
      params.push(category, category);
    }

    if (search && search.trim() !== '') {
      query += ` AND (p.name LIKE ? OR p.description LIKE ?)`;
      params.push(`%${search.trim()}%`, `%${search.trim()}%`);
    }

    if (sort === 'price_asc') {
      query += ` ORDER BY p.price ASC`;
    } else if (sort === 'price_desc') {
      query += ` ORDER BY p.price DESC`;
    } else if (sort === 'name') {
      query += ` ORDER BY p.name ASC`;
    } else {
      query += ` ORDER BY p.id DESC`;
    }

    const products = await all(query, params);
    res.json(products);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

const express = require('express');
const router = express.Router();
const { run, get, all } = require('../database');

function requireAuth(req, res, next) {
  if (!req.user) {
    return res.status(401).json({ error: 'Você precisa estar logado para acessar seu carrinho.' });
  }
  next();
}

router.use(requireAuth);

router.get('/', async (req, res) => {
  try {
    const items = await all(
      `SELECT c.id as cart_id, c.quantity, p.id as product_id, p.name, p.price, p.stock, p.image,
              (c.quantity * p.price) as subtotal
       FROM cart_items c
       JOIN products p ON c.product_id = p.id
       WHERE c.user_id = ?`,
      [req.user.id]
    );

    const total = items.reduce((sum, item) => sum + item.subtotal, 0);
    res.json({ items, total });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

router.post('/add', async (req, res) => {
  try {
    const { product_id, quantity = 1 } = req.body;
    const qty = parseInt(quantity) || 1;

    const product = await get(`SELECT * FROM products WHERE id = ?`, [product_id]);
    if (!product) {
      return res.status(404).json({ error: 'Produto não encontrado.' });
    }

    if (product.stock < qty) {
      return res.status(400).json({ error: 'Estoque insuficiente para a quantidade solicitada.' });
    }

    const existing = await get(
      `SELECT * FROM cart_items WHERE user_id = ? AND product_id = ?`,
      [req.user.id, product_id]
    );

    if (existing) {
      const newQty = existing.quantity + qty;
      if (product.stock < newQty) {
        return res.status(400).json({ error: `Você já tem ${existing.quantity} no carrinho. Estoque máximo disponível: ${product.stock}.` });
      }
      await run(`UPDATE cart_items SET quantity = ? WHERE id = ?`, [newQty, existing.id]);
    } else {
      await run(
        `INSERT INTO cart_items (user_id, product_id, quantity) VALUES (?, ?, ?)`,
        [req.user.id, product_id, qty]
      );
    }

    res.json({ message: 'Produto adicionado ao seu carrinho!' });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

router.put('/update', async (req, res) => {
  try {
    const { product_id, quantity } = req.body;
    const qty = parseInt(quantity);

    if (qty <= 0) {
      await run(`DELETE FROM cart_items WHERE user_id = ? AND product_id = ?`, [req.user.id, product_id]);
      return res.json({ message: 'Item removido do carrinho.' });
    }

    const product = await get(`SELECT stock FROM products WHERE id = ?`, [product_id]);
    if (!product || product.stock < qty) {
      return res.status(400).json({ error: 'Quantidade maior que o estoque disponível.' });
    }

    await run(
      `UPDATE cart_items SET quantity = ? WHERE user_id = ? AND product_id = ?`,
      [qty, req.user.id, product_id]
    );

    res.json({ message: 'Carrinho atualizado.' });
  } catch (err) {
    res.status(500).json({ error: err.message });

router.post('/checkout', async (req, res) => {
  try {
    const { shipping_address, payment_method } = req.body;

    if (!shipping_address || !shipping_address.trim()) {
      return res.status(400).json({ error: 'Informe o endereço completo para entrega.' });
    }

    const cartItems = await all(
      `SELECT c.quantity, p.id as product_id, p.name, p.price, p.stock
       FROM cart_items c
       JOIN products p ON c.product_id = p.id
       WHERE c.user_id = ?`,
      [req.user.id]
    );

    if (cartItems.length === 0) {
      return res.status(400).json({ error: 'Seu carrinho está vazio.' });
    }

    for (const item of cartItems) {
      if (item.stock < item.quantity) {
        return res.status(400).json({
          error: `O produto "${item.name}" só possui ${item.stock} unidades em estoque.`
        });
      }
    }

    const total = cartItems.reduce((acc, item) => acc + (item.price * item.quantity), 0);

    const orderRes = await run(
      `INSERT INTO orders (user_id, total, status, shipping_address, payment_method) VALUES (?, ?, 'Pendente', ?, ?)`,
      [req.user.id, total, shipping_address.trim(), payment_method || 'Cartão de Crédito']
    );

    const orderId = orderRes.lastID;

    for (const item of cartItems) {
      await run(
        `INSERT INTO order_items (order_id, product_id, product_name, price, quantity) VALUES (?, ?, ?, ?, ?)`,
        [orderId, item.product_id, item.name, item.price, item.quantity]
      );

      await run(
        `UPDATE products SET stock = stock - ? WHERE id = ?`,
        [item.quantity, item.product_id]
      );
    }

    await run(`DELETE FROM cart_items WHERE user_id = ?`, [req.user.id]);

    res.status(201).json({
      message: 'Pedido realizado com sucesso!',
      orderId,
      total
    });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

const app = {
  currentUser: null,
  categories: [],
  products: [],
  cart: { items: [], total: 0 },
  activeCategory: 'todos',
  activeSort: 'recent',
  authMode: 'login',

  async init() {
    await this.checkAuth();
    await this.loadCategories();
    await this.loadProducts();
    if (this.currentUser) {
      await this.loadCart();
    }
  },

  showToast(message, isError = false) {
    const toast = document.getElementById('toast');
    toast.textContent = message;
    toast.style.background = isError ? '#ef4444' : '#1e293b';
    toast.style.display = 'block';
    setTimeout(() => {
      toast.style.display = 'none';
    }, 3500);
  },

  async api(url, options = {}) {
    options.headers = options.headers || {};
    if (!(options.body instanceof FormData)) {
      options.headers['Content-Type'] = 'application/json';
    }
    const res = await fetch(url, options);
    const data = await res.json().catch(() => ({}));
    if (!res.ok) {
      throw new Error(data.error || 'Erro na requisição');
    }
    return data;
  },

  formatCurrency(value) {
    return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value);
  },

  async checkAuth() {
    try {
      const data = await this.api('/api/auth/me');
      this.currentUser = data.user;
      this.renderNav();
    } catch (e) {
      this.currentUser = null;
      this.renderNav();
    }
  },

  renderNav() {
    const nav = document.getElementById('userNavActions');
    const cartCount = this.cart.items.reduce((sum, item) => sum + item.quantity, 0);

    if (this.currentUser) {
      const isAdmin = this.currentUser.role === 'admin';
      nav.innerHTML = `
        ${isAdmin ? `
          <button class="action-btn" onclick="app.navigate('admin')">
            <i class="fa-solid fa-gauge-high"></i>
            <span>Painel do Dono</span>
          </button>
        ` : `
          <button class="action-btn" onclick="app.navigate('orders')">
            <i class="fa-solid fa-receipt"></i>
            <span>Meus Pedidos</span>
          </button>
        `}
        <button class="action-btn" onclick="app.toggleCart(true)">
          <i class="fa-solid fa-cart-shopping"></i>
          <span>Carrinho</span>
          <span class="badge-cart">${cartCount}</span>
        </button>
        <button class="action-btn" onclick="app.logout()" title="Sair">
          <i class="fa-solid fa-right-from-bracket"></i>
          <span>Sair (${this.currentUser.name.split(' ')[0]})</span>
        </button>
      `;
    } else {
      nav.innerHTML = `
        <button class="action-btn" onclick="app.openAuthModal('login')">

        </button>
      `;
    }
  },

  async loadCategories() {
    try {
      this.categories = await this.api('/api/products/categories');

      const quickDepts = document.getElementById('quickDepartments');

  async loadProducts() {

      if (this.products.length === 0) {
        grid.innerHTML = `

            <h3>Nenhum produto encontrado neste departamento.</h3>
            <p class="text-muted">Tente buscar por outro termo ou selecione outro departamento acima.</p>

            <span class="product-stock ${p.stock <= 3 ? 'low' : ''}">

            </span>
            <div class="product-price-row">
              <span class="product-price">${this.formatCurrency(p.price)}</span>
              <span class="product-installments">12x de ${this.formatCurrency(p.price / 12)}</span>

            <button class="btn btn-primary btn-block" onclick="app.addToCart(${p.id})" ${p.stock <= 0 ? 'disabled' : ''}>

          </div>
        </div>
      `).join('');
    } catch (e) {
      grid.innerHTML = `<p class="text-danger">Erro ao carregar produtos: ${e.message}</p>`;
    }
  },

  filterByCategory(slug) {
    this.activeCategory = slug;
    document.querySelectorAll('.categories-list li').forEach(li => li.classList.remove('active'));
    this.navigate('home');

    const catItems = document.querySelectorAll('.categories-list li');
    catItems.forEach(el => {
      if (el.textContent.includes(slug) || (slug === 'todos' && el.textContent.includes('Todos'))) {
        el.classList.add('active');
      }
    });

    const currentCatObj = this.categories.find(c => c.slug === slug);
    document.getElementById('catalogHeading').textContent = currentCatObj ? currentCatObj.name : 'Todos os Produtos';
    this.loadProducts();
    this.scrollToProducts();
  },

  sortProducts(val) {
    this.activeSort = val;
    this.loadProducts();
  },

  searchProducts() {
    this.navigate('home');
    this.loadProducts();
    this.scrollToProducts();
  },

  scrollToProducts() {
    const el = document.getElementById('productsAnchor');
    if (el) el.scrollIntoView({ behavior: 'smooth' });
  },

  navigate(view) {
    document.getElementById('viewHome').classList.add('hidden');
    document.getElementById('viewOrders').classList.add('hidden');
    document.getElementById('viewAdmin').classList.add('hidden');

    if (view === 'home') {
      document.getElementById('viewHome').classList.remove('hidden');
    } else if (view === 'orders') {
      if (!this.currentUser) return this.openAuthModal('login');
      document.getElementById('viewOrders').classList.remove('hidden');
      this.loadCustomerOrders();
    } else if (view === 'admin') {
      if (!this.currentUser || this.currentUser.role !== 'admin') {
        this.showToast('Acesso permitido apenas para o dono da loja.', true);
        return this.navigate('home');
      }
      document.getElementById('viewAdmin').classList.remove('hidden');
      this.loadAdminData();
    }
    window.scrollTo({ top: 0, behavior: 'smooth' });
  },

  // CARRINHO PERSISTENTE (CROSS-DEVICE)
  async loadCart() {
    if (!this.currentUser) return;
    try {
      this.cart = await this.api('/api/cart');
      this.renderCart();
      this.renderNav();
    } catch (e) {
      console.error(e);
    }
  },

  async addToCart(productId) {
    if (!this.currentUser) {
      this.showToast('Faça login para colocar produtos no carrinho!');
      return this.openAuthModal('login');
    }

    try {
      await this.api('/api/cart/add', {
        method: 'POST',
        body: JSON.stringify({ product_id: productId, quantity: 1 })
      });
      await this.loadCart();
      this.showToast('Produto colocado no seu carrinho!');
      this.toggleCart(true);
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  async updateCartQty(productId, qty) {
    try {
      await this.api('/api/cart/update', {
        method: 'PUT',
        body: JSON.stringify({ product_id: productId, quantity: qty })
      });
      await this.loadCart();
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  async removeFromCart(productId) {
    try {
      await this.api(`/api/cart/remove/${productId}`, { method: 'DELETE' });
      await this.loadCart();
      this.showToast('Produto retirado do carrinho.');
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  renderCart() {
    const body = document.getElementById('cartBody');
    const totalEl = document.getElementById('cartTotalPrice');

    if (!this.cart.items || this.cart.items.length === 0) {
      body.innerHTML = `
        <div style="text-align: center; padding: 40px 10px;">
          <i class="fa-solid fa-basket-shopping" style="font-size: 3rem; color: #cbd5e1; margin-bottom: 14px;"></i>
          <p style="font-weight: 600; color: #64748b;">Seu carrinho está vazio.</p>
          <small class="text-muted">Aproveite nossos departamentos e coloque ofertas incríveis!</small>
        </div>
      `;
      totalEl.textContent = this.formatCurrency(0);
      return;
    }

    body.innerHTML = this.cart.items.map(item => `
      <div class="cart-item">
        <img src="${item.image}" alt="${item.name}" onerror="this.src='https://images.unsplash.com/photo-1526170375885-4d8ecf77b99f?auto=format&fit=crop&w=700&q=80'">
        <div class="cart-item-details">
          <div class="cart-item-title">${item.name}</div>
          <div class="cart-item-price">${this.formatCurrency(item.price)}</div>
          <div class="cart-item-qty">
            <button class="qty-btn" onclick="app.updateCartQty(${item.product_id}, ${item.quantity - 1})">-</button>
            <span><strong>${item.quantity}</strong></span>
            <button class="qty-btn" onclick="app.updateCartQty(${item.product_id}, ${item.quantity + 1})">+</button>
            <button class="btn btn-sm btn-outline" style="margin-left: auto; color: #ef4444;" onclick="app.removeFromCart(${item.product_id})">
              <i class="fa-solid fa-trash-can"></i>
            </button>
          </div>
        </div>
      </div>
    `).join('');

    totalEl.textContent = this.formatCurrency(this.cart.total);
  },

  toggleCart(open) {
    const overlay = document.getElementById('cartOverlay');
    const drawer = document.getElementById('cartDrawer');
    if (open) {
      overlay.style.display = 'block';
      drawer.classList.add('open');
    } else {
      overlay.style.display = 'none';
      drawer.classList.remove('open');
    }
  },

  // AUTHENTICATION
  openAuthModal(mode = 'login') {
    this.setAuthMode(mode);
    this.openModal('authModal');
  },

  setAuthMode(mode) {
    this.authMode = mode;
    const isLogin = mode === 'login';
    document.getElementById('toggleLoginBtn').className = isLogin ? 'active' : '';
    document.getElementById('toggleRegisterBtn').className = isLogin ? '' : 'active';
    document.getElementById('registerNameGroup').classList.toggle('hidden', isLogin);
    document.getElementById('authModalTitle').innerHTML = isLogin ? '<i class="fa-solid fa-user"></i> Entrar na Loja' : '<i class="fa-solid fa-user-plus"></i> Criar Nova Conta';
    document.getElementById('authSubmitBtn').textContent = isLogin ? 'Entrar na Loja' : 'Finalizar Cadastro';
  },

  async handleAuthSubmit(event) {
    event.preventDefault();
    const email = document.getElementById('authEmail').value;
    const password = document.getElementById('authPassword').value;
    const name = document.getElementById('authName').value;

    const endpoint = this.authMode === 'login' ? '/api/auth/login' : '/api/auth/register';
    const payload = this.authMode === 'login' ? { email, password } : { name, email, password };

    try {
      const data = await this.api(endpoint, {
        method: 'POST',
        body: JSON.stringify(payload)
      });

      this.currentUser = data.user;
      this.closeModal('authModal');
      this.showToast(data.message);
      this.renderNav();
      await this.loadCart();

      if (this.currentUser.role === 'admin') {
        this.navigate('admin');
      } else {
        this.loadProducts();
      }
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  async logout() {
    try {
      await this.api('/api/auth/logout', { method: 'POST' });
      this.currentUser = null;
      this.cart = { items: [], total: 0 };
      this.renderNav();
      this.navigate('home');
      this.showToast('Sessão encerrada.');
    } catch (e) {
      console.error(e);
    }
  },

  // CHECKOUT
  openCheckoutModal() {
    if (!this.currentUser) return this.openAuthModal('login');
    if (!this.cart.items || this.cart.items.length === 0) {
      return this.showToast('Coloque itens no carrinho antes de finalizar a compra.', true);
    }

    this.toggleCart(false);
    document.getElementById('summarySubtotal').textContent = this.formatCurrency(this.cart.total);
    document.getElementById('summaryTotal').textContent = this.formatCurrency(this.cart.total);
    this.openModal('checkoutModal');
  },

  async handleCheckout(event) {
    event.preventDefault();
    const shipping_address = document.getElementById('checkoutAddress').value;
    const payment_method = document.getElementById('checkoutPayment').value;

    try {
      const res = await this.api('/api/orders/checkout', {
        method: 'POST',
        body: JSON.stringify({ shipping_address, payment_method })
      });

      this.closeModal('checkoutModal');
      await this.loadCart();
      await this.loadProducts();
      this.showToast(`🎉 Pedido #${res.orderId} realizado com sucesso!`);
      this.navigate('orders');
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

      document.querySelectorAll('.tab-btn')[1].classList.add('active');

  openNewProductModal() {
    document.getElementById('productForm').reset();

  async openEditProductModal(id) {
    try {
      const p = await this.api(`/api/products/${id}`);
      document.getElementById('prodEditId').value = p.id;
      document.getElementById('prodName').value = p.name;
      document.getElementById('prodCategory').value = p.category_id;
      document.getElementById('prodPrice').value = p.price;
      document.getElementById('prodStock').value = p.stock;
      document.getElementById('prodImageUrl').value = p.image.startsWith('http') ? p.image : '';
      document.getElementById('prodDescription').value = p.description || '';
      document.getElementById('productModalTitle').innerHTML = '<i class="fa-solid fa-pen-to-square"></i> Editar Produto';
      this.openModal('productModal');
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  async handleSaveProduct(event) {
    event.preventDefault();
    const editId = document.getElementById('prodEditId').value;
    const formData = new FormData();

    formData.append('name', document.getElementById('prodName').value);
    formData.append('category_id', document.getElementById('prodCategory').value);
    formData.append('price', document.getElementById('prodPrice').value);
    formData.append('stock', document.getElementById('prodStock').value);
    formData.append('description', document.getElementById('prodDescription').value);

    const imageUrl = document.getElementById('prodImageUrl').value;
    if (imageUrl) formData.append('image_url', imageUrl);

    const fileInput = document.getElementById('prodImageFile');
    if (fileInput.files.length > 0) {
      formData.append('image_file', fileInput.files[0]);
    }

    try {
      const url = editId ? `/api/products/${editId}` : '/api/products';
      const method = editId ? 'PUT' : 'POST';

      const res = await fetch(url, { method, body: formData });
      const data = await res.json();
      if (!res.ok) throw new Error(data.error || 'Erro ao salvar produto');

      this.closeModal('productModal');
      this.showToast(editId ? 'Produto alterado com sucesso!' : 'Novo produto cadastrado no shopping!');
      await this.loadAdminProducts();
      await this.loadProducts();
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  async deleteProduct(id) {
    if (!confirm('Deseja realmente remover este produto do shopping?')) return;
    try {
      await this.api(`/api/products/${id}`, { method: 'DELETE' });
      this.showToast('Produto removido com sucesso!');
      await this.loadAdminProducts();
      await this.loadProducts();
    } catch (e) {
      this.showToast(e.message, true);
    }
  },

  // MODAIS
  openModal(modalId) {
    document.getElementById(modalId).classList.add('show');
  },

  closeModal(modalId) {
    document.getElementById(modalId).classList.remove('show');
  }
};
