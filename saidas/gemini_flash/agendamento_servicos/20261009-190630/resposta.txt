function initDatabase() {
  db.serialize(() => {
    // Tabela de Usuários
    db.run(`
      CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        email TEXT UNIQUE NOT NULL,
        phone TEXT NOT NULL,
        password_hash TEXT NOT NULL,
        role TEXT NOT NULL DEFAULT 'client',
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `);

    // Tabela de Serviços
    db.run(`
      CREATE TABLE IF NOT EXISTS services (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        description TEXT,
        duration_minutes INTEGER NOT NULL DEFAULT 30,
        price REAL NOT NULL DEFAULT 0.00,
        active INTEGER NOT NULL DEFAULT 1
      )
    `);

    // Tabela de Horários de Funcionamento (dias da semana: 0 = Domingo, 1 = Segunda, etc.)
    db.run(`
      CREATE TABLE IF NOT EXISTS business_hours (
        day_of_week INTEGER PRIMARY KEY,
        day_name TEXT NOT NULL,
        is_open INTEGER NOT NULL DEFAULT 1,
        start_time TEXT NOT NULL DEFAULT '08:00',
        end_time TEXT NOT NULL DEFAULT '18:00',
        slot_interval INTEGER NOT NULL DEFAULT 30
      )
    `);

    // Tabela de Agendamentos
    db.run(`
      CREATE TABLE IF NOT EXISTS appointments (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        client_id INTEGER NOT NULL,
        service_id INTEGER NOT NULL,
        appointment_date TEXT NOT NULL,
        appointment_time TEXT NOT NULL,
        status TEXT NOT NULL DEFAULT 'pendente',
        notes TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (client_id) REFERENCES users (id),
        FOREIGN KEY (service_id) REFERENCES services (id)
      )
    `);

    // Criar Administrador padrão se não existir nenhum
    db.get(`SELECT id FROM users WHERE role = 'admin' LIMIT 1`, (err, row) => {
      if (!row) {
        const adminPassword = bcrypt.hashSync('admin123', 10);
        db.run(
          `INSERT INTO users (name, email, phone, password_hash, role) VALUES (?, ?, ?, ?, ?)`,
          ['Responsável (Admin)', 'admin@negocio.com', '11999999999', adminPassword, 'admin'],
          (insertErr) => {
            if (!insertErr) {
              console.log('Conta de administrador criada: admin@negocio.com / senha: admin123');
            }
          }
        );
      }
    });

    // Inserir dias padrão se a tabela de horários estiver vazia
    db.get(`SELECT COUNT(*) as count FROM business_hours`, (err, row) => {
      if (row && row.count === 0) {
        const days = [
          { id: 0, name: 'Domingo', open: 0, start: '08:00', end: '12:00' },
          { id: 1, name: 'Segunda-feira', open: 1, start: '08:00', end: '18:00' },
          { id: 2, name: 'Terça-feira', open: 1, start: '08:00', end: '18:00' },
          { id: 3, name: 'Quarta-feira', open: 1, start: '08:00', end: '18:00' },
          { id: 4, name: 'Quinta-feira', open: 1, start: '08:00', end: '18:00' },
          { id: 5, name: 'Sexta-feira', open: 1, start: '08:00', end: '18:00' },
          { id: 6, name: 'Sábado', open: 1, start: '08:00', end: '14:00' }
        ];

        const stmt = db.prepare(
          `INSERT INTO business_hours (day_of_week, day_name, is_open, start_time, end_time, slot_interval) VALUES (?, ?, ?, ?, ?, 30)`
        );
        days.forEach(d => {

// Middleware para verificar autenticação
function requireAuth(req, res, next) {
  if (!req.session.user) {
    return res.status(401).json({ error: 'Você precisa entrar na sua conta.' });
  }
  next();
}

// Middleware para verificar se é Admin/Responsável
function requireAdmin(req, res, next) {
  if (!req.session.user || req.session.user.role !== 'admin') {
    return res.status(403).json({ error: 'Acesso restrito ao responsável.' });
  }
  next();
}

app.post('/api/auth/register', (req, res) => {
  const { name, email, phone, password } = req.body;

  if (!name || !email || !phone || !password) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios.' });
  }

  const cleanEmail = email.toLowerCase().trim();

  db.get(`SELECT id FROM users WHERE email = ?`, [cleanEmail], (err, existing) => {
    if (err) return res.status(500).json({ error: 'Erro no servidor.' });
    if (existing) {
      return res.status(400).json({ error: 'Este e-mail já está cadastrado.' });
    }

    const passwordHash = bcrypt.hashSync(password, 10);
    db.run(
      `INSERT INTO users (name, email, phone, password_hash, role) VALUES (?, ?, ?, ?, 'client')`,
      [name.trim(), cleanEmail, phone.trim(), passwordHash],
      function (insertErr) {
        if (insertErr) {
          return res.status(500).json({ error: 'Erro ao cadastrar usuário.' });
        }

        const newUser = {
          id: this.lastID,
          name: name.trim(),
          email: cleanEmail,
          phone: phone.trim(),
          role: 'client'
        };

        req.session.user = newUser;
        res.status(201).json({ success: true, user: newUser });
      }
    );
  });
});

    const match = bcrypt.compareSync(password, user.password_hash);

      return res.status(400).json({ error: 'E-mail ou senha incorretos.' });
    }

    const sessionUser = {
      id: user.id,
      name: user.name,
      email: user.email,
      phone: user.phone,
      role: user.role
    };

    req.session.user = sessionUser;
    res.json({ success: true, user: sessionUser });
  });
});

app.get('/api/available-slots', (req, res) => {
  const { date, service_id } = req.query;

  if (!date) {
    return res.status(400).json({ error: 'Data não informada.' });
  }

  // Descobrir o dia da semana da data escolhida (YYYY-MM-DD)
  const parts = date.split('-');
  const targetDate = new Date(parts[0], parts[1] - 1, parts[2]);
  const dayOfWeek = targetDate.getDay();

  // Buscar horário de funcionamento desse dia
  db.get(`SELECT * FROM business_hours WHERE day_of_week = ?`, [dayOfWeek], (err, dayConfig) => {
    if (err) return res.status(500).json({ error: 'Erro ao consultar expediente.' });

    if (!dayConfig || dayConfig.is_open === 0) {
      return res.json({ slots: [], message: 'O negócio não abre neste dia da semana.' });
    }

    // Buscar agendamentos existentes neste dia que não foram recusados ou cancelados
    db.all(
      `SELECT appointment_time FROM appointments 
       WHERE appointment_date = ? AND status IN ('pendente', 'confirmado')`,
      [date],
      (errAppointments, bookedRows) => {
        if (errAppointments) {
          return res.status(500).json({ error: 'Erro ao verificar agendamentos existentes.' });
        }

        const bookedTimes = new Set(bookedRows.map(r => r.appointment_time));

        // Gerar todos os slots baseados no intervalo
        const slots = [];
        const [startHour, startMin] = dayConfig.start_time.split(':').map(Number);
        const [endHour, endMin] = dayConfig.end_time.split(':').map(Number);
        const interval = dayConfig.slot_interval || 30;

        let currentMinutes = startHour * 60 + startMin;
        const endMinutes = endHour * 60 + endMin;

        // Se a data for hoje, não mostrar horários passados
        const now = new Date();
        const isToday =
          now.getFullYear() === targetDate.getFullYear() &&
          now.getMonth() === targetDate.getMonth() &&
          now.getDate() === targetDate.getDate();
        const currentNowMinutes = now.getHours() * 60 + now.getMinutes();

        while (currentMinutes + interval <= endMinutes) {
          const h = Math.floor(currentMinutes / 60);
          const m = currentMinutes % 60;
          const timeString = `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;

          const isPast = isToday && currentMinutes <= currentNowMinutes;

          if (!bookedTimes.has(timeString) && !isPast) {
            slots.push(timeString);
          }

          currentMinutes += interval;
        }

        res.json({ slots, message: slots.length === 0 ? 'Nenhum horário livre restante neste dia.' : null });
      }
    );
  });
});

app.post('/api/appointments', requireAuth, (req, res) => {
  const { service_id, appointment_date, appointment_time, notes } = req.body;
  const client_id = req.session.user.id;

  if (!service_id || !appointment_date || !appointment_time) {
    return res.status(400).json({ error: 'Preencha serviço, data e horário.' });
  }

  // Verificar se o horário já não foi pego no mesmo milésimo de segundo
  db.get(
    `SELECT id FROM appointments 
     WHERE appointment_date = ? AND appointment_time = ? AND status IN ('pendente', 'confirmado')`,
    [appointment_date, appointment_time],
    (err, conflict) => {
      if (err) return res.status(500).json({ error: 'Erro no servidor.' });
      if (conflict) {
        return res.status(400).json({ error: 'Desculpe, este horário acabou de ser reservado. Por favor, selecione outro.' });
      }

      db.run(
        `INSERT INTO appointments (client_id, service_id, appointment_date, appointment_time, status, notes) 
         VALUES (?, ?, ?, ?, 'pendente', ?)`,
        [client_id, service_id, appointment_date, appointment_time, notes || ''],
        function (insertErr) {
          if (insertErr) {
            return res.status(500).json({ error: 'Não foi possível salvar o agendamento.' });
          }
          res.json({ success: true, appointment_id: this.lastID });
        }
      );
    }
  );
});

      res.json({ success: true });

app.get('/api/admin/appointments', requireAdmin, (req, res) => {
  const { status, date } = req.query;

  let query = `
    SELECT 
      a.id, a.appointment_date, a.appointment_time, a.status, a.notes, a.created_at,
      s.name as service_name, s.duration_minutes, s.price,
      u.name as client_name, u.phone as client_phone, u.email as client_email
    FROM appointments a
    JOIN services s ON a.service_id = s.id
    JOIN users u ON a.client_id = u.id
    WHERE 1=1
  `;
  const params = [];

  if (status && status !== 'todos') {
    query += ` AND a.status = ?`;
    params.push(status);
  }

  if (date) {
    query += ` AND a.appointment_date = ?`;
    params.push(date);
  }

  query += ` ORDER BY a.appointment_date ASC, a.appointment_time ASC`;

  db.all(query, params, (err, rows) => {
    if (err) return res.status(500).json({ error: 'Erro ao carregar agendamentos do sistema.' });
    res.json(rows);
  });
});

// SISTEMA DE NOTIFICAÇÕES (TOAST)
function showToast(message, type = 'info') {
  const container = document.getElementById('toast-container');
  const toast = document.createElement('div');
  toast.className = `toast toast-${type}`;
  toast.textContent = message;
  container.appendChild(toast);

  setTimeout(() => {
    toast.style.opacity = '0';
    toast.style.transform = 'translateY(-10px)';
    toast.style.transition = 'all 0.3s ease';
    setTimeout(() => toast.remove(), 300);
  }, 3500);
}

// NAVEGAÇÃO ENTRE TELAS
function navigate(viewName) {
  const sections = document.querySelectorAll('.view-section');
  sections.forEach(sec => sec.classList.remove('active'));

  const target = document.getElementById(`view-${viewName}`);
  if (target) {
    target.classList.add('active');
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  if (viewName === 'my-bookings') {
    loadMyBookings();
  } else if (viewName === 'admin') {
    loadAdminStats();
    loadAdminAppointments();
    renderAdminServices();
    renderBusinessHoursForm();
  } else if (viewName === 'services-list') {
    renderPublicServices();
  }
}

// VERIFICAR SESSÃO NO SERVIDOR
async function checkAuth() {
  try {
    const res = await fetch('/api/auth/me');
    const data = await res.json();
    currentUser = data.authenticated ? data.user : null;
    updateNavbar();
  } catch (err) {
    console.error('Erro ao verificar autenticação:', err);
  }
}

// MODAL DE AUTENTICAÇÃO
function openAuthModal(mode = 'login') {
  toggleAuthMode(mode);
  document.getElementById('modal-auth').classList.add('open');
}

function closeAuthModal() {
  document.getElementById('modal-auth').classList.remove('open');
}

function toggleAuthMode(mode) {
  authMode = mode;
  const loginTab = document.getElementById('auth-tab-login');
  const regTab = document.getElementById('auth-tab-register');
  const loginForm = document.getElementById('form-login');
  const regForm = document.getElementById('form-register');

  if (mode === 'login') {
    loginTab.classList.add('active');
    regTab.classList.remove('active');
    loginForm.style.display = 'block';
    regForm.style.display = 'none';
  } else {
    loginTab.classList.remove('active');
    regTab.classList.add('active');
    loginForm.style.display = 'none';
    regForm.style.display = 'block';
  }
}

async function submitLogin(e) {
  e.preventDefault();
  const email = document.getElementById('login-email').value;
  const password = document.getElementById('login-password').value;

  try {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao realizar login', 'error');
      return;
    }

    currentUser = data.user;
    updateNavbar();
    closeAuthModal();
    showToast(`Bem-vindo(a) de volta, ${currentUser.name}!`, 'success');

    if (currentUser.role === 'admin') {
      navigate('admin');
    } else {
      navigate('my-bookings');
    }
  } catch (err) {
    showToast('Falha na comunicação com o servidor.', 'error');
  }
}

async function submitRegister(e) {
  e.preventDefault();
  const name = document.getElementById('register-name').value;
  const phone = document.getElementById('register-phone').value;
  const email = document.getElementById('register-email').value;
  const password = document.getElementById('register-password').value;

  try {
    const res = await fetch('/api/auth/register', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, phone, email, password })

    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao criar conta', 'error');
      return;
    }

    currentUser = data.user;
    updateNavbar();
    closeAuthModal();
    showToast('Conta criada com sucesso!', 'success');
    navigate('new-booking');
  } catch (err) {
    showToast('Falha na conexão ao criar conta.', 'error');
  }
}

function onBookingServiceChange() {
  const select = document.getElementById('booking-service');
  const badge = document.getElementById('service-details-badge');
  const selected = select.options[select.selectedIndex];

  if (select.value) {
    badge.style.display = 'block';
    badge.textContent = `Duração aproximada: ${selected.dataset.duration} minutos | Valor: R$ ${Number(selected.dataset.price).toFixed(2).replace('.', ',')}`;
  } else {
    badge.style.display = 'none';
  }

  fetchAvailableSlots();
}

async function fetchAvailableSlots() {
  const dateInput = document.getElementById('booking-date');
  const serviceSelect = document.getElementById('booking-service');
  const slotsContainer = document.getElementById('slots-container');
  const loading = document.getElementById('slots-loading');
  const submitBtn = document.getElementById('btn-submit-booking');
  const selectedSlotInput = document.getElementById('selected-slot-time');

  selectedSlotInput.value = '';
  submitBtn.disabled = true;

  if (!dateInput.value) {
    slotsContainer.innerHTML = '<p class="text-muted">Selecione uma data para ver os horários.</p>';
    return;
  }

  loading.style.display = 'block';
  slotsContainer.innerHTML = '';

  try {
    const res = await fetch(`/api/available-slots?date=${dateInput.value}&service_id=${serviceSelect.value}`);
    const data = await res.json();
    loading.style.display = 'none';

    if (data.slots && data.slots.length > 0) {
      slotsContainer.innerHTML = data.slots.map(time => `
        <button type="button" class="slot-btn" onclick="selectSlot(this, '${time}')">${time}</button>
      `).join('');
    } else {
      slotsContainer.innerHTML = `<p class="text-muted">${data.message || 'Nenhum horário disponível para este dia.'}</p>`;
    }
  } catch (err) {
    loading.style.display = 'none';
    slotsContainer.innerHTML = '<p class="text-muted">Erro ao carregar horários livres.</p>';
  }
}

function selectSlot(button, time) {
  document.querySelectorAll('.slot-btn').forEach(b => b.classList.remove('selected'));
  button.classList.add('selected');
  document.getElementById('selected-slot-time').value = time;
  document.getElementById('btn-submit-booking').disabled = false;
}

async function submitBooking(e) {
  e.preventDefault();
  if (!currentUser) {
    showToast('Você precisa entrar na sua conta para confirmar.', 'error');
    openAuthModal('login');
    return;
  }

  const service_id = document.getElementById('booking-service').value;
  const appointment_date = document.getElementById('booking-date').value;
  const appointment_time = document.getElementById('selected-slot-time').value;
  const notes = document.getElementById('booking-notes').value;

  if (!service_id || !appointment_date || !appointment_time) {
    showToast('Por favor, escolha o serviço, o dia e o horário.', 'error');
    return;
  }

  try {
    const res = await fetch('/api/appointments', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ service_id, appointment_date, appointment_time, notes })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao realizar agendamento.', 'error');
      fetchAvailableSlots();
      return;
    }

    showToast('Agendamento realizado com sucesso!', 'success');
    document.getElementById('form-booking').reset();
    document.getElementById('selected-slot-time').value = '';
    document.getElementById('service-details-badge').style.display = 'none';
    navigate('my-bookings');
  } catch (err) {
    showToast('Erro de conexão com o servidor.', 'error');
  }
}

async function loadMyBookings() {
  const container = document.getElementById('my-bookings-list');
  container.innerHTML = '<p class="text-muted">Buscando seus agendamentos...</p>';

  try {
    const res = await fetch('/api/appointments/my');
    const list = await res.json();

    if (!list || list.length === 0) {
      container.innerHTML = `
        <div class="card text-center p-4">
          <p class="text-muted">Você ainda não tem nenhum horário agendado.</p>
          <button class="btn btn-primary btn-sm mt-3" onclick="navigate('new-booking')">Agendar Agora</button>
        </div>
      `;
      return;
    }

    container.innerHTML = list.map(item => {
      const formattedDate = formatDateBR(item.appointment_date);
      const canCancel = item.status === 'pendente' || item.status === 'confirmado';

      return `
        <div class="appointment-item">
          <div class="appointment-info">
            <h4>${escapeHtml(item.service_name)}</h4>
            <p>🗓 <strong>${formattedDate}</strong> às <strong>${item.appointment_time}</strong> (${item.duration_minutes} min)</p>
            <p>Valor: R$ ${Number(item.price).toFixed(2).replace('.', ',')} ${item.notes ? `| <em>"${escapeHtml(item.notes)}"</em>` : ''}</p>
          </div>
          <div style="display:flex; align-items:center; gap:0.75rem;">
            <span class="appointment-status status-${item.status}">${item.status}</span>
            ${canCancel ? `<button class="btn btn-danger btn-sm" onclick="cancelAppointment(${item.id})">Cancelar</button>` : ''}
          </div>
        </div>
      `;
    }).join('');
  } catch (err) {
    container.innerHTML = '<p class="text-muted">Erro ao carregar agendamentos.</p>';
  }
}

async function cancelAppointment(id) {
  if (!confirm('Tem certeza que deseja cancelar este agendamento?')) return;

  try {
    const res = await fetch(`/api/appointments/${id}/cancel`, { method: 'PATCH' });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao cancelar.', 'error');
      return;
    }

    showToast('Agendamento cancelado com sucesso.', 'info');
    if (currentUser.role === 'admin') {
      loadAdminAppointments();
      loadAdminStats();
    } else {
      loadMyBookings();
    }
  } catch (err) {
    showToast('Erro ao solicitar cancelamento.', 'error');
  }
}

                    <button class="btn btn-danger btn-sm" onclick="cancelAppointment(${item.id})">Cancelar</button>
                  ` : ''}
                  ${(item.status === 'recusado' || item.status === 'cancelado' || item.status === 'concluido') ? `
                    <span style="font-size:0.85rem; color:#94a3b8;">Finalizado</span>
                  ` : ''}
                </div>
              </td>
            </tr>
          `).join('')}
        </tbody>
      </table>
    `;
  } catch (err) {
    container.innerHTML = '<p class="p-3 text-muted">Erro ao carregar agendamentos.</p>';
  }
}

async function setAppointmentStatus(id, newStatus) {
  try {
    const res = await fetch(`/api/admin/appointments/${id}/status`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ status: newStatus })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao mudar status.', 'error');
      return;
    }

    showToast(`Agendamento marcado como: ${newStatus}`, 'success');
    loadAdminAppointments();
    loadAdminStats();
  } catch (err) {
    showToast('Falha na comunicação com o servidor.', 'error');
  }
}

function clearAdminFilters() {

  document.getElementById('admin-filter-date').value = '';
  loadAdminAppointments();
}

function openServiceModal(serviceId = null) {
  const form = document.getElementById('form-save-service');
  form.reset();

  const idInput = document.getElementById('service-id');
  const title = document.getElementById('service-modal-title');
  const activeToggle = document.getElementById('service-status-toggle');

  if (serviceId) {
    const s = currentServices.find(item => item.id === serviceId);
    if (!s) return;
    idInput.value = s.id;
    title.textContent = 'Editar Serviço';
    document.getElementById('service-name').value = s.name;
    document.getElementById('service-description').value = s.description;
    document.getElementById('service-duration').value = s.duration_minutes;
    document.getElementById('service-price').value = s.price;
    document.getElementById('service-active').checked = s.active === 1;
    activeToggle.style.display = 'block';
  } else {
    idInput.value = '';
    title.textContent = 'Cadastrar Novo Serviço';
    document.getElementById('service-active').checked = true;
    activeToggle.style.display = 'none';
  }

  document.getElementById('modal-service').classList.add('open');
}

function closeServiceModal() {
  document.getElementById('modal-service').classList.remove('open');
}

async function saveService(e) {
  e.preventDefault();
  const id = document.getElementById('service-id').value;
  const name = document.getElementById('service-name').value;
  const description = document.getElementById('service-description').value;
  const duration_minutes = document.getElementById('service-duration').value;
  const price = document.getElementById('service-price').value;
  const active = document.getElementById('service-active').checked ? 1 : 0;

  const url = id ? `/api/services/${id}` : '/api/services';
  const method = id ? 'PUT' : 'POST';

  try {
    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, description, duration_minutes, price, active })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao salvar serviço.', 'error');
      return;
    }

    showToast('Serviço gravado com sucesso!', 'success');
    closeServiceModal();
    await loadServices();
    renderAdminServices();
  } catch (err) {
    showToast('Erro ao conectar ao servidor.', 'error');
  }
}

async function deleteService(id) {
  if (!confirm('Deseja realmente excluir este serviço do catálogo?')) return;

  try {
    const res = await fetch(`/api/services/${id}`, { method: 'DELETE' });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao excluir.', 'error');
      return;
    }

    showToast('Serviço excluído.', 'info');
    await loadServices();
    renderAdminServices();
  } catch (err) {
    showToast('Erro ao excluir serviço.', 'error');
  }
}

async function saveBusinessHours(e) {
  e.preventDefault();
  const rows = document.querySelectorAll('.hour-row');
  const schedule = [];

  rows.forEach(row => {
    schedule.push({
      day_of_week: parseInt(row.dataset.day, 10),
      is_open: row.querySelector('.hour-is-open').checked ? 1 : 0,
      start_time: row.querySelector('.hour-start').value,
      end_time: row.querySelector('.hour-end').value,
      slot_interval: parseInt(row.querySelector('.hour-interval').value, 10) || 30
    });
  });

  try {
    const res = await fetch('/api/business-hours', {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ schedule })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao salvar expediente.', 'error');
      return;
    }

    showToast('Horários de funcionamento atualizados com sucesso!', 'success');
    await loadBusinessHours();
  } catch (err) {
    showToast('Falha ao gravar horários.', 'error');
  }
}

// UTILITÁRIOS
function formatDateBR(dateStr) {
  if (!dateStr) return '';
  const parts = dateStr.split('-');
  if (parts.length === 3) {
    return `${parts[2]}/${parts[1]}/${parts[0]}`;
  }
  return dateStr;
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

# Sistema de Agendamento de Serviços

Sistema completo, moderno e pronto para funcionar de agendamento de serviços online.
Ele roda direto no seu computador ou servidor web, com banco de dados próprio incluso e sem depender de contas ou integrações de terceiros.

---

## 🚀 Como instalar e colocar para funcionar (Passo a Passo)

### 1. Pré-requisito
Você só precisa ter o **Node.js** instalado no seu computador.
* Se ainda não tiver, baixe e instale a versão LTS oficial em: [https://nodejs.org](https://nodejs.org) (basta avançar e concluir a instalação normalmente).

---

### 2. Baixar as dependências e iniciar o site

Abra a pasta do projeto no Terminal (ou no Prompt de Comando / PowerShell do Windows) e execute apenas estes 2 comandos:

```bash
npm install
```

Quando a instalação terminar, inicie o sistema com:

```bash
npm start
```

Pronto! O site já está funcionando.

---

### 3. Acessar o site

Abra qualquer navegador (Chrome, Edge, Safari, Firefox, celular ou computador) e digite o endereço:

👉 **http://localhost:3000**

---

## 🔑 Contas de Acesso

O sistema já cria automaticamente no primeiro início a conta do responsável:

* **E-mail do Responsável:** `admin@negocio.com`
* **Senha:** `admin123`

> Os seus clientes podem clicar no botão **"Entrar / Cadastrar"** no topo da página e criar a própria conta em segundos informando apenas nome, telefone/WhatsApp, e-mail e senha.

---

## 🌟 O que o sistema faz

### Para os Clientes:
1. **Ver Serviços:** Acessar o catálogo com nome, descrição, duração e preço de cada serviço.
2. **Consultar Horários Livres:** Escolher um serviço e uma data. O sistema calcula automaticamente e exibe apenas os horários em que o negócio está aberto e que ainda não foram reservados por ninguém.
3. **Fazer o Agendamento:** Clicar no horário desejado e confirmar a reserva.
4. **Meus Horários:** Ver todos os agendamentos já feitos, acompanhar o status (Pendente, Confirmado, Recusado, Cancelado) e cancelar caso tenha um imprevisto.

### Para o Responsável pelo Negócio (Painel do Administrador):
1. **Painel Geral com Indicadores:** Quantidade de agendamentos pendentes, confirmados, agendados para hoje e total acumulado.
2. **Gerenciamento de Agendamentos:**
   * Ver todos os agendamentos de todos os clientes com nome, telefone e observações.
   * Botões para **Aprovar** ou **Recusar** agendamentos pendentes com 1 clique.
   * Filtros por status e por dia.
3. **Gerenciamento de Serviços:**
   * Cadastrar novos serviços.
   * Alterar nome, preço, duração em minutos e descrição.
   * Desativar ou excluir serviços.
4. **Horários de Funcionamento:**
   * Ativar ou desativar os dias da semana em que você atende.
   * Definir o horário de início e término de cada dia (ex: 08:00 às 18:00).
   * Definir a duração dos intervalos de horário (ex: 30 em 30 minutos).

---

## 💾 Onde as informações ficam salvas?

Tudo fica salvo em um banco de dados próprio (`agendamentos.db`) que é gerado na própria pasta do projeto. 
* Nada é perdido quando você desliga o computador ou reinicia o servidor.
* Se um cliente marcar um horário pelo celular, o responsável vê no computador imediatamente.
* Não é preciso pagar bancos de dados em nuvem nem criar contas externas.

---

## 🌐 Como colocar na internet para clientes reais usarem?

Basta subir esta mesma pasta para qualquer serviço de hospedagem simples de Node.js gratuito ou de baixo custo, como:
* **Render.com** (Basta conectar e clicar em Deploy)
* **Railway.app**
* Qualquer servidor VPS (Ubuntu/Debian com `npm start`)

O sistema lê a porta padrão da internet automaticamente (`process.env.PORT`) sem precisar mudar nenhuma linha de código.
