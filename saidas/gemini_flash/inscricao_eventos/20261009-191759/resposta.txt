      path: path.join(__dirname, 'data', 'sessions'),

function ensureDirSync(dir) {
  if (!fs.existsSync(dir)) {
    fs.mkdirSync(dir, { recursive: true });
  }
}

function readJSON(file, fallback = []) {
  try {
    if (!fs.existsSync(file)) {
      return fallback;
    }
    const content = fs.readFileSync(file, 'utf-8');
    return JSON.parse(content || '[]');
  } catch (err) {
    console.error(`Erro ao ler ${file}:`, err);
    return fallback;
  }
}

function writeJSON(file, data) {
  try {
    ensureDirSync(path.dirname(file));
    const tempFile = `${file}.tmp`;
    fs.writeFileSync(tempFile, JSON.stringify(data, null, 2), 'utf-8');
    fs.renameSync(tempFile, file);
  } catch (err) {
    console.error(`Erro ao escrever em ${file}:`, err);
  }
}

const express = require('express');
const router = express.Router();
const bcrypt = require('bcryptjs');
const { v4: uuidv4 } = require('uuid');
const db = require('../db');

// Obter usuário logado atual
router.get('/me', (req, res) => {
  if (!req.session || !req.session.userId) {
    return res.status(401).json({ error: 'Não autenticado' });
  }

  const users = db.getUsers();
  const user = users.find(u => u.id === req.session.userId);

  if (!user) {
    req.session.destroy();
    return res.status(401).json({ error: 'Usuário não encontrado' });
  }

  const { senhaHash, ...userSemSenha } = user;
  res.json({ user: userSemSenha });
});

// Cadastro de novos usuários
router.post('/register', (req, res) => {
  const { nome, email, senha, tipo } = req.body;

  if (!nome || !email || !senha) {
    return res.status(400).json({ error: 'Preencha todos os campos obrigatórios.' });
  }

  const emailFormatado = email.trim().toLowerCase();
  const users = db.getUsers();

  const usuarioExistente = users.find(u => u.email.toLowerCase() === emailFormatado);
  if (usuarioExistente) {
    return res.status(400).json({ error: 'Já existe uma conta com este e-mail.' });
  }

  const tipoFinal = tipo === 'organizador' ? 'organizador' : 'participante';
  const salt = bcrypt.genSaltSync(10);
  const senhaHash = bcrypt.hashSync(senha, salt);

  const novoUsuario = {
    id: 'user_' + uuidv4(),
    nome: nome.trim(),
    email: emailFormatado,
    senhaHash,
    tipo: tipoFinal,
    criadoEm: new Date().toISOString()
  };

  users.push(novoUsuario);
  db.saveUsers(users);

  // Iniciar sessão automaticamente
  req.session.userId = novoUsuario.id;

  const { senhaHash: _, ...userSemSenha } = novoUsuario;
  res.status(201).json({ mensagem: 'Conta criada com sucesso!', user: userSemSenha });
});

// Login
router.post('/login', (req, res) => {
  const { email, senha } = req.body;

  if (!email || !senha) {
    return res.status(400).json({ error: 'Informe e-mail e senha.' });
  }

  const emailFormatado = email.trim().toLowerCase();
  const users = db.getUsers();
  const user = users.find(u => u.email.toLowerCase() === emailFormatado);

  if (!user) {
    return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
  }

  const senhaValida = bcrypt.compareSync(senha, user.senhaHash);
  if (!senhaValida) {
    return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
  }

  req.session.userId = user.id;

  const { senhaHash, ...userSemSenha } = user;
  res.json({ mensagem: 'Login realizado com sucesso!', user: userSemSenha });
});

// Logout
router.post('/logout', (req, res) => {
  req.session.destroy(err => {
    if (err) {
      return res.status(500).json({ error: 'Erro ao sair' });
    }
    res.clearCookie('connect.sid');
    res.json({ mensagem: 'Desconectado com sucesso' });
  });
});

const express = require('express');
const router = express.Router();
const { v4: uuidv4 } = require('uuid');
const db = require('../db');

// Middleware para verificar se está autenticado
function autenticado(req, res, next) {
  if (!req.session || !req.session.userId) {
    return res.status(401).json({ error: 'Faça login para continuar.' });
  }
  const users = db.getUsers();
  const user = users.find(u => u.id === req.session.userId);
  if (!user) {
    return res.status(401).json({ error: 'Sessão inválida.' });
  }
  req.user = user;
  next();
}

// Middleware para verificar se é organizador
function apenasOrganizador(req, res, next) {
  if (req.user.tipo !== 'organizador') {
    return res.status(403).json({ error: 'Acesso permitido apenas para organizadores.' });
  }
  next();
}

// 1. Listar todos os eventos disponíveis com status da inscrição do usuário atual
router.get('/', (req, res) => {
  const events = db.getEvents();
  const registrations = db.getRegistrations();
  const userId = req.session ? req.session.userId : null;

  // Monta lista de eventos com contagem de vagas ocupadas
  const eventosCompletos = events.map(evt => {
    const inscricoesEvento = registrations.filter(r => r.eventoId === evt.id);
    const estaInscrito = userId ? inscricoesEvento.some(r => r.usuarioId === userId) : false;
    const minhaInscricao = userId ? inscricoesEvento.find(r => r.usuarioId === userId) : null;

    return {
      ...evt,
      totalInscritos: inscricoesEvento.length,
      vagasRestantes: Math.max(0, (parseInt(evt.vagas, 10) || 0) - inscricoesEvento.length),
      estaInscrito,
      presencaConfirmada: minhaInscricao ? minhaInscricao.presencaConfirmada : false
    };
  });

  // Ordena por data
  eventosCompletos.sort((a, b) => new Date(a.data) - new Date(b.data));

  res.json({ eventos: eventosCompletos });
});

// 2. Criar novo evento (Apenas organizador)
router.post('/', autenticado, apenasOrganizador, (req, res) => {
  const { titulo, descricao, categoria, data, local, vagas } = req.body;

  if (!titulo || !data || !local || !vagas) {
    return res.status(400).json({ error: 'Título, data, local e total de vagas são obrigatórios.' });
  }

  const events = db.getEvents();

  const novoEvento = {
    id: 'evt_' + uuidv4(),
    titulo: titulo.trim(),
    descricao: (descricao || '').trim(),
    categoria: (categoria || 'Geral').trim(),
    data: data,
    local: local.trim(),
    vagas: parseInt(vagas, 10) || 50,
    status: 'ativo',
    criadoPor: req.user.id,
    criadoEm: new Date().toISOString()
  };

  events.push(novoEvento);
  db.saveEvents(events);

  res.status(201).json({ mensagem: 'Evento criado com sucesso!', evento: novoEvento });
});

// 3. Editar evento existente (Apenas organizador)
router.put('/:id', autenticado, apenasOrganizador, (req, res) => {
  const { id } = req.params;
  const { titulo, descricao, categoria, data, local, vagas, status } = req.body;

  const events = db.getEvents();
  const index = events.findIndex(e => e.id === id);

  if (index === -1) {
    return res.status(404).json({ error: 'Evento não encontrado.' });
  }

  const eventoAtual = events[index];

  events[index] = {
    ...eventoAtual,
    titulo: titulo ? titulo.trim() : eventoAtual.titulo,
    descricao: descricao !== undefined ? descricao.trim() : eventoAtual.descricao,
    categoria: categoria ? categoria.trim() : eventoAtual.categoria,
    data: data || eventoAtual.data,
    local: local ? local.trim() : eventoAtual.local,
    vagas: vagas ? parseInt(vagas, 10) : eventoAtual.vagas,
    status: status || eventoAtual.status,
    atualizadoEm: new Date().toISOString()
  };

  db.saveEvents(events);
  res.json({ mensagem: 'Evento atualizado com sucesso!', evento: events[index] });
});

// 4. Cancelar / Alternar status do evento (Apenas organizador)
router.patch('/:id/toggle-status', autenticado, apenasOrganizador, (req, res) => {
  const { id } = req.params;
  const events = db.getEvents();
  const evento = events.find(e => e.id === id);

  if (!evento) {
    return res.status(404).json({ error: 'Evento não encontrado.' });
  }

  evento.status = evento.status === 'cancelado' ? 'ativo' : 'cancelado';
  evento.atualizadoEm = new Date().toISOString();

  db.saveEvents(events);
  res.json({ mensagem: `Evento ${evento.status === 'cancelado' ? 'cancelado' : 'reativado'} com sucesso!`, status: evento.status });
});

// 5. Inscrição do participante no evento
router.post('/:id/inscrever', autenticado, (req, res) => {
  const { id } = req.params;
  const events = db.getEvents();
  const evento = events.find(e => e.id === id);

  if (!evento) {
    return res.status(404).json({ error: 'Evento não encontrado.' });
  }

  if (evento.status === 'cancelado') {
    return res.status(400).json({ error: 'Este evento está cancelado e não aceita inscrições.' });
  }

  const registrations = db.getRegistrations();
  const jaInscrito = registrations.some(r => r.eventoId === id && r.usuarioId === req.user.id);

  if (jaInscrito) {
    return res.status(400).json({ error: 'Você já está inscrito neste evento.' });
  }

  // Verifica lotação
  const totalInscritos = registrations.filter(r => r.eventoId === id).length;
  if (totalInscritos >= parseInt(evento.vagas, 10)) {
    return res.status(400).json({ error: 'Desculpe, as vagas para este evento esgotaram!' });
  }

  const novaInscricao = {
    id: 'reg_' + uuidv4(),
    eventoId: id,
    usuarioId: req.user.id,
    presencaConfirmada: false,
    inscritoEm: new Date().toISOString()
  };

  registrations.push(novaInscricao);
  db.saveRegistrations(registrations);

  res.status(201).json({ mensagem: 'Inscrição confirmada com sucesso!' });
});

// 6. Cancelar inscrição do participante
router.delete('/:id/cancelar-inscricao', autenticado, (req, res) => {
  const { id } = req.params;
  let registrations = db.getRegistrations();

  const inscricao = registrations.find(r => r.eventoId === id && r.usuarioId === req.user.id);
  if (!inscricao) {
    return res.status(404).json({ error: 'Você não possui inscrição ativa neste evento.' });
  }

  registrations = registrations.filter(r => !(r.eventoId === id && r.usuarioId === req.user.id));
  db.saveRegistrations(registrations);

  res.json({ mensagem: 'Sua inscrição foi cancelada com sucesso.' });
});

// 7. Obter "Minhas Inscrições" (para o participante)
router.get('/minhas-inscricoes', autenticado, (req, res) => {
  const events = db.getEvents();
  const registrations = db.getRegistrations();

  const minhasInscricoes = registrations
    .filter(r => r.usuarioId === req.user.id)
    .map(r => {
      const evento = events.find(e => e.id === r.eventoId);
      return {
        inscricaoId: r.id,
        presencaConfirmada: r.presencaConfirmada,
        inscritoEm: r.inscritoEm,
        evento: evento || { titulo: 'Evento Removido', status: 'cancelado' }
      };
    });

  res.json({ inscricoes: minhasInscricoes });
});

// 8. Visualizar todas as inscrições de todos os eventos (Visão geral do Organizador)
router.get('/todas-inscricoes', autenticado, apenasOrganizador, (req, res) => {
  const events = db.getEvents();
  const users = db.getUsers();
  const registrations = db.getRegistrations();

  const listaCompleta = registrations.map(r => {
    const evento = events.find(e => e.id === r.eventoId);
    const participante = users.find(u => u.id === r.usuarioId);

    return {
      inscricaoId: r.id,
      presencaConfirmada: r.presencaConfirmada,
      inscritoEm: r.inscritoEm,
      eventoId: r.eventoId,
      eventoTitulo: evento ? evento.titulo : 'Evento Desconhecido',
      eventoData: evento ? evento.data : '',
      participanteNome: participante ? participante.nome : 'Usuário Desconhecido',
      participanteEmail: participante ? participante.email : '-'
    };
  });

  res.json({ inscricoes: listaCompleta });
});

// 9. Confirmar presença / Check-in de um participante (Apenas organizador)
router.patch('/inscricoes/:inscricaoId/confirmar-presenca', autenticado, apenasOrganizador, (req, res) => {
  const { inscricaoId } = req.params;
  const registrations = db.getRegistrations();
  const inscricao = registrations.find(r => r.id === inscricaoId);

  if (!inscricao) {
    return res.status(404).json({ error: 'Inscrição não encontrada.' });
  }

  // Alterna confirmação de presença (check-in)
  inscricao.presencaConfirmada = !inscricao.presencaConfirmada;
  inscricao.presencaAtualizadaEm = new Date().toISOString();

  db.saveRegistrations(registrations);

  res.json({
    mensagem: inscricao.presencaConfirmada ? 'Presença confirmada com sucesso!' : 'Confirmação de presença desmarcada.',
    presencaConfirmada: inscricao.presencaConfirmada
  });
});

// ESTADO GLOBAL DA APLICAÇÃO
let usuarioAtual = null;
let eventosGlobais = [];
let todasInscricoesGlobais = [];
let vistaAtual = 'eventos';

// INICIALIZAÇÃO
document.addEventListener('DOMContentLoaded', async () => {
  await checarSessao();
  await carregarEventos();
});

// 1. CHECAGEM DE AUTENTICAÇÃO
async function checarSessao() {
  try {
    const res = await fetch('/api/auth/me');
    if (res.ok) {
      const data = await res.json();
      usuarioAtual = data.user;
    } else {
      usuarioAtual = null;
    }
  } catch (err) {
    usuarioAtual = null;
  }
  atualizarInterfaceUsuario();
}

// 2. ATUALIZAR INTERFACE CONFORME O PAPEL DO USUÁRIO
function atualizarInterfaceUsuario() {
  const navDeslogado = document.getElementById('nav-deslogado');
  const navLogado = document.getElementById('nav-logado');
  const navParticipante = document.getElementById('nav-participante');
  const navOrganizador = document.getElementById('nav-organizador');
  const avisoUsuario = document.getElementById('aviso-usuario');

  if (usuarioAtual) {
    navDeslogado.classList.add('hidden');
    navLogado.classList.remove('hidden');

    document.getElementById('user-display-name').textContent = usuarioAtual.nome;
    document.getElementById('user-display-role').textContent =
      usuarioAtual.tipo === 'organizador' ? 'Organizador' : 'Participante';

    if (usuarioAtual.tipo === 'organizador') {
      navOrganizador.classList.remove('hidden');
      navParticipante.classList.add('hidden');
      avisoUsuario.innerHTML = `
        <div class="flex items-center gap-2">
          <i class="fa-solid fa-crown text-amber-300"></i>
          <span>Painel de <strong>Organizador</strong>: você pode criar, editar eventos e confirmar presença dos inscritos.</span>
        </div>`;
    } else {
      navOrganizador.classList.add('hidden');
      navParticipante.classList.remove('hidden');
      avisoUsuario.innerHTML = `
        <div class="flex items-center gap-2">
          <i class="fa-solid fa-circle-user text-indigo-300"></i>
          <span>Olá, <strong>${usuarioAtual.nome}</strong>! Escolha um evento abaixo para se inscrever.</span>
        </div>`;
    }
  } else {
    navDeslogado.classList.remove('hidden');
    navLogado.classList.add('hidden');
    navParticipante.classList.add('hidden');
    navOrganizador.classList.add('hidden');
    avisoUsuario.innerHTML = `
      <div class="flex items-center gap-2">
        <i class="fa-solid fa-info-circle text-indigo-200"></i>
        <span>Faça login ou crie uma conta grátis para se inscrever nos eventos.</span>
      </div>`;
  }
}

// 3. NAVEGAÇÃO ENTRE TELAS
function navegarPara(view) {
  vistaAtual = view;
  document.querySelectorAll('.view-section').forEach(el => el.classList.add('hidden'));

  const alvo = document.getElementById(`view-${view}`);
  if (alvo) alvo.classList.remove('hidden');

  if (view === 'eventos') {
    carregarEventos();
  } else if (view === 'minhas-inscricoes') {
    if (!usuarioAtual) {
      abrirModalAuth('login');
      return;
    }
    carregarMinhasInscricoes();
  } else if (view === 'gerenciar-inscritos') {
    if (!usuarioAtual || usuarioAtual.tipo !== 'organizador') {
      mostrarToast('Apenas organizadores podem acessar esta tela.', 'erro');
      navegarPara('eventos');
      return;
    }
    carregarTodasInscricoes();
  }
}

// 4. CARREGAR E RENDERIZAR EVENTOS
async function carregarEventos() {
  try {
    const res = await fetch('/api/events');
    const data = await res.json();
    eventosGlobais = data.eventos || [];
    renderizarCardsEventos(eventosGlobais);
  } catch (err) {
    mostrarToast('Erro ao carregar lista de eventos.', 'erro');
  }
}

function filtrarEventos() {
  const busca = (document.getElementById('filtro-busca').value || '').toLowerCase();
  const categoria = document.getElementById('filtro-categoria').value;

  const filtrados = eventosGlobais.filter(evt => {
    const matchBusca = evt.titulo.toLowerCase().includes(busca) || evt.local.toLowerCase().includes(busca);
    const matchCategoria = categoria === 'todos' || evt.categoria === categoria;
    return matchBusca && matchCategoria;
  });

  renderizarCardsEventos(filtrados);
}

function formatarData(dataISO) {
  if (!dataISO) return '';
  const d = new Date(dataISO);
  return d.toLocaleDateString('pt-BR', {
    day: '2-digit',
    month: 'short',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit'
  });
}

    .map(evt => {
      const cancelado = evt.status === 'cancelado';
      const esgotado = evt.vagasRestantes <= 0;
      const isOrg = usuarioAtual && usuarioAtual.tipo === 'organizador';

      let botaoAcao = '';

      if (isOrg) {
        botaoAcao = `
          <div class="flex gap-2 w-full mt-4">
            <button onclick="editarEvento('${evt.id}')" class="flex-1 py-2 bg-indigo-50 hover:bg-indigo-100 text-indigo-700 font-bold text-xs rounded-lg transition flex items-center justify-center gap-1">
              <i class="fa-solid fa-pen-to-square"></i> Editar
            </button>
            <button onclick="alternarStatusEvento('${evt.id}')" class="flex-1 py-2 ${cancelado ? 'bg-emerald-50 hover:bg-emerald-100 text-emerald-700' : 'bg-rose-50 hover:bg-rose-100 text-rose-700'} font-bold text-xs rounded-lg transition flex items-center justify-center gap-1">
              <i class="fa-solid ${cancelado ? 'fa-check' : 'fa-ban'}"></i> ${cancelado ? 'Reativar' : 'Cancelar Evento'}
            </button>
          </div>
        `;
      } else {
        if (cancelado) {
          botaoAcao = `
            <div class="w-full mt-4 py-2 bg-slate-100 text-slate-400 font-bold text-xs text-center rounded-lg cursor-not-allowed">
              Evento Cancelado
            </div>
          `;
        } else if (evt.estaInscrito) {
          botaoAcao = `
            <div class="w-full mt-4 space-y-2">
              <div class="py-2 bg-emerald-50 text-emerald-700 font-bold text-xs text-center rounded-lg border border-emerald-200 flex items-center justify-center gap-1">
                <i class="fa-solid fa-circle-check"></i> Você está inscrito! ${evt.presencaConfirmada ? '(Presença Confirmada)' : ''}
              </div>
              <button onclick="cancelarInscricao('${evt.id}')" class="w-full py-1.5 text-xs text-rose-600 hover:text-rose-800 hover:underline text-center">
                Cancelar minha inscrição
              </button>
            </div>
          `;
        } else if (esgotado) {
          botaoAcao = `
            <div class="w-full mt-4 py-2 bg-amber-50 text-amber-700 font-bold text-xs text-center rounded-lg border border-amber-200">
              Vagas Esgotadas
            </div>
          `;
        } else {
          botaoAcao = `
            <button onclick="inscreverNoEvento('${evt.id}')" class="w-full mt-4 py-2.5 bg-indigo-600 hover:bg-indigo-700 text-white font-bold text-xs rounded-lg shadow-sm transition flex items-center justify-center gap-1.5">
              <i class="fa-solid fa-ticket"></i> Inscrever-se Agora
            </button>
          `;
        }
      }

      return `
        <div class="card-evento bg-white rounded-2xl border border-slate-200 overflow-hidden flex flex-col justify-between p-5 relative ${cancelado ? 'opacity-70 bg-slate-50' : ''}">
          <div>
            <div class="flex justify-between items-start gap-2 mb-2">
              <span class="text-[10px] font-extrabold uppercase px-2.5 py-1 rounded-full bg-indigo-50 text-indigo-700">
                ${evt.categoria}
              </span>
              ${
                cancelado
                  ? `<span class="badge-status bg-rose-100 text-rose-700"><i class="fa-solid fa-ban"></i> Cancelado</span>`
                  : esgotado
                  ? `<span class="badge-status bg-amber-100 text-amber-800"><i class="fa-solid fa-lock"></i> Esgotado</span>`
                  : `<span class="badge-status bg-emerald-100 text-emerald-800"><i class="fa-solid fa-check"></i> Aberto</span>`
              }
            </div>

            <h3 class="text-base font-bold text-slate-800 mt-2 mb-1 line-clamp-2">${evt.titulo}</h3>
            <p class="text-xs text-slate-500 mb-4 line-clamp-3">${evt.descricao || 'Sem descrição detalhada.'}</p>

            <div class="space-y-1.5 text-xs text-slate-600 border-t border-slate-100 pt-3">
              <div class="flex items-center gap-2">
                <i class="fa-regular fa-clock text-indigo-500 w-4"></i>
                <span>${formatarData(evt.data)}</span>
              </div>
              <div class="flex items-center gap-2">
                <i class="fa-solid fa-location-dot text-rose-500 w-4"></i>
                <span class="truncate">${evt.local}</span>
              </div>
              <div class="flex items-center gap-2">
                <i class="fa-solid fa-users text-slate-400 w-4"></i>
                <span><strong>${evt.totalInscritos}</strong> de <strong>${evt.vagas}</strong> vagas (${evt.vagasRestantes} restantes)</span>
              </div>
            </div>
          </div>

          <div>
            ${botaoAcao}
          </div>
        </div>
      `;
    })
    .join('');
}

// 5. INSCRIÇÃO E CANCELAMENTO DE INSCRIÇÃO
async function inscreverNoEvento(eventoId) {
  if (!usuarioAtual) {
    mostrarToast('Você precisa entrar na sua conta para se inscrever.', 'aviso');
    abrirModalAuth('login');
    return;
  }

  try {
    const res = await fetch(`/api/events/${eventoId}/inscrever`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' }
    });
    const data = await res.json();

    if (res.ok) {
      mostrarToast(data.mensagem, 'sucesso');
      await carregarEventos();
    } else {
      mostrarToast(data.error || 'Erro ao realizar inscrição', 'erro');
    }
  } catch (err) {
    mostrarToast('Falha na comunicação com o servidor.', 'erro');
  }
}

async function cancelarInscricao(eventoId) {
  if (!confirm('Deseja realmente cancelar sua inscrição neste evento?')) return;

  try {
    const res = await fetch(`/api/events/${eventoId}/cancelar-inscricao`, {
      method: 'DELETE'
    });
    const data = await res.json();

    if (res.ok) {
      mostrarToast(data.mensagem, 'sucesso');
      await carregarEventos();
      if (vistaAtual === 'minhas-inscricoes') {
        carregarMinhasInscricoes();
      }
    } else {
      mostrarToast(data.error || 'Erro ao cancelar inscrição', 'erro');
    }
  } catch (err) {
    mostrarToast('Falha na comunicação com o servidor.', 'erro');
  }
}

// 6. MINHAS INSCRIÇÕES (PARTICIPANTE)

                  ${evt.categoria || 'Evento'}
                </span>

                  item.presencaConfirmada
                    ? `<span class="badge-status bg-emerald-100 text-emerald-800"><i class="fa-solid fa-circle-check"></i> Presença Confirmada</span>`

              <h3 class="text-base font-bold text-slate-800 mb-1">${evt.titulo}</h3>

              </button>
            </div>
          </div>

// 7. GERENCIAMENTO DE INSCRITOS & CHECK-IN (ORGANIZADOR)
async function carregarTodasInscricoes() {
  try {
    const res = await fetch('/api/events/todas-inscricoes');
    const data = await res.json();
    todasInscricoesGlobais = data.inscricoes || [];

    // Preenche seletor de eventos para filtro
    const select = document.getElementById('filtro-evento-presenca');
    const eventosMap = {};
    todasInscricoesGlobais.forEach(i => {
      eventosMap[i.eventoId] = i.eventoTitulo;
    });

    select.innerHTML = `<option value="todos">Todos os Eventos (${todasInscricoesGlobais.length} inscrições)</option>` +
      Object.keys(eventosMap).map(id => `<option value="${id}">${eventosMap[id]}</option>`).join('');

    renderizarTabelaInscritos();
  } catch (err) {
    mostrarToast('Erro ao carregar lista de presença.', 'erro');
  }
}

function renderizarTabelaInscritos() {
  const filtroId = document.getElementById('filtro-evento-presenca').value;
  const tbody = document.getElementById('tabela-inscritos-corpo');
  const vazia = document.getElementById('tabela-inscritos-vazia');

  const filtrados = todasInscricoesGlobais.filter(item => {

  });

  if (filtrados.length === 0) {
    tbody.innerHTML = '';
    vazia.classList.remove('hidden');

  }

  vazia.classList.add('hidden');
  tbody.innerHTML = filtrados
    .map(r => {

          <td class="px-6 py-4">

          <td class="px-6 py-4">
            <div class="font-semibold text-slate-700">${r.eventoTitulo}</div>
            <div class="text-xs text-slate-400">${formatarData(r.eventoData)}</div>
          </td>
          <td class="px-6 py-4 text-xs text-slate-500">

            </button>
          </td>

      `;

async function confirmarPresenca(inscricaoId) {
  try {
    const res = await fetch(`/api/events/inscricoes/${inscricaoId}/confirmar-presenca`, {

    const data = await res.json();

    if (res.ok) {
      mostrarToast(data.mensagem, 'sucesso');
      await carregarTodasInscricoes();
    } else {
      mostrarToast(data.error || 'Erro ao atualizar presença', 'erro');
    }
  } catch (err) {
    mostrarToast('Falha ao comunicar com o servidor.', 'erro');
  }
}

async function salvarEvento(e) {
  e.preventDefault();
  const id = document.getElementById('evento-id').value;
  const erroDiv = document.getElementById('evento-erro');
  erroDiv.classList.add('hidden');

  const payload = {
    titulo: document.getElementById('evento-titulo').value,
    descricao: document.getElementById('evento-descricao').value,
    categoria: document.getElementById('evento-categoria').value,
    data: document.getElementById('evento-data').value,
    local: document.getElementById('evento-local').value,
    vagas: document.getElementById('evento-vagas').value
  };

  const url = id ? `/api/events/${id}` : '/api/events';
  const method = id ? 'PUT' : 'POST';

  try {
    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });
    const data = await res.json();

    if (res.ok) {
      fecharModalEvento();
      mostrarToast(data.mensagem, 'sucesso');
      await carregarEventos();
    } else {
      erroDiv.textContent = data.error || 'Erro ao salvar evento.';
      erroDiv.classList.remove('hidden');
    }
  } catch (err) {
    erroDiv.textContent = 'Erro ao conectar ao servidor.';
    erroDiv.classList.remove('hidden');
  }
}

  }
}

async function submeterLogin(e) {
  e.preventDefault();
  const erroDiv = document.getElementById('login-erro');
  erroDiv.classList.add('hidden');

  const email = document.getElementById('login-email').value;
  const senha = document.getElementById('login-senha').value;

  try {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, senha })
    });
    const data = await res.json();

    if (res.ok) {
      usuarioAtual = data.user;
      fecharModalAuth();
      atualizarInterfaceUsuario();
      mostrarToast(`Bem-vindo de volta, ${usuarioAtual.nome}!`, 'sucesso');
      await carregarEventos();
    } else {
      erroDiv.textContent = data.error || 'Credenciais inválidas.';
      erroDiv.classList.remove('hidden');
    }
  } catch (err) {
    erroDiv.textContent = 'Erro ao tentar conectar ao servidor.';
    erroDiv.classList.remove('hidden');
  }
}

function mostrarToast(mensagem, tipo = 'sucesso') {
  const toast = document.getElementById('toast');
  const msg = document.getElementById('toast-msg');
  const icon = document.getElementById('toast-icon');

  msg.textContent = mensagem;

  if (tipo === 'sucesso') {
    icon.className = 'fa-solid fa-circle-check text-emerald-400 text-lg';
  } else if (tipo === 'erro') {
    icon.className = 'fa-solid fa-circle-exclamation text-rose-400 text-lg';
  } else {
    icon.className = 'fa-solid fa-circle-info text-amber-400 text-lg';
  }

  toast.classList.remove('hidden');

  setTimeout(() => {
    toast.classList.add('hidden');
  }, 4000);
}

# Portal de Inscrição em Eventos

Portal completo e pronto para uso de gestão e inscrição de eventos. Não depende de serviços externos, bancos de dados em nuvem ou cadastros em outros sites: tudo roda de forma 100% autônoma e os dados ficam salvos permanentemente no próprio servidor dentro da pasta `data/`.

---

## 🚀 Como Colocar para Rodar (Passo a Passo)

Você só precisa ter o **Node.js** instalado no seu computador. Se não tiver, baixe a versão recomendada (LTS) no site oficial: [https://nodejs.org](https://nodejs.org) (basta clicar em avançar e instalar).

### 1. Abra o Terminal ou Prompt de Comando
Navegue até a pasta onde estão os arquivos deste projeto:

```bash
cd caminho/para/a/pasta/do/projeto
```

### 2. Instale as dependências
Execute o comando:

```bash
npm install
```

### 3. Inicie o portal
Execute o comando:

```bash
npm start
```

Pronto! Você verá a mensagem:
`Portal de Eventos rodando em: http://localhost:3000`

Abra seu navegador (Chrome, Edge, Firefox, Safari) e acesse:
**[http://localhost:3000](http://localhost:3000)**

---

## 👥 Contas Prontas para Testar de Imediato

O sistema já inicia com duas contas pré-configuradas para você testar tudo de imediato:

### 1. Conta de Organizador
- **E-mail:** `organizador@portal.com`
- **Senha:** `senha123`
- **Permissões:**
  - Criar novos eventos;
  - Editar dados (título, descrição, categoria, data, local e vagas);
  - Cancelar ou reativar eventos;
  - Acessar a tela **"Presença & Inscritos"** com a lista de todos os inscritos;
  - Fazer check-in / confirmar a presença de cada participante com 1 clique.

### 2. Conta de Participante
- **E-mail:** `participante@portal.com`
- **Senha:** `senha123`
- **Permissões:**
  - Visualizar todos os eventos disponíveis e vagas restantes;
  - Inscrever-se em eventos com 1 clique;
  - Acessar a tela **"Minhas Inscrições"**;
  - Cancelar suas próprias inscrições quando quiser;
  - Ver se sua presença já foi confirmada pelo organizador.

> **Novas contas:** Qualquer pessoa pode clicar em "Cadastrar", escolher se quer ser **Participante** ou **Organizador** e criar sua própria conta.

---

## 📱 Como Usar no Celular e no Computador ao Mesmo Tempo

Tudo fica salvo no servidor central (pasta `data/`). Portanto, tudo o que for feito em um aparelho aparece imediatamente no outro:

1. No computador onde o servidor foi iniciado, abra o terminal e descubra seu IP local (ex: no Windows digite `ipconfig`; no Mac/Linux digite `ifconfig` ou `ip a`). Exemplo: `192.168.1.15`.
2. Conecte o celular na mesma rede Wi-Fi.
3. No navegador do celular, acesse: `http://SEU_IP:3000` (ex: `http://192.168.1.15:3000`).
4. Faça login com a mesma conta e você verá todos os dados sincronizados.

---

## 🌐 Como Colocar na Internet de Verdade (Gratuitamente)

Se você quiser colocar este site no ar para qualquer pessoa do mundo acessar sem depender do seu computador ligado, basta hospedar em qualquer plataforma como:
- **Render.com** (Basta conectar a pasta ou repositório GitHub e clicar em "New Web Service" -> Node -> `npm start`);
- **Railway.app** ou **Fly.io** ou qualquer VPS (Linux/Ubuntu).

Nenhum arquivo ou banco de dados externo é necessário. Tudo o que o sistema precisa já está contido neste projeto!
