      status TEXT DEFAULT 'active',
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP

  `);

  // Tabela de Categorias

    CREATE TABLE IF NOT EXISTS categories (
      id INTEGER PRIMARY KEY AUTOINCREMENT,

      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )
  `);

  // Tabela de Gastos / Despesas

      amount REAL NOT NULL,
      date TEXT NOT NULL,

      FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,

const express = require('express');
const cors = require('cors');

const db = require('./db');

const app = express();

const JWT_SECRET = process.env.JWT_SECRET || 'chave-secreta-despesas-2025-segura';

app.use(cors());

// Middleware de Autenticação
function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];

  if (!token) {
    return res.status(401).json({ error: 'Acesso não autorizado. Faça login novamente.' });
  }

  jwt.verify(token, JWT_SECRET, (err, decoded) => {
    if (err) {
      return res.status(403).json({ error: 'Sessão inválida ou expirada. Faça login novamente.' });
    }

    // Verificar se o usuário ainda está ativo no banco
    db.get('SELECT id, name, email, role, status FROM users WHERE id = ?', [decoded.id], (dbErr, user) => {
      if (dbErr || !user) {
        return res.status(404).json({ error: 'Usuário não encontrado.' });
      }
      if (user.status !== 'active') {
        return res.status(403).json({ error: 'Sua conta foi desativada pelo administrador.' });
      }
      req.user = user;
      next();
    });
  });
}

// Middleware para verificar se é Administrador
function requireAdmin(req, res, next) {
  if (req.user.role !== 'admin') {
    return res.status(403).json({ error: 'Acesso restrito para administradores.' });
  }
  next();
}

app.post('/api/auth/register', (req, res) => {
  const { name, email, password } = req.body;

  if (!name || !email || !password) {
    return res.status(400).json({ error: 'Preencha todos os campos obrigatórios.' });
  }

  if (password.length < 6) {
    return res.status(400).json({ error: 'A senha deve ter pelo menos 6 caracteres.' });
  }

  const normalizedEmail = email.trim().toLowerCase();
  const hashedPassword = bcrypt.hashSync(password, 10);

  db.run(
    'INSERT INTO users (name, email, password, role, status) VALUES (?, ?, ?, ?, ?)',
    [name.trim(), normalizedEmail, hashedPassword, 'user', 'active'],
    function (err) {
      if (err) {
        if (err.message.includes('UNIQUE constraint failed')) {
          return res.status(400).json({ error: 'Este e-mail já está cadastrado.' });
        }
        return res.status(500).json({ error: 'Erro ao cadastrar usuário.' });
      }

      const token = jwt.sign({ id: this.lastID, role: 'user' }, JWT_SECRET, { expiresIn: '30d' });
      res.status(201).json({
        message: 'Conta criada com sucesso!',
        token,
        user: { id: this.lastID, name: name.trim(), email: normalizedEmail, role: 'user', status: 'active' }
      });
    }
  );
});

// Login
app.post('/api/auth/login', (req, res) => {
  const { email, password } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: 'E-mail e senha são obrigatórios.' });
  }

  const normalizedEmail = email.trim().toLowerCase();

  db.get('SELECT * FROM users WHERE email = ?', [normalizedEmail], (err, user) => {
    if (err) {
      return res.status(500).json({ error: 'Erro no servidor.' });
    }
    if (!user) {
      return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
    }

    if (user.status !== 'active') {
      return res.status(403).json({ error: 'Esta conta foi desativada pelo administrador do sistema.' });
    }

    const passwordMatch = bcrypt.compareSync(password, user.password);
    if (!passwordMatch) {
      return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
    }

    const token = jwt.sign({ id: user.id, role: user.role }, JWT_SECRET, { expiresIn: '30d' });
    res.json({
      message: 'Login realizado com sucesso!',
      token,
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
        role: user.role,
        status: user.status
      }
    });
  });
});

// Listar gastos do próprio usuário com filtros de mês e categoria
app.get('/api/expenses', authenticateToken, (req, res) => {
  const { month, categoryId } = req.query;
  const userId = req.user.id;

  let query = `
    SELECT e.id, e.description, e.amount, e.date, e.created_at,
           c.id as category_id, c.name as category_name, c.color as category_color
    FROM expenses e
    JOIN categories c ON e.category_id = c.id
    WHERE e.user_id = ?
  `;
  const params = [userId];

  if (month) {
    // month no formato YYYY-MM
    query += ` AND strftime('%Y-%m', e.date) = ?`;
    params.push(month);
  }

  if (categoryId) {
    query += ` AND e.category_id = ?`;
    params.push(categoryId);
  }

  query += ` ORDER BY e.date DESC, e.id DESC`;

  db.all(query, params, (err, rows) => {
    if (err) {
      return res.status(500).json({ error: 'Erro ao carregar despesas.' });
    }
    res.json(rows);
  });
});

// Resumo do mês para o usuário logado (Total e Divisão por Categoria)
app.get('/api/expenses/summary', authenticateToken, (req, res) => {
  const { month } = req.query;
  const userId = req.user.id;

  const currentMonth = month || new Date().toISOString().slice(0, 7);

  // Total do mês
  const totalQuery = `
    SELECT COALESCE(SUM(amount), 0) as total
    FROM expenses
    WHERE user_id = ? AND strftime('%Y-%m', date) = ?
  `;

  // Total por categoria no mês
  const byCategoryQuery = `
    SELECT c.id, c.name, c.color, COALESCE(SUM(e.amount), 0) as total, COUNT(e.id) as count
    FROM categories c
    JOIN expenses e ON e.category_id = c.id
    WHERE e.user_id = ? AND strftime('%Y-%m', e.date) = ?
    GROUP BY c.id
    ORDER BY total DESC
  `;

  db.get(totalQuery, [userId, currentMonth], (err, totalRow) => {
    if (err) {
      return res.status(500).json({ error: 'Erro ao calcular total.' });
    }

    db.all(byCategoryQuery, [userId, currentMonth], (catErr, categoryRows) => {
      if (catErr) {
        return res.status(500).json({ error: 'Erro ao calcular categorias.' });
      }

      res.json({
        month: currentMonth,
        totalMonthly: totalRow.total,
        byCategory: categoryRows
      });
    });
  });
});

// Estatísticas globais do sistema
app.get('/api/admin/stats', authenticateToken, requireAdmin, (req, res) => {
  const totalExpensesQuery = `SELECT COALESCE(SUM(amount), 0) as totalGlobal, COUNT(id) as totalTransactions FROM expenses`;
  const totalUsersQuery = `SELECT COUNT(id) as totalUsers, SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) as activeUsers FROM users WHERE role != 'admin'`;
  const topCategoriesQuery = `
    SELECT c.name, c.color, COALESCE(SUM(e.amount), 0) as total
    FROM categories c
    LEFT JOIN expenses e ON e.category_id = c.id
    GROUP BY c.id
    HAVING total > 0
    ORDER BY total DESC
  `;

  db.get(totalExpensesQuery, [], (err, expRow) => {
    if (err) return res.status(500).json({ error: 'Erro nas estatísticas globais.' });

    db.get(totalUsersQuery, [], (userErr, uRow) => {
      if (userErr) return res.status(500).json({ error: 'Erro nas estatísticas de usuários.' });

      db.all(topCategoriesQuery, [], (catErr, catRows) => {
        if (catErr) return res.status(500).json({ error: 'Erro nas estatísticas de categoria.' });

        res.json({
          totalGlobal: expRow.totalGlobal,
          totalTransactions: expRow.totalTransactions,
          totalUsers: uRow.totalUsers || 0,
          activeUsers: uRow.activeUsers || 0,
          categories: catRows
        });
      });
    });
  });
});

app.patch('/api/admin/users/:id/status', authenticateToken, requireAdmin, (req, res) => {
  const userId = req.params.id;
  const { status } = req.body;

  if (!['active', 'disabled'].includes(status)) {
    return res.status(400).json({ error: 'Status inválido. Escolha "active" ou "disabled".' });
  }

  // Não permitir desativar a si próprio se for admin
  if (parseInt(userId, 10) === req.user.id) {
    return res.status(400).json({ error: 'Você não pode desativar seu próprio acesso de administrador.' });
  }

  db.run('UPDATE users SET status = ? WHERE id = ?', [status, userId], function (err) {
    if (err) {
      return res.status(500).json({ error: 'Erro ao atualizar status do usuário.' });
    }
    if (this.changes === 0) {
      return res.status(404).json({ error: 'Usuário não encontrado.' });
    }
    res.json({ message: `Conta ${status === 'active' ? 'ativada' : 'desativada'} com sucesso.` });
  });
});

// Formatar valor para moeda brasileira (R$)
function formatBRL(amount) {
  return new Intl.NumberFormat('pt-BR', {
    style: 'currency',
    currency: 'BRL'
  }).format(amount || 0);
}

// Formatar data YYYY-MM-DD para DD/MM/AAAA
function formatDateBR(dateString) {
  if (!dateString) return '-';
  const parts = dateString.split('-');
  if (parts.length === 3) {
    return `${parts[2]}/${parts[1]}/${parts[0]}`;
  }
  return dateString;
}

// --- FUNÇÃO AUXILIAR FETCH COM AUTH ---
async function apiFetch(url, options = {}) {
  const headers = options.headers || {};
  if (token) {
    headers['Authorization'] = `Bearer ${token}`;
  }
  if (options.body && typeof options.body === 'object') {
    headers['Content-Type'] = 'application/json';
    options.body = JSON.stringify(options.body);
  }

  options.headers = headers;
  const res = await fetch(url, options);
  const data = await res.json();

  if (!res.ok) {
    if (res.status === 401 || res.status === 403) {
      if (data.error && data.error.includes('desativada')) {
        showToast(data.error, 'error');
        handleLogout();
        return null;
      }
    }
    throw new Error(data.error || 'Erro na requisição');
  }

  return data;
}

async function checkCurrentUser() {
  try {
    const data = await apiFetch('/api/auth/me');
    if (data && data.user) {
      currentUser = data.user;
      showAppScreen();
    } else {
      handleLogout();
    }
  } catch (err) {
    handleLogout();
  }
}

async function handleLogin(e) {
  e.preventDefault();
  const email = document.getElementById('login-email').value;
  const password = document.getElementById('login-password').value;

  try {
    const data = await apiFetch('/api/auth/login', {
      method: 'POST',
      body: { email, password }
    });

    token = data.token;
    currentUser = data.user;
    localStorage.setItem('token', token);
    showToast(`Bem-vindo(a), ${currentUser.name}!`, 'success');
    showAppScreen();
  } catch (err) {
    showToast(err.message, 'error');
  }
}

async function loadUserExpenses() {
  const month = document.getElementById('filter-month').value;
  const categoryId = document.getElementById('filter-category').value;
  const tbody = document.getElementById('expenses-table-body');

  tbody.innerHTML = '<tr><td colspan="5" class="text-center">Carregando despesas...</td></tr>';

  try {
    // Carregar despesas filtradas
    let url = `/api/expenses?month=${month}`;
    if (categoryId) url += `&categoryId=${categoryId}`;

    const expenses = await apiFetch(url);

    // Carregar resumo do mês
    const summary = await apiFetch(`/api/expenses/summary?month=${month}`);

    // Atualizar Card de Total do Mês
    document.getElementById('total-month-value').innerText = formatBRL(summary.totalMonthly);
    const [year, m] = month.split('-');
    const dateObj = new Date(year, m - 1, 1);
    const monthName = dateObj.toLocaleDateString('pt-BR', { month: 'long', year: 'numeric' });
    document.getElementById('total-month-label').innerText = `Referência: ${monthName}`;

    // Renderizar Barras por Categoria no Mês
    renderCategoryBars(summary.byCategory, summary.totalMonthly);

    // Renderizar Tabela de Gastos
    if (expenses.length === 0) {
      tbody.innerHTML = '<tr><td colspan="5" class="text-center">Nenhum gasto encontrado para este período.</td></tr>';
      return;
    }

    tbody.innerHTML = expenses.map(exp => `
      <tr>
        <td>${formatDateBR(exp.date)}</td>
        <td><strong>${exp.description}</strong></td>
        <td>
          <span class="category-tag" style="background-color: ${exp.category_color};">
            ${exp.category_name}
          </span>
        </td>
        <td class="amount-text">${formatBRL(exp.amount)}</td>
        <td class="text-right">
          <button class="btn-danger-sm" onclick="handleDeleteExpense(${exp.id})">Excluir</button>
        </td>
      </tr>
    `).join('');

  } catch (err) {
    showToast('Erro ao carregar despesas.', 'error');
  }
}

function renderCategoryBars(categories, totalMonth) {
  const container = document.getElementById('category-bars-list');
  if (!categories || categories.length === 0 || totalMonth <= 0) {
    container.innerHTML = '<p class="empty-text">Nenhum gasto registrado neste mês.</p>';
    return;
  }

  container.innerHTML = categories.map(cat => {
    const percent = Math.min(100, Math.round((cat.total / totalMonth) * 100));
    return `
      <div class="bar-item">
        <div class="bar-info">
          <span class="bar-name">
            <span class="bar-color-dot" style="background-color: ${cat.color};"></span>
            ${cat.name} (${cat.count}x)
          </span>
          <span><strong>${formatBRL(cat.total)}</strong> (${percent}%)</span>
        </div>
        <div class="bar-track">
          <div class="bar-fill" style="width: ${percent}%; background-color: ${cat.color};"></div>
        </div>
      </div>
    `;
  }).join('');
}

async function loadAdminUsers() {
  const tbody = document.getElementById('admin-users-table-body');
  tbody.innerHTML = '<tr><td colspan="6" class="text-center">Carregando usuários...</td></tr>';

  try {
    const users = await apiFetch('/api/admin/users');

    if (users.length === 0) {
      tbody.innerHTML = '<tr><td colspan="6" class="text-center">Nenhum usuário comum registrado.</td></tr>';
      return;
    }

    tbody.innerHTML = users.map(u => {
      const isActive = u.status === 'active';
      const statusBadge = isActive
        ? '<span class="badge-status-active">Ativo</span>'
        : '<span class="badge-status-disabled">Desativado</span>';

      const actionBtn = isActive
        ? `<button class="btn-toggle-disable" onclick="toggleUserStatus(${u.id}, 'disabled')">Desativar Conta</button>`
        : `<button class="btn-toggle-active" onclick="toggleUserStatus(${u.id}, 'active')">Reativar Conta</button>`;

      return `
        <tr>
          <td><strong>${u.name}</strong></td>
          <td>${u.email}</td>
          <td>${statusBadge}</td>
          <td>${u.expense_count} despesa(s)</td>
          <td class="amount-text">${formatBRL(u.total_spent)}</td>
          <td class="text-right">${actionBtn}</td>
        </tr>
      `;
    }).join('');
  } catch (err) {
    showToast('Erro ao carregar lista de usuários.', 'error');
  }
}

async function toggleUserStatus(userId, newStatus) {
  const actionText = newStatus === 'disabled' ? 'desativar' : 'reativar';
  if (!confirm(`Deseja realmente ${actionText} o acesso deste usuário?`)) return;

  try {
    await apiFetch(`/api/admin/users/${userId}/status`, {
      method: 'PATCH',
      body: { status: newStatus }
    });

    showToast(`Conta ${newStatus === 'disabled' ? 'desativada' : 'reativada'} com sucesso!`, 'success');
    loadAdminUsers();
  } catch (err) {
    showToast(err.message, 'error');
  }
}

# Sistema de Controle de Despesas Pessoais

Sistema web completo de controle financeiro pessoal com suporte a múltiplos usuários e painel de administrador.

Todas as informações são gravadas automaticamente em um banco de dados SQLite (`despesas.db`) mantido no próprio servidor. As alterações feitas pelo computador ou pelo celular permanecem salvas de forma unificada e permanente.

---

## 🚀 Como Instalar e Rodar o Projeto

Você precisa ter o **Node.js** instalado no seu computador (versão 16 ou superior). Se ainda não tiver, baixe e instale em [nodejs.org](https://nodejs.org/).

### Passo 1: Abrir o terminal na pasta do projeto
Abra a pasta onde você descompactou os arquivos do projeto no seu terminal (Prompt de Comando, PowerShell ou Terminal).

### Passo 2: Instalar as dependências
Execute o comando abaixo e aguarde alguns segundos:
```bash
npm install
```

### Passo 3: Iniciar o site
Execute o comando para ligar o servidor:
```bash
npm start
```

### Passo 4: Acessar no navegador
Abra o seu navegador de internet e entre no endereço:
```
http://localhost:3000
```

Se quiser acessar do seu celular conectado na mesma rede Wi-Fi, basta abrir o navegador do celular e digitar o IP do seu computador com a porta 3000 (exemplo: `http://192.168.1.15:3000`).

---

## 🔑 Contas e Acessos

### 1. Conta do Administrador (criada automaticamente)
- **E-mail:** `admin@admin.com`
- **Senha:** `admin123`

O Administrador tem acesso a menus exclusivos:
- **Visão Geral:** visualiza a soma de todos os gastos já cadastrados no site por todos os usuários, número total de usuários e gráficos consolidados.
- **Categorias:** cadastra novas categorias de gastos com cores personalizadas ou remove categorias existentes.
- **Gerenciar Usuários:** visualiza quanto cada usuário cadastrado já gastou no sistema e possui o botão para **Desativar** ou **Reativar** a conta de qualquer pessoa.

### 2. Usuários Comuns (Você e outras pessoas)
- Qualquer pessoa pode clicar na aba **"Criar Conta"** na tela inicial, digitar seu nome, e-mail e senha.
- O usuário registra seus gastos diários informando descrição, valor, data e categoria.
- Pode selecionar o **mês de referência** para ver a soma total gasta naquele mês e o percentual gasto em cada categoria.
- Pode filtrar despesas por categoria ou excluir gastos que errou.
- Se a conta de um usuário for desativada pelo administrador, ele é impedido de entrar até que a conta seja reativada.
