# Sistema de Pedidos para Restaurante

Sistema completo de pedidos para restaurante, pronto para uso em computadores e celulares. Não necessita de serviços externos e armazena todos os dados de forma permanente no próprio servidor.

---

## 🚀 Como instalar e rodar (Passo a Passo)

Você só precisa ter o **Python** instalado no seu computador (se não tiver, baixe a versão mais recente em [python.org](https://www.python.org)).

### 1. Abra o Terminal ou Prompt de Comando
Navegue até a pasta onde descompactou os arquivos do projeto:
```bash
cd caminho/para/a/pasta
```

### 2. Instale os componentes necessários
Execute o comando:
```bash
pip install -r requirements.txt
```

### 3. Inicie o sistema
Execute o comando:
```bash
python app.py
```

### 4. Acesse no navegador
Abra seu navegador de internet e entre no endereço:
```
http://localhost:5000
```

Para acessar pelo **celular ou outro computador na mesma rede Wi-Fi**, basta abrir o navegador do celular e digitar o IP do computador onde o sistema está rodando (exemplo: `http://192.168.1.100:5000`). O próprio terminal mostra o endereço de rede ao iniciar.

---

## 💾 Onde ficam salvos os dados?

Todos os dados (usuários, cardápio, fotos e histórico de pedidos) são gravados no arquivo de banco de dados `restaurante.db`, criado automaticamente na mesma pasta. Mesmo se o computador for desligado, nada é perdido.

def current_user():
    user_id = session.get('user_id')
    if not user_id:
        return None
    conn = get_db()
    user = conn.execute("SELECT id, name, email, role, phone, address FROM users WHERE id = ?", (user_id,)).fetchone()
    conn.close()
    return dict(user) if user else None

@app.route('/api/register', methods=['POST'])
def api_register():
    data = request.get_json() or {}
    name = data.get('name', '').strip()
    email = data.get('email', '').strip().lower()
    password = data.get('password', '')
    phone = data.get('phone', '').strip()
    address = data.get('address', '').strip()

    if not name or not email or not password:
        return jsonify({'error': 'Nome, e-mail e senha são obrigatórios.'}), 400

    conn = get_db()
    existing = conn.execute("SELECT id FROM users WHERE email = ?", (email,)).fetchone()
    if existing:
        conn.close()
        return jsonify({'error': 'Este e-mail já está cadastrado.'}), 400

    hashed = generate_password_hash(password)
    cursor = conn.cursor()
    cursor.execute('''
        INSERT INTO users (name, email, password_hash, role, phone, address)
        VALUES (?, ?, ?, 'cliente', ?, ?)
    ''', (name, email, hashed, phone, address))
    user_id = cursor.lastrowid
    conn.commit()
    conn.close()

    session['user_id'] = user_id
    return jsonify({
        'success': True,
        'user': {
            'id': user_id,
            'name': name,
            'email': email,
            'role': 'cliente',
            'phone': phone,
            'address': address
        }
    })

@app.route('/api/login', methods=['POST'])
def api_login():
    data = request.get_json() or {}
    email = data.get('email', '').strip().lower()
    password = data.get('password', '')

    if not email or not password:
        return jsonify({'error': 'Preencha e-mail e senha.'}), 400

    conn = get_db()
    user = conn.execute("SELECT * FROM users WHERE email = ?", (email,)).fetchone()
    conn.close()

    if not user or not check_password_hash(user['password_hash'], password):
        return jsonify({'error': 'E-mail ou senha incorretos.'}), 401

    session['user_id'] = user['id']
    return jsonify({
        'success': True,
        'user': {
            'id': user['id'],
            'name': user['name'],
            'email': user['email'],
            'role': user['role'],
            'phone': user['phone'],
            'address': user['address']
        }
    })

@app.route('/api/menu')
def api_menu():
    conn = get_db()
    categories = [dict(c) for c in conn.execute("SELECT * FROM categories ORDER BY id ASC").fetchall()]
    
    # Gerente vê todos os itens (inclusive desativados), clientes veem apenas os disponíveis
    user = current_user()
    if user and user['role'] == 'gerente':
        dishes_raw = conn.execute('''
            SELECT d.*, c.name as category_name
            FROM dishes d
            JOIN categories c ON d.category_id = c.id
            ORDER BY d.category_id ASC, d.id ASC
        ''').fetchall()
    else:
        dishes_raw = conn.execute('''
            SELECT d.*, c.name as category_name
            FROM dishes d
            JOIN categories c ON d.category_id = c.id
            WHERE d.is_available = 1
            ORDER BY d.category_id ASC, d.id ASC
        ''').fetchall()

    dishes = [dict(d) for d in dishes_raw]
    conn.close()

    return jsonify({'categories': categories, 'dishes': dishes})

@app.route('/api/orders', methods=['POST'])
def api_create_order():
    user = current_user()
    if not user:
        return jsonify({'error': 'Faça login para realizar um pedido.'}), 401

    data = request.get_json() or {}
    items = data.get('items', [])
    delivery_address = data.get('delivery_address', '').strip()
    customer_phone = data.get('customer_phone', '').strip()
    payment_method = data.get('payment_method', 'PIX').strip()
    notes = data.get('notes', '').strip()

    if not items:
        return jsonify({'error': 'O carrinho está vazio.'}), 400
    if not delivery_address:
        return jsonify({'error': 'Endereço de entrega é obrigatório.'}), 400
    if not customer_phone:
        return jsonify({'error': 'Telefone para contato é obrigatório.'}), 400

    conn = get_db()
    cursor = conn.cursor()

    # Calcular total e validar itens
    total_price = 0.0
    validated_items = []

    for item in items:
        dish_id = item.get('dish_id')
        qty = int(item.get('quantity', 1))
        if qty <= 0:
            continue
        
        dish = conn.execute("SELECT * FROM dishes WHERE id = ?", (dish_id,)).fetchone()
        if not dish:
            continue
        
        subtotal = round(dish['price'] * qty, 2)
        total_price += subtotal
        validated_items.append({
            'dish_id': dish['id'],
            'dish_name': dish['name'],
            'unit_price': dish['price'],
            'quantity': qty,
            'subtotal': subtotal,
            'notes': item.get('notes', '')
        })

    if not validated_items:
        conn.close()
        return jsonify({'error': 'Nenhum prato válido no carrinho.'}), 400

    total_price = round(total_price, 2)

    # Inserir pedido
    cursor.execute('''
        INSERT INTO orders (user_id, customer_name, customer_phone, delivery_address, payment_method, notes, total_price, status)
        VALUES (?, ?, ?, ?, ?, ?, ?, 'recebido')
    ''', (user['id'], user['name'], customer_phone, delivery_address, payment_method, notes, total_price))
    order_id = cursor.lastrowid

    # Inserir itens
    for it in validated_items:
        cursor.execute('''
            INSERT INTO order_items (order_id, dish_id, dish_name, unit_price, quantity, subtotal, notes)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        ''', (order_id, it['dish_id'], it['dish_name'], it['unit_price'], it['quantity'], it['subtotal'], it['notes']))

    conn.commit()
    conn.close()

    return jsonify({'success': True, 'order_id': order_id})

@app.route('/api/my-orders')
def api_my_orders():
    user = current_user()
    if not user:
        return jsonify({'error': 'Não autenticado.'}), 401

    conn = get_db()
    orders_raw = conn.execute('''
        SELECT * FROM orders
        WHERE user_id = ?
        ORDER BY id DESC
    ''', (user['id'],)).fetchall()

    orders = []
    for o in orders_raw:
        o_dict = dict(o)
        items_raw = conn.execute('''
            SELECT * FROM order_items WHERE order_id = ?
        ''', (o['id'],)).fetchall()
        o_dict['items'] = [dict(it) for it in items_raw]
        orders.append(o_dict)

    conn.close()
    return jsonify({'orders': orders})

    function loadCartFromStorage() {
      try {
        const stored = localStorage.getItem('restaurante_carrinho');
        if (stored) cart = JSON.parse(stored);
      } catch (e) { cart = []; }
      updateCartBadge();
    }

    function saveCartToStorage() {
      localStorage.setItem('restaurante_carrinho', JSON.stringify(cart));
      updateCartBadge();
    }

    function addToCart(dishId) {
      const dish = menuData.dishes.find(d => d.id === dishId);
      if (!dish) return;

      const existing = cart.find(item => item.dish_id === dishId);
      if (existing) {
        existing.quantity += 1;
      } else {
        cart.push({
          dish_id: dish.id,
          name: dish.name,
          price: dish.price,
          quantity: 1,
          notes: ''
        });
      }

      saveCartToStorage();
      showToast(`"${dish.name}" adicionado ao carrinho!`);
    }

    function updateCartBadge() {
      const totalQty = cart.reduce((sum, item) => sum + item.quantity, 0);
      document.getElementById('cart-badge').innerText = totalQty;
    }

    function openCartModal() {
      renderCartItems();
      if (currentUser) {
        if (currentUser.phone) document.getElementById('order-phone').value = currentUser.phone;

    function renderCartItems() {
      const container = document.getElementById('cart-items-container');
      const checkoutSection = document.getElementById('cart-checkout-section');
      const totalSpan = document.getElementById('cart-total-price');

      if (cart.length === 0) {
        container.innerHTML = `
          <div class="empty-state">
            <div class="empty-state-icon">🛒</div>
            <h4>Seu carrinho está vazio</h4>
            <p>Escolha seus pratos favoritos no cardápio.</p>
          </div>
        `;
        checkoutSection.style.display = 'none';
        totalSpan.innerText = 'R$ 0,00';
        return;
      }

      checkoutSection.style.display = 'block';
      let total = 0;

      container.innerHTML = cart.map((item, index) => {
        const subtotal = item.price * item.quantity;
        total += subtotal;
        return `
          <div class="cart-item">
            <div class="cart-item-info">
              <div class="cart-item-name">${escapeHtml(item.name)}</div>
              <div class="cart-item-price">R$ ${item.price.toFixed(2).replace('.', ',')} cada</div>
              <input type="text" class="form-control" style="padding: 4px 8px; font-size: 0.8rem; margin-top: 4px;" 
                placeholder="Observação (ex: sem cebola)" 
                value="${escapeHtml(item.notes || '')}" 
                onchange="updateCartNotes(${index}, this.value)">
            </div>
            <div class="qty-controls">
              <button class="qty-btn" onclick="changeCartQty(${index}, -1)">-</button>
              <span style="font-weight: 700; font-size: 0.9rem; min-width: 20px; text-align: center;">${item.quantity}</span>
              <button class="qty-btn" onclick="changeCartQty(${index}, 1)">+</button>
            </div>
            <div style="font-weight: 700; font-size: 0.95rem; min-width: 70px; text-align: right;">
              R$ ${subtotal.toFixed(2).replace('.', ',')}
            </div>
          </div>
        `;
      }).join('');

      totalSpan.innerText = `R$ ${total.toFixed(2).replace('.', ',')}`;
    }

    function changeCartQty(index, delta) {
      if (!cart[index]) return;
      cart[index].quantity += delta;
      if (cart[index].quantity <= 0) {
        cart.splice(index, 1);
      }
      saveCartToStorage();
      renderCartItems();
    }

    function updateCartNotes(index, val) {
      if (cart[index]) {
        cart[index].notes = val.trim();
        saveCartToStorage();
      }
    }

    function renderOrderCard(order, isAdmin = false) {
      const statusMap = {
        'recebido': { label: 'Recebido', class: 'status-recebido', step: 1 },
        'preparando': { label: 'Em Preparo', class: 'status-preparando', step: 2 },
        'saiu_entrega': { label: 'Saiu para Entrega', class: 'status-saiu_entrega', step: 3 },
        'entregue': { label: 'Entregue', class: 'status-entregue', step: 4 },
        'cancelado': { label: 'Cancelado', class: 'status-cancelado', step: 0 }
      };

      const curr = statusMap[order.status] || { label: order.status, class: '', step: 1 };

      // Se for cliente e pedido não estiver cancelado, mostra a linha do tempo (stepper)
      let trackerHtml = '';
      if (!isAdmin && order.status !== 'cancelado') {
        const steps = [
          { num: 1, name: 'Recebido' },
          { num: 2, name: 'Na Cozinha' },
          { num: 3, name: 'A Caminho' },
          { num: 4, name: 'Entregue' }
        ];
        trackerHtml = `
          <div class="tracker-steps">
            ${steps.map(s => {
              const stateClass = s.num < curr.step ? 'completed' : (s.num === curr.step ? 'active' : '');
              return `
                <div class="tracker-step ${stateClass}">
                  <div class="step-circle">${s.num < curr.step ? '✓' : s.num}</div>
                  <div class="step-text">${s.name}</div>
                </div>
              `;
            }).join('')}
          </div>
        `;
      }

      const itemsRows = order.items.map(it => `
        <tr>
          <td><strong>${it.quantity}x</strong> ${escapeHtml(it.dish_name)} ${it.notes ? `<span class="cart-item-notes">(${escapeHtml(it.notes)})</span>` : ''}</td>
          <td style="text-align: right;">R$ ${it.subtotal.toFixed(2).replace('.', ',')}</td>
        </tr>
      `).join('');

      let adminActionsHtml = '';
      if (isAdmin) {
        adminActionsHtml = `
          <div class="order-actions">
            <span style="font-size: 0.82rem; font-weight: 700; color: var(--text-muted); align-self: center; margin-right: 6px;">Mudar Status:</span>
            <button class="btn btn-outline btn-sm" onclick="updateOrderStatus(${order.id}, 'recebido')">Recebido</button>
            <button class="btn btn-outline btn-sm" style="color: var(--info); border-color: var(--info);" onclick="updateOrderStatus(${order.id}, 'preparando')">👨‍🍳 Em Preparo</button>
            <button class="btn btn-outline btn-sm" style="color: #7b1fa2; border-color: #7b1fa2;" onclick="updateOrderStatus(${order.id}, 'saiu_entrega')">🛵 Saiu p/ Entrega</button>
            <button class="btn btn-outline btn-sm" style="color: var(--success); border-color: var(--success);" onclick="updateOrderStatus(${order.id}, 'entregue')">✅ Entregue</button>
            <button class="btn btn-outline btn-sm" style="color: var(--danger); border-color: var(--danger);" onclick="updateOrderStatus(${order.id}, 'cancelado')">❌ Cancelar</button>
          </div>
        `;
      }

      return `
        <div class="order-card">
          <div class="order-card-header">
            <div>
              <span class="order-id">Pedido #${order.id}</span>
              <div class="order-date">📅 ${new Date(order.created_at).toLocaleString('pt-BR')}</div>
            </div>
            <span class="status-badge ${curr.class}">${curr.label}</span>
          </div>

          ${trackerHtml}

          <div class="order-customer-box">
            <strong>Cliente:</strong> ${escapeHtml(order.customer_name)} | <strong>Tel:</strong> ${escapeHtml(order.customer_phone)}<br>
            <strong>Endereço de Entrega:</strong> ${escapeHtml(order.delivery_address)}<br>
            <strong>Pagamento:</strong> ${escapeHtml(order.payment_method)}
            ${order.notes ? `<br><strong>Obs do Cliente:</strong> <em>${escapeHtml(order.notes)}</em>` : ''}
          </div>

          <table class="order-items-table">
            <thead>
              <tr>
                <th>Item</th>
                <th style="text-align: right;">Subtotal</th>
              </tr>
            </thead>
            <tbody>
              ${itemsRows}
            </tbody>
          </table>

          <div class="order-total-bar">
            <span>Total Pago</span>
            <span style="color: var(--primary);">R$ ${order.total_price.toFixed(2).replace('.', ',')}</span>
          </div>

          ${adminActionsHtml}
        </div>
      `;
    }

    async function loadAdminOrders() {
      if (!currentUser || currentUser.role !== 'gerente') return;
      try {
        const res = await fetch('/api/admin/orders');
        const data = await res.json();
        const list = document.getElementById('admin-orders-list');

        let orders = data.orders || [];

        // Tocar alerta sonoro se novo pedido chegou
        if (orders.length > knownOrdersCount && knownOrdersCount !== 0) {
          playNotificationBeep();
          showToast('🔔 Novo pedido recebido!');
        }
        knownOrdersCount = orders.length;

        if (currentAdminOrdersFilter !== 'todos') {
          orders = orders.filter(o => o.status === currentAdminOrdersFilter);
        }

        if (orders.length === 0) {
          list.innerHTML = `
            <div class="empty-state">
              <div class="empty-state-icon">📋</div>
              <h3>Nenhum pedido nesta situação</h3>
              <p>Os pedidos dos clientes aparecerão aqui automaticamente.</p>
            </div>
          `;
          return;
        }

        list.innerHTML = orders.map(order => renderOrderCard(order, true)).join('');
      } catch (e) {
        console.error("Erro ao carregar pedidos administrativos", e);
      }
    }

          showToast('Erro ao atualizar status.');

      }

      if (!dish) return;

      document.getElementById('dish-modal-title').innerText = 'Editar Prato';
      document.getElementById('dish-id').value = dish.id;
      document.getElementById('dish-name').value = dish.name;
      document.getElementById('dish-category').value = dish.category_id;
      document.getElementById('dish-price').value = dish.price;
      document.getElementById('dish-desc').value = dish.description || '';
      document.getElementById('dish-image').value = dish.image_url || '';
      document.getElementById('dish-available').checked = !!dish.is_available;
      document.getElementById('modal-dish').classList.add('open');
    }

    function closeDishModal() {
      document.getElementById('modal-dish').classList.remove('open');
    }

    async function saveDish() {
      const id = document.getElementById('dish-id').value;
      const name = document.getElementById('dish-name').value.trim();
      const category_id = parseInt(document.getElementById('dish-category').value);
      const price = parseFloat(document.getElementById('dish-price').value);
      const description = document.getElementById('dish-desc').value.trim();
      const image_url = document.getElementById('dish-image').value.trim();
      const is_available = document.getElementById('dish-available').checked;

      if (!name || isNaN(price)) {
        alert('Preencha pelo menos o Nome e o Preço do prato.');
        return;
      }

      const payload = { name, category_id, price, description, image_url, is_available };

      try {
        let res;
        if (id) {
          res = await fetch(`/api/admin/dish/${id}`, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
        } else {
          res = await fetch('/api/admin/dish', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
        }

        if (res.ok) {
          closeDishModal();
          showToast(id ? 'Prato atualizado!' : 'Prato cadastrado com sucesso!');
          await loadMenu();
        } else {
          const err = await res.json();
          alert(err.error || 'Erro ao salvar prato.');
        }
      } catch (err) {
        alert('Erro ao conectar ao servidor.');
      }
    }

    async function deleteDish(dishId) {
      if (!confirm('Deseja realmente excluir este prato do cardápio?')) return;
      try {
        const res = await fetch(`/api/admin/dish/${dishId}`, { method: 'DELETE' });
        if (res.ok) {
          showToast('Prato removido.');
          await loadMenu();
        }
      } catch (err) {
        showToast('Erro ao excluir prato.');
      }
    }

    function switchView(viewName) {
      const views = ['cardapio', 'meus-pedidos', 'gerente-pedidos', 'gerente-cardapio'];
      views.forEach(v => {
        const el = document.getElementById(`view-${v}`);
        const tabBtn = document.getElementById(`tab-btn-${v}`);
        if (el) el.style.display = (v === viewName) ? 'block' : 'none';
        if (tabBtn) tabBtn.classList.toggle('active', v === viewName);
      });

      if (viewName === 'meus-pedidos') loadClientOrders();
      if (viewName === 'gerente-pedidos') loadAdminOrders();
      if (viewName === 'gerente-cardapio') renderAdminMenuTable();
    }

    function escapeHtml(text) {
      if (!text) return '';
      return String(text)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
    }

    function playNotificationBeep() {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.type = 'sine';
        osc.frequency.setValueAtTime(587.33, ctx.currentTime); // D5
        osc.frequency.setValueAtTime(880, ctx.currentTime + 0.15); // A5
        gain.gain.setValueAtTime(0.3, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.4);
        osc.start();
        osc.stop(ctx.currentTime + 0.4);
      } catch (e) {}
    }
