# Loja Virtual Completa

## Como instalar e executar

1. **Pré-requisitos**: Instale o Node.js (versão 14 ou superior) em seu computador/servidor

2. **Instalação**:
   ```bash
   npm install
   ```

3. **Executar**:
   ```bash
   npm start
   ```

4. **Acessar**: Abra o navegador em `http://localhost:3000`

## Para colocar na internet (hospedar)

Você precisará de um servidor (VPS) como DigitalOcean, AWS, ou similar:

1. Copie todos os arquivos para o servidor
2. Instale o Node.js no servidor
3. Execute `npm install`
4. Execute `npm start` (use PM2 para manter rodando: `pm2 start server.js`)
5. Configure um domínio apontando para o IP do servidor
6. Configure HTTPS com Let's Encrypt (recomendado)

## Primeiro acesso

- **Criar conta de dono**: Acesse `/admin-setup` e crie a primeira conta de administrador
- **Clientes**: Podem se registrar normalmente na página inicial

## Funcionalidades

### Cliente
- Navegar por produtos
- Adicionar ao carrinho
- Finalizar pedidos
- Ver histórico de pedidos

### Dono da loja
- Cadastrar/editar/remover produtos
- Ver todos os pedidos
- Alterar status dos pedidos

## Banco de dados

O sistema usa SQLite (arquivo `database.db`) que é criado automaticamente. Todos os dados ficam salvos permanentemente no servidor.

// Criar tabelas
db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT DEFAULT 'cliente',
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS produtos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    descricao TEXT,
    preco REAL NOT NULL,
    categoria TEXT,
    estoque INTEGER DEFAULT 0,
    imagem TEXT,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS pedidos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    total REAL NOT NULL,
    status TEXT DEFAULT 'pendente',
    endereco TEXT,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS itens_pedido (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pedido_id INTEGER NOT NULL,
    produto_id INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    preco REAL NOT NULL,
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
  )`);
});

// Middleware de autenticação
function requireAuth(req, res, next) {
  if (!req.session.usuarioId) {
    return res.status(401).json({ erro: 'Não autenticado' });
  }
  next();
}

function requireAdmin(req, res, next) {
  if (!req.session.usuarioId || req.session.tipo !== 'admin') {
    return res.status(403).json({ erro: 'Acesso negado' });
  }
  next();
}

app.post('/api/registro', async (req, res) => {
  const { nome, email, senha } = req.body;
  
  if (!nome || !email || !senha) {
    return res.status(400).json({ erro: 'Preencha todos os campos' });
  }

  try {
    const senhaHash = await bcrypt.hash(senha, 10);
    db.run(
      'INSERT INTO usuarios (nome, email, senha) VALUES (?, ?, ?)',
      [nome, email, senhaHash],
      function(err) {
        if (err) {
          return res.status(400).json({ erro: 'Email já cadastrado' });
        }
        res.json({ sucesso: true, id: this.lastID });
      }
    );
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao criar conta' });
  }
});

app.post('/api/registro-admin', async (req, res) => {
  const { nome, email, senha } = req.body;
  
  if (!nome || !email || !senha) {
    return res.status(400).json({ erro: 'Preencha todos os campos' });
  }

  // Verifica se já existe algum admin
  db.get('SELECT COUNT(*) as count FROM usuarios WHERE tipo = "admin"', async (err, row) => {
    if (row.count > 0) {
      return res.status(403).json({ erro: 'Já existe um administrador cadastrado' });
    }

    try {
      const senhaHash = await bcrypt.hash(senha, 10);
      db.run(
        'INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, "admin")',
        [nome, email, senhaHash],
        function(err) {
          if (err) {
            return res.status(400).json({ erro: 'Email já cadastrado' });
          }
          res.json({ sucesso: true, id: this.lastID });
        }
      );
    } catch (erro) {
      res.status(500).json({ erro: 'Erro ao criar conta' });
    }
  });
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;

  db.get('SELECT * FROM usuarios WHERE email = ?', [email], async (err, usuario) => {
    if (!usuario) {
      return res.status(401).json({ erro: 'Email ou senha incorretos' });
    }

    const senhaCorreta = await bcrypt.compare(senha, usuario.senha);
    if (!senhaCorreta) {
      return res.status(401).json({ erro: 'Email ou senha incorretos' });
    }

    req.session.usuarioId = usuario.id;
    req.session.nome = usuario.nome;
    req.session.tipo = usuario.tipo;

    res.json({ 
      sucesso: true, 
      usuario: { 
        id: usuario.id, 
        nome: usuario.nome, 
        email: usuario.email,
        tipo: usuario.tipo 
      } 
    });
  });
});

  if (categoria) {

    if (err) {
      return res.status(500).json({ erro: 'Erro ao buscar produtos' });
    }
    res.json(produtos);
  });
});

app.post('/api/pedidos', requireAuth, (req, res) => {
  const { itens, endereco } = req.body;

  if (!itens || itens.length === 0) {
    return res.status(400).json({ erro: 'Carrinho vazio' });
  }

  // Calcular total
  let total = 0;
  const produtoIds = itens.map(i => i.produto_id);
  
  db.all(
    `SELECT id, preco, estoque FROM produtos WHERE id IN (${produtoIds.join(',')})`,
    (err, produtos) => {
      if (err) {
        return res.status(500).json({ erro: 'Erro ao processar pedido' });
      }

      // Verificar estoque e calcular total
      for (let item of itens) {
        const produto = produtos.find(p => p.id === item.produto_id);
        if (!produto) {
          return res.status(400).json({ erro: 'Produto não encontrado' });
        }
        if (produto.estoque < item.quantidade) {
          return res.status(400).json({ erro: `Estoque insuficiente para o produto ID ${item.produto_id}` });
        }
        total += produto.preco * item.quantidade;
      }

      // Criar pedido
      db.run(
        'INSERT INTO pedidos (usuario_id, total, endereco) VALUES (?, ?, ?)',
        [req.session.usuarioId, total, endereco],
        function(err) {
          if (err) {
            return res.status(500).json({ erro: 'Erro ao criar pedido' });
          }

          const pedidoId = this.lastID;

          // Inserir itens e atualizar estoque
          const stmt = db.prepare('INSERT INTO itens_pedido (pedido_id, produto_id, quantidade, preco) VALUES (?, ?, ?, ?)');
          const updateEstoque = db.prepare('UPDATE produtos SET estoque = estoque - ? WHERE id = ?');

          itens.forEach(item => {
            const produto = produtos.find(p => p.id === item.produto_id);
            stmt.run(pedidoId, item.produto_id, item.quantidade, produto.preco);
            updateEstoque.run(item.quantidade, item.produto_id);
          });

          stmt.finalize();
          updateEstoque.finalize();

          res.json({ sucesso: true, pedido_id: pedidoId });
        }
      );
    }
  );
});

app.get('/api/pedidos/:id', requireAuth, (req, res) => {
  db.get(
    'SELECT * FROM pedidos WHERE id = ? AND (usuario_id = ? OR ? = "admin")',
    [req.params.id, req.session.usuarioId, req.session.tipo],
    (err, pedido) => {
      if (err || !pedido) {
        return res.status(404).json({ erro: 'Pedido não encontrado' });
      }

      db.all(
        `SELECT ip.*, p.nome, p.imagem 
         FROM itens_pedido ip 
         JOIN produtos p ON ip.produto_id = p.id 
         WHERE ip.pedido_id = ?`,
        [req.params.id],
        (err, itens) => {
          if (err) {
            return res.status(500).json({ erro: 'Erro ao buscar itens' });
          }
          res.json({ ...pedido, itens });
        }
      );
    }
  );
});

      res.json({ sucesso: true });

  );
});

async function verificarLogin() {
  try {
    const res = await fetch('/api/usuario-atual');
    const data = await res.json();
    
    if (data.usuario) {
      usuarioAtual = data.usuario;
      document.getElementById('menu-visitante').style.display = 'none';
      document.getElementById('menu-usuario').style.display = 'flex';
      document.getElementById('nome-usuario').textContent = `Olá, ${data.usuario.nome}`;
      
      if (data.usuario.tipo === 'admin') {
        document.getElementById('btn-admin').style.display = 'inline-block';
      }
    }
  } catch (erro) {
    console.error('Erro ao verificar login:', erro);
  }
}

async function fazerLogin(event) {
  event.preventDefault();
  
  const dados = {
    email: document.getElementById('login-email').value,
    senha: document.getElementById('login-senha').value
  };

  try {
    const res = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(dados)
    });

    const resultado = await res.json();

    if (resultado.sucesso) {
      mostrarMensagem('Login realizado com sucesso!', 'sucesso');
      usuarioAtual = resultado.usuario;
      setTimeout(() => {
        location.reload();
      }, 1000);
    } else {
      mostrarMensagem(resultado.erro, 'erro');
    }
  } catch (erro) {
    mostrarMensagem('Erro ao fazer login', 'erro');
  }
}

// ========== CARRINHO ==========

function carregarCarrinhoLocal() {
  const carrinhoSalvo = localStorage.getItem('carrinho');
  if (carrinhoSalvo) {
    carrinho = JSON.parse(carrinhoSalvo);
    atualizarQuantidadeCarrinho();
  }
}

function salvarCarrinhoLocal() {
  localStorage.setItem('carrinho', JSON.stringify(carrinho));
  atualizarQuantidadeCarrinho();
}

function atualizarQuantidadeCarrinho() {
  const total = carrinho.reduce((sum, item) => sum + item.quantidade, 0);
  document.getElementById('qtd-carrinho').textContent = total;
}

async function adicionarAoCarrinho(produtoId) {
  if (!usuarioAtual) {
    mostrarMensagem('Faça login para adicionar produtos ao carrinho', 'erro');
    mostrarLogin();
    return;
  }

  try {
    const res = await fetch(`/api/produtos/${produtoId}`);
    const produto = await res.json();

    const itemExistente = carrinho.find(item => item.produto_id === produtoId);
    
    if (itemExistente) {
      itemExistente.quantidade++;
    } else {
      carrinho.push({
        produto_id: produtoId,
        nome: produto.nome,
        preco: produto.preco,
        quantidade: 1
      });
    }

    salvarCarrinhoLocal();
    mostrarMensagem('Produto adicionado ao carrinho!', 'sucesso');
  } catch (erro) {
    mostrarMensagem('Erro ao adicionar produto', 'erro');
  }
}

function mostrarCarrinho() {
  esconderTodasTelas();
  document.getElementById('tela-carrinho').style.display = 'block';
  renderizarCarrinho();
}

function renderizarCarrinho() {
  const container = document.getElementById('itens-carrinho');
  const vazio = document.getElementById('carrinho-vazio');
  const comItens = document.getElementById('carrinho-com-itens');

  if (carrinho.length === 0) {
    container.innerHTML = '';
    vazio.style.display = 'block';
    comItens.style.display = 'none';
    return;
  }

  vazio.style.display = 'none';
  comItens.style.display = 'block';

  container.innerHTML = '';
  let total = 0;

  carrinho.forEach((item, index) => {
    total += item.preco * item.quantidade;

    const div = document.createElement('div');
    div.className = 'item-carrinho';
    div.innerHTML = `
      <div class="item-carrinho-info">
        <div class="item-carrinho-nome">${item.nome}</div>
        <div class="item-carrinho-preco">R$ ${item.preco.toFixed(2)}</div>
      </div>
      <div class="item-carrinho-acoes">
        <input type="number" min="1" value="${item.quantidade}" onchange="atualizarQuantidade(${index}, this.value)">
        <button class="btn-remover" onclick="removerDoCarrinho(${index})">Remover</button>
      </div>
    `;
    container.appendChild(div);
  });

  document.getElementById('total-carrinho').textContent = total.toFixed(2);
}

function atualizarQuantidade(index, novaQuantidade) {
  const qtd = parseInt(novaQuantidade);
  if (qtd > 0) {
    carrinho[index].quantidade = qtd;
    salvarCarrinhoLocal();
    renderizarCarrinho();
  }
}

function removerDoCarrinho(index) {
  carrinho.splice(index, 1);
  salvarCarrinhoLocal();
  renderizarCarrinho();
  mostrarMensagem('Produto removido do carrinho', 'sucesso');
}

async function finalizarPedido() {
  const endereco = document.getElementById('endereco').value;

  if (!endereco) {
    mostrarMensagem('Digite o endereço de entrega', 'erro');
    return;
  }

  const itens = carrinho.map(item => ({
    produto_id: item.produto_id,
    quantidade: item.quantidade
  }));

  try {
    const res = await fetch('/api/pedidos', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ itens, endereco })
    });

    const resultado = await res.json();

    if (resultado.sucesso) {
      mostrarMensagem('Pedido realizado com sucesso!', 'sucesso');
      carrinho = [];
      salvarCarrinhoLocal();
      document.getElementById('endereco').value = '';
      setTimeout(() => {
        mostrarMeusPedidos();
      }, 1500);
    } else {
      mostrarMensagem(resultado.erro, 'erro');
    }
  } catch (erro) {
    mostrarMensagem('Erro ao finalizar pedido', 'erro');
  }
}

function mostrarFormProduto() {

  document.getElementById('prod-id').value = '';
  document.querySelector('#form-produto form').reset();
}

function cancelarFormProduto() {
  document.getElementById('form-produto').style.display = 'none';
}

function editarProduto(produto) {
  document.getElementById('form-produto').style.display = 'block';
  document.getElementById('titulo-form-produto').textContent = 'Editar Produto';
  document.getElementById('prod-id').value = produto.id;
  document.getElementById('prod-nome').value = produto.nome;
  document.getElementById('prod-descricao').value = produto.descricao || '';
  document.getElementById('prod-preco').value = produto.preco;
  document.getElementById('prod-categoria').value = produto.categoria || '';
  document.getElementById('prod-estoque').value = produto.estoque;
  document.getElementById('prod-imagem').value = produto.imagem || '';
}

async function salvarProduto(event) {
  event.preventDefault();

  const id = document.getElementById('prod-id').value;
  const dados = {
    nome: document.getElementById('prod-nome').value,
    descricao: document.getElementById('prod-descricao').value,
    preco: parseFloat(document.getElementById('prod-preco').value),
    categoria: document.getElementById('prod-categoria').value,
    estoque: parseInt(document.getElementById('prod-estoque').value),
    imagem: document.getElementById('prod-imagem').value
  };

  try {
    const url = id ? `/api/produtos/${id}` : '/api/produtos';
    const method = id ? 'PUT' : 'POST';

    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(dados)
    });

    const resultado = await res.json();

    if (resultado.sucesso) {
      mostrarMensagem(id ? 'Produto atualizado!' : 'Produto criado!', 'sucesso');
      cancelarFormProduto();
      carregarAdminProdutos();
      carregarProdutos();
    } else {
      mostrarMensagem(resultado.erro, 'erro');
    }
  } catch (erro) {
    mostrarMensagem('Erro ao salvar produto', 'erro');
  }
}

async function excluirProduto(id) {
  if (!confirm('Tem certeza que deseja excluir este produto?')) return;

  try {
    const res = await fetch(`/api/produtos/${id}`, { method: 'DELETE' });
    const resultado = await res.json();

    if (resultado.sucesso) {
      mostrarMensagem('Produto excluído!', 'sucesso');
      carregarAdminProdutos();
      carregarProdutos();
    } else {
      mostrarMensagem(resultado.erro, 'erro');
    }
  } catch (erro) {
    mostrarMensagem('Erro ao excluir produto', 'erro');
  }
}

async function carregarAdminPedidos() {
  try {
    const res = await fetch('/api/todos-pedidos');
    const pedidos = await res.json();

    const container = document.getElementById('lista-admin-pedidos');
    container.innerHTML = '';

    if (pedidos.length === 0) {
      container.innerHTML = '<p>Nenhum pedido realizado ainda.</p>';
      return;
    }

    pedidos.forEach(pedido => {
      const card = document.createElement('div');
      card.className = 'pedido-card';
      
      const data = new Date(pedido.criado_em).toLocaleDateString('pt-BR');
      
      card.innerHTML = `
        <div class="pedido-header">
          <div class="pedido-numero">Pedido #${pedido.id}</div>
          <select class="select-status" onchange="alterarStatus(${pedido.id}, this.value)" onclick="event.stopPropagation()">
            <option value="pendente" ${pedido.status === 'pendente' ? 'selected' : ''}>Pendente</option>
            <option value="processando" ${pedido.status === 'processando' ? 'selected' : ''}>Processando</option>
            <option value="enviado" ${pedido.status === 'enviado' ? 'selected' : ''}>Enviado</option>
            <option value="entregue" ${pedido.status === 'entregue' ? 'selected' : ''}>Entregue</option>
            <option value="cancelado" ${pedido.status === 'cancelado' ? 'selected' : ''}>Cancelado</option>
          </select>
        </div>
        <div class="pedido-info">
          Cliente: ${pedido.cliente_nome} (${pedido.cliente_email})<br>
          Data: ${data}
        </div>
        <div class="pedido-total">Total: R$ ${pedido.total.toFixed(2)}</div>
      `;
      
      card.onclick = () => verDetalhesPedido(pedido.id);
      container.appendChild(card);
    });
  } catch (erro) {
    mostrarMensagem('Erro ao carregar pedidos', 'erro');
  }
}

async function alterarStatus(pedidoId, novoStatus) {
  try {
    const res = await fetch(`/api/pedidos/${pedidoId}/status`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ status: novoStatus })
    });

    const resultado = await res.json();

    if (resultado.sucesso) {
      mostrarMensagem('Status atualizado!', 'sucesso');
    } else {
      mostrarMensagem(resultado.erro, 'erro');
    }
  } catch (erro) {
    mostrarMensagem('Erro ao atualizar status', 'erro');
  }
}
