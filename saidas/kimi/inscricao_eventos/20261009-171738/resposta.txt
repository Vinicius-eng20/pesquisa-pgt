=== README.md ===
# Portal de Eventos

Um site simples para gerenciar inscrições em eventos. Funciona em computadores e celulares, com tudo salvo no próprio site (em um banco de dados SQLite no servidor).

## O que dá para fazer
- **Participantes**: podem ver os eventos, se inscrever, ver suas inscrições e cancelar.
- **Organizador**: pode criar, editar e cancelar eventos, ver todas as inscrições e confirmar presença dos participantes.

## Pré-requisitos
Você só precisa ter o **Node.js** instalado. Baixe em [https://nodejs.org](https://nodejs.org) (versão 18 ou superior). O resto o site instala sozinho.

## Como rodar
1. **Baixe o projeto**: copie todos os arquivos deste site para uma pasta no seu computador (por exemplo, `meu-portal`).
2. **Abra o terminal** (Prompt de Comando no Windows) dentro dessa pasta.
3. **Instale as dependências** (só na primeira vez):
   ```
   npm install
   ```
4. **Inicie o site**:
   ```
   npm start
   ```
5. Pronto! Acesse no navegador: `http://localhost:3000`.

## Primeiro acesso
- Na tela de login, clique em "Cadastre-se".
- Crie a conta do **Organizador** escolhendo a função "Organizador". O organizador pode criar os eventos.
- Crie contas de **Participantes** para as outras pessoas (ou elas mesmas criam), escolhendo a função "Participante".

## Para colocar na internet (opcional)
Se quiser que funcione na internet para outras pessoas acessarem de qualquer lugar, você precisa de um servidor (uma máquina virtual na nuvem, como na AWS, DigitalOcean, etc.).
1. Coloque os arquivos do projeto no servidor.
2. Instale o Node.js no servidor.
3. Rode `npm install` e `npm start`.
4. Libere a porta 3000 no firewall do servidor.

Feito isso, qualquer um com o endereço do servidor pode acessar.

## Detalhes técnicos (para curiosos)
- **Servidor**: Node.js com Express.
- **Banco de dados**: SQLite, salvo no arquivo `portal.db` na pasta do projeto. Tudo fica gravado aqui.
- **Login**: usa JWT para manter a sessão segura.
- **Frontend**: páginas HTML simples com JavaScript que conversa com o servidor.
- **Sem dependências externas**: todo o código e o banco de dados ficam na pasta do projeto.

=== server.js ===
```javascript
const express = require('express');
const sqlite3 = require('sqlite3').verbose();
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');
const path = require('path');

// Segredo para o token JWT. Em produção, use uma variável de ambiente.
const JWT_SECRET = 'segredo-super-secreto-mude-em-producao';

const app = express();
const PORT = 3000;

// Banco de dados SQLite
const db = new sqlite3.Database('./portal.db', (err) => {
  if (err) {
    console.error('Erro ao abrir o banco de dados:', err.message);
  } else {
    console.log('Conectado ao banco de dados SQLite.');
  }
});

// Criação das tabelas
db.serialize(() => {
  // Tabela de usuários
  db.run(`CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    password TEXT NOT NULL,
    role TEXT NOT NULL CHECK(role IN ('participante', 'organizador'))
  )`);

  // Tabela de eventos
  db.run(`CREATE TABLE IF NOT EXISTS events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    date TEXT NOT NULL,
    location TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'ativo' CHECK(status IN ('ativo', 'cancelado'))
  )`);

  // Tabela de inscrições
  db.run(`CREATE TABLE IF NOT EXISTS registrations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    event_id INTEGER NOT NULL,
    status TEXT NOT NULL DEFAULT 'inscrito' CHECK(status IN ('inscrito', 'presente', 'cancelado')),
    FOREIGN KEY (user_id) REFERENCES users (id),
    FOREIGN KEY (event_id) REFERENCES events (id),
    UNIQUE(user_id, event_id)
  )`);
});

// Middlewares
app.use(express.json());
app.use(express.static('public')); // Serve os arquivos da pasta 'public'

// Middleware para verificar o token de autenticação
function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1]; // Formato: "Bearer TOKEN"

  if (token == null) return res.sendStatus(401); // Não autorizado

  jwt.verify(token, JWT_SECRET, (err, user) => {
    if (err) return res.sendStatus(403); // Proibido
    req.user = user; // Adiciona os dados do usuário à requisição
    next();
  });
}

// Middleware para verificar se é organizador
function isOrganizer(req, res, next) {
  if (req.user.role !== 'organizador') {
    return res.status(403).send('Acesso negado. Somente organizadores.');
  }
  next();
}

// ====== ROTAS DE AUTENTICAÇÃO ======
app.post('/api/register', async (req, res) => {
  const { name, email, password, role } = req.body;
  if (!name || !email || !password || !role) {
    return res.status(400).json({ message: 'Todos os campos são obrigatórios.' });
  }

  try {
    const hashedPassword = await bcrypt.hash(password, 10);
    const sql = `INSERT INTO users (name, email, password, role) VALUES (?, ?, ?, ?)`;
    db.run(sql, [name, email, hashedPassword, role], function(err) {
      if (err) {
        return res.status(400).json({ message: 'Este e-mail já está em uso.' });
      }
      res.status(201).json({ message: 'Usuário criado com sucesso!' });
    });
  } catch (error) {
    res.status(500).json({ message: 'Erro no servidor.' });
  }
});

app.post('/api/login', (req, res) => {
  const { email, password } = req.body;
  const sql = `SELECT * FROM users WHERE email = ?`;
  db.get(sql, [email], async (err, user) => {
    if (err || !user) {
      return res.status(400).json({ message: 'E-mail ou senha inválidos.' });
    }
    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) {
      return res.status(400).json({ message: 'E-mail ou senha inválidos.' });
    }
    // Cria o token de acesso
    const accessToken = jwt.sign({ id: user.id, name: user.name, role: user.role }, JWT_SECRET, { expiresIn: '1d' });
    res.json({ accessToken, role: user.role, name: user.name });
  });
});

// ====== ROTAS DE EVENTOS ======
// Listar todos os eventos ativos (para participantes)
app.get('/api/events', authenticateToken, (req, res) => {
  const sql = `SELECT * FROM events WHERE status = 'ativo' ORDER BY date`;
  db.all(sql, [], (err, rows) => {
    if (err) return res.status(500).json({ message: 'Erro ao buscar eventos.' });
    res.json(rows);
  });
});

// Listar todos os eventos (para organizador, incluindo cancelados)
app.get('/api/admin/events', authenticateToken, isOrganizer, (req, res) => {
  const sql = `SELECT * FROM events ORDER BY date DESC`;
  db.all(sql, [], (err, rows) => {
    if (err) return res.status(500).json({ message: 'Erro ao buscar eventos.' });
    res.json(rows);
  });
});

// Criar um novo evento (somente organizador)
app.post('/api/events', authenticateToken, isOrganizer, (req, res) => {
  const { title, description, date, location } = req.body;
  const sql = `INSERT INTO events (title, description, date, location) VALUES (?, ?, ?, ?)`;
  db.run(sql, [title, description, date, location], function(err) {
    if (err) return res.status(500).json({ message: 'Erro ao criar evento.' });
    res.status(201).json({ id: this.lastID, message: 'Evento criado com sucesso!' });
  });
});

// Atualizar um evento (somente organizador)
app.put('/api/events/:id', authenticateToken, isOrganizer, (req, res) => {
  const { title, description, date, location } = req.body;
  const sql = `UPDATE events SET title = ?, description = ?, date = ?, location = ? WHERE id = ?`;
  db.run(sql, [title, description, date, location, req.params.id], function(err) {
    if (err) return res.status(500).json({ message: 'Erro ao atualizar evento.' });
    res.json({ message: 'Evento atualizado com sucesso!' });
  });
});

// Cancelar um evento (somente organizador)
app.put('/api/events/:id/cancel', authenticateToken, isOrganizer, (req, res) => {
  const sql = `UPDATE events SET status = 'cancelado' WHERE id = ?`;
  db.run(sql, [req.params.id], function(err) {
    if (err) return res.status(500).json({ message: 'Erro ao cancelar evento.' });
    res.json({ message: 'Evento cancelado com sucesso!' });
  });
});

// ====== ROTAS DE INSCRIÇÕES ======
// Participante se inscreve em um evento
app.post('/api/registrations', authenticateToken, (req, res) => {
  const { eventId } = req.body;
  const userId = req.user.id;
  const sql = `INSERT INTO registrations (user_id, event_id) VALUES (?, ?)`;
  db.run(sql, [userId, eventId], function(err) {
    if (err) {
      if (err.message.includes('UNIQUE constraint failed')) {
        return res.status(400).json({ message: 'Você já está inscrito neste evento.' });
      }
      return res.status(500).json({ message: 'Erro ao se inscrever.' });
    }
    res.status(201).json({ message: 'Inscrição realizada com sucesso!' });
  });
});

// Participante vê suas próprias inscrições
app.get('/api/my-registrations', authenticateToken, (req, res) => {
  const sql = `
    SELECT r.id as registration_id, r.status, e.id as event_id, e.title, e.date, e.location, e.status as event_status
    FROM registrations r
    JOIN events e ON r.event_id = e.id
    WHERE r.user_id = ? AND r.status != 'cancelado'
  `;
  db.all(sql, [req.user.id], (err, rows) => {
    if (err) return res.status(500).json({ message: 'Erro ao buscar inscrições.' });
    res.json(rows);
  });
});

// Participante cancela sua inscrição
app.put('/api/registrations/:id/cancel', authenticateToken, (req, res) => {
  const sql = `UPDATE registrations SET status = 'cancelado' WHERE id = ? AND user_id = ?`;
  db.run(sql, [req.params.id, req.user.id], function(err) {
    if (err) return res.status(500).json({ message: 'Erro ao cancelar inscrição.' });
    if (this.changes === 0) return res.status(404).json({ message: 'Inscrição não encontrada.' });
    res.json({ message: 'Inscrição cancelada com sucesso!' });
  });
});

// Organizador vê todas as inscrições de um evento
app.get('/api/events/:id/registrations', authenticateToken, isOrganizer, (req, res) => {
  const sql = `
    SELECT r.id, r.status, u.name, u.email
    FROM registrations r
    JOIN users u ON r.user_id = u.id
    WHERE r.event_id = ?
  `;
  db.all(sql, [req.params.id], (err, rows) => {
    if (err) return res.status(500).json({ message: 'Erro ao buscar inscrições.' });
    res.json(rows);
  });
});

// Organizador confirma a presença de um participante
app.put('/api/registrations/:id/confirm', authenticateToken, isOrganizer, (req, res) => {
  const sql = `UPDATE registrations SET status = 'presente' WHERE id = ?`;
  db.run(sql, [req.params.id], function(err) {
    if (err) return res.status(500).json({ message: 'Erro ao confirmar presença.' });
    if (this.changes === 0) return res.status(404).json({ message: 'Inscrição não encontrada.' });
    res.json({ message: 'Presença confirmada com sucesso!' });
  });
});

// Rota principal que serve o app
app.get('*', (req, res) => {
  res.sendFile(path.resolve(__dirname, 'public', 'index.html'));
});

app.listen(PORT, () => {
  console.log(`Servidor rodando na porta ${PORT}`);
  console.log(`Acesse http://localhost:${PORT} no seu navegador.`);
});
```

=== public/app.js ===
document.addEventListener('DOMContentLoaded', () => {
  const authSection = document.getElementById('auth-section');
  const mainSection = document.getElementById('main-section');
  const participantView = document.getElementById('participant-view');
  const organizerView = document.getElementById('organizer-view');

  const token = localStorage.getItem('token');
  const role = localStorage.getItem('role');
  const name = localStorage.getItem('name');

  // Se já estiver logado, mostra a tela principal
  if (token) {
    showMainSection(role, name);
  }

  // --- LÓGICA DE AUTENTICAÇÃO ---
  const showLoginBtn = document.getElementById('show-login-btn');
  const showRegisterBtn = document.getElementById('show-register-btn');
  const loginForm = document.getElementById('login-form');
  const registerForm = document.getElementById('register-form');

  showLoginBtn.addEventListener('click', () => {
    loginForm.style.display = 'block';
    registerForm.style.display = 'none';
    showLoginBtn.classList.add('active');
    showRegisterBtn.classList.remove('active');
  });

  showRegisterBtn.addEventListener('click', () => {
    loginForm.style.display = 'none';
    registerForm.style.display = 'block';
    showLoginBtn.classList.remove('active');
    showRegisterBtn.classList.add('active');
  });

  registerForm.addEventListener('submit', async (e) => {
    e.preventDefault();
    const userData = {
      name: document.getElementById('register-name').value,
      email: document.getElementById('register-email').value,
      password: document.getElementById('register-password').value,
      role: document.getElementById('register-role').value
    };
    try {
      const res = await fetch('/api/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(userData)
      });
      const data = await res.json();
      alert(data.message);
      if (res.ok) {
        registerForm.reset();
        showLoginBtn.click(); // Volta para a tela de login
      }
    } catch (error) {
      alert('Erro ao cadastrar.');
    }
  });

  loginForm.addEventListener('submit', async (e) => {
    e.preventDefault();
    const credentials = {
      email: document.getElementById('login-email').value,
      password: document.getElementById('login-password').value
    };
    try {
      const res = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(credentials)
      });
      const data = await res.json();
      if (res.ok) {
        localStorage.setItem('token', data.accessToken);
        localStorage.setItem('role', data.role);
        localStorage.setItem('name', data.name);
        showMainSection(data.role, data.name);
      } else {
        alert(data.message);
      }
    } catch (error) {
      alert('Erro ao fazer login.');
    }
  });

  document.getElementById('logout-btn').addEventListener('click', () => {
    localStorage.clear();
    window.location.reload();
  });

  function showMainSection(userRole, userName) {
    authSection.style.display = 'none';
    mainSection.style.display = 'block';
    document.getElementById('welcome-message').textContent = `Olá, ${userName}!`;

    if (userRole === 'participante') {
      participantView.style.display = 'block';
      loadAvailableEvents();
      loadMyRegistrations();
    } else if (userRole === 'organizador') {
      organizerView.style.display = 'block';
      loadAdminEvents();
    }
  }

  // --- LÓGICA DAS ABAS ---
  document.querySelectorAll('.tab-link').forEach(button => {
    button.addEventListener('click', () => {
      const tabId = button.getAttribute('data-tab');
      document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
      document.querySelectorAll('.tab-link').forEach(link => link.classList.remove('active'));
      document.getElementById(tabId).classList.add('active');
      button.classList.add('active');
    });
  });

  // --- FUNÇÕES DO PARTICIPANTE ---
  async function loadAvailableEvents() {
    const res = await fetch('/api/events', { headers: { 'Authorization': `Bearer ${token}` } });
    const events = await res.json();
    const list = document.getElementById('events-list');
    list.innerHTML = '';
    if (events.length === 0) {
      list.innerHTML = '<p>Nenhum evento disponível no momento.</p>';
      return;
    }
    events.forEach(event => {
      const card = document.createElement('div');
      card.className = 'event-card';
      card.innerHTML = `
        <h4>${event.title}</h4>
        <p><strong>Data:</strong> ${new Date(event.date).toLocaleString('pt-BR')}</p>
        <p><strong>Local:</strong> ${event.location}</p>
        <p>${event.description}</p>
        <button class="subscribe" data-event-id="${event.id}">Inscrever-se</button>
      `;
      list.appendChild(card);
    });
  }

  async function loadMyRegistrations() {
    const res = await fetch('/api/my-registrations', { headers: { 'Authorization': `Bearer ${token}` } });
    const registrations = await res.json();
    const list = document.getElementById('my-registrations-list');
    list.innerHTML = '';
    if (registrations.length === 0) {
      list.innerHTML = '<p>Você ainda não se inscreveu em nenhum evento.</p>';
      return;
    }
    registrations.forEach(reg => {
      const card = document.createElement('div');
      card.className = 'event-card';
      const statusText = reg.status === 'presente' ? '<span style="color: green;">Presença Confirmada</span>' : 'Inscrito';
      const eventStatus = reg.event_status === 'cancelado' ? '<p class="status-cancelado">Este evento foi cancelado.</p>' : '';
      card.innerHTML = `
        <h4>${reg.title}</h4>
        <p><strong>Data:</strong> ${new Date(reg.date).toLocaleString('pt-BR')}</p>
        <p><strong>Status:</strong> ${statusText}</p>
        ${eventStatus}
        <button class="cancel" data-reg-id="${reg.registration_id}">Cancelar Inscrição</button>
      `;
      list.appendChild(card);
    });
  }

  document.getElementById('events-list').addEventListener('click', async (e) => {
    if (e.target.classList.contains('subscribe')) {
      const eventId = e.target.getAttribute('data-event-id');
      const res = await fetch('/api/registrations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
        body: JSON.stringify({ eventId })
      });
      const data = await res.json();
      alert(data.message);
      if (res.ok) {
        loadMyRegistrations(); // Atualiza a lista de "Minhas Inscrições"
      }
    }
  });

  document.getElementById('my-registrations-list').addEventListener('click', async (e) => {
    if (e.target.classList.contains('cancel')) {
      const regId = e.target.getAttribute('data-reg-id');
      if (confirm('Tem certeza que deseja cancelar esta inscrição?')) {
        const res = await fetch(`/api/registrations/${regId}/cancel`, {
          method: 'PUT',
          headers: { 'Authorization': `Bearer ${token}` }
        });
        const data = await res.json();
        alert(data.message);
        if (res.ok) {
          loadMyRegistrations();
        }
      }
    }
  });

  // --- FUNÇÕES DO ORGANIZADOR ---
  const eventForm = document.getElementById('event-form');
  const cancelEditBtn = document.getElementById('cancel-edit-btn');

  async function loadAdminEvents() {
    const res = await fetch('/api/admin/events', { headers: { 'Authorization': `Bearer ${token}` } });
    const events = await res.json();
    const list = document.getElementById('admin-events-list');
    list.innerHTML = '';
    events.forEach(event => {
      const card = document.createElement('div');
      card.className = 'event-card';
      const status = event.status === 'cancelado' ? '<p class="status-cancelado">CANCELADO</p>' : '';
      card.innerHTML = `
        <h4>${event.title}</h4>
        <p><strong>Data:</strong> ${new Date(event.date).toLocaleString('pt-BR')}</p>
        ${status}
        <button class="edit" data-event='${JSON.stringify(event)}'>Editar</button>
        <button class="cancel" data-event-id="${event.id}" ${event.status === 'cancelado' ? 'disabled' : ''}>Cancelar Evento</button>
        <button class="details" data-event-id="${event.id}" data-event-title="${event.title}">Ver Inscrições</button>
      `;
      list.appendChild(card);
    });
  }

  eventForm.addEventListener('submit', async (e) => {
    e.preventDefault();
    const eventId = document.getElementById('event-id').value;
    const eventData = {
      title: document.getElementById('event-title').value,
      description: document.getElementById('event-description').value,
      date: document.getElementById('event-date').value,
      location: document.getElementById('event-location').value
    };

    const url = eventId ? `/api/events/${eventId}` : '/api/events';
    const method = eventId ? 'PUT' : 'POST';

    const res = await fetch(url, {
      method: method,
      headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
      body: JSON.stringify(eventData)
    });
    const data = await res.json();
    alert(data.message);
    if (res.ok) {
      resetEventForm();
      loadAdminEvents();
      // Muda para a aba de gerenciar para ver o resultado
      document.querySelector('[data-tab="manage-events"]').click();
    }
  });

  document.getElementById('admin-events-list').addEventListener('click', async (e) => {
    if (e.target.classList.contains('edit')) {
      const event = JSON.parse(e.target.getAttribute('data-event'));
      document.getElementById('form-title').textContent = 'Editar Evento';
      document.getElementById('event-id').value = event.id;
      document.getElementById('event-title').value = event.title;
      document.getElementById('event-description').value = event.description;
      document.getElementById('event-date').value = event.date;
      document.getElementById('event-location').value = event.location;
      cancelEditBtn.style.display = 'inline-block';
      document.querySelector('[data-tab="create-event"]').click();
    }

    if (e.target.classList.contains('cancel')) {
      const eventId = e.target.getAttribute('data-event-id');
      if (confirm('Tem certeza que deseja cancelar este evento?')) {
        const res = await fetch(`/api/events/${eventId}/cancel`, {
          method: 'PUT',
          headers: { 'Authorization': `Bearer ${token}` }
        });
        const data = await res.json();
        alert(data.message);
        if (res.ok) loadAdminEvents();
      }
    }

    if (e.target.classList.contains('details')) {
      const eventId = e.target.getAttribute('data-event-id');
      const eventTitle = e.target.getAttribute('data-event-title');
      openRegistrationsModal(eventId, eventTitle);
    }
  });

  cancelEditBtn.addEventListener('click', resetEventForm);

  function resetEventForm() {
    eventForm.reset();
    document.getElementById('event-id').value = '';
    document.getElementById('form-title').textContent = 'Criar Novo Evento';
    cancelEditBtn.style.display = 'none';
  }

  // --- LÓGICA DO MODAL DE INSCRIÇÕES ---
  const modal = document.getElementById('registrations-modal');
  const closeBtn = document.querySelector('.close-btn');

  async function openRegistrationsModal(eventId, eventTitle) {
    document.getElementById('modal-event-title').textContent = `Inscrições para: ${eventTitle}`;
    const res = await fetch(`/api/events/${eventId}/registrations`, { headers: { 'Authorization': `Bearer ${token}` } });
    const registrations = await res.json();
    const tableBody = document.querySelector('#registrations-table tbody');
    tableBody.innerHTML = '';
    if (registrations.length === 0) {
      tableBody.innerHTML = '<tr><td colspan="4">Nenhuma inscrição para este evento.</td></tr>';
    } else {
      registrations.forEach(reg => {
        const row = document.createElement('tr');
        row.innerHTML = `
          <td>${reg.name}</td>
          <td>${reg.email}</td>
          <td>${reg.status}</td>
          <td>
            ${reg.status === 'inscrito' ? `<button class="confirm-presence" data-reg-id="${reg.id}">Confirmar Presença</button>` : ''}
          </td>
        `;
        tableBody.appendChild(row);
      });
    }
    modal.style.display = 'block';
  }

  closeBtn.onclick = () => { modal.style.display = 'none'; }
  window.onclick = (event) => {
    if (event.target == modal) {
      modal.style.display = 'none';
    }
  }

  document.getElementById('registrations-table').addEventListener('click', async (e) => {
    if (e.target.classList.contains('confirm-presence')) {
      const regId = e.target.getAttribute('data-reg-id');
      const res = await fetch(`/api/registrations/${regId}/confirm`, {
        method: 'PUT',
        headers: { 'Authorization': `Bearer ${token}` }
      });
      const data = await res.json();
      alert(data.message);
      if (res.ok) {
        // Atualiza o modal sem fechá-lo
        const row = e.target.closest('tr');
        row.cells[2].textContent = 'presente';
        e.target.remove();
      }
    }
  });
});
