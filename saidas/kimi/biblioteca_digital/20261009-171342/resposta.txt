=== README.md ===
# Biblioteca Digital

Sistema completo de biblioteca digital com empréstimo de itens. Inclui área do leitor e do bibliotecário, com dados salvos no servidor (funciona em qualquer dispositivo).

## Conta do bibliotecário (já criada automaticamente)
- **E-mail:** admin@biblioteca.com
- **Senha:** admin123

Leitores se cadastram pela tela de "Cadastre-se".

## Como instalar e executar

### Pré-requisitos
- Node.js instalado (versão 16 ou superior). Baixe em: https://nodejs.org

### Passo a passo

1. Abra o terminal (ou Prompt de Comando) na pasta do projeto.

2. Instale as dependências:
```
npm install
```

3. Inicie o servidor:
```
npm start
```

4. Abra no navegador:
```
http://localhost:3000
```

## Como deixar disponível na internet (opcional)

Para que outras pessoas acessem de qualquer lugar, você pode usar o modo gratuito do Render, Railway ou similar. Como você pediu tudo dentro do próprio site sem criar conta em outros serviços, a forma mais simples é:

**Opção 1 - Na sua rede local (Wi-Fi):**
1. Descubra o IP do seu computador (ex: `ipconfig` no Windows ou `ifconfig` no Linux/Mac).
2. Inicie o servidor com: `npm start`
3. Outros aparelhos na mesma rede acessam: `http://SEU_IP:3000`

**Opção 2 - Na internet de verdade:**
Hospede em qualquer serviço de Node.js (Render, Railway, etc.) — o projeto está pronto, basta enviar os arquivos e definir o comando de start como `npm start`.

## Funcionalidades

### Leitor
- Criar conta e fazer login
- Buscar itens do acervo (por título, autor ou categoria)
- Ver detalhes e disponibilidade
- Pegar item emprestado (prazo de 14 dias)
- Ver seus empréstimos ativos e histórico
- Devolver itens

### Bibliotecário
- Cadastrar, editar e remover itens do acervo
- Ver todos os empréstimos de todos os leitores (ativos, atrasados, devolvidos)
- Bloquear/desbloquear leitores com atraso
- Ver painel com estatísticas

## Estrutura dos dados
Todos os dados ficam salvos no arquivo `biblioteca.db` (banco de dados SQLite), criado automaticamente na primeira execução. Nada se perde ao fechar o site.

## Para resetar tudo
Apague o arquivo `biblioteca.db` e reinicie o servidor.

CREATE TABLE IF NOT EXISTS usuarios (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  senha TEXT NOT NULL,
  papel TEXT NOT NULL DEFAULT 'leitor',

CREATE TABLE IF NOT EXISTS itens (
  id INTEGER PRIMARY KEY AUTOINCREMENT,

  autor TEXT NOT NULL DEFAULT '',
  categoria TEXT NOT NULL DEFAULT '',
  ano INTEGER,
  descricao TEXT NOT NULL DEFAULT '',
  quantidade INTEGER NOT NULL DEFAULT 1,

  criado_em TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE TABLE IF NOT EXISTS emprestimos (
  id INTEGER PRIMARY KEY AUTOINCREMENT,

  item_id INTEGER NOT NULL,

  data_prevista TEXT NOT NULL,
  data_devolucao TEXT,
  status TEXT NOT NULL DEFAULT 'ativo',
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
  FOREIGN KEY (item_id) REFERENCES itens(id)
);
`);

function limparAtrasados() {
  db.prepare(`UPDATE emprestimos SET status = 'atrasado'
    WHERE status = 'ativo' AND date(data_prevista) < date('now')`).run();
}

function reqLogin(req, res, next) {
  if (!req.session.usuario) return res.status(401).json({ erro: 'Faça login para continuar.' });
  next();
}

function reqBibliotecario(req, res, next) {
  if (!req.session.usuario) return res.status(401).json({ erro: 'Faça login para continuar.' });
  if (req.session.usuario.papel !== 'bibliotecario') return res.status(403).json({ erro: 'Acesso restrito ao bibliotecário.' });
  next();
}

function leitorTemAtraso(usuarioId) {
  limparAtrasados();
  const r = db.prepare(`SELECT COUNT(*) AS c FROM emprestimos WHERE usuario_id = ? AND status = 'atrasado'`).get(usuarioId);
  return r.c > 0;
}

app.post('/api/cadastro', (req, res) => {
  const { nome, email, senha } = req.body;
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha nome, e-mail e senha.' });
  if (senha.length < 4) return res.status(400).json({ erro: 'A senha precisa ter pelo menos 4 caracteres.' });
  const existe = db.prepare('SELECT id FROM usuarios WHERE email = ?').get(email.trim().toLowerCase());
  if (existe) return res.status(400).json({ erro: 'Este e-mail já está cadastrado.' });
  const hash = bcrypt.hashSync(senha, 10);
  const info = db.prepare('INSERT INTO usuarios (nome, email, senha, papel) VALUES (?, ?, ?, ?)')
    .run(nome.trim(), email.trim().toLowerCase(), hash, 'leitor');
  req.session.usuario = { id: info.lastInsertRowid, nome: nome.trim(), email: email.trim().toLowerCase(), papel: 'leitor' };
  res.json({ ok: true, usuario: req.session.usuario });
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;
  if (!email || !senha) return res.status(400).json({ erro: 'Informe e-mail e senha.' });
  const u = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email.trim().toLowerCase());
  if (!u || !bcrypt.compareSync(senha, u.senha)) return res.status(400).json({ erro: 'E-mail ou senha incorretos.' });
  req.session.usuario = { id: u.id, nome: u.nome, email: u.email, papel: u.papel };
  res.json({ ok: true, usuario: req.session.usuario });
});

app.post('/api/logout', (req, res) => {
  req.session.destroy(() => res.json({ ok: true }));

app.get('/api/itens', (req, res) => {
  limparAtrasados();
  const busca = (req.query.busca || '').trim();
  let itens;
  if (busca) {
    const like = `%${busca}%`;
    itens = db.prepare(`SELECT * FROM itens WHERE ativo = 1 AND (titulo LIKE ? OR autor LIKE ? OR categoria LIKE ?) ORDER BY titulo`).all(like, like, like);
  } else {
    itens = db.prepare('SELECT * FROM itens WHERE ativo = 1 ORDER BY titulo').all();
  }
  const disp = db.prepare(`SELECT item_id, COUNT(*) AS c FROM emprestimos WHERE status IN ('ativo','atrasado') GROUP BY item_id`).all();
  const mapa = {};
  disp.forEach(d => mapa[d.item_id] = d.c);
  itens = itens.map(i => ({
    ...i,
    emprestados: mapa[i.id] || 0,
    disponiveis: Math.max(0, i.quantidade - (mapa[i.id] || 0))
  }));
  res.json({ itens });
});

app.get('/api/itens/:id', (req, res) => {
  const item = db.prepare('SELECT * FROM itens WHERE id = ? AND ativo = 1').get(req.params.id);
  if (!item) return res.status(404).json({ erro: 'Item não encontrado.' });
  const emp = db.prepare(`SELECT COUNT(*) AS c FROM emprestimos WHERE item_id = ? AND status IN ('ativo','atrasado')`).get(item.id).c;
  res.json({ item: { ...item, emprestados: emp, disponiveis: Math.max(0, item.quantidade - emp) } });
});

// ============ EMPRÉSTIMOS (LEITOR) ============

app.post('/api/emprestar/:itemId', reqLogin, (req, res) => {
  limparAtrasados();
  const u = db.prepare('SELECT * FROM usuarios WHERE id = ?').get(req.session.usuario.id);
  if (u.bloqueado) return res.status(403).json({ erro: 'Sua conta está bloqueada para novos empréstimos. Procure o bibliotecário.' });
  if (leitorTemAtraso(u.id)) return res.status(403).json({ erro: 'Você possui itens em atraso. Devolva-os antes de pegar novos itens.' });

  const item = db.prepare('SELECT * FROM itens WHERE id = ? AND ativo = 1').get(req.params.itemId);
  if (!item) return res.status(404).json({ erro: 'Item não encontrado.' });

  const jaTem = db.prepare(`SELECT id FROM emprestimos WHERE usuario_id = ? AND item_id = ? AND status IN ('ativo','atrasado')`).get(u.id, item.id);
  if (jaTem) return res.status(400).json({ erro: 'Você já está com este item emprestado.' });

  const emp = db.prepare(`SELECT COUNT(*) AS c FROM emprestimos WHERE item_id = ? AND status IN ('ativo','atrasado')`).get(item.id).c;
  if (emp >= item.quantidade) return res.status(400).json({ erro: 'Não há exemplares disponíveis no momento.' });

  const ativos = db.prepare(`SELECT COUNT(*) AS c FROM emprestimos WHERE usuario_id = ? AND status IN ('ativo','atrasado')`).get(u.id).c;
  if (ativos >= 5) return res.status(400).json({ erro: 'Limite de 5 empréstimos simultâneos atingido.' });

  const prevista = new Date();
  prevista.setDate(prevista.getDate() + PRAZO_DIAS);
  db.prepare('INSERT INTO emprestimos (usuario_id, item_id, data_prevista) VALUES (?, ?, ?)')
    .run(u.id, item.id, prevista.toISOString().slice(0, 19).replace('T', ' '));
  res.json({ ok: true, mensagem: `Empréstimo realizado! Devolução até ${prevista.toLocaleDateString('pt-BR')}.` });
});

app.post('/api/devolver/:emprestimoId', reqLogin, (req, res) => {
  const emp = db.prepare('SELECT * FROM emprestimos WHERE id = ?').get(req.params.emprestimoId);
  if (!emp) return res.status(404).json({ erro: 'Empréstimo não encontrado.' });
  if (req.session.usuario.papel !== 'bibliotecario' && emp.usuario_id !== req.session.usuario.id) {
    return res.status(403).json({ erro: 'Você só pode devolver seus próprios itens.' });
  }
  if (emp.status === 'devolvido') return res.status(400).json({ erro: 'Este item já foi devolvido.' });
  db.prepare(`UPDATE emprestimos SET status = 'devolvido', data_devolucao = datetime('now') WHERE id = ?`).run(emp.id);
  res.json({ ok: true, mensagem: 'Devolução registrada com sucesso!' });
});

app.get('/api/meus-emprestimos', reqLogin, (req, res) => {
  limparAtrasados();
  const lista = db.prepare(`
    SELECT e.*, i.titulo, i.autor, i.categoria
    FROM emprestimos e JOIN itens i ON i.id = e.item_id
    WHERE e.usuario_id = ?
    ORDER BY e.status != 'devolvido' DESC, e.data_emprestimo DESC
  `).all(req.session.usuario.id);
  res.json({ emprestimos: lista });
});

app.post('/api/admin/itens', reqBibliotecario, (req, res) => {
  const { titulo, autor, categoria, ano, descricao, quantidade } = req.body;
  if (!titulo) return res.status(400).json({ erro: 'O título é obrigatório.' });
  const qtd = parseInt(quantidade) || 1;
  if (qtd < 1) return res.status(400).json({ erro: 'A quantidade deve ser pelo menos 1.' });
  const info = db.prepare('INSERT INTO itens (titulo, autor, categoria, ano, descricao, quantidade) VALUES (?, ?, ?, ?, ?, ?)')
    .run(titulo.trim(), (autor || '').trim(), (categoria || '').trim(), parseInt(ano) || null, (descricao || '').trim(), qtd);
  res.json({ ok: true, id: info.lastInsertRowid });
});

app.put('/api/admin/itens/:id', reqBibliotecario, (req, res) => {
  const item = db.prepare('SELECT * FROM itens WHERE id = ?').get(req.params.id);
  if (!item) return res.status(404).json({ erro: 'Item não encontrado.' });
  const { titulo, autor, categoria, ano, descricao, quantidade } = req.body;
  if (!titulo) return res.status(400).json({ erro: 'O título é obrigatório.' });
  const qtd = parseInt(quantidade);
  if (!qtd || qtd < 1) return res.status(400).json({ erro: 'A quantidade deve ser pelo menos 1.' });
  const emp = db.prepare(`SELECT COUNT(*) AS c FROM emprestimos WHERE item_id = ? AND status IN ('ativo','atrasado')`).get(item.id).c;
  if (qtd < emp) return res.status(400).json({ erro: `Há ${emp} exemplar(es) emprestado(s). A quantidade não pode ser menor que isso.` });
  db.prepare('UPDATE itens SET titulo=?, autor=?, categoria=?, ano=?, descricao=?, quantidade=? WHERE id=?')
    .run(titulo.trim(), (autor || '').trim(), (categoria || '').trim(), parseInt(ano) || null, (descricao || '').trim(), qtd, item.id);
  res.json({ ok: true });
});

app.delete('/api/admin/itens/:id', reqBibliotecario, (req, res) => {
  const item = db.prepare('SELECT * FROM itens WHERE id = ?').get(req.params.id);
  if (!item) return res.status(404).json({ erro: 'Item não encontrado.' });
  const emp = db.prepare(`SELECT COUNT(*) AS c FROM emprestimos WHERE item_id = ? AND status IN ('ativo','atrasado')`).get(item.id).c;
  if (emp > 0) return res.status(400).json({ erro: 'Não é possível remover: há exemplares emprestados. Aguarde as devoluções.' });
  db.prepare('UPDATE itens SET ativo = 0 WHERE id = ?').run(item.id);
  res.json({ ok: true });
});

app.get('/api/admin/emprestimos', reqBibliotecario, (req, res) => {
  limparAtrasados();
  const filtro = req.query.filtro || 'todos';
  let sql = `
    SELECT e.*, i.titulo, i.autor, u.nome AS leitor_nome, u.email AS leitor_email
    FROM emprestimos e
    JOIN itens i ON i.id = e.item_id
    JOIN usuarios u ON u.id = e.usuario_id
  `;
  if (filtro === 'ativos') sql += ` WHERE e.status = 'ativo'`;
  else if (filtro === 'atrasados') sql += ` WHERE e.status = 'atrasado'`;
  else if (filtro === 'devolvidos') sql += ` WHERE e.status = 'devolvido'`;
  sql += ` ORDER BY e.data_emprestimo DESC`;
  res.json({ emprestimos: db.prepare(sql).all() });
});

app.get('/api/admin/leitores', reqBibliotecario, (req, res) => {
  limparAtrasados();
  const leitores = db.prepare(`SELECT id, nome, email, bloqueado, criado_em FROM usuarios WHERE papel = 'leitor' ORDER BY nome`).all();
  const atrasos = db.prepare(`SELECT usuario_id, COUNT(*) AS c FROM emprestimos WHERE status = 'atrasado' GROUP BY usuario_id`).all();
  const emps = db.prepare(`SELECT usuario_id, COUNT(*) AS c FROM emprestimos WHERE status IN ('ativo','atrasado') GROUP BY usuario_id`).all();
  const ma = {}, me = {};
  atrasos.forEach(a => ma[a.usuario_id] = a.c);
  emps.forEach(e => me[e.usuario_id] = e.c);
  res.json({ leitores: leitores.map(l => ({ ...l, atrasados: ma[l.id] || 0, emprestimos_ativos: me[l.id] || 0 })) });
});

const areaUsuario = document.getElementById('area-usuario');

async function api(url, metodo = 'GET', corpo = null) {
  const opcoes = { method: metodo, headers: {} };
  if (corpo) {
    opcoes.headers['Content-Type'] = 'application/json';
    opcoes.body = JSON.stringify(corpo);
  }
  const resp = await fetch(url, opcoes);
  const dados = await resp.json().catch(() => ({}));
  if (!resp.ok) throw new Error(dados.erro || 'Erro inesperado. Tente novamente.');
  return dados;
}

function esc(t) {
  if (t === null || t === undefined) return '';
  return String(t).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

function fmtData(d) {
  if (!d) return '-';
  const dt = new Date(d.replace(' ', 'T') + (d.includes('Z') ? '' : 'Z'));
  return dt.toLocaleDateString('pt-BR');
}

      <button class="btn-sair btn-pequeno" onclick="sair()">Sair</button>
    `;
  } else {
    areaUsuario.innerHTML = '';
  }
}

}

async function renderAcervo(busca = '') {
  const cont = document.getElementById('conteudo');
  cont.innerHTML = `
    <div class="barra-busca">
      <input type="text" id="campo-busca" placeholder="Buscar por título, autor ou categoria..." value="${esc(busca)}">
      <button onclick="buscarItens()">Buscar</button>
    </div>
    <div id="lista-itens"><div class="vazio">Carregando...</div></div>
  `;
  const campo = document.getElementById('campo-busca');
  campo.addEventListener('keydown', e => { if (e.key === 'Enter') buscarItens(); });

  try {
    const dados = await api('/api/itens' + (busca ? '?busca=' + encodeURIComponent(busca) : ''));
    const lista = document.getElementById('lista-itens');
    if (dados.itens.length === 0) {
      lista.innerHTML = '<div class="caixa vazio">Nenhum item encontrado.</div>';
      return;
    }
    lista.innerHTML = '<div class="grade-itens">' + dados.itens.map(i => `
      <div class="cartao-item">
        <span class="categoria">${esc(i.categoria || 'Geral')}</span>
        <h3>${esc(i.titulo)}</h3>
        <div class="autor">${esc(i.autor)}${i.ano ? ' • ' + i.ano : ''}</div>
        <div class="descricao">${esc(i.descricao)}</div>
        <div class="disp ${i.disponiveis > 0 ? 'ok' : 'esgotado'}">
          ${i.disponiveis > 0 ? `✔ ${i.disponiveis} disponível(is)` : '✖ Indisponível no momento'}
        </div>
        <button ${i.disponiveis === 0 ? 'disabled' : ''} onclick="pegarEmprestado(${i.id})">Pegar emprestado</button>
      </div>
    `).join('') + '</div>';
  } catch (e) {
    document.getElementById('lista-itens').innerHTML = `<div class="caixa erro">${esc(e.message)}</div>`;
  }
}

async function renderMeusEmprestimos() {
  const cont = document.getElementById('conteudo');
  cont.innerHTML = '<div class="caixa vazio">Carregando...</div>';
  try {
    const dados = await api('/api/meus-emprestimos');
    const ativos = dados.emprestimos.filter(e => e.status !== 'devolvido');
    const historico = dados.emprestimos.filter(e => e.status === 'devolvido');

    let html = '<div class="caixa"><h2 class="secao">Itens comigo</h2>';
    if (ativos.length === 0) {
      html += '<div class="vazio">Você não tem itens emprestados no momento.</div>';
    } else {
      html += '<div class="rolagem-tabela"><table><tr><th>Item</th><th>Pego em</th><th>Devolver até</th><th>Situação</th><th></th></tr>';
      ativos.forEach(e => {
        html += `<tr>
          <td><strong>${esc(e.titulo)}</strong><br><small>${esc(e.autor)}</small></td>
          <td>${fmtData(e.data_emprestimo)}</td>
          <td>${fmtData(e.data_prevista)}</td>
          <td><span class="tag ${e.status}">${e.status === 'atrasado' ? 'Atrasado' : 'Em dia'}</span></td>
          <td><button class="btn-sucesso btn-pequeno" onclick="devolverItem(${e.id})">Devolver</button></td>
        </tr>`;
      });
      html += '</table></div>';
    }
    html += '</div>';

    html += '<div class="caixa"><h2 class="secao">Histórico de devoluções</h2>';
    if (historico.length === 0) {
      html += '<div class="vazio">Nenhuma devolução registrada ainda.</div>';
    } else {
      html += '<div class="rolagem-tabela"><table><tr><th>Item</th><th>Pego em</th><th>Devolvido em</th></tr>';
      historico.forEach(e => {
        html += `<tr>
          <td><strong>${esc(e.titulo)}</strong><br><small>${esc(e.autor)}</small></td>
          <td>${fmtData(e.data_emprestimo)}</td>
          <td>${fmtData(e.data_devolucao)}</td>
        </tr>`;
      });
      html += '</table></div>';
    }
    html += '</div>';
    cont.innerHTML = html;
  } catch (e) {
    cont.innerHTML = `<div class="caixa erro">${esc(e.message)}</div>`;
  }
}

async function salvarItem() {
  const id = document.getElementById('item-id').value;
  const corpo = {
    titulo: document.getElementById('item-titulo').value,

    categoria: document.getElementById('item-categoria').value,
    ano: document.getElementById('item-ano').value,
    descricao: document.getElementById('item-descricao').value,
    quantidade: document.getElementById('item-quantidade').value
  };
  const msg = document.getElementById('msg-form');
  msg.innerHTML = '';
  try {
    if (id) await api('/api/admin/itens/' + id, 'PUT', corpo);
    else await api('/api/admin/itens', 'POST', corpo);
    msg.innerHTML = `<div class="sucesso-msg">Item salvo com sucesso!</div>`;
    limparFormItem();
    carregarAdminItens();
    renderResumo();
  } catch (e) {
    msg.innerHTML = `<div class="erro">${esc(e.message)}</div>`;
  }
}

  }
}

async function renderAdminLeitores() {
  const cont = document.getElementById('conteudo');
  cont.innerHTML = '<div class="caixa"><h2 class="secao">Leitores cadastrados</h2><div id="lista-leitores"><div class="vazio">Carregando...</div></div></div>';
  try {
    const dados = await api('/api/admin/leitores');
    const lista = document.getElementById('lista-leitores');
    if (dados.leitores.length === 0) {
      lista.innerHTML = '<div class="vazio">Nenhum leitor cadastrado ainda.</div>';
      return;
    }
    lista.innerHTML = '<div class="rolagem-tabela"><table><tr><th>Nome</th><th>E-mail</th><th>Empréstimos ativos</th><th>Em atraso</th><th>Situação</th><th>Ações</th></tr>' +
      dados.leitores.map(l => `
        <tr>
          <td><strong>${esc(l.nome)}</strong></td>
          <td>${esc(l.email)}</td>
          <td>${l.emprestimos_ativos}</td>
          <td>${l.atrasados > 0 ? `<span class="tag atrasado">${l.atrasados}</span>` : '0'}</td>
          <td><span class="tag ${l.bloqueado ? 'bloqueado' : 'liberado'}">${l.bloqueado ? 'Bloqueado' : 'Liberado'}</span></td>
          <td>
            ${l.bloqueado
              ? `<button class="btn-sucesso btn-pequeno" onclick="desbloquearLeitor(${l.id})">Desbloquear</button>`
              : `<button class="btn-perigo btn-pequeno" onclick="bloquearLeitor(${l.id})">Bloquear</button>`}
          </td>
        </tr>
      `).join('') + '</table></div>';
  } catch (e) {
    document.getElementById('lista-leitores').innerHTML = `<div class="erro">${esc(e.message)}</div>`;
  }
}

async function bloquearLeitor(id) {
  try {
    const d = await api(`/api/admin/leitores/${id}/bloquear`, 'POST');
    alert(d.mensagem);
    renderAdminLeitores();
  } catch (e) {
    alert(e.message);
  }
}

async function desbloquearLeitor(id) {
  try {
    const d = await api(`/api/admin/leitores/${id}/desbloquear`, 'POST');
    alert(d.mensagem);
    renderAdminLeitores();
  } catch (e) {
    alert(e.message);
  }
}

// ============ INICIALIZAÇÃO ============

function render() {
  renderHeader();
  if (!estado.usuario) telaLogin();
  else if (estado.usuario.papel === 'bibliotecario') renderAdmin();
  else renderLeitor();
}

async function iniciar() {
  try {
    const dados = await api('/api/me');
    estado.usuario = dados.usuario;
    if (estado.usuario && estado.usuario.papel === 'bibliotecario' && !estado.aba.startsWith('admin')) {
      estado.aba = 'admin-acervo';
    }
    if (estado.usuario && estado.usuario.papel === 'leitor' && estado.aba.startsWith('admin')) {
      estado.aba = 'acervo';
    }
  } catch (e) {
    estado.usuario = null;
  }
  render();
}

iniciar();
