=== README.md ===
# Marketplace de Serviços Freelance

Projeto completo de um marketplace de serviços freelance com:

- cadastro e login de usuários;
- publicação de serviços;
- listagem de serviços aprovados;
- contratação de serviços;
- área do usuário com serviços e contratos;
- painel administrativo;
- aprovação e remoção de serviços;
- suspensão de usuários;
- persistência em banco de dados local SQLite;
- funcionamento em computador e celular.

## Requisitos

Instale no computador ou servidor:

- Node.js 18 ou superior

## Como instalar

```bash
npm install
```

## Como executar

```bash
npm start
```

Depois acesse no navegador:

```text
http://localhost:3000
```

## Conta de administrador padrão

Ao iniciar o sistema pela primeira vez, será criada automaticamente a conta:

```text
E-mail: admin@admin.com
Senha: admin123
```

Recomenda-se trocar a senha diretamente criando outro administrador ou alterando no banco em uma instalação real.

## Observações importantes

Este projeto salva os dados em um arquivo SQLite chamado:

```text
database.sqlite
```

Esse arquivo fica no próprio servidor onde o site está rodando. Portanto:

- se abrir em outro computador ou celular acessando o mesmo endereço do servidor, os dados serão os mesmos;
- se fechar e abrir novamente o site, os dados continuarão salvos;
- para funcionar na internet, coloque este projeto em um servidor com Node.js e acesse pelo domínio ou IP do servidor.

## Comandos úteis

Instalar dependências:

```bash
npm install
```

Rodar o projeto:

```bash
npm start
```

Rodar em modo desenvolvimento:

```bash
npm run dev
```

app.use(async (req, res, next) => {
  res.locals.user = req.session.user || null;
  res.locals.error = req.session.error || null;
  res.locals.success = req.session.success || null;
  delete req.session.error;
  delete req.session.success;

function requireAuth(req, res, next) {
  if (!req.session.user) {
    req.session.error = 'Entre na sua conta para continuar.';
    return res.redirect('/login');
  }
  next();
}

function requireAdmin(req, res, next) {
  if (!req.session.user || req.session.user.role !== 'admin') {
    req.session.error = 'Acesso restrito ao administrador.';
    return res.redirect('/');
  }
  next();
}

async function refreshUser(req) {
  if (!req.session.user) return null;
  const user = await db.get('SELECT id, name, email, role, suspended FROM users WHERE id = ?', [req.session.user.id]);
  if (!user || user.suspended) {
    req.session.destroy(() => {});
    return null;
  }
  req.session.user = user;
  return user;
}

  res.redirect('/login');

app.post('/login', async (req, res) => {
  const { email, password } = req.body;

  const user = await db.get('SELECT id, name, email, password_hash, role, suspended FROM users WHERE email = ?', [email]);
  if (!user) {
    req.session.error = 'E-mail ou senha inválidos.';
    return res.redirect('/login');
  }

  if (user.suspended) {
    req.session.error = 'Sua conta está suspensa.';
    return res.redirect('/login');
  }

  const ok = await bcrypt.compare(password, user.password_hash);
  if (!ok) {
    req.session.error = 'E-mail ou senha inválidos.';
    return res.redirect('/login');
  }

  req.session.user = {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role,
    suspended: user.suspended
  };

  res.redirect('/');
});

app.post('/services/:id/hire', requireAuth, async (req, res) => {
  const user = await refreshUser(req);
  if (!user) return res.redirect('/login');

  const service = await db.get(`
    SELECT services.*, users.suspended AS owner_suspended
    FROM services
    JOIN users ON users.id = services.user_id
    WHERE services.id = ?
  `, [req.params.id]);

  if (!service || service.status !== 'approved' || service.owner_suspended) {
    req.session.error = 'Serviço não encontrado.';
    return res.redirect('/');
  }

  if (service.user_id === user.id) {
    req.session.error = 'Você não pode contratar seu próprio serviço.';
    return res.redirect('/services/' + service.id);
  }

  await db.run(`
    INSERT INTO contracts (service_id, provider_id, client_id, status, created_at)
    VALUES (?, ?, ?, 'active', datetime("now"))
  `, [service.id, service.user_id, user.id]);

  req.session.success = 'Serviço contratado com sucesso.';
  res.redirect('/dashboard');
});

app.post('/contracts/:id/status', requireAuth, async (req, res) => {
  const user = await refreshUser(req);
  if (!user) return res.redirect('/login');

  const { status } = req.body;
  if (!['active', 'completed', 'cancelled'].includes(status)) {
    req.session.error = 'Status inválido.';
    return res.redirect('/dashboard');
  }

  const contract = await db.get('SELECT * FROM contracts WHERE id = ?', [req.params.id]);
  if (!contract || (contract.provider_id !== user.id && contract.client_id !== user.id)) {
    req.session.error = 'Contrato não encontrado.';
    return res.redirect('/dashboard');
  }

  await db.run('UPDATE contracts SET status = ? WHERE id = ?', [status, contract.id]);
  req.session.success = 'Contrato atualizado.';
  res.redirect('/dashboard');
});

=== src/db.js ===
const sqlite3 = require('sqlite3').verbose();
const bcrypt = require('bcryptjs');

const database = new sqlite3.Database('./database.sqlite');

function run(sql, params = []) {
  return new Promise((resolve, reject) => {
    database.run(sql, params, function (err) {
      if (err) reject(err);
      else resolve(this);
    });
  });
}

function get(sql, params = []) {
  return new Promise((resolve, reject) => {
    database.get(sql, params, function (err, row) {
      if (err) reject(err);
      else resolve(row);
    });
  });
}

function all(sql, params = []) {
  return new Promise((resolve, reject) => {
    database.all(sql, params, function (err, rows) {
      if (err) reject(err);
      else resolve(rows);
    });
  });
}

async function init() {
  await run(`
    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      email TEXT NOT NULL UNIQUE,
      password_hash TEXT NOT NULL,
      role TEXT NOT NULL DEFAULT 'user',
      suspended INTEGER NOT NULL DEFAULT 0,
      created_at TEXT NOT NULL
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS services (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      user_id INTEGER NOT NULL,
      title TEXT NOT NULL,
      description TEXT NOT NULL,
      price REAL NOT NULL,
      status TEXT NOT NULL DEFAULT 'pending',
      created_at TEXT NOT NULL,
      FOREIGN KEY (user_id) REFERENCES users(id)
    )
  `);

  await run(`
    CREATE TABLE IF NOT EXISTS contracts (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      service_id INTEGER NOT NULL,
      provider_id INTEGER NOT NULL,
      client_id INTEGER NOT NULL,
      status TEXT NOT NULL DEFAULT 'active',
      created_at TEXT NOT NULL,
      FOREIGN KEY (service_id) REFERENCES services(id),
      FOREIGN KEY (provider_id) REFERENCES users(id),
      FOREIGN KEY (client_id) REFERENCES users(id)
    )
  `);

  const admin = await get('SELECT id FROM users WHERE email = ?', ['admin@admin.com']);
  if (!admin) {
    const passwordHash = await bcrypt.hash('admin123', 10);
    await run(
      'INSERT INTO users (name, email, password_hash, role, suspended, created_at) VALUES (?, ?, ?, ?, 0, datetime("now"))',
      ['Administrador', 'admin@admin.com', passwordHash, 'admin']
    );
  }
}

module.exports = {
  run,
  get,
  all,
  init
};
