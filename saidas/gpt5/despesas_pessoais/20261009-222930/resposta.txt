=== README.md ===
# Controle de Despesas Pessoais

Site completo para controle de despesas pessoais com usuários comuns e administrador.

## O que o sistema faz

- Usuários podem:
  - Criar conta
  - Entrar no sistema
  - Registrar despesas
  - Escolher categoria da despesa
  - Ver total gasto no mês
  - Ver histórico de despesas

- Administrador pode:
  - Entrar no painel administrativo
  - Cadastrar categorias de gastos
  - Ativar ou desativar categorias
  - Desativar contas de usuários
  - Reativar contas de usuários
  - Ver total de gastos de todos os usuários
  - Ver total de usuários e despesas

- Os dados ficam salvos em banco SQLite no servidor.
- Funciona em computador e celular.
- Não depende de serviços externos.

## Requisitos

Você precisa ter instalado:

- Node.js 18 ou superior

Para verificar se tem Node.js instalado:

```bash
node -v
```

## Como instalar

Na pasta do projeto, execute:

```bash
npm install
```

## Como iniciar o site

Execute:

```bash
npm start
```

Depois abra no navegador:

```text
http://localhost:3000
```

## Conta de administrador inicial

Ao iniciar o sistema pela primeira vez, ele cria automaticamente um administrador:

```text
E-mail: admin@admin.com
Senha: admin123
```

Depois de entrar, recomenda-se alterar diretamente criando outro administrador no banco ou modificando a senha no código antes de usar publicamente.

## Observação importante para uso real na internet

Este projeto salva tudo em um arquivo chamado:

```text
database.sqlite
```

Ele fica na própria pasta do projeto no servidor.

Para colocar na internet, você precisa hospedar este projeto em um servidor que rode Node.js e mantenha o arquivo `database.sqlite` salvo entre reinicializações.

const Database = require("better-sqlite3");
const bcrypt = require("bcryptjs");

const db = new Database("database.sqlite");

db.pragma("journal_mode = WAL");

db.exec(`
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  role TEXT NOT NULL DEFAULT 'user',
  active INTEGER NOT NULL DEFAULT 1,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS categories (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL UNIQUE,
  active INTEGER NOT NULL DEFAULT 1,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS expenses (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  category_id INTEGER NOT NULL,
  description TEXT NOT NULL,
  amount INTEGER NOT NULL,
  expense_date TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY(user_id) REFERENCES users(id),
  FOREIGN KEY(category_id) REFERENCES categories(id)
);
`);

}

function requireLogin(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({ error: "Você precisa entrar no sistema." });
  }

  const user = db
    .prepare("SELECT id, name, email, role, active FROM users WHERE id = ?")
    .get(req.session.userId);

  if (!user || !user.active) {
    req.session.destroy(() => {});
    return res.status(403).json({ error: "Sua conta está desativada." });
  }

  req.user = user;
  next();
}

function requireAdmin(req, res, next) {
  requireLogin(req, res, () => {
    if (req.user.role !== "admin") {
      return res.status(403).json({ error: "Acesso permitido apenas para administrador." });
    }
    next();
  });
}

function centsFromValue(value) {
  const number = Number(value);
  if (!Number.isFinite(number) || number <= 0) {
    return null;
  }
  return Math.round(number * 100);
}

function formatExpense(row) {
  return {

    expense_date: row.expense_date,
    category_id: row.category_id,
    category_name: row.category_name,
    created_at: row.created_at
  };
}

app.post("/api/register", (req, res) => {
  const name = String(req.body.name || "").trim();
  const email = String(req.body.email || "").trim().toLowerCase();
  const password = String(req.body.password || "");

  if (!name || !email || !password) {
    return res.status(400).json({ error: "Preencha nome, e-mail e senha." });
  }

  if (password.length < 6) {
    return res.status(400).json({ error: "A senha precisa ter pelo menos 6 caracteres." });
  }

  const exists = db.prepare("SELECT id FROM users WHERE email = ?").get(email);
  if (exists) {
    return res.status(400).json({ error: "Este e-mail já está cadastrado." });
  }

  const passwordHash = bcrypt.hashSync(password, 10);

  const result = db.prepare(`
    INSERT INTO users (name, email, password_hash, role, active)
    VALUES (?, ?, ?, 'user', 1)
  `).run(name, email, passwordHash);

  req.session.userId = result.lastInsertRowid;

  res.json({ ok: true });
});

app.post("/api/login", (req, res) => {

  res.json({
    ok: true,

  });
});

app.get("/api/expenses", requireLogin, (req, res) => {
  const month = String(req.query.month || "").trim();

  let start;
  let end;

  if (/^\d{4}-\d{2}$/.test(month)) {
    start = `${month}-01`;
    const [year, monthNumber] = month.split("-").map(Number);
    end = new Date(year, monthNumber, 1).toISOString().slice(0, 10);
  } else {
    const now = new Date();
    const currentMonth = String(now.getMonth() + 1).padStart(2, "0");
    start = `${now.getFullYear()}-${currentMonth}-01`;
    end = new Date(now.getFullYear(), now.getMonth() + 1, 1).toISOString().slice(0, 10);
  }

  const rows = db.prepare(`
    SELECT e.*, c.name AS category_name
    FROM expenses e
    JOIN categories c ON c.id = e.category_id
    WHERE e.user_id = ?
      AND e.expense_date >= ?
      AND e.expense_date < ?
    ORDER BY e.expense_date DESC, e.id DESC
  `).all(req.user.id, start, end);

  const totalCents = rows.reduce((sum, row) => sum + row.amount, 0);

  res.json({
    expenses: rows.map(formatExpense),
    total: totalCents / 100
  });
});

function money(value) {
  return Number(value || 0).toLocaleString("pt-BR", {
    style: "currency",
    currency: "BRL"
  });
}

function today() {
  return new Date().toISOString().slice(0, 10);
}

function currentMonth() {
  const now = new Date();
  return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}`;
}

function showMessage(id, text, type = "error") {
  const el = document.getElementById(id);
  if (!el) return;
  el.className = `message ${type}`;
  el.textContent = text;
}

function clearMessage(id) {
  const el = document.getElementById(id);
  if (!el) return;
  el.className = "message";
  el.textContent = "";
}

async function api(path, options = {}) {
  const response = await fetch(path, {
    headers: {
      "Content-Type": "application/json",
      ...(options.headers || {})
    },
    credentials: "same-origin",
    ...options
  });

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(data.error || "Ocorreu um erro.");
  }

  return data;
}

async function logout() {
  await api("/api/logout", { method: "POST" });
  window.location.href = "/login.html";
}

async function requirePage(role) {
  try {
    const data = await api("/api/me");
    const userName = document.getElementById("userName");
    if (userName) userName.textContent = data.user.name;

    if (role && data.user.role !== role) {
      window.location.href = data.user.role === "admin" ? "/admin.html" : "/dashboard.html";
    }

    return data.user;
  } catch {
    window.location.href = "/login.html";
  }
}

async function initDashboard() {
  await requirePage("user");

  const dateInput = document.getElementById("expense_date");
  const monthInput = document.getElementById("month");
  dateInput.value = today();
  monthInput.value = currentMonth();

  await loadCategories();
  await loadExpenses();

  document.getElementById("expenseForm").addEventListener("submit", async (event) => {
    event.preventDefault();
    clearMessage("expenseMessage");

    try {
      await api("/api/expenses", {
        method: "POST",
        body: JSON.stringify({
          description: document.getElementById("description").value,
          amount: document.getElementById("amount").value,
          category_id: document.getElementById("category_id").value,
          expense_date: document.getElementById("expense_date").value
        })
      });

      document.getElementById("description").value = "";
      document.getElementById("amount").value = "";
      document.getElementById("expense_date").value = today();

      showMessage("expenseMessage", "Despesa registrada com sucesso.", "success");
      await loadExpenses();
    } catch (error) {
      showMessage("expenseMessage", error.message);
    }
  });

  monthInput.addEventListener("change", loadExpenses);
}

async function loadCategories() {
  const data = await api("/api/categories");
  const select = document.getElementById("category_id");
  select.innerHTML = "";

  data.categories.forEach((category) => {
    const option = document.createElement("option");
    option.value = category.id;
    option.textContent = category.name;
    select.appendChild(option);
  });
}

async function loadExpenses() {
  const month = document.getElementById("month").value;
  const data = await api(`/api/expenses?month=${encodeURIComponent(month)}`);

  document.getElementById("monthTotal").textContent = money(data.total);

  const tbody = document.getElementById("expensesBody");
  tbody.innerHTML = "";

  if (data.expenses.length === 0) {
    tbody.innerHTML = `<tr><td colspan="5">Nenhuma despesa encontrada neste mês.</td></tr>`;
    return;
  }

  data.expenses.forEach((expense) => {
    const tr = document.createElement("tr");
    tr.innerHTML = `
      <td>${expense.expense_date.split("-").reverse().join("/")}</td>
      <td>${escapeHtml(expense.description)}</td>
      <td>${escapeHtml(expense.category_name)}</td>
      <td>${money(expense.amount)}</td>
      <td><button class="danger" data-id="${expense.id}">Excluir</button></td>
    `;

    tr.querySelector("button").addEventListener("click", async () => {
      if (!confirm("Deseja excluir esta despesa?")) return;
      await api(`/api/expenses/${expense.id}`, { method: "DELETE" });
      await loadExpenses();
    });

    tbody.appendChild(tr);
  });
}

async function initAdmin() {
  await requirePage("admin");

  await loadAdminAll();

  document.getElementById("categoryForm").addEventListener("submit", async (event) => {
    event.preventDefault();
    clearMessage("categoryMessage");

    try {
      await api("/api/admin/categories", {
        method: "POST",
        body: JSON.stringify({
          name: document.getElementById("categoryName").value
        })
      });

      document.getElementById("categoryName").value = "";
      showMessage("categoryMessage", "Categoria criada com sucesso.", "success");
      await loadAdminAll();
    } catch (error) {
      showMessage("categoryMessage", error.message);
    }
  });
}

async function loadAdminAll() {
  await Promise.all([loadAdminSummary(), loadAdminUsers(), loadAdminCategories()]);
}

async function loadAdminSummary() {
  const data = await api("/api/admin/summary");

  document.getElementById("totalExpenses").textContent = money(data.totalExpenses);
  document.getElementById("totalUsers").textContent = data.totalUsers;
  document.getElementById("expenseCount").textContent = data.expenseCount;
}

async function loadAdminUsers() {
  const data = await api("/api/admin/users");
  const tbody = document.getElementById("usersBody");
  tbody.innerHTML = "";

  data.users.forEach((user) => {
    const tr = document.createElement("tr");

    const isAdmin = user.role === "admin";
    const nextActive = user.active ? false : true;

    tr.innerHTML = `
      <td>${escapeHtml(user.name)}</td>
      <td>${escapeHtml(user.email)}</td>
      <td>${isAdmin ? "Administrador" : "Usuário"}</td>
      <td><span class="badge ${user.active ? "active" : "inactive"}">${user.active ? "Ativo" : "Desativado"}</span></td>
      <td>${money(user.total_spent)}</td>
      <td>
        ${
          isAdmin
            ? "-"
            : `<button class="${user.active ? "danger" : "success"}">${user.active ? "Desativar" : "Reativar"}</button>`
        }
      </td>
    `;

    const button = tr.querySelector("button");
    if (button) {
      button.addEventListener("click", async () => {
        await api(`/api/admin/users/${user.id}/active`, {
          method: "PATCH",
          body: JSON.stringify({ active: nextActive })
        });
        await loadAdminUsers();
      });
    }

    tbody.appendChild(tr);
  });
}

async function loadAdminCategories() {
  const data = await api("/api/admin/categories");
  const tbody = document.getElementById("categoriesBody");
  tbody.innerHTML = "";

  data.categories.forEach((category) => {
    const tr = document.createElement("tr");
    const nextActive = category.active ? false : true;

    tr.innerHTML = `
      <td>${escapeHtml(category.name)}</td>
      <td><span class="badge ${category.active ? "active" : "inactive"}">${category.active ? "Ativa" : "Desativada"}</span></td>
      <td>${category.expense_count}</td>
      <td>${money(category.total_spent)}</td>
      <td><button class="${category.active ? "danger" : "success"}">${category.active ? "Desativar" : "Reativar"}</button></td>
    `;

    tr.querySelector("button").addEventListener("click", async () => {
      await api(`/api/admin/categories/${category.id}/active`, {
        method: "PATCH",
        body: JSON.stringify({ active: nextActive })
      });
      await loadAdminCategories();
      await loadCategoriesIfExists();
    });

    tbody.appendChild(tr);
  });
}

async function loadCategoriesIfExists() {
  if (document.getElementById("category_id")) {
    await loadCategories();
  }
}

function escapeHtml(value) {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}
