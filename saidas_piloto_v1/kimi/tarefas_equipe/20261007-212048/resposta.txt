=== README.md ===

# Sistema de Gestão de Tarefas de Equipe

Um sistema completo e autônomo de gestão de tarefas. Funciona 100% online, salva os dados de verdade e não precisa de nenhum serviço externo.

## O que você precisa ter instalado

Você só precisa de uma coisa: o **Node.js**.

1. Acesse [https://nodejs.org](https://nodejs.org)
2. Baixe a versão "LTS" (recomendada) e instale (apenas clique em "Avançar" até o fim).
3. Para confirmar que deu certo, abra o terminal (ou Prompt de Comando) e digite:
   ```
   node -v
   ```
   Se aparecer um número de versão (ex: `v20.11.0`), está tudo certo.

## Como executar o projeto no seu computador

1. Crie uma pasta no seu computador (ex: `gestao-tarefas`).
2. Coloque todos os arquivos deste projeto dentro dela.
3. Abra o terminal (ou Prompt de Comando) dentro dessa pasta.
4. Digite o comando abaixo e aperte Enter:
   ```
   npm install
   ```
5. Depois, digite este comando e aperte Enter:
   ```
   npm start
   ```
6. Pronto! Abra seu navegador e acesse:
   ```
   http://localhost:3000
   ```

**Importante:** Para parar o sistema, volte ao terminal e aperte `Ctrl + C`.

## Como colocar na internet (para a equipe usar de qualquer lugar)

Para o site funcionar 24h na internet, você precisa de um servidor. A forma mais fácil para quem não programa:

1. Crie uma conta no **Render.com** (é gratuito).
2. Crie um novo "Web Service" e conecte a este projeto.
3. Nas configurações, use:
   - **Build Command:** `npm install`
   - **Start Command:** `npm start`
4. O Render vai te dar um link (ex: `https://minha-equipe.onrender.com`) para você compartilhar com a equipe.

## Como usar pela primeira vez

1. Acesse o site e clique em **"Criar nova conta"**.
2. A primeira pessoa a se cadastrar deve escolher o perfil **"Gerente de Equipe"**.
3. Os outros membros se cadastram como **"Membro da Equipe"**.
4. O Gerente cria e distribui as tarefas. Os Membros veem e atualizam.

**Onde os dados ficam salvos?**
Tudo é salvo automaticamente em um arquivo chamado `banco-de-dados.json` dentro da pasta do projeto. Mesmo se você fechar e abrir o site, ou reiniciar o computador, os dados continuam lá.

=== server.js ===

```javascript
const express = require('express');
const session = require('express-session');
const bcrypt = require('bcryptjs');
const { v4: uuidv4 } = require('uuid');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;
const DATA_FILE = path.join(__dirname, 'banco-de-dados.json');

// Configuração do servidor
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static('public'));

// Configuração da sessão (mantém o usuário logado)
app.use(session({
  secret: 'chave-secreta-muito-segura-mude-isso-em-producao',
  resave: false,
  saveUninitialized: false,
  cookie: { maxAge: 24 * 60 * 60 * 1000 } // 24 horas
}));

// --- FUNÇÕES DO BANCO DE DADOS (Arquivo JSON) ---

// Inicializa o banco de dados se não existir
function initDatabase() {
  if (!fs.existsSync(DATA_FILE)) {
    const initialData = {
      users: [],
      tasks: []
    };
    fs.writeFileSync(DATA_FILE, JSON.stringify(initialData, null, 2));
    console.log('Banco de dados criado: ' + DATA_FILE);
  }
}

// Lê os dados do arquivo
function readDatabase() {
  try {
    const data = fs.readFileSync(DATA_FILE, 'utf8');
    return JSON.parse(data);
  } catch (err) {
    console.error('Erro ao ler banco de dados:', err);
    return { users: [], tasks: [] };
  }
}

// Salva os dados no arquivo
function writeDatabase(data) {
  try {
    fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2));
  } catch (err) {
    console.error('Erro ao salvar banco de dados:', err);
  }
}

// Inicializa o banco na inicialização do servidor
initDatabase();

// --- MIDDLEWARE DE AUTENTICAÇÃO ---

// Verifica se o usuário está logado
function requireAuth(req, res, next) {
  if (req.session.userId) {
    next();
  } else {
    res.status(401).json({ error: 'Não autorizado. Faça login.' });
  }
}

// Verifica se o usuário é gerente
function requireManager(req, res, next) {
  const db = readDatabase();
  const user = db.users.find(u => u.id === req.session.userId);
  
  if (user && user.role === 'manager') {
    next();
  } else {
    res.status(403).json({ error: 'Acesso negado. Apenas gerentes.' });
  }
}

// --- ROTAS DE AUTENTICAÇÃO ---

// Registro de novo usuário
app.post('/api/register', async (req, res) => {
  const { name, email, password, role } = req.body;
  
  if (!name || !email || !password || !role) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios.' });
  }
  
  const db = readDatabase();
  
  // Verifica se o email já existe
  if (db.users.some(u => u.email === email)) {
    return res.status(400).json({ error: 'Este email já está cadastrado.' });
  }
  
  // Criptografa a senha
  const hashedPassword = await bcrypt.hash(password, 10);
  
  const newUser = {
    id: uuidv4(),
    name,
    email,
    password: hashedPassword,
    role: role === 'manager' ? 'manager' : 'member', // 'manager' ou 'member'
    createdAt: new Date().toISOString()
  };
  
  db.users.push(newUser);
  writeDatabase(db);
  
  // Faz login automático
  req.session.userId = newUser.id;
  
  res.json({ 
    message: 'Conta criada com sucesso!', 
    user: { id: newUser.id, name: newUser.name, email: newUser.email, role: newUser.role } 
  });
});

// Login
app.post('/api/login', async (req, res) => {
  const { email, password } = req.body;
  
  const db = readDatabase();
  const user = db.users.find(u => u.email === email);
  
  if (!user) {
    return res.status(400).json({ error: 'Email ou senha incorretos.' });
  }
  
  const validPassword = await bcrypt.compare(password, user.password);
  
  if (!validPassword) {
    return res.status(400).json({ error: 'Email ou senha incorretos.' });
  }
  
  req.session.userId = user.id;
  
  res.json({ 
    message: 'Login realizado com sucesso!', 
    user: { id: user.id, name: user.name, email: user.email, role: user.role } 
  });
});

// Logout
app.post('/api/logout', (req, res) => {
  req.session.destroy();
  res.json({ message: 'Logout realizado com sucesso!' });
});

// Verifica quem está logado
app.get('/api/me', requireAuth, (req, res) => {
  const db = readDatabase();
  const user = db.users.find(u => u.id === req.session.userId);
  
  if (!user) {
    return res.status(404).json({ error: 'Usuário não encontrado.' });
  }
  
  res.json({ 
    user: { id: user.id, name: user.name, email: user.email, role: user.role } 
  });
});

// Lista todos os usuários (apenas gerente)
app.get('/api/users', requireAuth, requireManager, (req, res) => {
  const db = readDatabase();
  // Remove as senhas antes de enviar
  const safeUsers = db.users.map(u => ({
    id: u.id,
    name: u.name,
    email: u.email,
    role: u.role
  }));
  res.json({ users: safeUsers });
});

// --- ROTAS DE TAREFAS ---

// Lista tarefas (membros veem as suas, gerente vê todas)
app.get('/api/tasks', requireAuth, (req, res) => {
  const db = readDatabase();
  const user = db.users.find(u => u.id === req.session.userId);
  
  let tasks;
  
  if (user.role === 'manager') {
    // Gerente vê todas as tarefas
    tasks = db.tasks;
  } else {
    // Membro vê apenas as tarefas atribuídas a ele
    tasks = db.tasks.filter(t => t.assignedTo === user.id);
  }
  
  // Adiciona informações do responsável em cada tarefa
  const tasksWithUsers = tasks.map(task => {
    const assignedUser = db.users.find(u => u.id === task.assignedTo);
    return {
      ...task,
      assignedToName: assignedUser ? assignedUser.name : 'Desconhecido'
    };
  });
  
  res.json({ tasks: tasksWithUsers });
});

// Cria uma nova tarefa
app.post('/api/tasks', requireAuth, (req, res) => {
  const { title, description, assignedTo, priority, dueDate } = req.body;
  const db = readDatabase();
  const currentUser = db.users.find(u => u.id === req.session.userId);
  
  if (!title) {
    return res.status(400).json({ error: 'O título é obrigatório.' });
  }
  
  // Se for membro, só pode criar tarefa para si mesmo
  let finalAssignedTo = assignedTo;
  if (currentUser.role === 'member') {
    finalAssignedTo = currentUser.id;
  }
  
  // Se for gerente, pode atribuir a qualquer um (ou a si mesmo)
  if (!finalAssignedTo) {
    finalAssignedTo = currentUser.id;
  }
  
  // Verifica se o usuário atribuído existe
  const assignedUser = db.users.find(u => u.id === finalAssignedTo);
  if (!assignedUser) {
    return res.status(400).json({ error: 'Usuário atribuído não encontrado.' });
  }
  
  const newTask = {
    id: uuidv4(),
    title,
    description: description || '',
    status: 'pending', // pending, in_progress, completed
    priority: priority || 'medium', // low, medium, high
    assignedTo: finalAssignedTo,
    createdBy: currentUser.id,
    createdByName: currentUser.name,
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
    dueDate: dueDate || null
  };
  
  db.tasks.push(newTask);
  writeDatabase(db);
  
  res.json({ message: 'Tarefa criada com sucesso!', task: newTask });
});

// Atualiza uma tarefa (status ou edição completa)
app.put('/api/tasks/:id', requireAuth, (req, res) => {
  const { id } = req.params;
  const { title, description, status, priority, assignedTo, dueDate } = req.body;
  
  const db = readDatabase();
  const currentUser = db.users.find(u => u.id === req.session.userId);
  const taskIndex = db.tasks.findIndex(t => t.id === id);
  
  if (taskIndex === -1) {
    return res.status(404).json({ error: 'Tarefa não encontrada.' });
  }
  
  const task = db.tasks[taskIndex];
  
  // Verifica permissões
  const isManager = currentUser.role === 'manager';
  const isOwner = task.assignedTo === currentUser.id;
  
  if (!isManager && !isOwner) {
    return res.status(403).json({ error: 'Você não tem permissão para editar esta tarefa.' });
  }
  
  // Membros só podem mudar o status
  if (!isManager) {
    if (status) task.status = status;
  } else {
    // Gerente pode mudar tudo
    if (title) task.title = title;
    if (description !== undefined) task.description = description;
    if (status) task.status = status;
    if (priority) task.priority = priority;
    if (assignedTo) {
      // Verifica se o novo usuário existe
      const newUser = db.users.find(u => u.id === assignedTo);
      if (newUser) task.assignedTo = assignedTo;
    }
    if (dueDate !== undefined) task.dueDate = dueDate;
  }
  
  task.updatedAt = new Date().toISOString();
  
  db.tasks[taskIndex] = task;
  writeDatabase(db);
  
  res.json({ message: 'Tarefa atualizada com sucesso!', task });
});

// Remove uma tarefa (apenas gerente)
app.delete('/api/tasks/:id', requireAuth, requireManager, (req, res) => {
  const { id } = req.params;
  
  const db = readDatabase();
  const taskIndex = db.tasks.findIndex(t => t.id === id);
  
  if (taskIndex === -1) {
    return res.status(404).json({ error: 'Tarefa não encontrada.' });
  }
  
  db.tasks.splice(taskIndex, 1);
  writeDatabase(db);
  
  res.json({ message: 'Tarefa removida com sucesso!' });
});

// --- SERVIR ARQUIVOS ESTÁTICOS ---

app.get('/', (req, res) => {
  res.sendFile(path.join(__dirname, 'public', 'index.html'));
});

// Inicia o servidor
app.listen(PORT, () => {
  console.log(`Servidor rodando na porta ${PORT}`);
  console.log(`Acesse: http://localhost:${PORT}`);
});
```

```javascript
// Estado global da aplicação
let currentUser = null;
let allTasks = [];
let currentFilter = 'all';
let usersList = [];

// Inicialização
document.addEventListener('DOMContentLoaded', () => {
  checkAuth();
  
  // Event Listeners
  document.getElementById('login-form').addEventListener('submit', handleLogin);
  document.getElementById('register-form').addEventListener('submit', handleRegister);
  document.getElementById('task-form').addEventListener('submit', handleCreateTask);
  document.getElementById('edit-form').addEventListener('submit', handleEditTask);
});

// --- FUNÇÕES DE AUTENTICAÇÃO ---

async function checkAuth() {
  try {
    const response = await fetch('/api/me');
    if (response.ok) {
      const data = await response.json();
      currentUser = data.user;
      showApp();
      loadTasks();
      if (currentUser.role === 'manager') {
        loadUsers();
      }
    } else {
      showAuth();
    }
  } catch (err) {
    console.error('Erro ao verificar autenticação:', err);
    showAuth();
  }
}

async function handleLogin(e) {
  e.preventDefault();
  const email = document.getElementById('login-email').value;
  const password = document.getElementById('login-password').value;
  
  try {
    const response = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });
    
    const data = await response.json();
    
    if (response.ok) {
      currentUser = data.user;
      showApp();
      loadTasks();
      if (currentUser.role === 'manager') loadUsers();
      showError('');
    } else {
      showError(data.error);
    }
  } catch (err) {
    showError('Erro ao conectar com o servidor.');
  }
}

async function handleRegister(e) {
  e.preventDefault();
  const name = document.getElementById('reg-name').value;
  const email = document.getElementById('reg-email').value;
  const password = document.getElementById('reg-password').value;
  const role = document.getElementById('reg-role').value;
  
  try {
    const response = await fetch('/api/register', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, email, password, role })
    });
    
    const data = await response.json();
    
    if (response.ok) {
      currentUser = data.user;
      showApp();
      loadTasks();
      if (currentUser.role === 'manager') loadUsers();
      showError('');
    } else {
      showError(data.error);
    }
  } catch (err) {
    showError('Erro ao conectar com o servidor.');
  }
}

async function logout() {
  try {
    await fetch('/api/logout', { method: 'POST' });
    currentUser = null;
    location.reload();
  } catch (err) {
    console.error('Erro ao fazer logout:', err);
  }
}

function showAuth() {
  document.getElementById('auth-screen').style.display = 'flex';
  document.getElementById('app-screen').style.display = 'none';
}

function showApp() {
  document.getElementById('auth-screen').style.display = 'none';
  document.getElementById('app-screen').style.display = 'block';
  
  // Atualiza interface baseada no papel do usuário
  document.getElementById('user-info').textContent = 
    `${currentUser.name} (${currentUser.role === 'manager' ? 'Gerente' : 'Membro'})`;
  
  document.getElementById('tasks-title').textContent = 
    currentUser.role === 'manager' ? 'Todas as Tarefas' : 'Minhas Tarefas';
  
  // Mostra campo de atribuir apenas para gerente
  if (currentUser.role === 'manager') {
    document.getElementById('assign-section').style.display = 'block';
  }
}

function showError(msg) {
  document.getElementById('auth-error').textContent = msg;
}

function switchTab(tab) {
  document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
  document.querySelectorAll('.auth-form').forEach(form => form.classList.remove('active'));
  
  if (tab === 'login') {
    document.querySelectorAll('.tab-btn')[0].classList.add('active');
    document.getElementById('login-form').classList.add('active');
  } else {
    document.querySelectorAll('.tab-btn')[1].classList.add('active');
    document.getElementById('register-form').classList.add('active');
  }
}

// --- FUNÇÕES DE TAREFAS ---

async function loadUsers() {
  try {
    const response = await fetch('/api/users');
    if (response.ok) {
      const data = await response.json();
      usersList = data.users;
      
      // Preenche os selects de atribuição
      const assignSelect = document.getElementById('task-assign');
      const editAssignSelect = document.getElementById('edit-assign');
      
      const options = usersList.map(user => 
        `<option value="${user.id}">${user.name}</option>`
      ).join('');
      
      assignSelect.innerHTML = options;
      editAssignSelect.innerHTML = options;
    }
  } catch (err) {
    console.error('Erro ao carregar usuários:', err);
  }
}

async function loadTasks() {
  try {
    const response = await fetch('/api/tasks');
    if (response.ok) {
      const data = await response.json();
      allTasks = data.tasks;
      renderTasks();
      updateStats();
    }
  } catch (err) {
    console.error('Erro ao carregar tarefas:', err);
  }
}

async function handleCreateTask(e) {
  e.preventDefault();
  
  const title = document.getElementById('task-title').value;
  const description = document.getElementById('task-desc').value;
  const priority = document.getElementById('task-priority').value;
  const dueDate = document.getElementById('task-due-date').value;
  
  let assignedTo = null;
  if (currentUser.role === 'manager') {
    assignedTo = document.getElementById('task-assign').value;
  }
  
  try {
    const response = await fetch('/api/tasks', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ title, description, priority, assignedTo, dueDate })
    });
    
    if (response.ok) {
      document.getElementById('task-form').reset();
      loadTasks();
    } else {
      const data = await response.json();
      alert(data.error);
    }
  } catch (err) {
    alert('Erro ao criar tarefa.');
  }
}

async function handleEditTask(e) {
  e.preventDefault();
  
  const id = document.getElementById('edit-id').value;
  const title = document.getElementById('edit-title').value;
  const description = document.getElementById('edit-desc').value;
  const status = document.getElementById('edit-status').value;
  const priority = document.getElementById('edit-priority').value;
  const assignedTo = document.getElementById('edit-assign').value;
  const dueDate = document.getElementById('edit-due-date').value;
  
  try {
    const response = await fetch(`/api/tasks/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ title, description, status, priority, assignedTo, dueDate })
    });
    
    if (response.ok) {
      closeModal();
      loadTasks();
    } else {
      const data = await response.json();
      alert(data.error);
    }
  } catch (err) {
    alert('Erro ao editar tarefa.');
  }
}

async function updateTaskStatus(taskId, newStatus) {
  try {
    const response = await fetch(`/api/tasks/${taskId}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ status: newStatus })
    });
    
    if (response.ok) {
      loadTasks();
    } else {
      const data = await response.json();
      alert(data.error);
    }
  } catch (err) {
    alert('Erro ao atualizar status.');
  }
}

async function deleteTask(taskId) {
  if (!confirm('Tem certeza que deseja excluir esta tarefa?')) return;
  
  try {
    const response = await fetch(`/api/tasks/${taskId}`, {
      method: 'DELETE'
    });
    
    if (response.ok) {
      loadTasks();
    } else {
      const data = await response.json();
      alert(data.error);
    }
  } catch (err) {
    alert('Erro ao excluir tarefa.');
  }
}

// --- FUNÇÕES DE RENDERIZAÇÃO ---

function filterTasks(filter) {
  currentFilter = filter;
  document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
  event.target.classList.add('active');
  renderTasks();
}

function renderTasks() {
  const container = document.getElementById('tasks-list');
  
  let filteredTasks = allTasks;
  if (currentFilter !== 'all') {
    filteredTasks = allTasks.filter(t => t.status === currentFilter);
  }
  
  if (filteredTasks.length === 0) {
    container.innerHTML = '<p class="loading">Nenhuma tarefa encontrada.</p>';
    return;
  }
  
  container.innerHTML = filteredTasks.map(task => {
    const isManager = currentUser.role === 'manager';
    const isOwner = task.assignedTo === currentUser.id;
    const canEdit = isManager; // Apenas gerente pode editar completamente
    const canDelete = isManager;
    const canChangeStatus = isManager || isOwner;
    
    const priorityText = {
      low: 'Baixa',
      medium: 'Média',
      high: 'Alta'
    };
    
    const statusText = {
      pending: 'Pendente',
      in_progress: 'Em Andamento',
      completed: 'Concluída'
    };
    
    const dueDate = task.dueDate ? new Date(task.dueDate).toLocaleDateString('pt-BR') : 'Sem prazo';
    
    return `
      <div class="task-card priority-${task.priority}">
        <div class="task-header">
          <div class="task-title">${escapeHtml(task.title)}</div>
          <div class="task-badges">
            <span class="badge badge-priority-${task.priority}">${priorityText[task.priority]}</span>
            <span class="badge badge-status-${task.status}">${statusText[task.status]}</span>
          </div>
        </div>
        
        <div class="task-desc">${escapeHtml(task.description || 'Sem descrição')}</div>
        
        <div class="task-meta">
          <span>Responsável: <span class="task-assigned">${escapeHtml(task.assignedToName)}</span></span>
          <span>Prazo: ${dueDate}</span>
          <span>Criada por: ${escapeHtml(task.createdByName)}</span>
        </div>
        
        <div class="task-actions">
          ${canChangeStatus ? `
            <select class="status-select" onchange="updateTaskStatus('${task.id}', this.value)">
              <option value="pending" ${task.status === 'pending' ? 'selected' : ''}>Pendente</option>
              <option value="in_progress" ${task.status === 'in_progress' ? 'selected' : ''}>Em Andamento</option>
              <option value="completed" ${task.status === 'completed' ? 'selected' : ''}>Concluída</option>
            </select>
          ` : ''}
          
          ${canEdit ? `
            <button class="btn-small btn-edit" onclick="openEditModal('${task.id}')">✏️ Editar</button>
          ` : ''}
          
          ${canDelete ? `
            <button class="btn-small btn-delete" onclick="deleteTask('${task.id}')">🗑️ Excluir</button>
          ` : ''}
        </div>
      </div>
    `;
  }).join('');
}

function updateStats() {
  const pending = allTasks.filter(t => t.status === 'pending').length;
  const inProgress = allTasks.filter(t => t.status === 'in_progress').length;
  const completed = allTasks.filter(t => t.status === 'completed').length;
  
  document.getElementById('stat-pending').textContent = pending;
  document.getElementById('stat-progress').textContent = inProgress;
  document.getElementById('stat-done').textContent = completed;
}

// --- FUNÇÕES DO MODAL ---

function openEditModal(taskId) {
  const task = allTasks.find(t => t.id === taskId);
  if (!task) return;
  
  document.getElementById('edit-id').value = task.id;
  document.getElementById('edit-title').value = task.title;
  document.getElementById('edit-desc').value = task.description;
  document.getElementById('edit-status').value = task.status;
  document.getElementById('edit-priority').value = task.priority;
  document.getElementById('edit-assign').value = task.assignedTo;
  document.getElementById('edit-due-date').value = task.dueDate || '';
  
  document.getElementById('edit-modal').style.display = 'block';
}

function closeModal() {
  document.getElementById('edit-modal').style.display = 'none';
}

// Fecha modal se clicar fora dele
window.onclick = function(event) {
  const modal = document.getElementById('edit-modal');
  if (event.target === modal) {
    closeModal();
  }
}

// --- UTILITÁRIOS ---

function escapeHtml(text) {
  if (!text) return '';
  const div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;
}
```
