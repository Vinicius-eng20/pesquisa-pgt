const express = require('express');

const app = express();
const db = new Database(path.join(__dirname, 'biblioteca.db'));

db.pragma('journal_mode = WAL');

db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    senha TEXT NOT NULL,
    papel TEXT NOT NULL CHECK(papel IN ('leitor','bibliotecario'))
  );

  CREATE TABLE IF NOT EXISTS itens (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    autor TEXT NOT NULL,
    categoria TEXT NOT NULL,
    ano INTEGER,
    disponivel INTEGER NOT NULL DEFAULT 1
  );

  CREATE TABLE IF NOT EXISTS emprestimos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    item_id INTEGER NOT NULL,
    usuario_id INTEGER NOT NULL,
    data_emprestimo TEXT NOT NULL,
    data_prevista TEXT NOT NULL,
    data_devolucao TEXT,
    FOREIGN KEY(item_id) REFERENCES itens(id),
    FOREIGN KEY(usuario_id) REFERENCES usuarios(id)
  );
`);

function exigirLogin(req, res, next) {
  if (!req.session.usuario) return res.status(401).json({ erro: 'Faça login primeiro.' });

}

function exigirBibliotecario(req, res, next) {
  if (!req.session.usuario || req.session.usuario.papel !== 'bibliotecario')

app.post('/api/cadastro', (req, res) => {
  const { nome, email, senha, papel } = req.body;
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha todos os campos.' });
  const p = papel === 'bibliotecario' ? 'bibliotecario' : 'leitor';
  try {
    const hash = bcrypt.hashSync(senha, 10);
    const info = db.prepare("INSERT INTO usuarios (nome,email,senha,papel) VALUES (?,?,?,?)")
      .run(nome, email.toLowerCase(), hash, p);
    req.session.usuario = { id: info.lastInsertRowid, nome, email: email.toLowerCase(), papel: p };
    res.json({ ok: true, usuario: req.session.usuario });
  } catch (e) {
    res.status(400).json({ erro: 'Este email já está cadastrado.' });
  }
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;
  const u = db.prepare("SELECT * FROM usuarios WHERE email=?").get((email || '').toLowerCase());
  if (!u || !bcrypt.compareSync(senha || '', u.senha))
    return res.status(401).json({ erro: 'Email ou senha incorretos.' });
  req.session.usuario = { id: u.id, nome: u.nome, email: u.email, papel: u.papel };
  res.json({ ok: true, usuario: req.session.usuario });
});

function temAtraso(usuarioId) {
  const hoje = new Date().toISOString().slice(0, 10);
  const atrasado = db.prepare(
    "SELECT id FROM emprestimos WHERE usuario_id=? AND data_devolucao IS NULL AND data_prevista < ?"
  ).get(usuarioId, hoje);
  return !!atrasado;
}

app.post('/api/emprestimos', exigirLogin, (req, res) => {
  const { item_id } = req.body;
  const item = db.prepare("SELECT * FROM itens WHERE id=?").get(item_id);
  if (!item) return res.status(404).json({ erro: 'Item não encontrado.' });
  if (!item.disponivel) return res.status(400).json({ erro: 'Item indisponível.' });

  if (temAtraso(req.session.usuario.id))
    return res.status(403).json({ erro: 'Você tem empréstimos em atraso. Devolva-os antes de pegar novos.' });

  const hoje = new Date();
  const prevista = new Date();
  prevista.setDate(hoje.getDate() + 14);

  db.prepare("INSERT INTO emprestimos (item_id,usuario_id,data_emprestimo,data_prevista) VALUES (?,?,?,?)")
    .run(item_id, req.session.usuario.id, hoje.toISOString().slice(0, 10), prevista.toISOString().slice(0, 10));
  db.prepare("UPDATE itens SET disponivel=0 WHERE id=?").run(item_id);
  res.json({ ok: true });
});

app.post('/api/emprestimos/:id/devolver', exigirLogin, (req, res) => {
  const emp = db.prepare("SELECT * FROM emprestimos WHERE id=?").get(req.params.id);
  if (!emp) return res.status(404).json({ erro: 'Empréstimo não encontrado.' });
  if (emp.usuario_id !== req.session.usuario.id && req.session.usuario.papel !== 'bibliotecario')
    return res.status(403).json({ erro: 'Sem permissão.' });
  if (emp.data_devolucao) return res.status(400).json({ erro: 'Já devolvido.' });
  db.prepare("UPDATE emprestimos SET data_devolucao=? WHERE id=?")
    .run(new Date().toISOString().slice(0, 10), req.params.id);
  db.prepare("UPDATE itens SET disponivel=1 WHERE id=?").run(emp.item_id);
  res.json({ ok: true });
});

    SELECT e.*, i.titulo, i.autor, u.nome AS leitor, u.email AS leitor_email
    FROM emprestimos e
    JOIN itens i ON i.id = e.item_id
    JOIN usuarios u ON u.id = e.usuario_id
    ORDER BY e.data_emprestimo DESC
  `).all();
  res.json(rows);

app.get('/api/usuarios', exigirBibliotecario, (req, res) => {
  const rows = db.prepare("SELECT id,nome,email,papel FROM usuarios ORDER BY nome").all();
  const hoje = new Date().toISOString().slice(0, 10);
  const result = rows.map(u => {
    const atrasados = db.prepare(
      "SELECT COUNT(*) c FROM emprestimos WHERE usuario_id=? AND data_devolucao IS NULL AND data_prevista < ?"
    ).get(u.id, hoje).c;
    const ativos = db.prepare(
      "SELECT COUNT(*) c FROM emprestimos WHERE usuario_id=? AND data_devolucao IS NULL"
    ).get(u.id).c;
    return { ...u, atrasados, ativos };
  });
  res.json(result);
});

const app = document.getElementById('app');
let usuario = null;

async function api(url, opts = {}) {
  const r = await fetch(url, {
    headers: { 'Content-Type': 'application/json' },
    credentials: 'same-origin',
    ...opts,
    body: opts.body ? JSON.stringify(opts.body) : undefined
  });
  const data = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(data.erro || 'Erro na operação.');
  return data;
}

function esc(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  })[c]);
}

function msg(texto, tipo = 'erro') {
  const el = document.createElement('div');
  el.className = 'msg ' + tipo;
  el.textContent = texto;
  return el;
}

let abaAtiva = 'acervo';

function renderApp() {
  const isBib = usuario.papel === 'bibliotecario';
  const abas = isBib
    ? [['acervo','Acervo'], ['gerenciar','Gerenciar Itens'], ['historico','Histórico'], ['usuarios','Leitores']]
    : [['acervo','Acervo'], ['meus','Meus Empréstimos']];

  if (!abas.find(a => a[0] === abaAtiva)) abaAtiva = 'acervo';

  app.innerHTML = `
    <header>
      <h1>📚 Biblioteca Digital</h1>
      <div class="user-info">
        <span>${esc(usuario.nome)} (${isBib ? 'Bibliotecário' : 'Leitor'})</span>
        <button class="small secondary" id="logout">Sair</button>
      </div>
    </header>
    <div class="container">
      <div class="tabs">
        ${abas.map(([id,label]) => `<div class="tab ${abaAtiva===id?'active':''}" data-aba="${id}">${label}</div>`).join('')}
      </div>
      <div id="conteudo"></div>
    </div>
  `;
  document.getElementById('logout').onclick = async () => {
    await api('/api/logout', { method: 'POST' });
    usuario = null;
    abaAtiva = 'acervo';
    renderLogin();
  };
  document.querySelectorAll('.tab').forEach(t => {
    t.onclick = () => { abaAtiva = t.dataset.aba; renderApp(); };
  });
  carregarAba();
}

function carregarAba() {
  if (abaAtiva === 'acervo') renderAcervo();
  else if (abaAtiva === 'meus') renderMeusEmprestimos();
  else if (abaAtiva === 'gerenciar') renderGerenciar();
  else if (abaAtiva === 'historico') renderHistorico();
  else if (abaAtiva === 'usuarios') renderUsuarios();
}

async function renderAcervo() {
  const c = document.getElementById('conteudo');
  c.innerHTML = `
    <div class="card">
      <h2>Buscar no acervo</h2>
      <div class="row">
        <input id="busca" type="text" placeholder="Título, autor ou categoria...">
        <button id="btn-buscar">Buscar</button>
        <button class="secondary" id="btn-limpar">Limpar</button>
      </div>
    </div>
    <div class="card">
      <h2>Itens disponíveis</h2>
      <div id="lista-itens" class="grid"><div class="empty">Carregando...</div></div>
    </div>
  `;
  const carregar = async (busca = '') => {
    const lista = document.getElementById('lista-itens');
    try {
      const itens = await api('/api/itens' + (busca ? '?busca=' + encodeURIComponent(busca) : ''));
      if (!itens.length) { lista.innerHTML = '<div class="empty">Nenhum item encontrado.</div>'; return; }
      lista.innerHTML = itens.map(it => `
        <div class="item">
          <span class="badge ${it.disponivel ? 'ok' : 'no'}">${it.disponivel ? 'Disponível' : 'Emprestado'}</span>
          <h3>${esc(it.titulo)}</h3>
          <div class="meta">${esc(it.autor)} • ${esc(it.categoria)}${it.ano ? ' • ' + it.ano : ''}</div>
          ${it.disponivel ? `<button class="small" data-emp="${it.id}">Pegar emprestado</button>` : ''}
        </div>
      `).join('');
      lista.querySelectorAll('[data-emp]').forEach(b => {
        b.onclick = async () => {
          try {
            await api('/api/emprestimos', { method: 'POST', body: { item_id: Number(b.dataset.emp) } });
            alert('Empréstimo realizado! Devolução em 14 dias.');
            carregar(document.getElementById('busca').value);
          } catch (e) { alert(e.message); }
        };
      });
    } catch (e) { lista.innerHTML = `<div class="empty">${esc(e.message)}</div>`; }
  };
  document.getElementById('btn-buscar').onclick = () => carregar(document.getElementById('busca').value);
  document.getElementById('busca').onkeydown = e => { if (e.key === 'Enter') carregar(e.target.value); };
  document.getElementById('btn-limpar').onclick = () => { document.getElementById('busca').value = ''; carregar(''); };
  carregar();
}

async function renderMeusEmprestimos() {
  const c = document.getElementById('conteudo');
  c.innerHTML = `<div class="card"><h2>Meus Empréstimos</h2><div id="lista"><div class="empty">Carregando...</div></div></div>`;
  try {
    const lista = await api('/api/meus-emprestimos');
    const el = document.getElementById('lista');
    if (!lista.length) { el.innerHTML = '<div class="empty">Você não tem empréstimos.</div>'; return; }
    const hoje = new Date().toISOString().slice(0, 10);
    el.innerHTML = `
      <table>
        <thead><tr><th>Título</th><th>Autor</th><th>Empréstimo</th><th>Devolução prevista</th><th>Status</th><th></th></tr></thead>
        <tbody>
        ${lista.map(e => {
          let status;
          if (e.data_devolucao) status = `<span class="badge ok">Devolvido em ${e.data_devolucao}</span>`;
          else if (e.data_prevista < hoje) status = '<span class="badge no">Em atraso</span>';
          else status = '<span class="badge warn">Em andamento</span>';
          return `<tr>
            <td>${esc(e.titulo)}</td>
            <td>${esc(e.autor)}</td>
            <td>${e.data_emprestimo}</td>
            <td>${e.data_prevista}</td>
            <td>${status}</td>
            <td>${!e.data_devolucao ? `<button class="small" data-dev="${e.id}">Devolver</button>` : ''}</td>
          </tr>`;
        }).join('')}
        </tbody>
      </table>
    `;
    el.querySelectorAll('[data-dev]').forEach(b => {
      b.onclick = async () => {
        if (!confirm('Confirmar devolução?')) return;
        try {
          await api(`/api/emprestimos/${b.dataset.dev}/devolver`, { method: 'POST' });
          renderMeusEmprestimos();
        } catch (e) { alert(e.message); }
      };
    });
  } catch (e) {
    document.getElementById('lista').innerHTML = `<div class="empty">${esc(e.message)}</div>`;
  }
}

async function renderGerenciar() {
  const c = document.getElementById('conteudo');
  c.innerHTML = `
    <div class="card">
      <h2 id="form-titulo">Novo Item</h2>
      <div id="msg-area"></div>
      <input type="hidden" id="edit-id">
      <label>Título</label>
      <input id="f-titulo" type="text">
      <label>Autor</label>
      <input id="f-autor" type="text">
      <label>Categoria</label>
      <input id="f-categoria" type="text">
      <label>Ano (opcional)</label>
      <input id="f-ano" type="number">
      <div class="row">
        <button id="btn-salvar">Salvar</button>
        <button class="secondary" id="btn-cancelar" style="display:none">Cancelar edição</button>
      </div>
    </div>
    <div class="card">
      <h2>Itens cadastrados</h2>
      <div id="lista"><div class="empty">Carregando...</div></div>
    </div>
  `;
  const limparForm = () => {
    ['f-titulo','f-autor','f-categoria','f-ano'].forEach(id => document.getElementById(id).value = '');
    document.getElementById('edit-id').value = '';
    document.getElementById('form-titulo').textContent = 'Novo Item';
    document.getElementById('btn-cancelar').style.display = 'none';
  };
  const carregar = async () => {
    const lista = await api('/api/itens');
    const el = document.getElementById('lista');
    if (!lista.length) { el.innerHTML = '<div class="empty">Nenhum item cadastrado.</div>'; return; }
    el.innerHTML = `
      <table>
        <thead><tr><th>Título</th><th>Autor</th><th>Categoria</th><th>Ano</th><th>Status</th><th></th></tr></thead>
        <tbody>
        ${lista.map(it => `<tr>
          <td>${esc(it.titulo)}</td>
          <td>${esc(it.autor)}</td>
          <td>${esc(it.categoria)}</td>
          <td>${it.ano || '-'}</td>
          <td><span class="badge ${it.disponivel?'ok':'no'}">${it.disponivel?'Disponível':'Emprestado'}</span></td>
          <td>
            <button class="small secondary" data-edit="${it.id}">Editar</button>
            <button class="small danger" data-del="${it.id}">Remover</button>
          </td>
        </tr>`).join('')}
        </tbody>
      </table>
    `;
    el.querySelectorAll('[data-edit]').forEach(b => {
      b.onclick = () => {
        const it = lista.find(x => x.id == b.dataset.edit);
        document.getElementById('f-titulo').value = it.titulo;
        document.getElementById('f-autor').value = it.autor;
        document.getElementById('f-categoria').value = it.categoria;
        document.getElementById('f-ano').value = it.ano || '';
        document.getElementById('edit-id').value = it.id;
        document.getElementById('form-titulo').textContent = 'Editar Item';
        document.getElementById('btn-cancelar').style.display = 'inline-block';
        window.scrollTo(0, 0);
      };
    });
    el.querySelectorAll('[data-del]').forEach(b => {
      b.onclick = async () => {
        if (!confirm('Remover este item?')) return;
        try {
          await api('/api/itens/' + b.dataset.del, { method: 'DELETE' });
          carregar();
        } catch (e) { alert(e.message); }
      };
    });
  };
  document.getElementById('btn-cancelar').onclick = limparForm;
  document.getElementById('btn-salvar').onclick = async () => {
    const area = document.getElementById('msg-area');
    area.innerHTML = '';
    const body = {
      titulo: document.getElementById('f-titulo').value,
      autor: document.getElementById('f-autor').value,
      categoria: document.getElementById('f-categoria').value,
      ano: document.getElementById('f-ano').value || null
    };
    const id = document.getElementById('edit-id').value;
    try {
      if (id) await api('/api/itens/' + id, { method: 'PUT', body });
      else await api('/api/itens', { method: 'POST', body });
      limparForm();
      carregar();
    } catch (e) { area.appendChild(msg(e.message)); }
  };
  carregar();
}

  const c = document.getElementById('conteudo');
  c.innerHTML = `<div class="card"><h2>Histórico de Empréstimos</h2><div id="lista"><div class="empty">Carregando...</div></div></div>`;

    const lista = await api('/api/emprestimos');
    const el = document.getElementById('lista');
    if (!lista.length) { el.innerHTML = '<div class="empty">Nenhum empréstimo registrado.</div>'; return; }
    const hoje = new Date().toISOString().slice(0, 10);
    el.innerHTML = `
      <table>
        <thead><tr><th>Leitor</th><th>Título</th><th>Empréstimo</th><th>Prevista</th><th>Devolução</th><th>Status</th></tr></thead>
        <tbody>
        ${lista.map(e => {
          let status;
          if (e.data_devolucao) status = '<span class="badge ok">Devolvido</span>';
          else if (e.data_prevista < hoje) status = '<span class="badge no">Em atraso</span>';
          else status = '<span class="badge warn">Em andamento</span>';
          return `<tr>
            <td>${esc(e.leitor)}<br><small style="color:#718096">${esc(e.leitor_email)}</small></td>
            <td>${esc(e.titulo)}</td>
            <td>${e.data_emprestimo}</td>
            <td>${e.data_prevista}</td>
            <td>${e.data_devolucao || '-'}</td>
            <td>${status}</td>
          </tr>`;
        }).join('')}
        </tbody>
      </table>
    `;
  } catch (e) {
    document.getElementById('lista').innerHTML = `<div class="empty">${esc(e.message)}</div>`;
  }
}

# 📚 Biblioteca Digital

Site completo de biblioteca com empréstimo de itens, contas de usuário (leitores e bibliotecário) e dados salvos em banco local (SQLite). Tudo funciona sem depender de serviços externos.

## Como instalar e executar

Você precisa ter o **Node.js** instalado (versão 18 ou superior). Baixe em https://nodejs.org.

1. Abra o terminal na pasta do projeto.
2. Instale as dependências:
   ```
   npm install
   ```
3. Inicie o site:
   ```
   npm start
   ```
4. Abra o navegador em: **http://localhost:3000**

## Conta de bibliotecário padrão

- **Email:** admin@biblioteca.com
- **Senha:** admin123

## Como usar

### Leitores
- Clicam em **Cadastre-se** na tela inicial, escolhem o tipo "Leitor".
- Aba **Acervo**: buscam itens por título, autor ou categoria e pegam emprestado.
- Aba **Meus Empréstimos**: veem o que pegaram e podem devolver.
- Prazo de devolução: 14 dias. Se ficar em atraso, não conseguem pegar novos itens até devolver.

### Bibliotecário
- Aba **Acervo**: mesmo que os leitores.
- Aba **Gerenciar Itens**: cadastrar, editar e remover itens.
- Aba **Histórico**: vê todos os empréstimos de todos os leitores.
- Aba **Leitores**: vê a situação de cada leitor (quem está bloqueado por atraso).

## Onde ficam os dados

Todos os dados (contas, itens e empréstimos) ficam salvos no arquivo `biblioteca.db` na pasta do projeto. Mesmo fechando o site e abrindo de novo, tudo permanece.

## Colocar na internet

Para publicar, hospede em um serviço como Render, Railway ou Fly.io. Envie os arquivos do projeto, defina o comando de start como `npm start` e o site estará acessível publicamente. O banco `biblioteca.db` ficará salvo junto com a aplicação.
