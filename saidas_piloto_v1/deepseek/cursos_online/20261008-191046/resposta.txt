const express = require('express');
const session = require('express-session');
const bcrypt = require('bcryptjs');
const Database = require('better-sqlite3');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

const db = new Database('cursos.db');

db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT NOT NULL DEFAULT 'aluno'
  );

  CREATE TABLE IF NOT EXISTS cursos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    descricao TEXT NOT NULL,
    instrutor TEXT NOT NULL,
    carga_horaria INTEGER NOT NULL,
    preco REAL NOT NULL DEFAULT 0
  );

  CREATE TABLE IF NOT EXISTS matriculas (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    curso_id INTEGER NOT NULL,
    data_matricula TEXT NOT NULL,
    UNIQUE(usuario_id, curso_id),
    FOREIGN KEY(usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY(curso_id) REFERENCES cursos(id)
  );
`);

const adminExiste = db.prepare('SELECT id FROM usuarios WHERE email = ?').get('admin@admin.com');
if (!adminExiste) {
  db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)')
    .run('Administrador', 'admin@admin.com', bcrypt.hashSync('admin123', 10), 'admin');
}

app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(session({

  resave: false,
  saveUninitialized: false,
  cookie: { maxAge: 1000 * 60 * 60 * 24 * 7 }
}));

function requireAuth(req, res, next) {
  if (!req.session.usuario) return res.status(401).json({ erro: 'Não autenticado' });
  next();
}

function requireAdmin(req, res, next) {
  if (!req.session.usuario || req.session.usuario.tipo !== 'admin') {
    return res.status(403).json({ erro: 'Acesso negado' });
  }
  next();
}

app.post('/api/registrar', (req, res) => {
  const { nome, email, senha } = req.body;
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });
  if (senha.length < 4) return res.status(400).json({ erro: 'Senha muito curta' });
  try {
    const hash = bcrypt.hashSync(senha, 10);
    const info = db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)')
      .run(nome, email.toLowerCase(), hash, 'aluno');
    req.session.usuario = { id: info.lastInsertRowid, nome, email: email.toLowerCase(), tipo: 'aluno' };
    res.json({ ok: true, usuario: req.session.usuario });
  } catch (e) {
    res.status(400).json({ erro: 'Email já cadastrado' });
  }
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;
  const u = db.prepare('SELECT * FROM usuarios WHERE email = ?').get((email || '').toLowerCase());
  if (!u || !bcrypt.compareSync(senha, u.senha)) {
    return res.status(401).json({ erro: 'Email ou senha inválidos' });
  }
  req.session.usuario = { id: u.id, nome: u.nome, email: u.email, tipo: u.tipo };
  res.json({ ok: true, usuario: req.session.usuario });
});

app.post('/api/logout', (req, res) => {
  req.session.destroy(() => res.json({ ok: true }));
});

app.get('/api/eu', (req, res) => {
  res.json({ usuario: req.session.usuario || null });
});

app.get('/api/cursos', (req, res) => {
  const cursos = db.prepare('SELECT * FROM cursos ORDER BY id DESC').all();
  res.json(cursos);
});

app.get('/api/cursos/:id', (req, res) => {
  const curso = db.prepare('SELECT * FROM cursos WHERE id = ?').get(req.params.id);
  if (!curso) return res.status(404).json({ erro: 'Curso não encontrado' });
  res.json(curso);
});

app.post('/api/cursos', requireAdmin, (req, res) => {
  const { titulo, descricao, instrutor, carga_horaria, preco } = req.body;
  if (!titulo || !descricao || !instrutor) return res.status(400).json({ erro: 'Preencha todos os campos' });
  const info = db.prepare('INSERT INTO cursos (titulo, descricao, instrutor, carga_horaria, preco) VALUES (?, ?, ?, ?, ?)')
    .run(titulo, descricao, instrutor, Number(carga_horaria) || 0, Number(preco) || 0);
  res.json({ ok: true, id: info.lastInsertRowid });
});

app.put('/api/cursos/:id', requireAdmin, (req, res) => {
  const { titulo, descricao, instrutor, carga_horaria, preco } = req.body;
  db.prepare('UPDATE cursos SET titulo=?, descricao=?, instrutor=?, carga_horaria=?, preco=? WHERE id=?')
    .run(titulo, descricao, instrutor, Number(carga_horaria) || 0, Number(preco) || 0, req.params.id);
  res.json({ ok: true });
});

app.delete('/api/cursos/:id', requireAdmin, (req, res) => {
  db.prepare('DELETE FROM matriculas WHERE curso_id = ?').run(req.params.id);
  db.prepare('DELETE FROM cursos WHERE id = ?').run(req.params.id);
  res.json({ ok: true });
});

app.post('/api/matriculas', requireAuth, (req, res) => {
  const { curso_id } = req.body;
  const curso = db.prepare('SELECT id FROM cursos WHERE id = ?').get(curso_id);
  if (!curso) return res.status(404).json({ erro: 'Curso não encontrado' });
  try {
    db.prepare('INSERT INTO matriculas (usuario_id, curso_id, data_matricula) VALUES (?, ?, ?)')
      .run(req.session.usuario.id, curso_id, new Date().toISOString());
    res.json({ ok: true });
  } catch (e) {
    res.status(400).json({ erro: 'Você já está matriculado neste curso' });
  }
});

app.get('/api/minhas-matriculas', requireAuth, (req, res) => {
  const rows = db.prepare(`
    SELECT m.id as matricula_id, m.data_matricula, c.*
    FROM matriculas m JOIN cursos c ON c.id = m.curso_id
    WHERE m.usuario_id = ?
    ORDER BY m.id DESC
  `).all(req.session.usuario.id);
  res.json(rows);
});

app.get('/api/admin/matriculas', requireAdmin, (req, res) => {
  const rows = db.prepare(`
    SELECT m.id as matricula_id, m.data_matricula,
           u.nome as aluno_nome, u.email as aluno_email,
           c.titulo as curso_titulo, c.id as curso_id
    FROM matriculas m
    JOIN usuarios u ON u.id = m.usuario_id
    JOIN cursos c ON c.id = m.curso_id
    ORDER BY m.id DESC
  `).all();
  res.json(rows);
});

app.delete('/api/admin/matriculas/:id', requireAdmin, (req, res) => {
  db.prepare('DELETE FROM matriculas WHERE id = ?').run(req.params.id);
  res.json({ ok: true });
});

app.use(express.static(path.join(__dirname, 'public')));

app.listen(PORT, () => {
  console.log(`Servidor rodando em http://localhost:${PORT}`);
});

let usuario = null;

async function api(url, options = {}) {
  const res = await fetch(url, {
    headers: { 'Content-Type': 'application/json' },
    credentials: 'same-origin',
    ...options,
    body: options.body ? JSON.stringify(options.body) : undefined
  });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.erro || 'Erro');
  return data;
}

function el(id) { return document.getElementById(id); }

function show(view) {
  ['authView', 'alunoView', 'adminView'].forEach(v => el(v).classList.add('hidden'));
  el(view).classList.remove('hidden');
}

function setMsg(text, type) {
  const m = el('authMsg');
  m.textContent = text || '';
  m.className = 'msg ' + (type || '');
}

async function init() {
  try {
    const data = await api('/api/eu');
    usuario = data.usuario;
  } catch { usuario = null; }
  renderHeader();
  if (!usuario) {
    show('authView');
  } else if (usuario.tipo === 'admin') {
    show('adminView');
    carregarCursosAdmin();
  } else {
    show('alunoView');
    carregarCursosDisponiveis();
  }
}

function renderHeader() {
  const area = el('userArea');
  if (!usuario) {
    area.innerHTML = '';
    return;
  }
  area.innerHTML = `
    <span>Olá, <strong>${usuario.nome}</strong> ${usuario.tipo === 'admin' ? '(Admin)' : ''}</span>
    <button class="btn-secondary btn-small" onclick="logout()">Sair</button>
  `;
}

async function logout() {
  await api('/api/logout', { method: 'POST' });
  usuario = null;
  renderHeader();
  show('authView');
}

async function carregarCursosDisponiveis() {
  const cursos = await api('/api/cursos');
  const el2 = el('cursosDisponiveis');
  if (!cursos.length) {
    el2.innerHTML = '<div class="empty">Nenhum curso disponível no momento.</div>';
    return;
  }
  el2.innerHTML = cursos.map(c => `
    <div class="card">
      <h3>${escapeHtml(c.titulo)}</h3>
      <p>${escapeHtml(c.descricao)}</p>
      <div class="meta">
        <span>👨‍🏫 ${escapeHtml(c.instrutor)}</span>
        <span>⏱ ${c.carga_horaria}h</span>
      </div>
      <div class="preco">${c.preco > 0 ? 'R$ ' + c.preco.toFixed(2) : 'Gratuito'}</div>
      <div class="actions">
        <button onclick="matricular(${c.id})">Matricular-se</button>
      </div>
    </div>
  `).join('');
}

async function matricular(cursoId) {
  try {
    await api('/api/matriculas', { method: 'POST', body: { curso_id: cursoId } });
    alert('Matrícula realizada com sucesso!');
  } catch (err) {
    alert(err.message);
  }
}

async function editarCurso(id) {
  const c = await api('/api/cursos/' + id);
  el('cursoId').value = c.id;
  el('cursoTitulo').value = c.titulo;
  el('cursoDescricao').value = c.descricao;
  el('cursoInstrutor').value = c.instrutor;
  el('cursoCarga').value = c.carga_horaria;
  el('cursoPreco').value = c.preco;
  el('cursoFormTitle').textContent = 'Editar Curso';
  el('cancelCurso').classList.remove('hidden');
  el('cursoForm').scrollIntoView({ behavior: 'smooth' });
}

    descricao: el('cursoDescricao').value,
    instrutor: el('cursoInstrutor').value,
    carga_horaria: el('cursoCarga').value,
    preco: el('cursoPreco').value
  };
  try {
    if (id) {
      await api('/api/cursos/' + id, { method: 'PUT', body });
    } else {
      await api('/api/cursos', { method: 'POST', body });
    }
    el('cancelCurso').onclick();
    carregarCursosAdmin();
  } catch (err) {
    alert(err.message);
  }
};

async function removerCurso(id) {
  if (!confirm('Remover este curso? Todas as matrículas associadas também serão removidas.')) return;
  await api('/api/cursos/' + id, { method: 'DELETE' });
  carregarCursosAdmin();
}

async function carregarMatriculasAdmin() {
  const lista = await api('/api/admin/matriculas');
  const el2 = el('listaMatriculas');
  if (!lista.length) {
    el2.innerHTML = '<div class="empty">Nenhuma matrícula registrada.</div>';
    return;
  }
  el2.innerHTML = lista.map(m => `
    <div class="card">
      <h3>${escapeHtml(m.curso_titulo)}</h3>
      <p><strong>Aluno:</strong> ${escapeHtml(m.aluno_nome)}</p>
      <p><strong>Email:</strong> ${escapeHtml(m.aluno_email)}</p>
      <div class="meta"><span>📅 ${new Date(m.data_matricula).toLocaleDateString('pt-BR')}</span></div>
      <div class="actions">
        <button class="btn-danger" onclick="cancelarMatricula(${m.matricula_id})">Cancelar Matrícula</button>
      </div>
    </div>
  `).join('');
}

async function cancelarMatricula(id) {
  if (!confirm('Cancelar esta matrícula?')) return;
  await api('/api/admin/matriculas/' + id, { method: 'DELETE' });
  carregarMatriculasAdmin();
}

function escapeHtml(s) {
  return String(s || '').replace(/[&<>"']/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[c]));
}

# Plataforma de Cursos Online

Site completo com cadastro de alunos, matrícula em cursos e painel de administração. Os dados ficam salvos em um banco de dados SQLite (arquivo `cursos.db`), ou seja, tudo permanece salvo mesmo ao fechar e reabrir o site.

## O que vem pronto

- **Alunos**: podem criar conta, ver cursos disponíveis, se matricular e ver os cursos em que estão matriculados.
- **Administrador**: pode cadastrar, editar e remover cursos, ver todas as matrículas de todos os alunos e cancelar matrículas.
- **Login do administrador padrão** (já criado automaticamente):
  - Email: `admin@admin.com`
  - Senha: `admin123`

## Como instalar e executar

Você precisa ter o **Node.js** instalado (baixe em https://nodejs.org — escolha a versão LTS e instale com as opções padrão).

Depois, abra o terminal (Prompt de Comando no Windows, ou Terminal no Mac/Linux) na pasta do projeto e rode:

```
npm install
```

Isso instala tudo que o site precisa (só precisa fazer uma vez).

Depois, para iniciar o site:

```
npm start
```

Você verá a mensagem: `Servidor rodando em http://localhost:3000`

Abra o navegador e acesse:

```
http://localhost:3000
```

Pronto! O site está funcionando.

## Como publicar na internet

Para que o site fique acessível pela internet, hospede em um serviço que suporte Node.js. Recomendamos o **Render** (render.com) ou **Railway** (railway.app) — ambos possuem planos gratuitos:

1. Crie uma conta nesses serviços.
2. Faça upload do projeto (ou conecte um repositório no GitHub).
3. Configure:
   - **Build command**: `npm install`
   - **Start command**: `npm start`
4. O serviço irá gerar uma URL pública do tipo `https://seu-site.onrender.com`. Basta compartilhar essa URL.

O banco de dados continuará funcionando automaticamente, pois fica salvo junto com o site.

## Como usar

1. Acesse o site e **crie uma conta** (aba "Criar Conta") ou entre com uma conta existente.
2. Como aluno, veja os cursos em **Cursos Disponíveis** e clique em **Matricular-se**.
3. Veja seus cursos em **Meus Cursos**.
4. Para entrar como administrador, faça login com `admin@admin.com` / `admin123`. No painel de admin você poderá criar/editar/remover cursos e gerenciar todas as matrículas.
