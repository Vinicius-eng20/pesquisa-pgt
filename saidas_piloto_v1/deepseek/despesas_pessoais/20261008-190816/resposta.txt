# Controle de Despesas Pessoais

Sistema web completo de controle de despesas pessoais com autenticação, categorias e painel administrativo.

## Requisitos

- Node.js 18 ou superior instalado no computador
- Nenhum outro serviço externo é necessário (banco de dados fica em arquivo local)

## Como instalar

1. Abra o terminal (Prompt de Comando no Windows, Terminal no Mac/Linux) na pasta do projeto.
2. Execute o comando abaixo para instalar as dependências:

```
npm install
```

## Como executar

```
npm start
```

Depois de executar, abra o navegador e acesse:

```
http://localhost:3000
```

## Usuário administrador padrão

Na primeira vez que o site rodar, será criado automaticamente um administrador:

- **E-mail:** admin@admin.com
- **Senha:** admin123

**Importante:** troque a senha do administrador após o primeiro acesso (você pode criar um novo admin e desativar o antigo, ou editar o arquivo `data/db.json`).

## Como usar

### Para quem controla os gastos
1. Acesse a página inicial e clique em **Cadastrar**.
2. Crie sua conta com e-mail e senha.
3. Faça login e use o painel para:
   - Registrar gastos (valor, descrição, categoria e data)
   - Ver o total gasto no mês
   - Ver gastos agrupados por categoria
   - Excluir gastos lançados por engano

### Para o administrador
1. Faça login com a conta de administrador.
2. No painel administrativo você pode:
   - Cadastrar novas categorias de gasto
   - Ativar/desativar usuários
   - Ver o total de gastos de todos os usuários

## Onde ficam os dados

Todos os dados ficam salvos no arquivo `data/db.json` dentro da pasta do projeto. Isso significa que mesmo fechando o navegador ou desligando o computador, tudo continua salvo.

Para colocar na internet, hospede este projeto em qualquer serviço que rode Node.js (Render, Railway, Fly.io, VPS, etc). O arquivo de dados persistirá enquanto o disco do servidor for persistente.

## Estrutura

- `server.js` — servidor principal (Express + API + páginas)
- `data/db.json` — banco de dados em arquivo (criado automaticamente)
- `public/` — arquivos estáticos (HTML, CSS, JS do navegador)
- `package.json` — configuração do projeto

const express = require('express');
const session = require('express-session');
const bcrypt = require('bcryptjs');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;
const DATA_DIR = path.join(__dirname, 'data');
const DB_FILE = path.join(DATA_DIR, 'db.json');

// ---------- Banco de dados em arquivo ----------
function ensureDataDir() {
  if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });
}

function loadDB() {
  ensureDataDir();
  if (!fs.existsSync(DB_FILE)) {
    return { users: [], categories: [], expenses: [], nextUserId: 1, nextCategoryId: 1, nextExpenseId: 1 };
  }
  try {
    return JSON.parse(fs.readFileSync(DB_FILE, 'utf8'));
  } catch (e) {
    return { users: [], categories: [], expenses: [], nextUserId: 1, nextCategoryId: 1, nextExpenseId: 1 };
  }
}

function saveDB(db) {
  ensureDataDir();
  fs.writeFileSync(DB_FILE, JSON.stringify(db, null, 2), 'utf8');
}

function seed() {
  const db = loadDB();
  let changed = false;

  if (db.categories.length === 0) {
    const defaultCats = ['Alimentação', 'Transporte', 'Moradia', 'Lazer', 'Saúde', 'Outros'];
    for (const name of defaultCats) {
      db.categories.push({ id: db.nextCategoryId++, name, active: true });
    }
    changed = true;
  }

  if (!db.users.find(u => u.role === 'admin')) {
    const hash = bcrypt.hashSync('admin123', 10);
    db.users.push({
      id: db.nextUserId++,
      name: 'Administrador',
      email: 'admin@admin.com',
      password: hash,
      role: 'admin',
      active: true,
      createdAt: new Date().toISOString()
    });
    changed = true;
  }

  if (changed) saveDB(db);
}

seed();

// ---------- Helpers de autenticação ----------
function requireAuth(req, res, next) {
  if (!req.session.userId) return res.status(401).json({ error: 'Não autenticado' });
  const db = loadDB();
  const user = db.users.find(u => u.id === req.session.userId);
  if (!user || !user.active) {
    req.session.destroy(() => {});
    return res.status(401).json({ error: 'Conta inativa ou inexistente' });
  }
  req.user = user;
  next();
}

function requireAdmin(req, res, next) {
  requireAuth(req, res, () => {
    if (req.user.role !== 'admin') return res.status(403).json({ error: 'Acesso restrito ao administrador' });
    next();
  });
}

app.post('/api/register', (req, res) => {
  const { name, email, password } = req.body || {};
  if (!name || !email || !password) return res.status(400).json({ error: 'Preencha nome, e-mail e senha' });
  if (password.length < 6) return res.status(400).json({ error: 'A senha precisa ter pelo menos 6 caracteres' });

  const db = loadDB();
  const exists = db.users.find(u => u.email.toLowerCase() === email.toLowerCase());
  if (exists) return res.status(400).json({ error: 'Já existe uma conta com este e-mail' });

  const hash = bcrypt.hashSync(password, 10);
  const user = {
    id: db.nextUserId++,
    name,
    email: email.toLowerCase(),
    password: hash,
    role: 'user',
    active: true,
    createdAt: new Date().toISOString()
  };
  db.users.push(user);
  saveDB(db);

  req.session.userId = user.id;
  res.json({ id: user.id, name: user.name, email: user.email, role: user.role });
});

app.post('/api/login', (req, res) => {
  const { email, password } = req.body || {};
  if (!email || !password) return res.status(400).json({ error: 'Informe e-mail e senha' });

  const db = loadDB();
  const user = db.users.find(u => u.email.toLowerCase() === email.toLowerCase());
  if (!user) return res.status(400).json({ error: 'E-mail ou senha incorretos' });
  if (!user.active) return res.status(403).json({ error: 'Esta conta está desativada' });
  if (!bcrypt.compareSync(password, user.password)) return res.status(400).json({ error: 'E-mail ou senha incorretos' });

  req.session.userId = user.id;
  res.json({ id: user.id, name: user.name, email: user.email, role: user.role });
});

app.post('/api/logout', (req, res) => {

app.post('/api/categories', requireAdmin, (req, res) => {
  const { name } = req.body || {};

  const db = loadDB();
  const exists = db.categories.find(c => c.name.toLowerCase() === name.trim().toLowerCase());
  if (exists) return res.status(400).json({ error: 'Já existe uma categoria com este nome' });

  db.categories.push(cat);

  res.json(cat);

app.patch('/api/categories/:id', requireAdmin, (req, res) => {
  const id = parseInt(req.params.id, 10);
  const db = loadDB();
  const cat = db.categories.find(c => c.id === id);
  if (!cat) return res.status(404).json({ error: 'Categoria não encontrada' });
  if (typeof req.body.active === 'boolean') cat.active = req.body.active;
  if (typeof req.body.name === 'string' && req.body.name.trim()) cat.name = req.body.name.trim();
  saveDB(db);
  res.json(cat);
});

// ---------- API de despesas ----------
app.get('/api/expenses', requireAuth, (req, res) => {
  const db = loadDB();
  const list = db.expenses
    .filter(e => e.userId === req.user.id)
    .sort((a, b) => (a.date < b.date ? 1 : -1));
  res.json(list);
});

app.post('/api/expenses', requireAuth, (req, res) => {
  const { amount, description, categoryId, date } = req.body || {};
  const value = parseFloat(amount);
  if (!value || value <= 0) return res.status(400).json({ error: 'Informe um valor maior que zero' });
  if (!date) return res.status(400).json({ error: 'Informe a data do gasto' });

  const db = loadDB();
  const cat = db.categories.find(c => c.id === parseInt(categoryId, 10));
  if (!cat || !cat.active) return res.status(400).json({ error: 'Categoria inválida' });

  const exp = {
    id: db.nextExpenseId++,
    userId: req.user.id,
    amount: Math.round(value * 100) / 100,
    description: (description || '').trim(),
    categoryId: cat.id,
    date,
    createdAt: new Date().toISOString()
  };
  db.expenses.push(exp);
  saveDB(db);
  res.json(exp);
});

app.delete('/api/expenses/:id', requireAuth, (req, res) => {
  const id = parseInt(req.params.id, 10);
  const db = loadDB();
  const idx = db.expenses.findIndex(e => e.id === id && e.userId === req.user.id);
  if (idx === -1) return res.status(404).json({ error: 'Gasto não encontrado' });
  db.expenses.splice(idx, 1);
  saveDB(db);
  res.json({ ok: true });
});

app.get('/api/summary', requireAuth, (req, res) => {
  const db = loadDB();
  const mine = db.expenses.filter(e => e.userId === req.user.id);
  const now = new Date();
  const monthKey = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;

  let totalMonth = 0;
  const byCategory = {};
  for (const e of mine) {
    if ((e.date || '').startsWith(monthKey)) totalMonth += e.amount;
    byCategory[e.categoryId] = (byCategory[e.categoryId] || 0) + e.amount;
  }
  const categories = db.categories.map(c => ({
    id: c.id,
    name: c.name,
    total: Math.round((byCategory[c.id] || 0) * 100) / 100
  }));

  res.json({
    totalMonth: Math.round(totalMonth * 100) / 100,
    totalAll: Math.round(mine.reduce((s, e) => s + e.amount, 0) * 100) / 100,
    count: mine.length,
    byCategory: categories
  });
});

  const id = parseInt(req.params.id, 10);
  const db = loadDB();
  const u = db.users.find(x => x.id === id);
  if (!u) return res.status(404).json({ error: 'Usuário não encontrado' });
  if (u.id === req.user.id && req.body.active === false) {
    return res.status(400).json({ error: 'Você não pode desativar sua própria conta' });
  }
  if (typeof req.body.active === 'boolean') u.active = req.body.active;
  saveDB(db);
  res.json({ id: u.id, name: u.name, email: u.email, role: u.role, active: u.active });
});

app.get('/api/admin/summary', requireAdmin, (req, res) => {
  const db = loadDB();
  const totalAll = db.expenses.reduce((s, e) => s + e.amount, 0);
  const perUser = db.users.map(u => {
    const total = db.expenses.filter(e => e.userId === u.id).reduce((s, e) => s + e.amount, 0);
    return {
      id: u.id,
      name: u.name,
      email: u.email,
      active: u.active,
      total: Math.round(total * 100) / 100
    };
  });
  res.json({
    totalAll: Math.round(totalAll * 100) / 100,
    totalUsers: db.users.length,
    totalExpenses: db.expenses.length,
    perUser
  });
});

  const d = new Date();

async function init() {
  try {
    const me = await api('/api/me');

      const a = document.createElement('a');
      a.href = '/admin';
      a.textContent = 'Painel Admin';
      document.querySelector('header nav').insertBefore(a, document.getElementById('logout'));
    }
  } catch (e) {
    window.location.href = '/';
    return;
  }

  document.getElementById('expDate').value = todayISO();

  categories = await api('/api/categories');
  const sel = document.getElementById('expCategory');

    sel.appendChild(o);
  });

}

function catName(id) {
  const c = categories.find(x => x.id === id);
  return c ? c.name : '—';
}

async function refresh() {
  const expenses = await api('/api/expenses');
  const tbody = document.getElementById('expTbody');
  tbody.innerHTML = '';
  if (expenses.length === 0) {
    tbody.innerHTML = '<tr><td colspan="5" class="muted">Nenhum gasto registrado.</td></tr>';
  }
  for (const e of expenses) {
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${e.date.split('-').reverse().join('/')}</td>
      <td>${e.description || '—'}</td>
      <td>${catName(e.categoryId)}</td>
      <td style="text-align:right">${money(e.amount)}</td>
      <td style="text-align:right"><button class="danger small" data-id="${e.id}">Excluir</button></td>
    `;
    tr.querySelector('button').onclick = async () => {
      if (!confirm('Excluir este gasto?')) return;
      await api('/api/expenses/' + e.id, { method: 'DELETE' });
      refresh();
    };
    tbody.appendChild(tr);
  }

  const s = await api('/api/summary');
  document.getElementById('totalMonth').textContent = money(s.totalMonth);
  document.getElementById('totalAll').textContent = money(s.totalAll);

  const catList = document.getElementById('catList');
  const withValues = s.byCategory.filter(c => c.total > 0).sort((a, b) => b.total - a.total);
  if (withValues.length === 0) {
    catList.innerHTML = '<div class="muted">Sem dados ainda.</div>';
  } else {
    const max = withValues[0].total;
    catList.innerHTML = withValues.map(c => `
      <div style="margin-bottom:10px">
        <div style="display:flex;justify-content:space-between;font-size:14px">
          <span>${c.name}</span>
          <strong>${money(c.total)}</strong>
        </div>
        <div class="bar"><div style="width:${(c.total / max * 100).toFixed(1)}%"></div></div>
      </div>
    `).join('');
  }
}

async function api(url, opts) {
  const r = await fetch(url, opts);
  const json = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(json.error || 'Erro');
  return json;
}

function money(v) {
  return 'R$ ' + Number(v).toFixed(2).replace('.', ',');
}

async function init() {
  try {
    const me = await api('/api/me');

  } catch (e) {
    window.location.href = '/';
    return;
  }
  await loadAll();
}

async function loadCategories() {
  const cats = await api('/api/categories');
  const tbody = document.getElementById('catTbody');
  tbody.innerHTML = '';
  cats.forEach(c => {
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${c.id}</td>
      <td id="name-${c.id}">${c.name}</td>
      <td>${c.active ? '<span class="badge ok">Ativa</span>' : '<span class="badge off">Inativa</span>'}</td>
      <td style="text-align:right">
        <button class="secondary small" data-act="rename" data-id="${c.id}">Renomear</button>
        <button class="${c.active ? 'danger' : ''} small" data-act="toggle" data-id="${c.id}" data-active="${c.active}">
          ${c.active ? 'Desativar' : 'Ativar'}
        </button>
      </td>
    `;
    tr.querySelectorAll('button').forEach(b => {
      b.onclick = async () => {
        const id = parseInt(b.dataset.id, 10);
        if (b.dataset.act === 'rename') {
          const novo = prompt('Novo nome da categoria:');
          if (!novo || !novo.trim()) return;
          await api('/api/categories/' + id, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ name: novo.trim() })
          });
        } else {
          const novoStatus = b.dataset.active === 'true' ? false : true;
          await api('/api/categories/' + id, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ active: novoStatus })
          });
        }
        loadAll();
      };
    });
    tbody.appendChild(tr);
  });
}

async function loadUsers() {
  const users = await api('/api/admin/users');
  const summary = await api('/api/admin/summary');
  const totals = {};
  summary.perUser.forEach(p => totals[p.id] = p.total);

  const me = await api('/api/me');
  const tbody = document.getElementById('userTbody');
  tbody.innerHTML = '';
  users.forEach(u => {
    const tr = document.createElement('tr');
    const isMe = u.id === me.id;
    tr.innerHTML = `
      <td>${u.id}</td>
      <td>${u.name}${isMe ? ' <span class="muted">(você)</span>' : ''}</td>
      <td>${u.email}</td>
      <td>${u.role === 'admin' ? 'Administrador' : 'Usuário'}</td>
      <td>${u.active ? '<span class="badge ok">Ativo</span>' : '<span class="badge off">Desativado</span>'}</td>
      <td style="text-align:right">${money(totals[u.id] || 0)}</td>
      <td style="text-align:right">
        ${isMe ? '' : `<button class="${u.active ? 'danger' : ''} small" data-id="${u.id}" data-active="${u.active}">
          ${u.active ? 'Desativar' : 'Ativar'}
        </button>`}
      </td>
    `;
    const btn = tr.querySelector('button');
    if (btn) {
      btn.onclick = async () => {
        const id = parseInt(btn.dataset.id, 10);
        const novoStatus = btn.dataset.active === 'true' ? false : true;
        if (novoStatus === false && !confirm('Desativar esta conta? O usuário não poderá mais entrar.')) return;
        try {
          await api('/api/admin/users/' + id, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ active: novoStatus })
          });
          loadAll();
        } catch (e) {
          alert(e.message);
        }
      };
    }
    tbody.appendChild(tr);
  });
}
