db.exec(`
CREATE TABLE IF NOT EXISTS usuarios (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  senha TEXT NOT NULL,
  tipo TEXT NOT NULL DEFAULT 'cliente'
);
CREATE TABLE IF NOT EXISTS pratos (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,
  descricao TEXT NOT NULL DEFAULT '',
  preco REAL NOT NULL,
  categoria TEXT NOT NULL DEFAULT 'Principal',
  disponivel INTEGER NOT NULL DEFAULT 1
);
CREATE TABLE IF NOT EXISTS pedidos (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  usuario_id INTEGER NOT NULL,
  status TEXT NOT NULL DEFAULT 'Pendente',
  total REAL NOT NULL,
  criado_em TEXT NOT NULL,
  FOREIGN KEY(usuario_id) REFERENCES usuarios(id)
);
CREATE TABLE IF NOT EXISTS itens_pedido (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  pedido_id INTEGER NOT NULL,
  prato_id INTEGER NOT NULL,
  nome_prato TEXT NOT NULL,
  preco REAL NOT NULL,
  quantidade INTEGER NOT NULL,
  FOREIGN KEY(pedido_id) REFERENCES pedidos(id)
);
`);

app.post('/api/cadastro', (req, res) => {
  const { nome, email, senha } = req.body;
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });
  try {
    const hash = bcrypt.hashSync(senha, 10);
    const info = db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)').run(nome, email.toLowerCase(), hash, 'cliente');
    req.session.usuario = { id: info.lastInsertRowid, nome, email: email.toLowerCase(), tipo: 'cliente' };
    res.json(req.session.usuario);
  } catch (e) {
    res.status(400).json({ erro: 'Email já cadastrado' });
  }
});

app.post('/api/login', (req, res) => {

  req.session.usuario = { id: u.id, nome: u.nome, email: u.email, tipo: u.tipo };
  res.json(req.session.usuario);
});

  db.prepare('UPDATE pratos SET nome=?, descricao=?, preco=?, categoria=?, disponivel=? WHERE id=?')
    .run(nome, descricao || '', Number(preco), categoria || 'Principal', disponivel ? 1 : 0, req.params.id);
  res.json(db.prepare('SELECT * FROM pratos WHERE id = ?').get(req.params.id));
});

// Pedidos
app.post('/api/pedidos', requerLogin, (req, res) => {
  const { itens } = req.body;
  if (!Array.isArray(itens) || itens.length === 0) return res.status(400).json({ erro: 'Pedido vazio' });
  let total = 0;
  const itensFinal = [];
  for (const it of itens) {
    const prato = db.prepare('SELECT * FROM pratos WHERE id = ?').get(it.prato_id);
    if (!prato || !prato.disponivel) return res.status(400).json({ erro: 'Prato indisponível' });
    const qtd = Number(it.quantidade) || 1;
    total += prato.preco * qtd;
    itensFinal.push({ prato_id: prato.id, nome_prato: prato.nome, preco: prato.preco, quantidade: qtd });
  }
  const agora = new Date().toISOString();
  const pedidoInfo = db.prepare('INSERT INTO pedidos (usuario_id, status, total, criado_em) VALUES (?, ?, ?, ?)')
    .run(req.session.usuario.id, 'Pendente', total, agora);
  const pedidoId = pedidoInfo.lastInsertRowid;
  const insItem = db.prepare('INSERT INTO itens_pedido (pedido_id, prato_id, nome_prato, preco, quantidade) VALUES (?, ?, ?, ?, ?)');
  for (const it of itensFinal) insItem.run(pedidoId, it.prato_id, it.nome_prato, it.preco, it.quantidade);
  res.json({ id: pedidoId });
});

function pedidoCompleto(p) {
  const itens = db.prepare('SELECT * FROM itens_pedido WHERE pedido_id = ?').all(p.id);
  return { ...p, itens };
}

app.get('/api/meus-pedidos', requerLogin, (req, res) => {
  const pedidos = db.prepare('SELECT * FROM pedidos WHERE usuario_id = ? ORDER BY id DESC').all(req.session.usuario.id);
  res.json(pedidos.map(pedidoCompleto));
});

app.get('/api/pedidos', requerGerente, (req, res) => {
  const pedidos = db.prepare(`
    SELECT p.*, u.nome as cliente_nome, u.email as cliente_email
    FROM pedidos p JOIN usuarios u ON u.id = p.usuario_id
    ORDER BY p.id DESC
  `).all();
  res.json(pedidos.map(pedidoCompleto));
});

app.put('/api/pedidos/:id/status', requerGerente, (req, res) => {
  const { status } = req.body;
  const permitidos = ['Pendente', 'Preparando', 'Pronto', 'Entregue', 'Cancelado'];
  if (!permitidos.includes(status)) return res.status(400).json({ erro: 'Status inválido' });
  db.prepare('UPDATE pedidos SET status = ? WHERE id = ?').run(status, req.params.id);
  res.json({ ok: true });
});

app.listen(PORT, () => {
  console.log('Servidor rodando em http://localhost:' + PORT);

let usuarioAtual = null;
let pratosCache = [];
let carrinho = JSON.parse(localStorage.getItem('carrinho') || '[]');

function salvarCarrinho() {
  localStorage.setItem('carrinho', JSON.stringify(carrinho));
  atualizarCartCount();
}
function atualizarCartCount() {
  const total = carrinho.reduce((s, i) => s + i.quantidade, 0);
  const el = document.getElementById('cartCount');
  if (el) el.textContent = total;
}

async function api(url, opts = {}) {
  const res = await fetch(url, {
    headers: { 'Content-Type': 'application/json' },
    credentials: 'same-origin',
    ...opts,
    body: opts.body ? JSON.stringify(opts.body) : undefined
  });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.erro || 'Erro');
  return data;
}

async function carregarUsuario() {
  usuarioAtual = await api('/api/eu');
  renderizarTela();
}

function renderizarTela() {
  const auth = document.getElementById('authScreen');
  const cli = document.getElementById('clienteScreen');
  const ger = document.getElementById('gerenteScreen');
  const userArea = document.getElementById('userArea');
  auth.classList.add('hidden');
  cli.classList.add('hidden');
  ger.classList.add('hidden');

  if (!usuarioAtual) {
    auth.classList.remove('hidden');
    userArea.innerHTML = '';
    return;
  }
  userArea.innerHTML = `<span>Olá, ${usuarioAtual.nome}</span> <button class="small" onclick="logout()">Sair</button>`;
  if (usuarioAtual.tipo === 'gerente') {
    ger.classList.remove('hidden');
    carregarPratosGerente();
    carregarTodosPedidos();
  } else {
    cli.classList.remove('hidden');
    carregarCardapio();
    carregarMeusPedidos();
    atualizarCartCount();
  }
}

// CARDAPIO
async function carregarCardapio() {
  pratosCache = await api('/api/pratos');
  const el = document.getElementById('listaPratos');
  el.innerHTML = pratosCache.map(p => `
    <div class="prato-card">
      <span class="categoria">${p.categoria}</span>
      <h4>${p.nome}</h4>
      <p class="descricao">${p.descricao || ''}</p>
      <p class="preco">R$ ${p.preco.toFixed(2)}</p>
      ${p.disponivel
        ? `<button class="primary small" onclick="adicionarAoCarrinho(${p.id})">Adicionar</button>`
        : `<p class="indisponivel">Indisponível</p>`}
    </div>
  `).join('');
}

function adicionarAoCarrinho(id) {
  const prato = pratosCache.find(p => p.id === id);
  if (!prato) return;
  const item = carrinho.find(i => i.prato_id === id);
  if (item) item.quantidade++;
  else carrinho.push({ prato_id: id, nome: prato.nome, preco: prato.preco, quantidade: 1 });
  salvarCarrinho();
  renderizarCarrinho();
  alert(prato.nome + ' adicionado ao carrinho!');
}

function renderizarCarrinho() {
  const el = document.getElementById('carrinhoItens');
  if (carrinho.length === 0) {
    el.innerHTML = '<p>Seu carrinho está vazio.</p>';
    document.getElementById('carrinhoTotal').textContent = '';
    return;
  }
  el.innerHTML = carrinho.map((i, idx) => `
    <div class="carrinho-item">
      <span class="nome">${i.nome} - R$ ${i.preco.toFixed(2)}</span>
      <div class="qtd-ctrl">
        <button class="small" onclick="alterarQtd(${idx}, -1)">-</button>
        <span>${i.quantidade}</span>
        <button class="small" onclick="alterarQtd(${idx}, 1)">+</button>
        <button class="danger" onclick="removerItem(${idx})">Remover</button>
      </div>
    </div>
  `).join('');
  const total = carrinho.reduce((s, i) => s + i.preco * i.quantidade, 0);
  document.getElementById('carrinhoTotal').textContent = 'Total: R$ ' + total.toFixed(2);
}

function alterarQtd(idx, delta) {
  carrinho[idx].quantidade += delta;
  if (carrinho[idx].quantidade <= 0) carrinho.splice(idx, 1);
  salvarCarrinho();
  renderizarCarrinho();
}
function removerItem(idx) {
  carrinho.splice(idx, 1);
  salvarCarrinho();
  renderizarCarrinho();
}

async function finalizarPedido() {
  if (carrinho.length === 0) return alert('Carrinho vazio');
  try {
    await api('/api/pedidos', {
      method: 'POST',
      body: { itens: carrinho.map(i => ({ prato_id: i.prato_id, quantidade: i.quantidade })) }
    });
    carrinho = [];
    salvarCarrinho();
    renderizarCarrinho();
    alert('Pedido realizado com sucesso!');
    carregarMeusPedidos();
    mostrarTab('meusPedidos');
  } catch (e) {
    alert('Erro: ' + e.message);
  }
}

async function carregarMeusPedidos() {
  const pedidos = await api('/api/meus-pedidos');
  const el = document.getElementById('meusPedidos');
  if (pedidos.length === 0) {
    el.innerHTML = '<p>Você ainda não fez nenhum pedido.</p>';
    return;
  }
  el.innerHTML = pedidos.map(p => `
    <div class="pedido-card">
      <div class="pedido-header">
        <strong>Pedido #${p.id}</strong>
        <span class="status ${p.status}">${p.status}</span>
        <span>${new Date(p.criado_em).toLocaleString('pt-BR')}</span>
      </div>
      <ul class="itens-lista">
        ${p.itens.map(it => `<li>${it.quantidade}x ${it.nome_prato} - R$ ${(it.preco * it.quantidade).toFixed(2)}</li>`).join('')}
      </ul>
      <div class="total">Total: R$ ${p.total.toFixed(2)}</div>
    </div>
  `).join('');
}

function editarPrato(p) {
  document.getElementById('formTitulo').textContent = 'Editar Prato';
  document.getElementById('pratoId').value = p.id;
  document.getElementById('pratoNome').value = p.nome;
  document.getElementById('pratoDescricao').value = p.descricao || '';
  document.getElementById('pratoPreco').value = p.preco;
  document.getElementById('pratoCategoria').value = p.categoria;
  document.getElementById('pratoDisponivel').checked = !!p.disponivel;
  window.scrollTo(0, 0);
}

function limparFormPrato() {
  document.getElementById('formTitulo').textContent = 'Adicionar Prato';
  document.getElementById('pratoId').value = '';
  document.getElementById('pratoNome').value = '';
  document.getElementById('pratoDescricao').value = '';
  document.getElementById('pratoPreco').value = '';
  document.getElementById('pratoCategoria').value = 'Principal';
  document.getElementById('pratoDisponivel').checked = true;
}

async function salvarPrato() {
  const id = document.getElementById('pratoId').value;
  const body = {
    nome: document.getElementById('pratoNome').value,
    descricao: document.getElementById('pratoDescricao').value,
    preco: parseFloat(document.getElementById('pratoPreco').value),
    categoria: document.getElementById('pratoCategoria').value,
    disponivel: document.getElementById('pratoDisponivel').checked
  };
  if (!body.nome || isNaN(body.preco)) return alert('Preencha nome e preço');
  try {
    if (id) {
      await api('/api/pratos/' + id, { method: 'PUT', body });
    } else {
      await api('/api/pratos', { method: 'POST', body });
    }
    limparFormPrato();
    carregarPratosGerente();
  } catch (e) {
    alert('Erro: ' + e.message);
  }
}

async function removerPrato(id) {
  if (!confirm('Remover este prato?')) return;
  await api('/api/pratos/' + id, { method: 'DELETE' });
  carregarPratosGerente();
}

# Sistema de Pedidos para Restaurante

Site completo com cadastro de clientes, cardápio, pedidos e painel do gerente. Tudo salvo em banco de dados local (SQLite), sem depender de serviços externos.

## Como instalar e executar

Você precisa ter o **Node.js** instalado (baixe em https://nodejs.org).

1. Abra o terminal (Prompt de Comando ou Terminal) na pasta do projeto.
2. Instale as dependências:

```
npm install
```

3. Inicie o servidor:

```
npm start
```

4. Abra o navegador em:

```
http://localhost:3000
```

## Contas

- **Gerente padrão** (criado automaticamente):
  - Email: `gerente@restaurante.com`
  - Senha: `admin123`

- **Clientes**: crie a conta na tela inicial (aba de cadastro).

## O que cada um pode fazer

**Cliente:**
- Ver o cardápio
- Adicionar pratos ao carrinho e finalizar pedido
- Acompanhar o status de seus pedidos

**Gerente:**
- Cadastrar, editar e remover pratos do cardápio
- Ver todos os pedidos de todos os clientes
- Alterar a situação de qualquer pedido (Pendente, Preparando, Pronto, Entregue, Cancelado)

## Colocar na internet

Para publicar na internet de verdade, hospede em serviços como Render, Railway ou Fly.io (todos aceitam projetos Node.js). O arquivo `restaurante.db` guarda todos os dados permanentemente. Em produção, recomenda-se usar um disco persistente para manter o banco.
