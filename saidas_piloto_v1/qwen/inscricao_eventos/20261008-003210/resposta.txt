=== src/lib/db.js ===
// Banco de dados simples usando o localStorage do navegador.
// Tudo fica salvo no seu computador e continua lá mesmo fechando o site.
// Para que várias pessoas diferentes consigam usar, o site precisa ser
// publicado na internet (veja o README.md) para todos verem a mesma página.

const K = {
  users: 'portal.eventos.users',
  events: 'portal.eventos.eventos',
  insc: 'portal.eventos.inscricoes',
  session: 'portal.eventos.sessao'
};

function read(key, fallback) {
  try {
    const raw = localStorage.getItem(key);
    if (!raw) return fallback;
    return JSON.parse(raw);
  } catch (e) {
    return fallback;
  }
}

function write(key, value) {
  localStorage.setItem(key, JSON.stringify(value));
}

export function getUsers() {
  return read(K.users, []);
}

export function getEvents() {
  return read(K.events, []);
}

export function getInscriptions() {
  return read(K.insc, []);
}

export function setUsers(u) { write(K.users, u); }
export function setEvents(e) { write(K.events, e); }
export function setInscriptions(i) { write(K.insc, i); }

export function getSession() {
  return read(K.session, null);
}

export function setSession(userId) {
  if (userId) {
    write(K.session, userId);
  } else {
    localStorage.removeItem(K.session);
  }
}

// --- Usuários ---

export function createUser({ name, email, password, role }) {
  const users = getUsers();
  if (users.some(u => u.email.toLowerCase() === email.toLowerCase())) {
    return { ok: false, error: 'Já existe uma conta com este e-mail.' };
  }
  const user = { id: 'u_' + Date.now() + Math.random().toString(16).slice(2, 7), name, email, password, role };
  users.push(user);
  setUsers(users);
  return { ok: true, user };
}

export function login(email, password) {
  const users = getUsers();
  const user = users.find(u => u.email.toLowerCase() === email.toLowerCase());
  if (!user) return { ok: false, error: 'Conta não encontrada. Verifique o e-mail.' };
  if (user.password !== password) return { ok: false, error: 'Senha incorreta.' };
  setSession(user.id);
  return { ok: true, user };
}

export function getUserById(id) {
  return getUsers().find(u => u.id === id) || null;
}

// --- Eventos ---

export function createEvent(data) {
  const events = getEvents();
  const ev = {
    id: 'e_' + Date.now() + Math.random().toString(16).slice(2, 7),
    status: 'aberto',
    createdAt: Date.now(),
    ...data
  };
  events.unshift(ev);
  setEvents(events);
  return ev;
}

export function updateEvent(id, patch) {
  const events = getEvents();
  const idx = events.findIndex(e => e.id === id);
  if (idx === -1) return null;
  events[idx] = { ...events[idx], ...patch };
  setEvents(events);
  return events[idx];
}

export function setEventStatus(id, status) {
  return updateEvent(id, { status });
}

// --- Inscrições ---

export function addInscription({ eventId, userId, name, email, phone, note }) {
  const insc = getInscriptions();
  // Evita a mesma pessoa se inscrever duas vezes no mesmo evento
  const exists = insc.some(i => i.eventId === eventId && i.userId === userId);
  if (exists) return { ok: false, error: 'Você já está inscrito(a) neste evento.' };
  const item = {
    id: 'i_' + Date.now() + Math.random().toString(16).slice(2, 7),
    eventId, userId,
    name, email, phone: phone || '', note: note || '',
    status: 'pendente',
    createdAt: Date.now()
  };
  insc.push(item);
  setInscriptions(insc);
  return { ok: true, item };
}

export function cancelInscription(id) {
  setInscriptions(getInscriptions().filter(i => i.id !== id));
}

export function setInscriptionStatus(id, status) {
  const insc = getInscriptions();
  const idx = insc.findIndex(i => i.id === id);
  if (idx === -1) return null;
  insc[idx] = { ...insc[idx], status };
  setInscriptions(insc);
  return insc[idx];
}

export function getEventsForUser() {
  const insc = getInscriptions();
  const ids = new Set(insc.filter(i => i.status !== 'cancelado').map(i => i.eventId));
  return getEvents().filter(e => ids.has(e.id));
}

export function getInscriptionsForUser(userId) {
  return getInscriptions().filter(i => i.userId === userId);
}

// --- Utilidades de exibição ---

export function formatDate(s) {
  if (!s) return 'Data a definir';
  const d = new Date(s);
  if (isNaN(d.getTime())) return s;
  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: 'long', year: 'numeric' });
}

export function formatTime(s) {
  if (!s) return '';
  const d = new Date(s);
  if (isNaN(d.getTime())) return s;
  return d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
}

export function formatDateTime(s) {
  if (!s) return '';
  const d = new Date(s);
  if (isNaN(d.getTime())) return s;
  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit', year: 'numeric' }) +
    ' às ' + d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
}

import React, { useState, useEffect, useCallback } from 'react';
import { Routes, Route, Link, Navigate, useLocation } from 'react-router-dom';
import * as db from './lib/db.js';
import Topbar from './components/Topbar.jsx';
import LoginPage from './pages/LoginPage.jsx';
import ListEventsPage from './pages/ListEventsPage.jsx';
import RegisterPage from './pages/RegisterPage.jsx';
import MyInscriptionsPage from './pages/MyInscriptionsPage.jsx';
import ManageEventsPage from './pages/ManageEventsPage.jsx';
import EventFormPage from './pages/EventFormPage.jsx';
import AllInscriptionsPage from './pages/AllInscriptionsPage.jsx';
import NotFoundPage from './pages/NotFoundPage.jsx';

function useSession() {
  const [user, setUser] = useState(() => {
    const id = db.getSession();
    return id ? db.getUserById(id) : null;
  });

  const login = (u) => setUser(u);
  const logout = () => {
    db.setSession(null);
    setUser(null);
  };
  return { user, login, logout };
}

function RequireAuth({ user, children }) {
  const loc = useLocation();
  if (!user) return <Navigate to="/entrar" state={{ from: loc }} replace />;
  return children;
}

export default function App() {
  const { user, login, logout } = useSession();

  return (
    <div>
      <Topbar user={user} onLogout={logout} />
      <div className="container">
        <Routes>
          <Route path="/" element={user ? <Navigate to="/eventos" replace /> : <Navigate to="/entrar" replace />} />
          <Route path="/entrar" element={<LoginPage user={user} onLogin={login} />} />

          <Route path="/eventos" element={
            <RequireAuth user={user}>
              <ListEventsPage user={user} />
            </RequireAuth>
          } />

          <Route path="/inscrever/:id" element={
            <RequireAuth user={user}>
              <RegisterPage user={user} />
            </RequireAuth>
          } />

          <Route path="/minhas-inscricoes" element={
            <RequireAuth user={user}>
              <MyInscriptionsPage user={user} />
            </RequireAuth>
          } />

          <Route path="/organizar/eventos" element={
            <RequireAuth user={user}>
              <ManageEventsPage user={user} />
            </RequireAuth>
          } />

          <Route path="/organizar/eventos/novo" element={
            <RequireAuth user={user}>
              <EventFormPage user={user} mode="novo" />
            </RequireAuth>
          } />

          <Route path="/organizar/eventos/:id" element={
            <RequireAuth user={user}>
              <EventFormPage user={user} mode="editar" />
            </RequireAuth>
          } />

          <Route path="/organizar/inscricoes" element={
            <RequireAuth user={user}>
              <AllInscriptionsPage user={user} />
            </RequireAuth>
          } />

          <Route path="*" element={<NotFoundPage />} />
        </Routes>
      </div>
    </div>
  );
}

import React, { useState } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import * as db from '../lib/db.js';

const ADMIN_EMAIL = 'organizador@evento.com';
const ADMIN_PASS = 'organizar123';

export default function LoginPage({ user, onLogin }) {
  const nav = useNavigate();
  const loc = useLocation();
  const [mode, setMode] = useState('login');
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');

  if (user) {
    const dest = loc.state?.from?.pathname || (user.role === 'admin' ? '/organizar/eventos' : '/eventos');
    return <RedirectTo path={dest} />;
  }

  function finish(u) {
    onLogin(u);
    const dest = loc.state?.from?.pathname || (u.role === 'admin' ? '/organizar/eventos' : '/eventos');
    nav(dest, { replace: true });
  }

  function doLogin(e) {
    e.preventDefault();
    setError('');
    const res = db.login(email.trim(), password);
    if (!res.ok) { setError(res.error); return; }
    finish(res.user);
  }

  function doRegister(e) {
    e.preventDefault();
    setError('');
    const res = db.createUser({ name: name.trim(), email: email.trim(), password, role: 'participante' });
    if (!res.ok) { setError(res.error); return; }
    finish(res.user);
  }

  return (
    <main>
      <div className="card auth-card">
        <div className="card__body">
          <h2 style={{ fontSize: 22, marginBottom: 4 }}>{mode === 'login' ? 'Entrar' : 'Criar conta'}</h2>
          <p className="sub" style={{ marginBottom: 18 }}>
            {mode === 'login' ? 'Use sua conta para acessar o portal.' : 'Cadastre-se para ver e se inscrever nos eventos.'}
          </p>

          <div className="tabs">
            <div className={'tab ' + (mode === 'login' ? 'active' : '')} onClick={() => { setMode('login'); setError(''); }}>Entrar</div>
            <div className={'tab ' + (mode === 'register' ? 'active' : '')} onClick={() => { setMode('register'); setError(''); }}>Criar conta</div>
          </div>

          {error && <div className="form-error">{error}</div>}

          {mode === 'login' ? (
            <form onSubmit={doLogin}>
              <div className="field">
                <label>E-mail</label>
                <input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required />
              </div>
              <div className="field">
                <label>Senha</label>
                <input type="password" value={password} onChange={(e) => setPassword(e.target.value)} required />
              </div>
              <button className="btn btn--primary" type="submit" style={{ width: '100%', justifyContent: 'center' }}>Entrar</button>
            </form>
          ) : (
            <form onSubmit={doRegister}>
              <div className="field">
                <label>Nome completo</label>
                <input value={name} onChange={(e) => setName(e.target.value)} required />
              </div>
              <div className="field">
                <label>E-mail</label>
                <input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required />
              </div>
              <div className="field">
                <label>Senha</label>
                <input type="password" value={password} onChange={(e) => setPassword(e.target.value)} required />
                <span className="hint">Só você vai ver sua senha. Guarde-a bem.</span>
              </div>
              <button className="btn btn--primary" type="submit" style={{ width: '100%', justifyContent: 'center' }}>Criar conta</button>
            </form>
          )}

                            <button className="btn btn--primary btn--sm" onClick={() => setStatus(r.id, 'confirmada')}>Confirmar</button>
                          )}
                          {r.status !== 'cancelada' && (
                            <button className="btn btn--danger btn--sm" onClick={() => cancel(r.id)}>Cancelar</button>
                          )}
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>
      )}
    </main>
  );
}

=== README.md ===
# Portal de Inscrições para Eventos

Um site completo para **participantes** e **organizador** de eventos.

- **Participantes:** entram com a própria conta, veem os eventos disponíveis, se inscrevem e veem/cancelam as inscrições.
- **Organizador:** cria, edita e cancela eventos, confirma a presença dos inscritos e vê todas as inscrições de todos os participantes.

## Como rodar no seu computador

Você precisa ter o **Node.js** instalado (versão 18 ou mais nova). Baixe em https://nodejs.org

Abra o terminal na pasta do projeto e digite:

```bash
npm install
npm run dev
```

Depois abra no navegador o endereço que aparecer no terminal, normalmente:

```
http://localhost:5173
```

## Publicar na internet (para ficar acessível para todo mundo)

Como o site precisa "funcionar de verdade na internet", publique o código abaixo na **plataforma Vercel** (gratuita):

1. Crie uma conta grátis em https://vercel.com
2. Faça o upload (ou conecte o GitHub) desta pasta de projeto
3. A Vercel vai detectar o projeto e gerar um link como `https://seu-nome.vercel.app`

Quem abrir esse link vai usar o **mesmo site**.

## Como funcionam as contas

- **Participantes:** crie quantas contas quiser usando a tela “Criar conta”.
- **Organizador (pronta para testar):**
  - **E-mail:** `organizador@evento.com`
  - **Senha:** `organizar123`

> A conta de organizador é criada automaticamente no primeiro acesso. Você pode criar outros organizadores depois, se quiser.

## Onde ficam guardados os dados

Tudo fica salvo **dentro do próprio site**, no armazenamento do navegador (localStorage). Os dados continuam lá mesmo fechando e abrindo o site novamente.

**Importante sobre “tudo dentro do próprio site”:** o navegador guarda os dados de cada dispositivo à parte. Para que **mais de uma pessoa** veja as mesmas informações ao mesmo tempo, o ideal é publicar o site (na Vercel, por exemplo) para que todas usem o mesmo endereço. Se cada pessoa abrir no seu próprio computador sem publicar, cada uma verá os seus próprios dados.
