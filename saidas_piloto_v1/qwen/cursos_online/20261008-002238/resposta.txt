# Plataforma de Cursos Online

Site completo para cadastro e matrícula em cursos online, com área de aluno e área de administrador. Todo o conteúdo (cursos, matrículas e contas) fica salvo dentro do próprio navegador, então as informações permanecem mesmo se você fechar e abrir o site de novo.

## Como instalar e executar

Você precisa ter o **Node.js** instalado (instale em https://nodejs.org se ainda não tiver).

1. Abra o terminal dentro da pasta do projeto.
2. Rode o comando:
   ```
   npm install
   ```
3. Para abrir o site no seu computador (modo de desenvolvimento):
   ```
   npm run dev
   ```
   Depois abra o navegador em `http://localhost:5173`.

4. Para publicar na internet (produção):
   ```
   npm run build
   ```
   O site pronto para publicar fica na pasta `dist`.
   Você pode publicar a pasta `dist` em qualquer hospedagem estática como **Netlify**, **Vercel** ou **GitHub Pages**.

> Dica rápida para Netlify: no painel, crie um site novo e arraste a pasta `dist`. Pronto.

## Como funciona

- Na tela inicial, escolha **Entrar como Aluno** ou **Entrar como Administrador**.
- **Conta padrão do administrador** (já vem criada):
  - Usuário: `admin`
  - Senha: `admin123`
- Os alunos escolhem seu nome, usuário e senha quando entram pela primeira vez (é criado o cadastro automaticamente).
- Depois de entrar, o aluno pode: ver os cursos disponíveis, se matricular e ver seus cursos matriculados.
- O administrador pode: cadastrar, editar e remover cursos, ver todas as matrículas de todos os alunos e cancelar matrículas.

## Observação importante sobre "funcionar na internet"

Nesta versão, os dados ficam guardados no navegador de cada pessoa (localStorage). Ou seja:

- O site funciona de verdade na internet, mas cada visitante vê e salva os dados no seu próprio navegador.
- Um aluno que matricula em um curso não vê essa matrícula na tela do administrador (e vice-versa), pois não há um servidor compartilhando os dados entre todos.

Se você quiser que **todos os usuários vejam os mesmos cursos e matrículas de verdade** (administrador vê as matrículas de todos os alunos), é necessário uma base de dados em um servidor (backend). Posso criar essa versão também se você me pedir.

// Camadas de armazenamento usando o localStorage do navegador.
// Tudo fica salvo no próprio navegador e continua disponível
// mesmo depois de fechar e reabrir o site.

const KEYS = {
  courses: 'plataforma_cursos',
  students: 'plataforma_alunos',
  enrollments: 'plataforma_matriculas',
  session: 'plataforma_sessao',
}

function read(key, fallback) {
  try {
    const raw = localStorage.getItem(key)
    return raw ? JSON.parse(raw) : fallback
  } catch {
    return fallback
  }
}

function write(key, value) {
  localStorage.setItem(key, JSON.stringify(value))
}

function uid() {
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 8)
}

// Garante que o administrador padrão exista na primeira vez.
function ensureAdmin() {
  const students = read(KEYS.students, [])
  if (!students.some((s) => s.login === 'admin')) {
    students.push({
      id: 'admin',
      name: 'Administrador',
      login: 'admin',
      password: 'admin123',
      isAdmin: true,
    })
    write(KEYS.students, students)
  }
}

export const db = {
  // Cursos
  getCourses() {
    return read(KEYS.courses, [])
  },
  saveCourses(courses) {
    write(KEYS.courses, courses)
  },
  addCourse(course) {
    const courses = this.getCourses()
    course.id = uid()
    course.createdAt = new Date().toISOString()
    courses.push(course)
    this.saveCourses(courses)
    return course
  },
  updateCourse(id, changes) {
    const courses = this.getCourses().map((c) =>
      c.id === id ? { ...c, ...changes } : c
    )
    this.saveCourses(courses)
    return courses.find((c) => c.id === id)
  },
  removeCourse(id) {
    this.saveCourses(this.getCourses().filter((c) => c.id !== id))
    // Remove também as matrículas do curso
    this.saveEnrollments(this.getEnrollments().filter((e) => e.courseId !== id))
  },

  // Alunos (e admin)
  getStudents() {
    ensureAdmin()
    return read(KEYS.students, [])
  },
  saveStudents(students) {
    write(KEYS.students, students)
  },
  findStudentByLogin(login) {
    return this.getStudents().find((s) => s.login.toLowerCase() === login.toLowerCase())
  },
  addStudent(data) {
    const students = this.getStudents()
    const student = {
      id: uid(),
      name: data.name,
      login: data.login,
      password: data.password,
      isAdmin: false,
    }
    students.push(student)
    this.saveStudents(students)
    return student
  },

  // Matrículas
  getEnrollments() {
    return read(KEYS.enrollments, [])
  },
  saveEnrollments(enrollments) {
    write(KEYS.enrollments, enrollments)
  },
  addEnrollment(courseId, studentId) {
    const enrollments = this.getEnrollments()
    if (enrollments.some((e) => e.courseId === courseId && e.studentId === studentId)) {
      return null // já matriculado
    }
    const enrollment = {
      id: uid(),
      courseId,
      studentId,
      createdAt: new Date().toISOString(),
    }
    enrollments.push(enrollment)
    this.saveEnrollments(enrollments)
    return enrollment
  },
  removeEnrollment(id) {
    this.saveEnrollments(this.getEnrollments().filter((e) => e.id !== id))
  },
  hasEnrollment(courseId, studentId) {
    return this.getEnrollments().some(
      (e) => e.courseId === courseId && e.studentId === studentId
    )
  },
  enrollmentsForStudent(studentId) {
    return this.getEnrollments().filter((e) => e.studentId === studentId)
  },

  // Sessão (quem está logado)
  getSession() {
    return read(KEYS.session, null)
  },
  setSession(session) {
    write(KEYS.session, session)
  },
  clearSession() {
    localStorage.removeItem(KEYS.session)
  },
}

import React from 'react'
import { Routes, Route, Navigate, useNavigate, useLocation } from 'react-router-dom'
import { db } from './storage.js'
import Home from './pages/Home.jsx'
import StudentLogin from './pages/StudentLogin.jsx'
import AdminLogin from './pages/AdminLogin.jsx'
import StudentDashboard from './pages/StudentDashboard.jsx'
import EnrolledCourses from './pages/EnrolledCourses.jsx'
import AdminDashboard from './pages/AdminDashboard.jsx'
import EnrollmentsAdmin from './pages/EnrollmentsAdmin.jsx'
import Layout from './components/Layout.jsx'

function useAuth() {
  const [session, setSession] = React.useState(db.getSession())
  const login = (student) => {
    const s = { studentId: student.id, isAdmin: student.isAdmin }
    db.setSession(s)
    setSession(s)
  }
  const logout = () => {
    db.clearSession()
    setSession(null)
  }
  const currentStudent = session ? db.getStudents().find((s) => s.id === session.studentId) : null
  return { session, login, logout, currentStudent }
}

const AuthContext = React.createContext(null)

export function AuthProvider({ children }) {
  const auth = useAuth()
  return <AuthContext.Provider value={auth}>{children}</AuthContext.Provider>
}

export function useAuthContext() {
  return React.useContext(AuthContext)
}

function RequireAuth({ children, role }) {
  const { session, currentStudent } = useAuthContext()
  const navigate = useNavigate()
  const location = useLocation()

  if (!session || !currentStudent) {
    return <Navigate to="/" replace />
  }
  if (role && role !== 'any' && currentStudent.isAdmin !== (role === 'admin')) {
    return <Navigate to={currentStudent.isAdmin ? '/admin' : '/aluno'} replace />
  }
  return children
}

function AppRoutes() {
  const { session, currentStudent } = useAuthContext()
  const defaultPath = session
    ? currentStudent?.isAdmin
      ? '/admin'
      : '/aluno'
    : '/'

  return (
    <Routes>
      <Route path="/" element={<Home />} />
      <Route path="/entrar/aluno" element={<StudentLogin />} />
      <Route path="/entrar/admin" element={<AdminLogin />} />

      <Route
        path="/aluno"
        element={
          <RequireAuth role="student">
            <StudentDashboard />
          </RequireAuth>
        }
      />
      <Route
        path="/aluno/meus-cursos"
        element={
          <RequireAuth role="student">
            <EnrolledCourses />
          </RequireAuth>
        }
      />
      <Route
        path="/admin"
        element={
          <RequireAuth role="admin">
            <AdminDashboard />
          </RequireAuth>
        }
      />
      <Route
        path="/admin/matriculas"
        element={
          <RequireAuth role="admin">
            <EnrollmentsAdmin />
          </RequireAuth>
        }
      />

      <Route path="*" element={<Navigate to={defaultPath} replace />} />
    </Routes>
  )
}

export default function App() {
  return (
    <AuthProvider>
      <AppRoutes />
    </AuthProvider>
  )
}

import React, { useState } from 'react'
import { useNavigate } from 'react-router-dom'
import { db } from '../storage.js'
import { useAuthContext } from '../App.jsx'

export default function StudentLogin() {
  const { login } = useAuthContext()
  const navigate = useNavigate()
  const [mode, setMode] = useState('login')
  const [name, setName] = useState('')
  const [loginField, setLoginField] = useState('')
  const [password, setPassword] = useState('')
  const [error, setError] = useState('')

  const handle = (e) => {
    e.preventDefault()
    setError('')

    if (mode === 'register') {
      if (!name.trim() || !loginField.trim() || !password.trim()) {
        setError('Preencha nome, usuário e senha.')
        return
      }
      if (password.length < 4) {
        setError('A senha precisa ter pelo menos 4 caracteres.')
        return
      }
      if (db.findStudentByLogin(loginField)) {
        setError('Este usuário já existe. Faça login.')
        return
      }
      const student = db.addStudent({
        name: name.trim(),
        login: loginField.trim(),
        password,
      })
      login(student)
      navigate('/aluno')
    } else {
      const student = db.findStudentByLogin(loginField)
      if (!student || student.password !== password) {
        setError('Usuário ou senha incorretos.')
        return
      }
      if (student.isAdmin) {
        setError('Este é um usuário de administrador. Entre pela área do administrador.')
        return
      }
      login(student)
      navigate('/aluno')
    }

  const [message, setMessage] = useState('')

    setEditing(null)
    setShowForm(true)
  }
  const openEdit = (course) => {
    setEditing(course)
    setShowForm(true)
  }
  const closeForm = () => {
    setShowForm(false)
    setEditing(null)
    refresh()
  }
  const save = (data) => {
    if (editing) {
      db.updateCourse(editing.id, data)
      setMessage('Curso atualizado!')
    } else {
      db.addCourse(data)
      setMessage('Curso cadastrado!')

    closeForm()
    setTimeout(() => setMessage(''), 2500)
  }
  const remove = (course) => {
    if (
      window.confirm(

      )
    ) {
      db.removeCourse(course.id)
      refresh()
      setMessage('Curso removido.')

    }
  }

  return (
    <Layout
      title="Gestão de Cursos"
      subtitle="Cadastre, edite e remova os cursos da plataforma."
      actions={
        !showForm && (

export default function CourseForm({ initial, onSave, onCancel }) {
  const [title, setTitle] = useState(initial?.title || '')
