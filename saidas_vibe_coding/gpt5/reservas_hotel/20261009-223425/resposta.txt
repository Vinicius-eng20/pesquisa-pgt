=== README.md ===
# Sistema de Reservas de Hotel

Este projeto é um site completo de reservas de hotel, com:

- Cadastro e login de hóspedes
- Login de gerente
- Quartos cadastrados no banco de dados
- Reservas salvas no servidor
- Consulta de disponibilidade por datas
- Cancelamento de reservas por hóspedes
- Confirmação ou recusa de reservas pelo gerente
- Cadastro, edição e remoção de quartos pelo gerente
- Interface responsiva para computador e celular

Os dados ficam salvos em um banco SQLite no servidor, no arquivo `hotel.db`.

## Requisitos

Instale antes:

- Node.js versão 18 ou superior

## Como instalar

Abra o terminal na pasta do projeto e execute:

```bash
npm install
```

## Como iniciar

```bash
npm start
```

Depois abra no navegador:

```text
http://localhost:3000
```

## Conta inicial do gerente

Ao iniciar o projeto pela primeira vez, o sistema cria automaticamente uma conta de gerente:

```text
E-mail: gerente@hotel.com
Senha: gerente123
```

## Observações importantes

- Para usar na internet de verdade, este projeto precisa ser colocado em um servidor com Node.js.
- O banco de dados fica no arquivo `hotel.db`.
- Não depende de serviços externos.
- As informações não ficam apenas no navegador: ficam salvas no servidor.

function requireLogin(req, res, next) {
  if (!req.session.user) {
    return res.status(401).json({ error: "Você precisa entrar na sua conta." });
  }
  next();
}

function requireManager(req, res, next) {
  if (!req.session.user || req.session.user.role !== "manager") {
    return res.status(403).json({ error: "Acesso permitido apenas ao gerente." });
  }
  next();
}

function isValidDate(value) {
  return /^\d{4}-\d{2}-\d{2}$/.test(value) && !Number.isNaN(new Date(value).getTime());
}

function validateReservationDates(checkin, checkout) {
  if (!isValidDate(checkin) || !isValidDate(checkout)) {
    return "Informe datas válidas.";
  }

  const inDate = new Date(checkin + "T00:00:00");
  const outDate = new Date(checkout + "T00:00:00");
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  if (inDate < today) {
    return "A data de entrada não pode ser no passado.";
  }

  if (outDate <= inDate) {
    return "A data de saída precisa ser depois da data de entrada.";
  }

  return null;
}

function roomIsAvailable(roomId, checkin, checkout, ignoreReservationId = null) {
  let sql = `
    SELECT COUNT(*) AS total
    FROM reservations
    WHERE room_id = ?
      AND status IN ('pending', 'confirmed')
      AND checkin < ?
      AND checkout > ?
  `;
  const params = [roomId, checkout, checkin];

  if (ignoreReservationId) {
    sql += " AND id != ?";
    params.push(ignoreReservationId);
  }

  const result = db.prepare(sql).get(...params);
  return result.total === 0;
}

app.post("/api/register", (req, res) => {
  const name = String(req.body.name || "").trim();
  const email = String(req.body.email || "").trim().toLowerCase();
  const password = String(req.body.password || "");

  if (!name || !email || !password) {
    return res.status(400).json({ error: "Preencha nome, e-mail e senha." });
  }

  if (password.length < 6) {
    return res.status(400).json({ error: "A senha precisa ter pelo menos 6 caracteres." });
  }

  const exists = db.prepare("SELECT id FROM users WHERE email = ?").get(email);
  if (exists) {
    return res.status(400).json({ error: "Já existe uma conta com este e-mail." });
  }

  const hash = bcrypt.hashSync(password, 10);
  const result = db.prepare("INSERT INTO users (name, email, password_hash, role) VALUES (?, ?, ?, 'guest')").run(
    name,
    email,
    hash
  );

  req.session.user = {
    id: result.lastInsertRowid,
    name,
    email,
    role: "guest"
  };

  res.json({ user: req.session.user });
});

app.post("/api/login", (req, res) => {
  const email = String(req.body.email || "").trim().toLowerCase();
  const password = String(req.body.password || "");

  const user = db.prepare("SELECT * FROM users WHERE email = ?").get(email);
  if (!user || !bcrypt.compareSync(password, user.password_hash)) {
    return res.status(400).json({ error: "E-mail ou senha incorretos." });
  }

  req.session.user = {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role
  };

  res.json({ user: req.session.user });
});

app.get("/api/rooms/available", (req, res) => {
  const checkin = String(req.query.checkin || "");
  const checkout = String(req.query.checkout || "");
  const guests = Number(req.query.guests || 1);

  const dateError = validateReservationDates(checkin, checkout);
  if (dateError) {
    return res.status(400).json({ error: dateError });
  }

  if (!Number.isInteger(guests) || guests < 1) {
    return res.status(400).json({ error: "Informe a quantidade de hóspedes." });
  }

  const rooms = db.prepare(`
    SELECT *
    FROM rooms
    WHERE active = 1
      AND capacity >= ?
      AND id NOT IN (
        SELECT room_id
        FROM reservations
        WHERE status IN ('pending', 'confirmed')
          AND checkin < ?
          AND checkout > ?
      )
    ORDER BY price ASC
  `).all(guests, checkout, checkin);

  res.json({ rooms });
});

app.post("/api/reservations", requireLogin, (req, res) => {
  if (req.session.user.role !== "guest") {
    return res.status(403).json({ error: "Apenas hóspedes podem criar reservas." });
  }

  const roomId = Number(req.body.roomId);
  const checkin = String(req.body.checkin || "");
  const checkout = String(req.body.checkout || "");
  const guests = Number(req.body.guests || 1);

  const dateError = validateReservationDates(checkin, checkout);
  if (dateError) {
    return res.status(400).json({ error: dateError });
  }

  if (!Number.isInteger(roomId) || !Number.isInteger(guests) || guests < 1) {
    return res.status(400).json({ error: "Dados da reserva inválidos." });
  }

  const room = db.prepare("SELECT * FROM rooms WHERE id = ? AND active = 1").get(roomId);
  if (!room) {
    return res.status(404).json({ error: "Quarto não encontrado." });
  }

  if (guests > room.capacity) {
    return res.status(400).json({ error: "Este quarto não comporta essa quantidade de hóspedes." });
  }

  if (!roomIsAvailable(roomId, checkin, checkout)) {
    return res.status(400).json({ error: "Este quarto não está mais disponível nessas datas." });
  }

  const result = db.prepare(`
    INSERT INTO reservations (user_id, room_id, checkin, checkout, guests, status)
    VALUES (?, ?, ?, ?, ?, 'pending')
  `).run(req.session.user.id, roomId, checkin, checkout, guests);

  res.json({ id: result.lastInsertRowid, message: "Reserva criada. Aguarde confirmação do gerente." });
});

app.get("/api/my-reservations", requireLogin, (req, res) => {
  if (req.session.user.role !== "guest") {
    return res.status(403).json({ error: "Apenas hóspedes acessam esta área." });
  }

  const reservations = db.prepare(`
    SELECT r.*, rooms.name AS room_name, rooms.price AS room_price, rooms.image AS room_image
    FROM reservations r
    JOIN rooms ON rooms.id = r.room_id
    WHERE r.user_id = ?
    ORDER BY r.created_at DESC
  `).all(req.session.user.id);

  res.json({ reservations });
});

app.post("/api/my-reservations/:id/cancel", requireLogin, (req, res) => {
  if (req.session.user.role !== "guest") {
    return res.status(403).json({ error: "Apenas hóspedes podem cancelar suas reservas." });
  }

  const id = Number(req.params.id);
  const reservation = db.prepare("SELECT * FROM reservations WHERE id = ? AND user_id = ?").get(id, req.session.user.id);

  if (!reservation) {
    return res.status(404).json({ error: "Reserva não encontrada." });
  }

  if (reservation.status === "cancelled") {
    return res.status(400).json({ error: "Esta reserva já foi cancelada." });
  }

  if (reservation.status === "rejected") {
    return res.status(400).json({ error: "Esta reserva já foi recusada." });
  }

  db.prepare("UPDATE reservations SET status = 'cancelled' WHERE id = ?").run(id);
  res.json({ ok: true });
});

    UPDATE rooms
    SET name = ?, description = ?, capacity = ?, price = ?, image = ?, active = ?

  `).run(name, description, capacity, price, image, active, id);

  res.json({ ok: true });
});

app.delete("/api/manager/rooms/:id", requireManager, (req, res) => {
  const id = Number(req.params.id);

  const room = db.prepare("SELECT id FROM rooms WHERE id = ?").get(id);
  if (!room) {
    return res.status(404).json({ error: "Quarto não encontrado." });
  }

  const hasReservations = db.prepare("SELECT COUNT(*) AS total FROM reservations WHERE room_id = ?").get(id).total;

  if (hasReservations > 0) {
    db.prepare("UPDATE rooms SET active = 0 WHERE id = ?").run(id);
  } else {
    db.prepare("DELETE FROM rooms WHERE id = ?").run(id);
  }

  res.json({ ok: true });
});

app.post("/api/manager/reservations/:id/status", requireManager, (req, res) => {
  const id = Number(req.params.id);
  const status = String(req.body.status || "");

  if (!["confirmed", "rejected"].includes(status)) {
    return res.status(400).json({ error: "Status inválido." });
  }

  const reservation = db.prepare("SELECT * FROM reservations WHERE id = ?").get(id);
  if (!reservation) {
    return res.status(404).json({ error: "Reserva não encontrada." });
  }

  if (reservation.status === "cancelled") {
    return res.status(400).json({ error: "Não é possível alterar uma reserva cancelada." });
  }

  if (status === "confirmed" && !roomIsAvailable(reservation.room_id, reservation.checkin, reservation.checkout, reservation.id)) {
    return res.status(400).json({ error: "Não é possível confirmar: existe outra reserva pendente ou confirmada no mesmo período." });
  }

  db.prepare("UPDATE reservations SET status = ? WHERE id = ?").run(status, id);
  res.json({ ok: true });
});

let lastSearch = null;

function showToast(message) {
  toast.textContent = message;
  toast.classList.remove("hidden");
  setTimeout(() => toast.classList.add("hidden"), 3500);
}

async function api(url, options = {}) {
  const response = await fetch(url, {
    headers: {
      "Content-Type": "application/json",
      ...(options.headers || {})
    },
    ...options
  });

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(data.error || "Erro ao comunicar com o servidor.");
  }

  return data;
}

function money(value) {
  return Number(value).toLocaleString("pt-BR", {
    style: "currency",
    currency: "BRL"
  });
}

function statusText(status) {
  return {
    pending: "Pendente",
    confirmed: "Confirmada",
    rejected: "Recusada",
    cancelled: "Cancelada"
  }[status] || status;
}

function formatDate(date) {
  return new Date(date + "T00:00:00").toLocaleDateString("pt-BR");
}

function setTodayLimits() {
  const today = new Date().toISOString().slice(0, 10);
  searchForm.checkin.min = today;
  searchForm.checkout.min = today;
}

function renderNav() {
  if (!currentUser) {
    nav.innerHTML = "";
    return;
  }

  nav.innerHTML = `
    <span>Olá, <strong>${escapeHtml(currentUser.name)}</strong></span>
    <button class="secondary" id="logoutBtn">Sair</button>
  `;

  document.getElementById("logoutBtn").onclick = async () => {
    await api("/api/logout", { method: "POST" });
    currentUser = null;
    renderApp();
  };
}

function renderApp() {
  renderNav();

  authView.classList.toggle("hidden", !!currentUser);
  guestView.classList.toggle("hidden", !currentUser || currentUser.role !== "guest");
  managerView.classList.toggle("hidden", !currentUser || currentUser.role !== "manager");

  if (currentUser?.role === "guest") {
    loadMyReservations();
  }

  if (currentUser?.role === "manager") {
    loadManagerRooms();
    loadManagerReservations();
  }
}

function escapeHtml(text) {
  return String(text || "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

loginForm.addEventListener("submit", async (event) => {
  event.preventDefault();

  try {
    const form = new FormData(loginForm);
    const data = await api("/api/login", {
      method: "POST",
      body: JSON.stringify({
        email: form.get("email"),
        password: form.get("password")
      })
    });

    currentUser = data.user;
    loginForm.reset();
    renderApp();
    showToast("Login realizado com sucesso.");
  } catch (error) {
    showToast(error.message);
  }
});

searchForm.addEventListener("submit", async (event) => {

  const form = new FormData(searchForm);
  lastSearch = {
    checkin: form.get("checkin"),
    checkout: form.get("checkout"),
    guests: form.get("guests")

  await loadAvailableRooms();

  searchForm.checkout.min = searchForm.checkin.value;
});

async function loadAvailableRooms() {
  if (!lastSearch) return;

  try {
    availableRooms.innerHTML = `<div class="empty">Buscando quartos disponíveis...</div>`;

    const params = new URLSearchParams(lastSearch);
    const data = await api(`/api/rooms/available?${params}`);

    if (data.rooms.length === 0) {
      availableRooms.innerHTML = `<div class="empty">Nenhum quarto disponível para estas datas.</div>`;
      return;
    }

    availableRooms.innerHTML = data.rooms.map(room => `
      <article class="card">
        ${room.image ? `<img class="room-img" src="${escapeHtml(room.image)}" alt="${escapeHtml(room.name)}">` : ""}
        <h3>${escapeHtml(room.name)}</h3>
        <p>${escapeHtml(room.description)}</p>
        <div class="meta">
          <span class="badge">Até ${room.capacity} hóspedes</span>
          <span class="badge">${money(room.price)} / diária</span>
        </div>
        <button onclick="createReservation(${room.id})">Reservar</button>
      </article>
    `).join("");
  } catch (error) {
    availableRooms.innerHTML = `<div class="empty">${escapeHtml(error.message)}</div>`;
  }
}

async function createReservation(roomId) {
  if (!lastSearch) {
    showToast("Escolha as datas antes de reservar.");
    return;
  }

  try {
    await api("/api/reservations", {
      method: "POST",
      body: JSON.stringify({
        roomId,
        checkin: lastSearch.checkin,
        checkout: lastSearch.checkout,
        guests: Number(lastSearch.guests)
      })
    });

    showToast("Reserva criada. Aguarde confirmação do gerente.");
    await loadAvailableRooms();
    await loadMyReservations();
  } catch (error) {
    showToast(error.message);
  }
}

async function loadMyReservations() {
  try {
    const data = await api("/api/my-reservations");

    if (data.reservations.length === 0) {
      myReservations.innerHTML = `<div class="empty">Você ainda não tem reservas.</div>`;
      return;
    }

    myReservations.innerHTML = data.reservations.map(res => `
      <article class="card">
        ${res.room_image ? `<img class="room-img" src="${escapeHtml(res.room_image)}" alt="${escapeHtml(res.room_name)}">` : ""}
        <h3>${escapeHtml(res.room_name)}</h3>
        <div class="meta">
          <span class="badge ${res.status}">${statusText(res.status)}</span>
          <span class="badge">${formatDate(res.checkin)} até ${formatDate(res.checkout)}</span>
          <span class="badge">${res.guests} hóspede(s)</span>
          <span class="badge">${money(res.room_price)} / diária</span>
        </div>
        ${["pending", "confirmed"].includes(res.status) ? `
          <button class="danger" onclick="cancelReservation(${res.id})">Cancelar reserva</button>
        ` : ""}
      </article>
    `).join("");
  } catch (error) {
    myReservations.innerHTML = `<div class="empty">${escapeHtml(error.message)}</div>`;
  }
}

async function cancelReservation(id) {
  if (!confirm("Tem certeza que deseja cancelar esta reserva?")) return;

  try {
    await api(`/api/my-reservations/${id}/cancel`, { method: "POST" });
    showToast("Reserva cancelada.");
    await loadMyReservations();
    if (lastSearch) await loadAvailableRooms();
  } catch (error) {
    showToast(error.message);
  }
}

roomForm.addEventListener("submit", async (event) => {
  event.preventDefault();

  const form = new FormData(roomForm);
  const id = form.get("id");

  const payload = {
    name: form.get("name"),
    description: form.get("description"),
    capacity: Number(form.get("capacity")),
    price: Number(form.get("price")),
    image: form.get("image"),
    active: roomForm.active.checked
  };

  try {
    if (id) {
      await api(`/api/manager/rooms/${id}`, {
        method: "PUT",
        body: JSON.stringify(payload)
      });
      showToast("Quarto atualizado.");
    } else {
      await api("/api/manager/rooms", {
        method: "POST",
        body: JSON.stringify(payload)
      });
      showToast("Quarto cadastrado.");
    }

    resetRoomForm();
    await loadManagerRooms();
  } catch (error) {
    showToast(error.message);
  }
});

cancelEditRoom.addEventListener("click", resetRoomForm);

function resetRoomForm() {
  roomForm.reset();
  roomForm.id.value = "";
  roomForm.active.checked = true;
  roomFormTitle.textContent = "Cadastrar quarto";
  cancelEditRoom.classList.add("hidden");
}

async function loadManagerRooms() {
  try {
    const data = await api("/api/manager/rooms");

    if (data.rooms.length === 0) {
      managerRooms.innerHTML = `<div class="empty">Nenhum quarto cadastrado.</div>`;
      return;
    }

    managerRooms.innerHTML = data.rooms.map(room => `
      <article class="card">
        ${room.image ? `<img class="room-img" src="${escapeHtml(room.image)}" alt="${escapeHtml(room.name)}">` : ""}
        <h3>${escapeHtml(room.name)}</h3>
        <p>${escapeHtml(room.description)}</p>
        <div class="meta">
          <span class="badge">Até ${room.capacity} hóspedes</span>
          <span class="badge">${money(room.price)} / diária</span>
          <span class="badge ${room.active ? "confirmed" : "cancelled"}">${room.active ? "Ativo" : "Inativo"}</span>
        </div>
        <div class="row">
          <button onclick='editRoom(${JSON.stringify(room).replaceAll("'", "&apos;")})'>Editar</button>
          <button class="danger" onclick="deleteRoom(${room.id})">Remover</button>
        </div>
      </article>
    `).join("");
  } catch (error) {
    managerRooms.innerHTML = `<div class="empty">${escapeHtml(error.message)}</div>`;
  }
}

function editRoom(room) {
  roomForm.id.value = room.id;
  roomForm.name.value = room.name;
  roomForm.description.value = room.description;
  roomForm.capacity.value = room.capacity;
  roomForm.price.value = room.price;
  roomForm.image.value = room.image || "";
  roomForm.active.checked = !!room.active;
  roomFormTitle.textContent = "Editar quarto";
  cancelEditRoom.classList.remove("hidden");
  roomForm.scrollIntoView({ behavior: "smooth" });
}

async function deleteRoom(id) {
  if (!confirm("Tem certeza que deseja remover este quarto?")) return;

  try {
    await api(`/api/manager/rooms/${id}`, { method: "DELETE" });
    showToast("Quarto removido ou desativado.");
    await loadManagerRooms();
  } catch (error) {
    showToast(error.message);
  }
}

async function loadManagerReservations() {
  try {
    const data = await api("/api/manager/reservations");

    if (data.reservations.length === 0) {
      managerReservations.innerHTML = `<div class="empty">Nenhuma reserva encontrada.</div>`;
      return;
    }

    managerReservations.innerHTML = data.reservations.map(res => `
      <article class="card">
        <h3>Reserva #${res.id} - ${escapeHtml(res.room_name)}</h3>
        <p><strong>Hóspede:</strong> ${escapeHtml(res.guest_name)} (${escapeHtml(res.guest_email)})</p>
        <div class="meta">
          <span class="badge ${res.status}">${statusText(res.status)}</span>
          <span class="badge">${formatDate(res.checkin)} até ${formatDate(res.checkout)}</span>
          <span class="badge">${res.guests} hóspede(s)</span>
          <span class="badge">${money(res.room_price)} / diária</span>
        </div>
        ${res.status === "pending" ? `
          <div class="row">
            <button class="success" onclick="changeReservationStatus(${res.id}, 'confirmed')">Confirmar</button>
            <button class="warning" onclick="changeReservationStatus(${res.id}, 'rejected')">Recusar</button>
          </div>
        ` : ""}
      </article>
    `).join("");
  } catch (error) {
    managerReservations.innerHTML = `<div class="empty">${escapeHtml(error.message)}</div>`;
  }
}

async function changeReservationStatus(id, status) {
  try {
    await api(`/api/manager/reservations/${id}/status`, {
      method: "POST",
      body: JSON.stringify({ status })
    });

    showToast(status === "confirmed" ? "Reserva confirmada." : "Reserva recusada.");
    await loadManagerReservations();
  } catch (error) {
    showToast(error.message);
  }
}

  } catch {

  renderApp();
}
