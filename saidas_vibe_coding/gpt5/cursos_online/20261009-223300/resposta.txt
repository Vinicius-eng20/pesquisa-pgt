=== README.md ===
# Plataforma de Cursos Online com Matrícula

Este projeto é um site completo com backend, frontend, login, cadastro, administração, cursos e matrículas.

Os dados ficam salvos em um banco SQLite no próprio projeto (`data/app.db`), então não somem ao fechar o navegador ou reiniciar o site.

## Requisitos

Instale o Node.js:

https://nodejs.org/

Recomendado: Node.js 20 ou superior.

## Como instalar

Abra o terminal na pasta do projeto e execute:

```bash
npm install
```

## Como iniciar

```bash
npm start
```

Depois acesse no navegador:

```text
http://localhost:3000
```

## Conta de administrador

Ao iniciar pela primeira vez, o sistema cria automaticamente uma conta de administrador:

```text
Email: admin@site.com
Senha: admin123
```

## Alunos

Alunos podem criar conta pelo botão "Criar conta" na tela inicial.

## Funcionalidades

### Aluno
- Criar conta
- Entrar com email e senha
- Ver cursos disponíveis
- Matricular-se em cursos
- Ver suas matrículas
- Cancelar sua própria matrícula

### Administrador
- Entrar com a conta admin
- Criar cursos
- Editar cursos
- Remover cursos
- Ver todas as matrículas
- Cancelar matrículas de qualquer aluno

## Observações

- O banco de dados fica em `data/app.db`.
- Para usar em outro computador na mesma rede, acesse pelo IP da máquina que está rodando o projeto.
- Para publicar na internet de verdade, é necessário colocar este projeto em um servidor/VPS próprio com Node.js. Não é necessário usar serviços externos de banco de dados.

db.pragma('foreign_keys = ON');

db.exec(`
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  role TEXT NOT NULL CHECK(role IN ('admin', 'student')),
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS courses (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  description TEXT NOT NULL,
  workload TEXT NOT NULL,
  teacher TEXT NOT NULL,
  active INTEGER NOT NULL DEFAULT 1,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS enrollments (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  course_id INTEGER NOT NULL,
  status TEXT NOT NULL DEFAULT 'active' CHECK(status IN ('active', 'cancelled')),
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  cancelled_at TEXT,
  FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY(course_id) REFERENCES courses(id) ON DELETE CASCADE,
  UNIQUE(user_id, course_id)
);
`);

function requireLogin(req, res, next) {
  if (!req.session.user) {
    flash(req, 'error', 'Entre com sua conta para continuar.');
    return res.redirect('/login');

function requireAdmin(req, res, next) {

  next();

app.post('/register', (req, res) => {
  const name = String(req.body.name || '').trim();
  const email = String(req.body.email || '').trim().toLowerCase();
  const password = String(req.body.password || '');
  const passwordConfirm = String(req.body.passwordConfirm || '');

  if (!name || !email || !password || !passwordConfirm) {
    flash(req, 'error', 'Preencha todos os campos.');
    return res.render('register', { values: { name, email } });
  }

  if (password.length < 6) {
    flash(req, 'error', 'A senha precisa ter pelo menos 6 caracteres.');
    return res.render('register', { values: { name, email } });
  }

  if (password !== passwordConfirm) {
    flash(req, 'error', 'As senhas não são iguais.');
    return res.render('register', { values: { name, email } });
  }

  const exists = db.prepare(`SELECT id FROM users WHERE email = ?`).get(email);
  if (exists) {
    flash(req, 'error', 'Já existe uma conta com este email.');
    return res.render('register', { values: { name, email } });
  }

  const hash = bcrypt.hashSync(password, 10);
  const result = db.prepare(`
    INSERT INTO users (name, email, password_hash, role)
    VALUES (?, ?, ?, 'student')
  `).run(name, email, hash);

  req.session.user = {
    id: result.lastInsertRowid,
    name,
    email,
    role: 'student'
  };

  flash(req, 'success', 'Conta criada com sucesso.');
  res.redirect('/courses');
});

app.get('/login', (req, res) => {

app.post('/login', (req, res) => {
  const email = String(req.body.email || '').trim().toLowerCase();
  const password = String(req.body.password || '');

  if (!email || !password) {
    flash(req, 'error', 'Informe email e senha.');
    return res.render('login', { values: { email } });
  }

  const user = db.prepare(`SELECT * FROM users WHERE email = ?`).get(email);
  if (!user || !bcrypt.compareSync(password, user.password_hash)) {
    flash(req, 'error', 'Email ou senha inválidos.');
    return res.render('login', { values: { email } });
  }

  req.session.user = {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role
  };

  flash(req, 'success', 'Entrada realizada com sucesso.');
  res.redirect('/');
});

app.get('/courses', requireStudent, (req, res) => {
  const courses = db.prepare(`
    SELECT
      c.*,
      e.id AS enrollment_id,
      e.status AS enrollment_status
    FROM courses c
    LEFT JOIN enrollments e
      ON e.course_id = c.id
     AND e.user_id = ?
    WHERE c.active = 1
    ORDER BY c.created_at DESC, c.id DESC
  `).all(req.session.user.id);

  res.render('courses', { courses });
});

app.post('/courses/:id/enroll', requireStudent, (req, res) => {
  const courseId = Number(req.params.id);
  const course = db.prepare(`SELECT * FROM courses WHERE id = ? AND active = 1`).get(courseId);

  if (!course) {
    flash(req, 'error', 'Curso não encontrado.');
    return res.redirect('/courses');
  }

  const existing = db.prepare(`
    SELECT * FROM enrollments
    WHERE user_id = ? AND course_id = ?
  `).get(req.session.user.id, courseId);

  if (existing && existing.status === 'active') {
    flash(req, 'error', 'Você já está matriculado neste curso.');
    return res.redirect('/courses');
  }

  if (existing && existing.status === 'cancelled') {
    db.prepare(`
      UPDATE enrollments
      SET status = 'active', cancelled_at = NULL, created_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(existing.id);

    flash(req, 'success', 'Matrícula reativada com sucesso.');
    return res.redirect('/my-enrollments');
  }

  db.prepare(`
    INSERT INTO enrollments (user_id, course_id)
    VALUES (?, ?)
  `).run(req.session.user.id, courseId);

  flash(req, 'success', 'Matrícula realizada com sucesso.');
  res.redirect('/my-enrollments');
});

app.get('/admin/courses', requireAdmin, (req, res) => {
  const courses = db.prepare(`
    SELECT
      c.*,
      COUNT(e.id) AS enrollment_count,
      SUM(CASE WHEN e.status = 'active' THEN 1 ELSE 0 END) AS active_enrollment_count
    FROM courses c
    LEFT JOIN enrollments e ON e.course_id = c.id
    GROUP BY c.id
    ORDER BY c.created_at DESC, c.id DESC
  `).all();

  res.render('admin/courses', { courses });
});

  if (!existing) {
    flash(req, 'error', 'Curso não encontrado.');

  }

  const title = String(req.body.title || '').trim();
  const description = String(req.body.description || '').trim();
  const workload = String(req.body.workload || '').trim();
  const teacher = String(req.body.teacher || '').trim();
  const active = req.body.active ? 1 : 0;

app.get('/admin/enrollments', requireAdmin, (req, res) => {
  const status = String(req.query.status || 'all');

  let query = `
    SELECT
      e.*,
      u.name AS student_name,
      u.email AS student_email,
      c.title AS course_title,
      c.workload AS course_workload
    FROM enrollments e
    JOIN users u ON u.id = e.user_id
    JOIN courses c ON c.id = e.course_id
  `;

  const params = [];

  if (status === 'active' || status === 'cancelled') {
    query += ` WHERE e.status = ? `;
    params.push(status);
  }

  query += ` ORDER BY e.created_at DESC, e.id DESC `;

  const enrollments = db.prepare(query).all(...params);

  res.render('admin/enrollments', { enrollments, status });
});

app.post('/admin/enrollments/:id/cancel', requireAdmin, (req, res) => {
  const enrollment = db.prepare(`SELECT * FROM enrollments WHERE id = ?`).get(Number(req.params.id));

  if (!enrollment) {
    flash(req, 'error', 'Matrícula não encontrada.');
    return res.redirect('/admin/enrollments');
  }

  if (enrollment.status === 'cancelled') {
    flash(req, 'error', 'Matrícula já está cancelada.');
    return res.redirect('/admin/enrollments');
  }

  db.prepare(`
    UPDATE enrollments
    SET status = 'cancelled', cancelled_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `).run(enrollment.id);

  flash(req, 'success', 'Matrícula cancelada pelo administrador.');
  res.redirect('/admin/enrollments');
});

document.addEventListener('DOMContentLoaded', () => {
  const toggle = document.querySelector('[data-menu-toggle]');
  const menu = document.querySelector('[data-menu]');

  if (toggle && menu) {
    toggle.addEventListener('click', () => {
      menu.classList.toggle('open');
    });
  }

  document.querySelectorAll('form[data-confirm]').forEach((form) => {
    form.addEventListener('submit', (event) => {
      const message = form.getAttribute('data-confirm') || 'Tem certeza?';
      if (!window.confirm(message)) {
        event.preventDefault();
      }
    });
  });
});
