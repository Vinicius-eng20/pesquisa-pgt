# Portal de Inscrição para Eventos

Site completo para participantes e organizador criarem contas, entrarem no sistema, criarem eventos, se inscreverem, cancelarem inscrições, confirmarem presença e acompanharem tudo com dados salvos em banco SQLite no próprio projeto.

## Requisitos

Instale o Node.js versão 18 ou superior:

- https://nodejs.org/

## Como instalar

Abra o terminal dentro da pasta do projeto e rode:

```bash
npm install
```

Depois crie o banco de dados inicial:

```bash
npm run init-db
```

## Como executar

```bash
npm start
```

Acesse no navegador:

```text
http://localhost:3000
```

## Conta inicial do organizador

Ao rodar `npm run init-db`, uma conta de organizador é criada automaticamente:

- E-mail: `organizador@eventos.local`
- Senha: `admin123`

Você pode alterar a senha depois editando diretamente pelo banco ou criando outra conta organizadora no script, se desejar.

## Como publicar na internet

Este projeto funciona com Node.js e SQLite. Para colocar na internet sem serviços externos, use um servidor/VPS próprio com Node.js instalado.

Passos gerais no servidor:

```bash
npm install
npm run init-db
npm start
```

Para uso real em produção, configure também:

- HTTPS
- domínio
- proxy reverso, como Nginx
- backup periódico do arquivo `data/app.db`

## Observação importante

Tudo fica salvo no arquivo:

```text
data/app.db
```

Não apague esse arquivo se quiser manter usuários, eventos e inscrições.

function flash(req, type, message) {
  req.session.flash = { type, message };
}

function requireLogin(req, res, next) {
  if (!req.session.user) {
    flash(req, 'error', 'Entre na sua conta para continuar.');
    return res.redirect('/login');
  }
  next();
}

function requireOrganizer(req, res, next) {
  if (!req.session.user) {
    flash(req, 'error', 'Entre na sua conta para continuar.');
    return res.redirect('/login');
  }
  if (req.session.user.role !== 'organizer') {
    flash(req, 'error', 'Apenas o organizador pode acessar esta área.');
    return res.redirect('/dashboard');
  }
  next();
}

function normalizeText(value) {
  return String(value || '').trim();

function normalizeDate(value) {
  return String(value || '').trim();

app.post('/register', (req, res) => {
  const name = normalizeText(req.body.name);
  const email = normalizeText(req.body.email).toLowerCase();
  const password = String(req.body.password || '');

  if (!name || !email || !password) {
    flash(req, 'error', 'Preencha nome, e-mail e senha.');
    return res.redirect('/register');
  }

  if (password.length < 6) {
    flash(req, 'error', 'A senha precisa ter pelo menos 6 caracteres.');
    return res.redirect('/register');
  }

  const existing = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
  if (existing) {
    flash(req, 'error', 'Já existe uma conta com esse e-mail.');
    return res.redirect('/register');
  }

  const passwordHash = bcrypt.hashSync(password, 12);
  const result = db.prepare(`
    INSERT INTO users (name, email, password_hash, role)
    VALUES (?, ?, ?, 'participant')
  `).run(name, email, passwordHash);

  req.session.user = { id: result.lastInsertRowid, name, email, role: 'participant' };
  flash(req, 'success', 'Conta criada com sucesso.');
  res.redirect('/dashboard');
});

app.post('/login', (req, res) => {
  const email = normalizeText(req.body.email).toLowerCase();
  const password = String(req.body.password || '');

  const user = db.prepare('SELECT * FROM users WHERE email = ?').get(email);

  if (!user || !bcrypt.compareSync(password, user.password_hash)) {
    flash(req, 'error', 'E-mail ou senha inválidos.');
    return res.redirect('/login');
  }

  req.session.user = {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role
  };

  flash(req, 'success', 'Entrada realizada com sucesso.');
  res.redirect('/dashboard');
});

app.get('/events/:id', (req, res) => {
  const event = db.prepare(`
    SELECT e.*,
      (SELECT COUNT(*) FROM registrations r WHERE r.event_id = e.id AND r.status != 'cancelada') AS active_registrations
    FROM events e
    WHERE e.id = ?
  `).get(req.params.id);

  if (!event) {
    flash(req, 'error', 'Evento não encontrado.');
    return res.redirect('/');
  }

    registration = db.prepare(`
      SELECT * FROM registrations
      WHERE event_id = ? AND user_id = ?
      ORDER BY id DESC
      LIMIT 1

  }

  res.render('event-detail', { title: event.title, event, registration });

app.post('/events/:id/register', requireLogin, (req, res) => {
  if (req.session.user.role !== 'participant') {
    flash(req, 'error', 'O organizador não pode se inscrever em eventos.');
    return res.redirect(`/events/${req.params.id}`);
  }

  const event = db.prepare('SELECT * FROM events WHERE id = ?').get(req.params.id);
  if (!event || event.status !== 'ativo') {
    flash(req, 'error', 'Este evento não está disponível para inscrição.');
    return res.redirect('/');
  }

  const activeCount = db.prepare(`
    SELECT COUNT(*) AS total FROM registrations
    WHERE event_id = ? AND status != 'cancelada'
  `).get(event.id).total;

  if (event.capacity && activeCount >= event.capacity) {
    flash(req, 'error', 'Este evento está sem vagas.');
    return res.redirect(`/events/${event.id}`);
  }

  const existing = db.prepare(`
    SELECT * FROM registrations
    WHERE event_id = ? AND user_id = ? AND status != 'cancelada'
  `).get(event.id, req.session.user.id);

  if (existing) {
    flash(req, 'error', 'Você já está inscrito neste evento.');
    return res.redirect(`/events/${event.id}`);
  }

  db.prepare(`
    INSERT INTO registrations (event_id, user_id, status, attendance_confirmed)
    VALUES (?, ?, 'inscrita', 0)
  `).run(event.id, req.session.user.id);

  flash(req, 'success', 'Inscrição realizada com sucesso.');
  res.redirect('/my-registrations');
});

const fs = require('fs');
const path = require('path');
const Database = require('better-sqlite3');

const dataDir = path.join(__dirname, '..', 'data');
if (!fs.existsSync(dataDir)) {
  fs.mkdirSync(dataDir, { recursive: true });
}

const dbPath = path.join(dataDir, 'app.db');
const db = new Database(dbPath);

db.pragma('journal_mode = WAL');
db.pragma('foreign_keys = ON');

db.exec(`
  CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    role TEXT NOT NULL CHECK(role IN ('participant', 'organizer')),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    location TEXT NOT NULL,
    event_date TEXT NOT NULL,
    capacity INTEGER,
    status TEXT NOT NULL DEFAULT 'ativo' CHECK(status IN ('ativo', 'cancelado')),
    created_by INTEGER,
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(created_by) REFERENCES users(id)
  );

  CREATE TABLE IF NOT EXISTS registrations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    event_id INTEGER NOT NULL,
    user_id INTEGER NOT NULL,
    status TEXT NOT NULL DEFAULT 'inscrita' CHECK(status IN ('inscrita', 'confirmada', 'cancelada')),
    attendance_confirmed INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(event_id) REFERENCES events(id),
    FOREIGN KEY(user_id) REFERENCES users(id)
  );
`);

module.exports = db;
