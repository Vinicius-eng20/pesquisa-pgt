# Sistema de Agendamento de Serviços

Sistema completo de agendamento que funciona de verdade na internet, com dados salvos em um banco de dados SQLite no próprio servidor.

## O que o sistema faz

**Clientes podem:**
- Criar conta e fazer login
- Ver todos os serviços disponíveis
- Ver horários livres e marcar um horário
- Ver seus agendamentos
- Cancelar agendamentos

**Administrador (responsável pelo negócio) pode:**
- Fazer login
- Cadastrar e editar serviços
- Cadastrar e editar horários disponíveis
- Confirmar ou recusar agendamentos
- Ver todos os agendamentos de todos os clientes

## Como instalar e executar

Você precisa ter o **Node.js** instalado (versão 18 ou superior). Baixe em https://nodejs.org

1. Abra o terminal (ou Prompt de Comando) na pasta do projeto
2. Instale as dependências:
```
npm install
```
3. Inicie o servidor:
```
npm start
```
4. Abra o navegador em: **http://localhost:3000**

## Conta do administrador

Na primeira vez que o site roda, uma conta de administrador é criada automaticamente:

- **E-mail:** admin@admin.com
- **Senha:** admin123

## Como colocar na internet (de verdade)

Como o projeto já é um servidor Node.js com banco de dados próprio, você pode hospedá-lo em qualquer serviço de hospedagem de Node.js. Alguns gratuitos ou baratos:

- **Render.com** (tem plano gratuito)
- **Railway.app**
- **Fly.io**

Passos gerais:
1. Suba o projeto para um repositório no GitHub
2. No serviço de hospedagem, conecte o repositório
3. Comando de build: `npm install`
4. Comando de start: `npm start`
5. O site vai ganhar um endereço na internet que funciona em qualquer computador ou celular

## Observação importante sobre o banco de dados

Os dados ficam salvos no arquivo `agendamentos.db` que é criado automaticamente na pasta do projeto. Enquanto esse arquivo existir, nada se perde. Ao hospedar na internet, verifique se o serviço de hospedagem salva os arquivos permanentemente (a maioria tem uma pasta de dados que persiste).

const express = require('express');
const session = require('express-session');

const app = express();
const db = new Database('agendamentos.db');

// Criar tabelas
db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT NOT NULL DEFAULT 'cliente'
  );

  CREATE TABLE IF NOT EXISTS servicos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    descricao TEXT,
    preco REAL,
    duracao INTEGER DEFAULT 30
  );

  CREATE TABLE IF NOT EXISTS horarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    data TEXT NOT NULL,
    hora TEXT NOT NULL,
    disponivel INTEGER DEFAULT 1,
    UNIQUE(data, hora)
  );

  CREATE TABLE IF NOT EXISTS agendamentos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    usuario_id INTEGER NOT NULL,
    servico_id INTEGER NOT NULL,
    horario_id INTEGER NOT NULL,
    status TEXT DEFAULT 'pendente',
    criado_em TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (servico_id) REFERENCES servicos(id),
    FOREIGN KEY (horario_id) REFERENCES horarios(id)
  );
`);

app.post('/api/registrar', (req, res) => {
  const { nome, email, senha } = req.body;
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });
  try {
    const hash = bcrypt.hashSync(senha, 10);
    const info = db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)').run(nome, email, hash, 'cliente');
    req.session.usuario = { id: info.lastInsertRowid, nome, email, tipo: 'cliente' };
    res.json({ ok: true, usuario: req.session.usuario });
  } catch (e) {
    res.status(400).json({ erro: 'Este e-mail já está cadastrado' });
  }
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;
  const usuario = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email);
  if (!usuario || !bcrypt.compareSync(senha, usuario.senha)) {
    return res.status(400).json({ erro: 'E-mail ou senha incorretos' });

  req.session.usuario = { id: usuario.id, nome: usuario.nome, email: usuario.email, tipo: usuario.tipo };
  res.json({ ok: true, usuario: req.session.usuario });
});

app.get('/api/meus-agendamentos', precisaLogin, (req, res) => {
  const lista = db.prepare(`
    SELECT a.*, s.nome AS servico_nome, s.preco AS servico_preco, h.data, h.hora
    FROM agendamentos a
    JOIN servicos s ON s.id = a.servico_id
    JOIN horarios h ON h.id = a.horario_id
    WHERE a.usuario_id = ?
    ORDER BY h.data DESC, h.hora DESC
  `).all(req.session.usuario.id);
  res.json(lista);
});

app.post('/api/agendamentos', precisaLogin, (req, res) => {
  const { servico_id, horario_id } = req.body;
  if (!servico_id || !horario_id) return res.status(400).json({ erro: 'Selecione serviço e horário' });

  const ocupado = db.prepare(`
    SELECT id FROM agendamentos WHERE horario_id = ? AND status IN ('pendente', 'confirmado')
  `).get(horario_id);
  if (ocupado) return res.status(400).json({ erro: 'Este horário já está ocupado' });

  const info = db.prepare('INSERT INTO agendamentos (usuario_id, servico_id, horario_id) VALUES (?, ?, ?)').run(
    req.session.usuario.id, servico_id, horario_id
  );
  res.json({ ok: true, id: info.lastInsertRowid });
});

app.delete('/api/agendamentos/:id', precisaLogin, (req, res) => {
  const ag = db.prepare('SELECT * FROM agendamentos WHERE id = ?').get(req.params.id);
  if (!ag) return res.status(404).json({ erro: 'Agendamento não encontrado' });
  if (ag.usuario_id !== req.session.usuario.id && req.session.usuario.tipo !== 'admin') {
    return res.status(403).json({ erro: 'Sem permissão' });
  }
  db.prepare('DELETE FROM agendamentos WHERE id = ?').run(req.params.id);
  res.json({ ok: true });
});

app.get('/api/todos-agendamentos', precisaAdmin, (req, res) => {
  const lista = db.prepare(`
    SELECT a.*, u.nome AS cliente_nome, u.email AS cliente_email,

    FROM agendamentos a
    JOIN usuarios u ON u.id = a.usuario_id
    JOIN servicos s ON s.id = a.servico_id
    JOIN horarios h ON h.id = a.horario_id
    ORDER BY h.data DESC, h.hora DESC

async function api(url, metodo = 'GET', corpo = null) {
  const opcoes = { method: metodo, headers: { 'Content-Type': 'application/json' } };
  if (corpo) opcoes.body = JSON.stringify(corpo);
  const resp = await fetch(url, opcoes);
  const dados = await resp.json().catch(() => ({}));
  if (!resp.ok) throw new Error(dados.erro || 'Erro ao processar');
  return dados;
}

function mostrarMsg(elementId, texto, tipo = 'erro') {
  const el = document.getElementById(elementId);
  el.innerHTML = '<div class="msg msg-' + tipo + '">' + texto + '</div>';
  setTimeout(() => { el.innerHTML = ''; }, 5000);

function mostrarTab(id, el) {
  const tabs = el.parentElement.querySelectorAll('.tab');
  tabs.forEach(t => t.classList.remove('active'));
  el.classList.add('active');
  const area = el.closest('#clienteArea, #adminArea');
  area.querySelectorAll('[id^="tab-"]').forEach(d => {
    if (d.classList.contains('tabs')) return;
    d.classList.add('hidden');
  });
  document.getElementById(id).classList.remove('hidden');
  if (id === 'tab-meus') carregarMeusAgendamentos();
  if (id === 'tab-agendamentos-admin') carregarTodosAgendamentos();
  if (id === 'tab-servicos-admin') carregarServicosAdmin();
  if (id === 'tab-horarios-admin') carregarHorariosAdmin();
  if (id === 'tab-horarios') { carregarServicosCliente(); carregarHorariosLivres(); }
}

async function fazerLogin() {
  const email = document.getElementById('loginEmail').value.trim();
  const senha = document.getElementById('loginSenha').value;
  if (!email || !senha) return mostrarMsg('authMsg', 'Preencha e-mail e senha');
  try {
    const r = await api('/api/login', 'POST', { email, senha });
    usuarioAtual = r.usuario;
    iniciar();
  } catch (e) { mostrarMsg('authMsg', e.message); }
}

async function fazerCadastro() {
  const nome = document.getElementById('cadNome').value.trim();
  const email = document.getElementById('cadEmail').value.trim();
  const senha = document.getElementById('cadSenha').value;
  if (!nome || !email || !senha) return mostrarMsg('authMsg', 'Preencha todos os campos');
  if (senha.length < 4) return mostrarMsg('authMsg', 'Senha muito curta (mínimo 4)');
  try {
    const r = await api('/api/registrar', 'POST', { nome, email, senha });
    usuarioAtual = r.usuario;
    iniciar();
  } catch (e) { mostrarMsg('authMsg', e.message); }
}

async function logout() {
  await api('/api/logout', 'POST');
  usuarioAtual = null;
  location.reload();
}

function iniciar() {
  document.getElementById('authArea').classList.add('hidden');
  document.getElementById('topbar').classList.remove('hidden');
  document.getElementById('nomeUsuario').textContent = usuarioAtual.nome;

  if (usuarioAtual.tipo === 'admin') {
    document.getElementById('tituloHeader').textContent = 'Painel do Administrador';
    document.getElementById('adminArea').classList.remove('hidden');
    carregarTodosAgendamentos();
    carregarServicosAdmin();
    carregarHorariosAdmin();
  } else {
    document.getElementById('tituloHeader').textContent = 'Área do Cliente';
    document.getElementById('clienteArea').classList.remove('hidden');
    carregarServicosCliente();
    carregarHorariosLivres();
  }
}

// ===== CLIENTE =====
async function carregarServicosCliente() {
  const servicos = await api('/api/servicos');
  const sel = document.getElementById('selServico');
  sel.innerHTML = '<option value="">Selecione um serviço</option>';
  const lista = document.getElementById('listaServicosCliente');
  if (servicos.length === 0) {
    lista.innerHTML = '<div class="empty">Nenhum serviço disponível ainda</div>';
    sel.innerHTML = '<option value="">Nenhum serviço cadastrado</option>';
    return;
  }
  servicos.forEach(s => {
    sel.innerHTML += '<option value="' + s.id + '">' + escapar(s.nome) + (s.preco ? ' - R$ ' + Number(s.preco).toFixed(2) : '') + '</option>';
  });
  lista.innerHTML = '<table><thead><tr><th>Serviço</th><th>Descrição</th><th>Preço</th></tr></thead><tbody>' +
    servicos.map(s => '<tr><td><strong>' + escapar(s.nome) + '</strong></td><td>' + escapar(s.descricao || '-') + '</td><td>' + (s.preco ? 'R$ ' + Number(s.preco).toFixed(2) : '-') + '</td></tr>').join('') +
    '</tbody></table>';
}

async function carregarHorariosLivres() {
  const horarios = await api('/api/horarios-livres');
  const sel = document.getElementById('selHorario');
  sel.innerHTML = '<option value="">Selecione um horário</option>';
  if (horarios.length === 0) {
    sel.innerHTML = '<option value="">Nenhum horário disponível</option>';
    return;
  }
  horarios.forEach(h => {
    const dataFmt = formatarData(h.data);
    sel.innerHTML += '<option value="' + h.id + '">' + dataFmt + ' às ' + h.hora + '</option>';
  });
}

async function salvarServico() {
  const id = document.getElementById('servicoEditId').value;
  const dados = {
    nome: document.getElementById('servNome').value.trim(),
    descricao: document.getElementById('servDesc').value.trim(),
    preco: parseFloat(document.getElementById('servPreco').value) || 0
  };
  if (!dados.nome) return mostrarMsg('msgAdminServ', 'Informe o nome do serviço');
  try {
    if (id) await api('/api/servicos/' + id, 'PUT', dados);

    cancelarEdicaoServico();
    carregarServicosAdmin();
  } catch (e) { mostrarMsg('msgAdminServ', e.message); }
}

async function toggleHorario(id, ativo) {
  const horarios = await api('/api/horarios');
  const h = horarios.find(x => x.id === id);
  if (!h) return;
  try {
    await api('/api/horarios/' + id, 'PUT', { data: h.data, hora: h.hora, disponivel: ativo });
    carregarHorariosAdmin();
  } catch (e) { mostrarMsg('msgAdminHor', e.message); }
}

// ===== UTIL =====
function escapar(s) {
  if (s === null || s === undefined) return '';
  return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

function formatarData(iso) {
  if (!iso) return '';
  const [y, m, d] = iso.split('-');
  return d + '/' + m + '/' + y;
}
