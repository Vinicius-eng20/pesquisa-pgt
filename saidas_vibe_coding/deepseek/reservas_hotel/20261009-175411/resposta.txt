# Sistema de Reservas de Hotel

Sistema completo de reservas de hotel com armazenamento em banco de dados SQLite (no próprio servidor), autenticação de usuários e perfis de hóspede/gerente.

## Como instalar e executar

1. Instale o [Node.js](https://nodejs.org/) (versão 18 ou superior).
2. Abra o terminal na pasta do projeto.
3. Instale as dependências:

```bash
npm install
```

4. Inicie o servidor:

```bash
npm start
```

5. Abra o navegador em: http://localhost:3000

## Contas pré-criadas

Ao iniciar pela primeira vez, o sistema cria automaticamente:

- **Gerente**: usuário `gerente` / senha `gerente123`
- **Hóspede**: usuário `hospede` / senha `hospede123`

Novos hóspedes podem se cadastrar livremente pela tela inicial.

## Observações

- Todos os dados ficam salvos no arquivo `hotel.db` (SQLite), no próprio servidor.
- Para usar na internet, hospede este projeto em qualquer serviço que rode Node.js (Render, Railway, Fly.io, VPS, etc.). Basta subir os arquivos e rodar `npm install && npm start`.
- O banco de dados persiste entre reinicializações.

  next();

app.post('/api/registrar', (req, res) => {
  const { usuario, senha, nome } = req.body;
  if (!usuario || !senha || !nome)
    return res.status(400).json({ erro: 'Preencha todos os campos' });
  if (usuario.length < 3 || senha.length < 6)
    return res.status(400).json({ erro: 'Usuário com pelo menos 3 e senha com pelo menos 6 caracteres' });
  const existe = db.prepare('SELECT id FROM usuarios WHERE usuario = ?').get(usuario);
  if (existe) return res.status(400).json({ erro: 'Usuário já existe' });
  const hash = bcrypt.hashSync(senha, 10);
  const info = db.prepare('INSERT INTO usuarios (usuario, senha, nome, tipo) VALUES (?, ?, ?, ?)')
    .run(usuario, hash, nome, 'hospede');
  req.session.usuario = { id: info.lastInsertRowid, usuario, nome, tipo: 'hospede' };
  res.json({ ok: true, usuario: req.session.usuario });
});

app.post('/api/login', (req, res) => {
  const { usuario, senha } = req.body;
  if (!usuario || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });
  const u = db.prepare('SELECT * FROM usuarios WHERE usuario = ?').get(usuario);
  if (!u || !bcrypt.compareSync(senha, u.senha))
    return res.status(400).json({ erro: 'Usuário ou senha inválidos' });
  req.session.usuario = { id: u.id, usuario: u.usuario, nome: u.nome, tipo: u.tipo };
  res.json({ ok: true, usuario: req.session.usuario });
});

    SELECT * FROM quartos
    WHERE ativo = 1
    AND id NOT IN (

      WHERE status IN ('pendente', 'confirmada')
      AND NOT (data_saida <= ? OR data_entrada >= ?)
    )
    ORDER BY numero
  `).all(entrada, saida);
  res.json(quartos);
});

app.post('/api/quartos', autenticado, apenasGerente, (req, res) => {
  const { numero, tipo, capacidade, preco, descricao } = req.body;
  if (!numero || !tipo || !capacidade || !preco)
    return res.status(400).json({ erro: 'Preencha os campos obrigatórios' });
  try {
    const info = db.prepare('INSERT INTO quartos (numero, tipo, capacidade, preco, descricao) VALUES (?, ?, ?, ?, ?)')
      .run(numero, tipo, capacidade, preco, descricao || '');
    res.json({ ok: true, id: info.lastInsertRowid });
  } catch (e) {
    res.status(400).json({ erro: 'Número de quarto já existe' });
  }
});

app.put('/api/quartos/:id', autenticado, apenasGerente, (req, res) => {
  const { numero, tipo, capacidade, preco, descricao, ativo } = req.body;
  try {
    db.prepare('UPDATE quartos SET numero=?, tipo=?, capacidade=?, preco=?, descricao=?, ativo=? WHERE id=?')
      .run(numero, tipo, capacidade, preco, descricao || '', ativo ? 1 : 0, req.params.id);
    res.json({ ok: true });
  } catch (e) {
    res.status(400).json({ erro: 'Erro ao salvar quarto' });
  }
});

app.delete('/api/quartos/:id', autenticado, apenasGerente, (req, res) => {
  const reservas = db.prepare("SELECT COUNT(*) as c FROM reservas WHERE quarto_id = ? AND status IN ('pendente','confirmada')").get(req.params.id).c;
  if (reservas > 0) return res.status(400).json({ erro: 'Há reservas ativas para este quarto. Desative-o em vez de remover.' });
  db.prepare('DELETE FROM quartos WHERE id = ?').run(req.params.id);
  res.json({ ok: true });
});

    FROM reservas r JOIN quartos q ON q.id = r.quarto_id
    WHERE r.usuario_id = ? ORDER BY r.criado_em DESC
  `).all(req.session.usuario.id);
  res.json(reservas);
});

app.get('/api/reservas', autenticado, apenasGerente, (req, res) => {
  const reservas = db.prepare(`
    SELECT r.*, q.numero as quarto_numero, q.tipo as quarto_tipo, q.preco as quarto_preco,
           u.nome as hospede_nome, u.usuario as hospede_usuario
    FROM reservas r
    JOIN quartos q ON q.id = r.quarto_id
    JOIN usuarios u ON u.id = r.usuario_id
    ORDER BY r.criado_em DESC
  `).all();
  res.json(reservas);
});

app.post('/api/reservas', autenticado, (req, res) => {
  const { quarto_id, data_entrada, data_saida } = req.body;
  if (!quarto_id || !data_entrada || !data_saida)
    return res.status(400).json({ erro: 'Preencha todos os campos' });
  if (data_entrada >= data_saida)
    return res.status(400).json({ erro: 'Data de saída deve ser depois da entrada' });
  const conflito = db.prepare(`
    SELECT COUNT(*) as c FROM reservas
    WHERE quarto_id = ? AND status IN ('pendente','confirmada')
    AND NOT (data_saida <= ? OR data_entrada >= ?)
  `).get(quarto_id, data_entrada, data_saida).c;
  if (conflito > 0) return res.status(400).json({ erro: 'Quarto indisponível nesse período' });
  const info = db.prepare('INSERT INTO reservas (usuario_id, quarto_id, data_entrada, data_saida, status, criado_em) VALUES (?, ?, ?, ?, ?, ?)')
    .run(req.session.usuario.id, quarto_id, data_entrada, data_saida, 'pendente', new Date().toISOString());
  res.json({ ok: true, id: info.lastInsertRowid });
});

app.post('/api/reservas/:id/status', autenticado, apenasGerente, (req, res) => {
  const { status } = req.body;
  if (!['confirmada', 'recusada', 'cancelada', 'pendente'].includes(status))
    return res.status(400).json({ erro: 'Status inválido' });
  db.prepare('UPDATE reservas SET status = ? WHERE id = ?').run(status, req.params.id);
  res.json({ ok: true });
});

app.post('/api/reservas/:id/cancelar', autenticado, (req, res) => {
  const r = db.prepare('SELECT * FROM reservas WHERE id = ?').get(req.params.id);
  if (!r) return res.status(404).json({ erro: 'Reserva não encontrada' });
  if (r.usuario_id !== req.session.usuario.id && req.session.usuario.tipo !== 'gerente')
    return res.status(403).json({ erro: 'Sem permissão' });
  db.prepare("UPDATE reservas SET status = 'cancelada' WHERE id = ?").run(req.params.id);
  res.json({ ok: true });
});

async function api(url, metodo = 'GET', corpo = null) {
  const opcoes = { method: metodo, headers: { 'Content-Type': 'application/json' } };
  if (corpo) opcoes.body = JSON.stringify(corpo);
  const r = await fetch(url, opcoes);
  const d = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(d.erro || 'Erro');
  return d;
}

function aviso(texto, tipo = 'ok') { msg = { texto, tipo }; render(); }

function render() {
  if (!usuario) return renderLogin();
  if (usuario.tipo === 'gerente') return renderGerente();
  return renderHospede();
}

function renderCabecalho() {
  return `<header>

function renderMsg() {
  if (!msg) return '';

  return `<div class="msg ${m.tipo}">${m.texto}</div>`;
}

async function fazerLogin() {
  try {
    const usuario_ = document.getElementById('usuario_login').value.trim();
    const senha = document.getElementById('senha_login').value;
    const r = await api('/api/login', 'POST', { usuario: usuario_, senha });
    usuario = r.usuario;
    msg = { texto: 'Bem-vindo!', tipo: 'ok' };
    await carregarTudo();
    render();
  } catch (e) { aviso(e.message, 'erro'); }
}

    await carregarTudo();
    render();

function htmlBusca() {
  let lista = '';
  if (quartosDisponiveis.length === 0) {
    lista = '<p class="vazio">Informe as datas e clique em "Buscar" para ver os quartos disponíveis.</p>';
  } else {
    lista = '<div class="grid">' + quartosDisponiveis.map(q => `
      <div class="quarto">
        <h3>Quarto ${q.numero}</h3>
        <div class="tipo">${q.tipo} • até ${q.capacidade} pessoa(s)</div>
        <div class="preco">R$ ${q.preco.toFixed(2)} / noite</div>
        <div class="tipo">${q.descricao || ''}</div>
        <button onclick="abrirReserva(${q.id})">Reservar</button>
      </div>`).join('') + '</div>';
  }
  return `<div class="card">
    <h2>Buscar quartos disponíveis</h2>
    <div class="flex-datas">
      <div><label>Entrada</label><input type="date" id="busca_entrada"></div>
      <div><label>Saída</label><input type="date" id="busca_saida"></div>
      <div><button onclick="buscar()">Buscar</button></div>
    </div>
  </div>
  <div class="card">
    <h2>Resultados</h2>
    ${lista}
  </div>`;

async function buscar() {
  try {
    const entrada = document.getElementById('busca_entrada').value;
    const saida = document.getElementById('busca_saida').value;
    if (!entrada || !saida) return aviso('Informe as datas', 'erro');
    quartosDisponiveis = await api(`/api/quartos/disponiveis?entrada=${entrada}&saida=${saida}`);
    render();
  } catch (e) { aviso(e.message, 'erro'); }
}

function abrirReserva(quartoId) {
  const q = quartosDisponiveis.find(x => x.id === quartoId);
  const entrada = document.getElementById('busca_entrada').value;
  const saida = document.getElementById('busca_saida').value;
  const diarias = Math.max(1, Math.round((new Date(saida) - new Date(entrada)) / 86400000));
  const total = diarias * q.preco;
  if (!confirm(`Confirmar reserva do Quarto ${q.numero}?\nEntrada: ${entrada}\nSaída: ${saida}\n${diarias} noite(s)\nTotal: R$ ${total.toFixed(2)}`)) return;
  api('/api/reservas', 'POST', { quarto_id: q.id, data_entrada: entrada, data_saida: saida })
    .then(async () => {
      msg = { texto: 'Reserva solicitada! Aguarde a confirmação do gerente.', tipo: 'ok' };
      await carregarTudo();
      abaAtual = 'minhas';
      render();
    })
    .catch(e => aviso(e.message, 'erro'));
}

    <div class="reserva">
      <div class="linha"><strong>Quarto ${r.quarto_numero} (${r.quarto_tipo})</strong>
        <span class="status ${r.status}">${r.status}</span></div>

function htmlGerenciarQuartos() {
  return `<div class="card">
    <h2>Novo quarto</h2>
    <div class="grid">
      <div><label>Número</label><input id="q_numero"></div>
      <div><label>Tipo</label><input id="q_tipo" placeholder="Solteiro, Casal, Família..."></div>

          <button class="perigo" onclick="removerQuarto(${q.id})">Remover</button>
        </div>
      </div>`).join('') + `</div>

async function criarQuarto() {
  try {
    const numero = document.getElementById('q_numero').value.trim();
    const tipo = document.getElementById('q_tipo').value.trim();
    const capacidade = parseInt(document.getElementById('q_capacidade').value);
    const preco = parseFloat(document.getElementById('q_preco').value);
    const descricao = document.getElementById('q_descricao').value.trim();
    if (!numero || !tipo || !capacidade || isNaN(preco)) return aviso('Preencha os campos', 'erro');
    await api('/api/quartos', 'POST', { numero, tipo, capacidade, preco, descricao });
    msg = { texto: 'Quarto cadastrado.', tipo: 'ok' };
    await carregarTudo();
    render();
  } catch (e) { aviso(e.message, 'erro'); }
}

async function editarQuarto(id) {
  const q = quartos.find(x => x.id === id);
  const numero = prompt('Número do quarto:', q.numero);
  if (numero === null) return;
  const tipo = prompt('Tipo:', q.tipo);
  if (tipo === null) return;
  const capacidade = prompt('Capacidade:', q.capacidade);
  if (capacidade === null) return;
  const preco = prompt('Preço por noite:', q.preco);
  if (preco === null) return;
  const descricao = prompt('Descrição:', q.descricao || '');
  if (descricao === null) return;
  const ativo = confirm('Quarto está ativo? OK = sim, Cancelar = não');
  try {
    await api(`/api/quartos/${id}`, 'PUT', {
      numero, tipo, capacidade: parseInt(capacidade), preco: parseFloat(preco), descricao, ativo
    });
    msg = { texto: 'Quarto atualizado.', tipo: 'ok' };
    await carregarTudo();
    render();
  } catch (e) { aviso(e.message, 'erro'); }
}

async function removerQuarto(id) {
  if (!confirm('Remover este quarto?')) return;
  try {
    await api(`/api/quartos/${id}`, 'DELETE');
    msg = { texto: 'Quarto removido.', tipo: 'ok' };
    await carregarTudo();
    render();
  } catch (e) { aviso(e.message, 'erro'); }
}
