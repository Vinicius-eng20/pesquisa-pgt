=== README.md ===
# Biblioteca Digital

Sistema de biblioteca digital com empréstimo de itens, controle de acervo e usuários.

## Como usar

### Instalação

1. Instale o [Node.js](https://nodejs.org) (versão 18 ou superior).
2. Abra o terminal (Prompt de Comando no Windows) na pasta do projeto.
3. Execute:

```
npm install
```

### Executar

```
npm start
```

4. Abra o navegador em: **http://localhost:3000**

### Contas de teste

- **Bibliotecário**: usuário `admin` / senha `admin123`
- **Leitor**: usuário `joao` / senha `joao123`
- **Leitor**: usuário `maria` / senha `maria123`

### Sobre os dados

Todos os dados (usuários, itens do acervo, empréstimos) ficam salvos no arquivo `dados.json` dentro da pasta do projeto. Nada depende de serviços externos — tudo funciona apenas com o Node.js instalado.

const cookieParser = require('cookie-parser');
const bcrypt = require('bcryptjs');

const path = require('path');
const crypto = require('crypto');

const app = express();
const PORTA = process.env.PORT || 3000;
const ARQUIVO_DADOS = path.join(__dirname, 'dados.json');

app.use(express.json());
app.use(cookieParser());

      { id: 2, titulo: 'O Cortiço', autor: 'Aluísio Azevedo', categoria: 'Livro', ano: 1890, disponivel: true },

      { id: 4, titulo: 'A Origem das Espécies', autor: 'Charles Darwin', categoria: 'Livro', ano: 1859, disponivel: true }

    proximoIdEmprestimo: 1

function carregarDados() {
  if (!fs.existsSync(ARQUIVO_DADOS)) {
    const d = dadosIniciais();
    fs.writeFileSync(ARQUIVO_DADOS, JSON.stringify(d, null, 2));
    return d;
  }
  try {
    return JSON.parse(fs.readFileSync(ARQUIVO_DADOS, 'utf8'));
  } catch (e) {
    const d = dadosIniciais();
    fs.writeFileSync(ARQUIVO_DADOS, JSON.stringify(d, null, 2));
    return d;
  }
}

function salvarDados(d) {
  fs.writeFileSync(ARQUIVO_DADOS, JSON.stringify(d, null, 2));
}

// ------- Autenticação -------
function usuarioLogado(req, dados) {
  const token = req.cookies['sessao'];
  if (!token) return null;
  const id = dados.sessoes[token];
  if (!id) return null;
  return dados.usuarios.find(u => u.id === id) || null;
}

function exigirLogin(req, res, next) {
  const dados = carregarDados();
  const u = usuarioLogado(req, dados);
  if (!u) return res.status(401).json({ erro: 'Não autenticado' });
  req.dados = dados;
  req.usuario = u;
  next();
}

function exigirBibliotecario(req, res, next) {
  if (req.usuario.papel !== 'bibliotecario') {
    return res.status(403).json({ erro: 'Apenas bibliotecário pode fazer isso' });
  }
  next();
}

// ------- Rotas de autenticação -------
app.post('/api/login', (req, res) => {
  const { usuario, senha } = req.body;
  const dados = carregarDados();
  const u = dados.usuarios.find(x => x.usuario === usuario);
  if (!u || !bcrypt.compareSync(senha, u.senha)) {
    return res.status(401).json({ erro: 'Usuário ou senha inválidos' });
  }
  const token = crypto.randomBytes(24).toString('hex');
  dados.sessoes[token] = u.id;
  salvarDados(dados);
  res.cookie('sessao', token, { httpOnly: true, maxAge: 1000 * 60 * 60 * 24 * 30 });
  res.json({ id: u.id, nome: u.nome, usuario: u.usuario, papel: u.papel });
});

app.post('/api/logout', (req, res) => {
  const token = req.cookies['sessao'];
  if (token) {
    const dados = carregarDados();
    delete dados.sessoes[token];
    salvarDados(dados);
  }
  res.clearCookie('sessao');
  res.json({ ok: true });
});

app.get('/api/eu', exigirLogin, (req, res) => {
  res.json({ id: req.usuario.id, nome: req.usuario.nome, usuario: req.usuario.usuario, papel: req.usuario.papel });
});

app.post('/api/registrar', (req, res) => {
  const { nome, usuario, senha } = req.body;
  if (!nome || !usuario || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });
  const dados = carregarDados();
  if (dados.usuarios.find(u => u.usuario === usuario)) {
    return res.status(400).json({ erro: 'Usuário já existe' });
  }
  const novo = {
    id: dados.proximoIdUsuario++,
    nome,
    usuario,
    senha: bcrypt.hashSync(senha, 8),
    papel: 'leitor'
  };
  dados.usuarios.push(novo);
  salvarDados(dados);
  res.json({ ok: true });
});

app.get('/api/itens', exigirLogin, (req, res) => {
  const busca = (req.query.busca || '').toString().toLowerCase();
  let lista = req.dados.itens;
  if (busca) {
    lista = lista.filter(i =>
      i.titulo.toLowerCase().includes(busca) ||
      i.autor.toLowerCase().includes(busca) ||
      i.categoria.toLowerCase().includes(busca)
    );
  }
  res.json(lista);
});

app.post('/api/itens', exigirLogin, exigirBibliotecario, (req, res) => {
  const { titulo, autor, categoria, ano } = req.body;
  if (!titulo || !autor || !categoria) return res.status(400).json({ erro: 'Preencha os campos' });
  const item = {
    id: req.dados.proximoIdItem++,
    titulo,
    autor,
    categoria,
    ano: parseInt(ano) || null,
    disponivel: true
  };
  req.dados.itens.push(item);
  salvarDados(req.dados);
  res.json(item);
});

app.put('/api/itens/:id', exigirLogin, exigirBibliotecario, (req, res) => {
  const id = parseInt(req.params.id);
  const item = req.dados.itens.find(i => i.id === id);
  if (!item) return res.status(404).json({ erro: 'Item não encontrado' });
  const { titulo, autor, categoria, ano } = req.body;
  if (titulo !== undefined) item.titulo = titulo;
  if (autor !== undefined) item.autor = autor;
  if (categoria !== undefined) item.categoria = categoria;
  if (ano !== undefined) item.ano = parseInt(ano) || null;
  salvarDados(req.dados);
  res.json(item);
});

app.delete('/api/itens/:id', exigirLogin, exigirBibliotecario, (req, res) => {
  const id = parseInt(req.params.id);
  const emprestado = req.dados.emprestimos.find(e => e.itemId === id && !e.devolvido);
  if (emprestado) return res.status(400).json({ erro: 'Item está emprestado no momento' });
  const idx = req.dados.itens.findIndex(i => i.id === id);
  if (idx === -1) return res.status(404).json({ erro: 'Item não encontrado' });
  req.dados.itens.splice(idx, 1);
  salvarDados(req.dados);
  res.json({ ok: true });
});

// ------- Empréstimos -------
const DIAS_PRAZO = 14;

app.post('/api/emprestimos', exigirLogin, (req, res) => {
  if (req.usuario.papel !== 'leitor') return res.status(403).json({ erro: 'Apenas leitores podem pegar emprestado' });
  const { itemId } = req.body;
  const item = req.dados.itens.find(i => i.id === parseInt(itemId));
  if (!item) return res.status(404).json({ erro: 'Item não encontrado' });
  if (!item.disponivel) return res.status(400).json({ erro: 'Item indisponível' });

  // Bloquear se houver atraso
  const hoje = new Date();
  const atrasos = req.dados.emprestimos.filter(e =>
    e.usuarioId === req.usuario.id && !e.devolvido && new Date(e.prazo) < hoje
  );
  if (atrasos.length > 0) {
    return res.status(403).json({ erro: 'Você tem empréstimos em atraso. Devolva-os antes de pegar novos.' });
  }

  const empr = {
    id: req.dados.proximoIdEmprestimo++,
    usuarioId: req.usuario.id,
    usuarioNome: req.usuario.nome,
    itemId: item.id,
    itemTitulo: item.titulo,
    dataEmprestimo: hoje.toISOString(),
    prazo: new Date(hoje.getTime() + DIAS_PRAZO * 86400000).toISOString(),
    devolvido: false,
    dataDevolucao: null
  };
  item.disponivel = false;
  req.dados.emprestimos.push(empr);
  salvarDados(req.dados);
  res.json(empr);
});

app.post('/api/emprestimos/:id/devolver', exigirLogin, (req, res) => {
  const id = parseInt(req.params.id);
  const empr = req.dados.emprestimos.find(e => e.id === id);
  if (!empr) return res.status(404).json({ erro: 'Empréstimo não encontrado' });
  if (req.usuario.papel !== 'bibliotecario' && empr.usuarioId !== req.usuario.id) {
    return res.status(403).json({ erro: 'Sem permissão' });
  }
  if (empr.devolvido) return res.status(400).json({ erro: 'Já devolvido' });
  empr.devolvido = true;
  empr.dataDevolucao = new Date().toISOString();
  const item = req.dados.itens.find(i => i.id === empr.itemId);
  if (item) item.disponivel = true;
  salvarDados(req.dados);
  res.json(empr);
});

app.get('/api/meus-emprestimos', exigirLogin, (req, res) => {
  const lista = req.dados.emprestimos
    .filter(e => e.usuarioId === req.usuario.id)
    .sort((a, b) => new Date(b.dataEmprestimo) - new Date(a.dataEmprestimo));
  res.json(lista);
});

app.get('/api/emprestimos', exigirLogin, exigirBibliotecario, (req, res) => {
  const lista = req.dados.emprestimos
    .slice()
    .sort((a, b) => new Date(b.dataEmprestimo) - new Date(a.dataEmprestimo));
  res.json(lista);
});

app.get('/api/leitores', exigirLogin, exigirBibliotecario, (req, res) => {
  const hoje = new Date();
  const leitores = req.dados.usuarios
    .filter(u => u.papel === 'leitor')
    .map(u => {
      const atrasados = req.dados.emprestimos.filter(e =>
        e.usuarioId === u.id && !e.devolvido && new Date(e.prazo) < hoje
      ).length;
      const ativos = req.dados.emprestimos.filter(e =>
        e.usuarioId === u.id && !e.devolvido
      ).length;
      return { id: u.id, nome: u.nome, usuario: u.usuario, ativos, atrasados };
    });
  res.json(leitores);
});

app.listen(PORTA, () => {
  console.log('Biblioteca Digital rodando em http://localhost:' + PORTA);

// ---------- Utilidades ----------
async function api(url, opcoes = {}) {
  const cfg = { credentials: 'same-origin', headers: { 'Content-Type': 'application/json' }, ...opcoes };
  if (cfg.body && typeof cfg.body !== 'string') cfg.body = JSON.stringify(cfg.body);
  const r = await fetch(url, cfg);
  let dados = null;
  try { dados = await r.json(); } catch (e) {}
  if (!r.ok) throw new Error((dados && dados.erro) || 'Erro na requisição');
  return dados;
}

function fmtData(iso) {
  if (!iso) return '—';
  const d = new Date(iso);
  return d.toLocaleDateString('pt-BR');
}

function esc(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  })[c]);
}

async function entrarApp(u) {
  usuarioAtual = u;

  document.getElementById('tela-app').style.display = '';
  document.getElementById('nome-usuario').textContent = u.nome;

function trocarView(nome) {
  document.querySelectorAll('.menu-btn').forEach(b => {
    b.classList.toggle('ativa', b.dataset.view === nome);
  });
  document.querySelectorAll('.view').forEach(v => v.style.display = 'none');
  document.getElementById('view-' + nome).style.display = '';
  if (nome === 'acervo') carregarAcervo();
  if (nome === 'meus') carregarMeus();
  if (nome === 'gerenciar') carregarAdmin();
  if (nome === 'leitores') carregarLeitores();
  if (nome === 'historico') carregarHistorico();
}

let timerBusca = null;
document.getElementById('busca').oninput = () => {

  timerBusca = setTimeout(carregarAcervo, 250);

async function carregarAcervo() {
  const busca = document.getElementById('busca').value;
  const itens = await api('/api/itens?busca=' + encodeURIComponent(busca));
  const div = document.getElementById('lista-acervo');
  if (!itens.length) { div.innerHTML = '<p>Nenhum item encontrado.</p>'; return; }
  div.innerHTML = itens.map(i => `
    <div class="card">
      <h4>${esc(i.titulo)}</h4>
      <div class="info">${esc(i.autor)} — ${esc(i.categoria)}${i.ano ? ' (' + i.ano + ')' : ''}</div>
      <div class="status ${i.disponivel ? 'ok' : 'indisponivel'}">${i.disponivel ? 'Disponível' : 'Indisponível'}</div>
      ${usuarioAtual.papel === 'leitor' && i.disponivel ?
        `<div class="acoes"><button class="btn-emprestar" onclick="emprestar(${i.id})">Pegar emprestado</button></div>` : ''}
    </div>
  `).join('');
}

async function emprestar(id) {
  try {
    await api('/api/emprestimos', { method: 'POST', body: { itemId: id } });
    alert('Empréstimo realizado! Prazo: 14 dias.');
    carregarAcervo();
  } catch (e) { alert(e.message); }
}

// ---------- Meus empréstimos ----------
async function carregarMeus() {
  const lista = await api('/api/meus-emprestimos');
  const div = document.getElementById('lista-meus');
  if (!lista.length) { div.innerHTML = '<p>Você ainda não pegou nenhum item.</p>'; return; }
  const hoje = new Date();
  div.innerHTML = `
    <table class="tabela">
      <thead><tr><th>Item</th><th>Emprestado</th><th>Prazo</th><th>Status</th><th></th></tr></thead>
      <tbody>${lista.map(e => {
        const atrasado = !e.devolvido && new Date(e.prazo) < hoje;
        const status = e.devolvido ? 'Devolvido' : (atrasado ? 'Atrasado' : 'Em andamento');
        const classe = e.devolvido ? 'ok' : (atrasado ? 'atrasado' : 'ok');
        return `<tr>
          <td>${esc(e.itemTitulo)}</td>
          <td>${fmtData(e.dataEmprestimo)}</td>
          <td>${fmtData(e.prazo)}</td>
          <td><span class="status ${classe}">${status}</span></td>
          <td>${!e.devolvido ? `<button class="btn-devolver" onclick="devolver(${e.id})">Devolver</button>` : ''}</td>
        </tr>`;
      }).join('')}</tbody>
    </table>`;
}

async function devolver(id) {
  try {
    await api('/api/emprestimos/' + id + '/devolver', { method: 'POST' });
    alert('Devolução registrada!');
    carregarMeus();
  } catch (e) { alert(e.message); }
}

let editandoId = null;

document.getElementById('btn-salvar-item').onclick = async () => {
  const err = document.getElementById('item-erro');
  err.textContent = '';
  const body = {
    titulo: document.getElementById('item-titulo').value,
    autor: document.getElementById('item-autor').value,
    categoria: document.getElementById('item-categoria').value,
    ano: document.getElementById('item-ano').value
  };
  try {
    if (editandoId) {
      await api('/api/itens/' + editandoId, { method: 'PUT', body });
    } else {
      await api('/api/itens', { method: 'POST', body });
    }
    cancelarEdicao();
    carregarAdmin();
  } catch (e) { err.textContent = e.message; }
};

document.getElementById('btn-cancelar-edicao').onclick = cancelarEdicao;

function cancelarEdicao() {
  editandoId = null;
  document.getElementById('titulo-form-item').textContent = 'Cadastrar novo item';
  document.getElementById('item-titulo').value = '';
  document.getElementById('item-autor').value = '';
  document.getElementById('item-categoria').value = '';
  document.getElementById('item-ano').value = '';
  document.getElementById('btn-cancelar-edicao').style.display = 'none';
  document.getElementById('btn-salvar-item').textContent = 'Salvar';
}

function editarItem(id) {
  api('/api/itens').then(itens => {
    const i = itens.find(x => x.id === id);
    if (!i) return;
    editandoId = id;
    document.getElementById('titulo-form-item').textContent = 'Editar item';
    document.getElementById('item-titulo').value = i.titulo;
    document.getElementById('item-autor').value = i.autor;
    document.getElementById('item-categoria').value = i.categoria;
    document.getElementById('item-ano').value = i.ano || '';
    document.getElementById('btn-cancelar-edicao').style.display = '';
    document.getElementById('btn-salvar-item').textContent = 'Atualizar';
  });
}

async function excluirItem(id) {
  if (!confirm('Remover este item do acervo?')) return;
  try {
    await api('/api/itens/' + id, { method: 'DELETE' });
    carregarAdmin();
  } catch (e) { alert(e.message); }
}

async function carregarAdmin() {
  const itens = await api('/api/itens');
  const div = document.getElementById('lista-admin');
  if (!itens.length) { div.innerHTML = '<p>Nenhum item cadastrado.</p>'; return; }
  div.innerHTML = itens.map(i => `
    <div class="card">
      <h4>${esc(i.titulo)}</h4>
      <div class="info">${esc(i.autor)} — ${esc(i.categoria)}${i.ano ? ' (' + i.ano + ')' : ''}</div>
      <div class="status ${i.disponivel ? 'ok' : 'indisponivel'}">${i.disponivel ? 'Disponível' : 'Emprestado'}</div>
      <div class="acoes">
        <button class="btn-editar" onclick="editarItem(${i.id})">Editar</button>
        <button class="btn-excluir" onclick="excluirItem(${i.id})">Excluir</button>
      </div>
    </div>
  `).join('');
}
