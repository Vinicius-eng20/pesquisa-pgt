# Sistema de Pedidos para Restaurante

Site completo de pedidos para restaurante, com área do cliente e do gerente.

## Como rodar no seu computador

1. Instale o [Node.js](https://nodejs.org) (versão 18 ou superior). É só baixar e instalar, tudo padrão.
2. Abra o terminal (Prompt de Comando no Windows) dentro desta pasta.
3. Rode:

```
npm install
```

4. Depois rode:

```
npm start
```

5. Abra o navegador em: http://localhost:3000

## Como usar na primeira vez

Na primeira vez que o site abre, ele cria automaticamente um usuário gerente:

- **E-mail:** gerente@restaurante.com
- **Senha:** gerente123

Entre com esse usuário, cadastre os pratos do cardápio e depois crie contas de clientes (pela tela de cadastro normal).

**Importante:** troque a senha do gerente depois (ou crie outro gerente e apague este) — mas para começar está funcional.

## Colocar na internet de verdade

O site guarda tudo em um banco de dados SQLite (arquivo `restaurante.db`), dentro da própria pasta do projeto. Para publicar de verdade:

1. Suba este projeto num serviço como **Render**, **Railway** ou **Fly.io** (todos têm plano gratuito).
2. O comando de start é `npm start`. O site usa a variável `PORT` automaticamente.
3. Os dados ficam salvos no arquivo `restaurante.db` — em serviços com disco persistente eles não somem.

## O que cada um pode fazer

**Cliente:**
- Ver o cardápio
- Fazer pedidos (escolher prato e quantidade)
- Acompanhar seus pedidos e o status de cada um

**Gerente:**
- Cadastrar, editar e remover pratos
- Ver todos os pedidos de todos os clientes
- Mudar o status de qualquer pedido

## Estrutura

- `server.js` — servidor principal
- `database.js` — banco de dados e dados iniciais
- `public/` — páginas do site (HTML, CSS, JS)
- `restaurante.db` — arquivo do banco de dados (criado automaticamente)

const app = express();

const JWT_SECRET = process.env.JWT_SECRET || 'troque-este-segredo-em-producao-restaurante-xyz';

app.use(express.json());
app.use(cookieParser());

function gerarToken(user) {
  return jwt.sign({ id: user.id, email: user.email, role: user.role, nome: user.nome }, JWT_SECRET, {
    expiresIn: '30d'
  });
}

function auth(req, res, next) {
  const token = req.cookies.token;
  if (!token) return res.status(401).json({ erro: 'Não autenticado' });
  try {
    req.user = jwt.verify(token, JWT_SECRET);
    next();
  } catch (e) {
    return res.status(401).json({ erro: 'Sessão inválida' });
  }
}

function soGerente(req, res, next) {
  if (req.user.role !== 'gerente') return res.status(403).json({ erro: 'Acesso restrito ao gerente' });
  next();
}

app.post('/api/registrar', (req, res) => {
  const { nome, email, senha, role } = req.body || {};
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha nome, email e senha' });
  if (senha.length < 4) return res.status(400).json({ erro: 'Senha muito curta' });

  const existente = db.prepare('SELECT id FROM usuarios WHERE email = ?').get(email.toLowerCase());
  if (existente) return res.status(400).json({ erro: 'Este email já está cadastrado' });

  // Só permite criar gerente se não existir nenhum gerente ainda
  let userRole = 'cliente';
  if (role === 'gerente') {
    const existeGerente = db.prepare("SELECT id FROM usuarios WHERE role='gerente'").get();
    if (existeGerente) return res.status(403).json({ erro: 'Já existe um gerente cadastrado' });
    userRole = 'gerente';
  }

  const hash = bcrypt.hashSync(senha, 10);
  const info = db.prepare('INSERT INTO usuarios (nome, email, senha, role) VALUES (?, ?, ?, ?)')
    .run(nome, email.toLowerCase(), hash, userRole);

  const user = { id: info.lastInsertRowid, nome, email: email.toLowerCase(), role: userRole };
  const token = gerarToken(user);
  res.cookie('token', token, { httpOnly: true, sameSite: 'lax', maxAge: 30 * 24 * 60 * 60 * 1000 });
  res.json({ ok: true, user });
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body || {};
  if (!email || !senha) return res.status(400).json({ erro: 'Preencha email e senha' });
  const user = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email.toLowerCase());
  if (!user) return res.status(401).json({ erro: 'Email ou senha incorretos' });
  if (!bcrypt.compareSync(senha, user.senha)) return res.status(401).json({ erro: 'Email ou senha incorretos' });

  const token = gerarToken(user);
  res.cookie('token', token, { httpOnly: true, sameSite: 'lax', maxAge: 30 * 24 * 60 * 60 * 1000 });
  res.json({ ok: true, user: { id: user.id, nome: user.nome, email: user.email, role: user.role } });
});

app.post('/api/logout', (req, res) => {
  res.clearCookie('token');
  res.json({ ok: true });
});

app.post('/api/pratos', auth, soGerente, (req, res) => {

  const info = db.prepare('INSERT INTO pratos (nome, descricao, preco) VALUES (?, ?, ?)')

  const prato = db.prepare('SELECT * FROM pratos WHERE id = ?').get(info.lastInsertRowid);

app.put('/api/pratos/:id', auth, soGerente, (req, res) => {
  const { nome, descricao, preco } = req.body || {};
  const id = Number(req.params.id);
  const existe = db.prepare('SELECT id FROM pratos WHERE id = ?').get(id);
  if (!existe) return res.status(404).json({ erro: 'Prato não encontrado' });
  db.prepare('UPDATE pratos SET nome = ?, descricao = ?, preco = ? WHERE id = ?')
    .run(nome, descricao || '', Number(preco), id);
  res.json(db.prepare('SELECT * FROM pratos WHERE id = ?').get(id));
});

app.delete('/api/pratos/:id', auth, soGerente, (req, res) => {
  const id = Number(req.params.id);
  db.prepare('DELETE FROM pratos WHERE id = ?').run(id);
  res.json({ ok: true });
});

app.post('/api/pedidos', auth, (req, res) => {
  const { itens } = req.body || {};
  if (!Array.isArray(itens) || itens.length === 0) return res.status(400).json({ erro: 'Pedido vazio' });

  const itensValidos = [];
  let total = 0;
  for (const it of itens) {
    const prato = db.prepare('SELECT * FROM pratos WHERE id = ?').get(Number(it.prato_id));
    const qtd = Math.max(1, parseInt(it.quantidade) || 1);
    if (!prato) continue;
    itensValidos.push({ prato_id: prato.id, nome: prato.nome, preco: prato.preco, quantidade: qtd });
    total += prato.preco * qtd;
  }
  if (itensValidos.length === 0) return res.status(400).json({ erro: 'Nenhum prato válido' });

  const info = db.prepare('INSERT INTO pedidos (usuario_id, total, status, itens) VALUES (?, ?, ?, ?)')
    .run(req.user.id, total, 'recebido', JSON.stringify(itensValidos));

  res.json({ ok: true, pedidoId: info.lastInsertRowid });
});

app.get('/api/meus-pedidos', auth, (req, res) => {
  const pedidos = db.prepare('SELECT * FROM pedidos WHERE usuario_id = ? ORDER BY id DESC').all(req.user.id);
  res.json(pedidos.map(p => ({ ...p, itens: JSON.parse(p.itens) })));
});

app.get('/api/pedidos', auth, soGerente, (req, res) => {
  const pedidos = db.prepare(`
    SELECT p.*, u.nome AS cliente_nome, u.email AS cliente_email
    FROM pedidos p JOIN usuarios u ON u.id = p.usuario_id
    ORDER BY p.id DESC
  `).all();
  res.json(pedidos.map(p => ({ ...p, itens: JSON.parse(p.itens) })));
});

app.put('/api/pedidos/:id/status', auth, soGerente, (req, res) => {
  const { status } = req.body || {};
  const validos = ['recebido', 'em_preparo', 'pronto', 'entregue', 'cancelado'];
  if (!validos.includes(status)) return res.status(400).json({ erro: 'Status inválido' });
  const id = Number(req.params.id);
  const existe = db.prepare('SELECT id FROM pedidos WHERE id = ?').get(id);
  if (!existe) return res.status(404).json({ erro: 'Pedido não encontrado' });
  db.prepare('UPDATE pedidos SET status = ? WHERE id = ?').run(status, id);
  res.json({ ok: true });
});

});

const Database = require('better-sqlite3');
const bcrypt = require('bcryptjs');

const db = new Database(path.join(__dirname, 'restaurante.db'));
db.pragma('journal_mode = WAL');

db.exec(`
CREATE TABLE IF NOT EXISTS usuarios (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  senha TEXT NOT NULL,
  role TEXT NOT NULL DEFAULT 'cliente',
  criado_em TEXT DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS pratos (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,
  descricao TEXT,
  preco REAL NOT NULL,
  criado_em TEXT DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS pedidos (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  usuario_id INTEGER NOT NULL,
  total REAL NOT NULL,
  status TEXT NOT NULL DEFAULT 'recebido',
  itens TEXT NOT NULL,
  criado_em TEXT DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);
`);

(function () {
  const app = document.getElementById('app');
  const nav = document.getElementById('menu-nav');
  const toast = document.getElementById('toast');

  let usuario = null;
  let paginaAtual = 'entrar';
  let carrinho = [];

  function mostrarToast(msg) {
    toast.textContent = msg;
    toast.classList.add('show');
    setTimeout(() => toast.classList.remove('show'), 3000);
  }

  async function api(url, opcoes = {}) {
    const res = await fetch(url, {
      headers: { 'Content-Type': 'application/json' },
      credentials: 'same-origin',
      ...opcoes,
      body: opcoes.body ? JSON.stringify(opcoes.body) : undefined
    });
    let dados = null;
    try { dados = await res.json(); } catch (e) {}
    if (!res.ok) throw new Error((dados && dados.erro) || 'Erro na operação');
    return dados;
  }

  function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
      '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
    }[c]));
  }

  function dinheiro(v) { return 'R$ ' + Number(v).toFixed(2).replace('.', ','); }

  const nomesStatus = {
    recebido: 'Recebido',
    em_preparo: 'Em preparo',
    pronto: 'Pronto',
    entregue: 'Entregue',
    cancelado: 'Cancelado'
  };

  async function init() {
    try {
      const r = await api('/api/eu');
      usuario = r.user;
    } catch (e) {
      usuario = null;
    }
    renderNav();
    if (usuario) paginaAtual = usuario.role === 'gerente' ? 'pedidos-admin' : 'cardapio';
    else paginaAtual = 'entrar';

    nav.innerHTML = '';

      addNavBtn('Cardápio', 'cardapio');
      addNavBtn('Gerenciar Pratos', 'gerenciar-pratos');
      addNavBtn('Todos os Pedidos', 'pedidos-admin');
    }
    const b = document.createElement('button');
    b.textContent = 'Sair (' + usuario.nome + ')';
    b.onclick = async () => {
      await api('/api/logout', { method: 'POST' });

      carrinho = [];
      paginaAtual = 'entrar';

      render();

    nav.appendChild(b);

      <div id="carrinho"></div>`;
    let pratos = [];
    try { pratos = await api('/api/pratos'); } catch (e) { app.innerHTML = '<div class="card erro">' + esc(e.message) + '</div>'; return; }

    const lista = document.getElementById('lista');
    if (!pratos.length) {
      lista.innerHTML = '<p class="aviso">Nenhum prato cadastrado ainda.</p>';
      return;
    }
    lista.innerHTML = '';
    pratos.forEach(p => {
      const div = document.createElement('div');
      div.className = 'prato';
      div.innerHTML = `
        <div><strong>${esc(p.nome)}</strong></div>
        <div class="desc">${esc(p.descricao || '')}</div>
        <div class="preco">${dinheiro(p.preco)}</div>
      `;
      const btn = document.createElement('button');
      btn.className = 'principal';
      btn.textContent = 'Adicionar';
      btn.onclick = () => {
        const item = carrinho.find(i => i.prato_id === p.id);
        if (item) item.quantidade++;
        else carrinho.push({ prato_id: p.id, nome: p.nome, preco: p.preco, quantidade: 1 });
        renderCarrinho();
        mostrarToast(p.nome + ' adicionado');
      };
      div.appendChild(btn);
      lista.appendChild(div);
    });
    renderCarrinho();
  }

  function renderCarrinho() {
    const el = document.getElementById('carrinho');
    if (!el) return;
    if (!carrinho.length) {
      el.innerHTML = '';
      return;
    }
    const total = carrinho.reduce((s, i) => s + i.preco * i.quantidade, 0);
    el.innerHTML = `
      <div class="card">
        <h2>Seu pedido</h2>
        <div id="itens-carrinho"></div>
        <div class="linha" style="margin-top:10px;justify-content:space-between;">
          <strong>Total: ${dinheiro(total)}</strong>
          <button class="principal" id="enviar">Fazer pedido</button>
        </div>
      </div>
    `;
    const cont = document.getElementById('itens-carrinho');
    carrinho.forEach((it, idx) => {
      const div = document.createElement('div');
      div.className = 'pedido';
      div.innerHTML = `
        <div class="topo">
          <strong>${esc(it.nome)}</strong>
          <span>${dinheiro(it.preco)} × ${it.quantidade} = <span class="total">${dinheiro(it.preco * it.quantidade)}</span></span>
        </div>
      `;
      const acoes = document.createElement('div');
      acoes.className = 'acoes';
      const menos = document.createElement('button');
      menos.className = 'secundario'; menos.textContent = '−';
      menos.onclick = () => { it.quantidade--; if (it.quantidade <= 0) carrinho.splice(idx, 1); renderCarrinho(); };
      const mais = document.createElement('button');
      mais.className = 'secundario'; mais.textContent = '+';
      mais.onclick = () => { it.quantidade++; renderCarrinho(); };
      const rem = document.createElement('button');
      rem.className = 'danger'; rem.textContent = 'Remover';
      rem.onclick = () => { carrinho.splice(idx, 1); renderCarrinho(); };
      acoes.appendChild(menos); acoes.appendChild(mais); acoes.appendChild(rem);
      div.appendChild(acoes);
      cont.appendChild(div);
    });
    document.getElementById('enviar').onclick = async () => {
      try {
        await api('/api/pedidos', { method: 'POST', body: { itens: carrinho.map(i => ({ prato_id: i.prato_id, quantidade: i.quantidade })) } });
        carrinho = [];
        mostrarToast('Pedido enviado!');
        paginaAtual = 'meus-pedidos';
        renderNav();
        render();
      } catch (e) { mostrarToast(e.message); }
    };
  }

  async function renderMeusPedidos() {

    let pedidos = [];

  async function renderGerenciarPratos() {
    app.innerHTML = `
      <div class="card">
        <h2>Novo prato</h2>
        <div id="err" class="erro"></div>
        <input id="nome" placeholder="Nome do prato">
        <textarea id="desc" placeholder="Descrição (opcional)"></textarea>
        <input id="preco" type="number" step="0.01" min="0" placeholder="Preço (ex: 29.90)">
        <button class="principal" id="add">Adicionar prato</button>
      </div>
      <div class="card"><h2>Pratos cadastrados</h2><div id="lista">Carregando...</div></div>
    `;
    document.getElementById('add').onclick = async () => {
      const nome = document.getElementById('nome').value.trim();
      const descricao = document.getElementById('desc').value.trim();
      const preco = parseFloat(document.getElementById('preco').value);
      if (!nome || isNaN(preco)) { document.getElementById('err').textContent = 'Informe nome e preço válidos'; return; }
      try {
        await api('/api/pratos', { method: 'POST', body: { nome, descricao, preco } });
        mostrarToast('Prato adicionado');
        renderGerenciarPratos();
      } catch (e) { document.getElementById('err').textContent = e.message; }
    };
    await listarPratosAdmin();
  }

  async function listarPratosAdmin() {
    let pratos = [];
    try { pratos = await api('/api/pratos'); } catch (e) {}
    const lista = document.getElementById('lista');
    if (!pratos.length) { lista.innerHTML = '<p class="aviso">Nenhum prato.</p>'; return; }
    lista.innerHTML = '';
    pratos.forEach(p => {
      const div = document.createElement('div');
      div.className = 'pedido';
      div.innerHTML = `
        <div class="topo"><strong>${esc(p.nome)}</strong><span>${dinheiro(p.preco)}</span></div>
        <div class="itens">${esc(p.descricao || '')}</div>
      `;
      const acoes = document.createElement('div');
      acoes.className = 'acoes';
      const ed = document.createElement('button');
      ed.className = 'secundario'; ed.textContent = 'Editar';
      ed.onclick = () => editarPrato(p);
      const del = document.createElement('button');
      del.className = 'danger'; del.textContent = 'Remover';
      del.onclick = async () => {
        if (!confirm('Remover o prato "' + p.nome + '"?')) return;
        try { await api('/api/pratos/' + p.id, { method: 'DELETE' }); mostrarToast('Prato removido'); renderGerenciarPratos(); }
        catch (e) { mostrarToast(e.message); }
      };
      acoes.appendChild(ed); acoes.appendChild(del);
      div.appendChild(acoes);
      lista.appendChild(div);
    });
  }

  function editarPrato(p) {
    app.innerHTML = `
      <div class="card">
        <h2>Editar prato</h2>
        <div id="err" class="erro"></div>
        <input id="nome" value="${esc(p.nome)}">
        <textarea id="desc">${esc(p.descricao || '')}</textarea>
        <input id="preco" type="number" step="0.01" min="0" value="${p.preco}">
        <div class="acoes">
          <button class="principal" id="salvar">Salvar</button>
          <button class="secundario" id="cancelar">Cancelar</button>
        </div>
      </div>
    `;
    document.getElementById('cancelar').onclick = () => { paginaAtual = 'gerenciar-pratos'; render(); };
    document.getElementById('salvar').onclick = async () => {
      const nome = document.getElementById('nome').value.trim();
      const descricao = document.getElementById('desc').value.trim();
      const preco = parseFloat(document.getElementById('preco').value);
      if (!nome || isNaN(preco)) { document.getElementById('err').textContent = 'Informe nome e preço válidos'; return; }
      try {
        await api('/api/pratos/' + p.id, { method: 'PUT', body: { nome, descricao, preco } });
        mostrarToast('Prato atualizado');
        paginaAtual = 'gerenciar-pratos';
        render();
      } catch (e) { document.getElementById('err').textContent = e.message; }
    };
  }

  async function renderPedidosAdmin() {
    app.innerHTML = `<div class="card"><h2>Todos os Pedidos</h2><div id="lista">Carregando...</div></div>`;
    let pedidos = [];
    try { pedidos = await api('/api/pedidos'); } catch (e) { app.innerHTML = '<div class="card erro">' + esc(e.message) + '</div>'; return; }
    const lista = document.getElementById('lista');
    if (!pedidos.length) { lista.innerHTML = '<p class="aviso">Ainda não há pedidos.</p>'; return; }
    lista.innerHTML = '';
    pedidos.forEach(p => {
      const div = document.createElement('div');
      div.className = 'pedido';
      const itensHtml = p.itens.map(i => `${esc(i.nome)} × ${i.quantidade}`).join('<br>');
      div.innerHTML = `
        <div class="topo">
          <div>
            <strong>Pedido #${p.id}</strong><br>
            <small>Cliente: ${esc(p.cliente_nome)} (${esc(p.cliente_email)})</small>
          </div>
          <span class="status ${p.status}">${nomesStatus[p.status] || p.status}</span>
        </div>
        <div class="itens">${itensHtml}</div>
        <div class="total">Total: ${dinheiro(p.total)}</div>
      `;
      const acoes = document.createElement('div');
      acoes.className = 'acoes';
      const sel = document.createElement('select');
      sel.style.maxWidth = '200px';
      sel.style.marginBottom = '0';
      Object.keys(nomesStatus).forEach(k => {
        const opt = document.createElement('option');
        opt.value = k; opt.textContent = nomesStatus[k];
        if (k === p.status) opt.selected = true;
        sel.appendChild(opt);
      });
      const btn = document.createElement('button');
      btn.className = 'principal'; btn.textContent = 'Atualizar status';
      btn.onclick = async () => {
        try {
          await api('/api/pedidos/' + p.id + '/status', { method: 'PUT', body: { status: sel.value } });
          mostrarToast('Status atualizado');
          renderPedidosAdmin();
        } catch (e) { mostrarToast(e.message); }
      };
      acoes.appendChild(sel); acoes.appendChild(btn);
      div.appendChild(acoes);
      lista.appendChild(div);
    });
  }
