const express = require('express');

const path = require('path');

const PORT = process.env.PORT || 3000;

// Banco de dados persistente em arquivo
const db = new sqlite3.Database(path.join(__dirname, 'dados.db'));

db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    senha TEXT NOT NULL,
    papel TEXT NOT NULL
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS tarefas (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    descricao TEXT,
    status TEXT NOT NULL DEFAULT 'pendente',
    membro_id INTEGER,
    criado_por INTEGER,
    criado_em TEXT NOT NULL,
    FOREIGN KEY (membro_id) REFERENCES usuarios(id),
    FOREIGN KEY (criado_por) REFERENCES usuarios(id)
  )`);

  // Cria gerente padrão se não existir
  db.get(`SELECT COUNT(*) as n FROM usuarios WHERE papel = 'gerente'`, (err, row) => {
    if (row && row.n === 0) {
      const hash = bcrypt.hashSync('admin123', 10);
      db.run(`INSERT INTO usuarios (nome, email, senha, papel) VALUES (?, ?, ?, ?)`,
        ['Gerente', 'gerente@equipe.com', hash, 'gerente']);
      console.log('Gerente padrão criado: gerente@equipe.com / admin123');
    }
  });
});

function precisaLogin(req, res, next) {
  if (!req.session.userId) return res.status(401).json({ erro: 'Não autenticado' });
  next();
}

function apenasGerente(req, res, next) {
  if (!req.session.userId) return res.status(401).json({ erro: 'Não autenticado' });
  if (req.session.papel !== 'gerente') return res.status(403).json({ erro: 'Apenas gerente' });
  next();
}

app.post('/api/registrar', (req, res) => {
  const { nome, email, senha, papel } = req.body;
  if (!nome || !email || !senha) return res.status(400).json({ erro: 'Preencha todos os campos' });
  const papelFinal = papel === 'gerente' ? 'gerente' : 'membro';
  
  // Só permite criar gerente se não existir nenhum
  if (papelFinal === 'gerente') {
    db.get(`SELECT COUNT(*) as n FROM usuarios WHERE papel = 'gerente'`, (err, row) => {
      if (row && row.n > 0) return res.status(403).json({ erro: 'Já existe um gerente' });
      continuar();
    });
  } else continuar();

  function continuar() {
    const hash = bcrypt.hashSync(senha, 10);
    db.run(`INSERT INTO usuarios (nome, email, senha, papel) VALUES (?, ?, ?, ?)`,
      [nome, email.toLowerCase(), hash, papelFinal],
      function(err) {
        if (err) return res.status(400).json({ erro: 'Email já cadastrado' });
        req.session.userId = this.lastID;
        req.session.nome = nome;
        req.session.papel = papelFinal;
        res.json({ ok: true, papel: papelFinal });
      });
  }
});

app.post('/api/login', (req, res) => {
  const { email, senha } = req.body;
  db.get(`SELECT * FROM usuarios WHERE email = ?`, [email.toLowerCase()], (err, user) => {
    if (!user) return res.status(400).json({ erro: 'Email ou senha inválidos' });
    if (!bcrypt.compareSync(senha, user.senha)) return res.status(400).json({ erro: 'Email ou senha inválidos' });
    req.session.userId = user.id;
    req.session.nome = user.nome;
    req.session.papel = user.papel;
    res.json({ ok: true, papel: user.papel });
  });
});

app.post('/api/logout', (req, res) => {
  req.session.destroy(() => res.json({ ok: true }));
});

app.get('/api/tarefas', precisaLogin, (req, res) => {
  const sql = `SELECT t.*, u.nome as membro_nome, c.nome as criador_nome 
    FROM tarefas t 
    LEFT JOIN usuarios u ON t.membro_id = u.id 
    LEFT JOIN usuarios c ON t.criado_por = c.id`;
  
  if (req.session.papel === 'gerente') {
    db.all(sql + ' ORDER BY t.criado_em DESC', (err, rows) => res.json(rows || []));
  } else {
    db.all(sql + ' WHERE t.membro_id = ? ORDER BY t.criado_em DESC', [req.session.userId], (err, rows) => res.json(rows || []));
  }
});

app.post('/api/tarefas', precisaLogin, (req, res) => {
  const { titulo, descricao, membro_id } = req.body;
  if (!titulo) return res.status(400).json({ erro: 'Título obrigatório' });

  let destino;
  if (req.session.papel === 'gerente') {
    destino = membro_id || null;
  } else {
    // Membro só pode criar para si mesmo
    destino = req.session.userId;
  }

  const agora = new Date().toISOString();
  db.run(`INSERT INTO tarefas (titulo, descricao, status, membro_id, criado_por, criado_em) 
    VALUES (?, ?, 'pendente', ?, ?, ?)`,
    [titulo, descricao || '', destino, req.session.userId, agora],
    function(err) {
      if (err) return res.status(500).json({ erro: 'Erro ao criar' });
      res.json({ ok: true, id: this.lastID });
    });
});

app.put('/api/tarefas/:id', precisaLogin, (req, res) => {
  const id = req.params.id;
  const { titulo, descricao, status, membro_id } = req.body;

  db.get(`SELECT * FROM tarefas WHERE id = ?`, [id], (err, tarefa) => {
    if (!tarefa) return res.status(404).json({ erro: 'Tarefa não encontrada' });

    if (req.session.papel === 'membro') {
      // Membro só pode mudar status da própria tarefa
      if (tarefa.membro_id !== req.session.userId) return res.status(403).json({ erro: 'Sem permissão' });
      if (status && ['pendente', 'em_andamento', 'concluida'].includes(status)) {
        db.run(`UPDATE tarefas SET status = ? WHERE id = ?`, [status, id], () => res.json({ ok: true }));
      } else {
        res.status(400).json({ erro: 'Status inválido' });
      }
    } else {
      // Gerente pode editar tudo
      db.run(`UPDATE tarefas SET titulo = ?, descricao = ?, status = ?, membro_id = ? WHERE id = ?`,
        [titulo, descricao, status, membro_id, id],
        () => res.json({ ok: true }));
    }
  });
});

app.delete('/api/tarefas/:id', apenasGerente, (req, res) => {
  db.run(`DELETE FROM tarefas WHERE id = ?`, [req.params.id], () => res.json({ ok: true }));
});

app.listen(PORT, () => console.log(`Servidor rodando em http://localhost:${PORT}`));

const app = document.getElementById('app');
let estadoAtual = { usuario: null, tarefas: [], usuarios: [], filtro: 'todas', aberturas: null };

async function api(url, opts) {
  const r = await fetch(url, {
    headers: { 'Content-Type': 'application/json' },
    credentials: 'same-origin',
    ...opts
  });
  const data = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(data.erro || 'Erro');
  return data;
}

async function iniciar() {
  try {
    const eu = await api('/api/eu');
    if (eu.logado) {
      estadoAtual.usuario = eu;
      await carregarDados();
      renderApp();
    } else renderLogin();
  } catch (e) { renderLogin(); }
}

async function carregarDados() {
  estadoAtual.tarefas = await api('/api/tarefas');
  if (estadoAtual.usuario.papel === 'gerente') {
    estadoAtual.usuarios = await api('/api/usuarios');
  }
}

function renderLogin() {

          <button id="tab-reg">Criar conta</button>

  const area = document.getElementById('form-area');

    area.innerHTML = `
      <input id="email" type="email" placeholder="Email">
      <input id="senha" type="password" placeholder="Senha">
      <div id="msg-erro" class="erro"></div>
      <button class="principal" id="btn-entrar">Entrar</button>
      <p class="info">Se você é o gerente e ainda não configurou:<br>gerente@equipe.com / admin123</p>`;

function renderFiltros() {
  const f = document.getElementById('filtros');
  const opcoes = [['todas','Todas'],['pendente','Pendentes'],['em_andamento','Em andamento'],['concluida','Concluídas']];
  f.innerHTML = opcoes.map(([v,l]) => `<button data-f="${v}" class="${estadoAtual.filtro === v ? 'ativo':''}">${l}</button>`).join('');
  f.querySelectorAll('button').forEach(b => b.onclick = () => { estadoAtual.filtro = b.dataset.f; renderFiltros(); renderLista(); });
}

function renderLista() {
  const lista = document.getElementById('lista');
  let t = estadoAtual.tarefas;
  if (estadoAtual.filtro !== 'todas') t = t.filter(x => x.status === estadoAtual.filtro);
  if (t.length === 0) {
    lista.innerHTML = `<div class="vazio">Nenhuma tarefa encontrada.</div>`;
    return;
  }
  const ehGerente = estadoAtual.usuario.papel === 'gerente';
  lista.innerHTML = t.map(tarefa => {
    const statusLabel = { pendente: 'Pendente', em_andamento: 'Em andamento', concluida: 'Concluída' }[tarefa.status];
    return `
    <div class="tarefa ${tarefa.status === 'concluida' ? 'concluida' : ''}">
      <div class="linha-topo">
        <h3>${escapeHtml(tarefa.titulo)}</h3>
        <span class="badge ${tarefa.status}">${statusLabel}</span>
      </div>
      ${tarefa.descricao ? `<div class="desc">${escapeHtml(tarefa.descricao)}</div>` : ''}
      <div class="meta">
        ${ehGerente ? `<span>👤 ${tarefa.membro_nome ? escapeHtml(tarefa.membro_nome) : 'Não atribuída'}</span>` : ''}
        <span>📅 ${new Date(tarefa.criado_em).toLocaleDateString('pt-BR')}</span>
      </div>
      <div class="acoes">
        <select data-status="${tarefa.id}">
          <option value="pendente" ${tarefa.status==='pendente'?'selected':''}>Pendente</option>
          <option value="em_andamento" ${tarefa.status==='em_andamento'?'selected':''}>Em andamento</option>
          <option value="concluida" ${tarefa.status==='concluida'?'selected':''}>Concluída</option>
        </select>
        ${ehGerente ? `<button data-editar="${tarefa.id}">✏️ Editar</button><button class="btn-excluir" data-excluir="${tarefa.id}">🗑️ Excluir</button>` : ''}
      </div>
    </div>`;
  }).join('');

  lista.querySelectorAll('[data-status]').forEach(sel => {
    sel.onchange = async () => {
      try {
        await api('/api/tarefas/' + sel.dataset.status, { method: 'PUT', body: JSON.stringify({ status: sel.value }) });
        await carregarDados(); renderLista();
      } catch (e) { alert(e.message); }
    };
  });
  lista.querySelectorAll('[data-editar]').forEach(b => b.onclick = () => abrirModalEditar(+b.dataset.editar));
  lista.querySelectorAll('[data-excluir]').forEach(b => b.onclick = async () => {
    if (!confirm('Excluir esta tarefa?')) return;
    await api('/api/tarefas/' + b.dataset.excluir, { method: 'DELETE' });
    await carregarDados(); renderLista();
  });
}

function abrirModalNova() {
  const ehGerente = estadoAtual.usuario.papel === 'gerente';
  const modal = document.getElementById('modal-area');
  modal.innerHTML = `
    <div class="modal-fundo">
      <div class="modal">
        <h2>Nova tarefa</h2>
        <label>Título</label>
        <input id="m-titulo" placeholder="Ex: Enviar relatório">
        <label>Descrição</label>
        <textarea id="m-desc" placeholder="Detalhes da tarefa..."></textarea>
        ${ehGerente ? `
          <label>Atribuir a</label>
          <select id="m-membro">
            <option value="">— Sem atribuição —</option>
            ${estadoAtual.usuarios.map(u => `<option value="${u.id}">${escapeHtml(u.nome)}</option>`).join('')}
          </select>` : ''}
        <div class="botoes">
          <button class="cancelar" id="m-cancelar">Cancelar</button>
          <button class="salvar" id="m-salvar">Criar tarefa</button>
        </div>
      </div>
    </div>`;
  document.getElementById('m-cancelar').onclick = () => modal.innerHTML = '';
  document.getElementById('m-salvar').onclick = async () => {
    try {
      const body = {
        titulo: document.getElementById('m-titulo').value,
        descricao: document.getElementById('m-desc').value
      };
      if (ehGerente) {
        const mid = document.getElementById('m-membro').value;
        body.membro_id = mid ? +mid : null;
      }
      await api('/api/tarefas', { method: 'POST', body: JSON.stringify(body) });
      modal.innerHTML = '';
      await carregarDados(); renderLista();
    } catch (e) { alert(e.message); }
  };
}

function abrirModalEditar(id) {
  const t = estadoAtual.tarefas.find(x => x.id === id);
  if (!t) return;
  const modal = document.getElementById('modal-area');
  modal.innerHTML = `
    <div class="modal-fundo">
      <div class="modal">
        <h2>Editar tarefa</h2>
        <label>Título</label>
        <input id="m-titulo" value="${escapeHtml(t.titulo)}">
        <label>Descrição</label>
        <textarea id="m-desc">${escapeHtml(t.descricao || '')}</textarea>
        <label>Status</label>
        <select id="m-status">
          <option value="pendente" ${t.status==='pendente'?'selected':''}>Pendente</option>
          <option value="em_andamento" ${t.status==='em_andamento'?'selected':''}>Em andamento</option>
          <option value="concluida" ${t.status==='concluida'?'selected':''}>Concluída</option>
        </select>
        <label>Atribuir a</label>
        <select id="m-membro">
          <option value="">— Sem atribuição —</option>
          ${estadoAtual.usuarios.map(u => `<option value="${u.id}" ${t.membro_id === u.id ? 'selected':''}>${escapeHtml(u.nome)}</option>`).join('')}
        </select>
        <div class="botoes">
          <button class="cancelar" id="m-cancelar">Cancelar</button>
          <button class="salvar" id="m-salvar">Salvar</button>
        </div>
      </div>
    </div>`;
  document.getElementById('m-cancelar').onclick = () => modal.innerHTML = '';
  document.getElementById('m-salvar').onclick = async () => {
    try {
      const mid = document.getElementById('m-membro').value;
      await api('/api/tarefas/' + id, { method: 'PUT', body: JSON.stringify({
        titulo: document.getElementById('m-titulo').value,
        descricao: document.getElementById('m-desc').value,
        status: document.getElementById('m-status').value,
        membro_id: mid ? +mid : null
      })});
      modal.innerHTML = '';
      await carregarDados(); renderLista();
    } catch (e) { alert(e.message); }
  };
}

function escapeHtml(s) {
  return String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

# Gestão de Tarefas da Equipe

Sistema completo de gestão de tarefas para equipes, com contas de usuário, gerente e membros.

## Como instalar e executar

Você precisa ter o **Node.js** instalado (baixe em https://nodejs.org).

1. Abra o terminal (ou prompt de comando) na pasta do projeto.
2. Instale as dependências:
   ```
   npm install
   ```
3. Inicie o servidor:
   ```
   npm start
   ```
4. Abra o navegador em: **http://localhost:3000**

## Primeiro acesso (gerente)

- **Email:** gerente@equipe.com
- **Senha:** admin123

Troque a senha depois criando seu próprio gerente (só é possível ter um gerente por vez).

## Como usar

**Gerente:**
- Cria contas de membros pela tela inicial (aba "Criar conta" → "Sou gerente" — mas só o primeiro pode ser gerente).
- Cria tarefas e atribui aos membros.
- Edita, remove e vê todas as tarefas de todos.

**Membro:**
- Cria sua própria conta (aba "Criar conta" → "Sou membro").
- Vê apenas as tarefas atribuídas a ele.
- Pode mudar o status (Pendente → Em andamento → Concluída).
- Pode criar tarefas para si mesmo.

## Funciona na internet (deploy)

Os dados ficam salvos no arquivo `dados.db` na pasta do projeto (não somem ao fechar). Para colocar online:

1. Suba o projeto para um serviço gratuito de hospedagem Node.js (ex: **Render**, **Railway**, **Fly.io**).
2. Configure a variável `PORT` (a maioria já define automaticamente).
3. Use um disco persistente (Persistent Disk) para que o arquivo `dados.db` não seja apagado em reinícios.

Comandos de build/start:
- Build: `npm install`
- Start: `npm start`

## Acesso pelo celular

Como o site roda no servidor, qualquer pessoa pode acessar do celular ou computador usando o mesmo endereço. Tudo é salvo no servidor — o que você faz no celular aparece no computador e vice-versa.
