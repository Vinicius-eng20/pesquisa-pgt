function readData() {
  if (!fs.existsSync(DB_FILE)) {
    const initial = getInitialData();
    writeData(initial);
    return initial;
  }
  try {
    const raw = fs.readFileSync(DB_FILE, 'utf-8');
    return JSON.parse(raw);
  } catch (err) {
    const initial = getInitialData();
    writeData(initial);
    return initial;
  }
}

function writeData(data) {
  fs.writeFileSync(DB_FILE, JSON.stringify(data, null, 2), 'utf-8');
}

module.exports = {
  getUsers: () => readData().users,
  findUserByEmail: (email) => {
    const users = readData().users;
    return users.find((u) => u.email.toLowerCase() === email.toLowerCase());
  },
  findUserById: (id) => {
    const users = readData().users;
    return users.find((u) => u.id === id);
  },
  createUser: (userData) => {
    const data = readData();
    const newUser = {
      id: 'user-' + Date.now() + '-' + Math.random().toString(36).substr(2, 6),
      createdAt: new Date().toISOString(),
      role: 'student',
      ...userData
    };
    data.users.push(newUser);
    writeData(data);
    return newUser;
  },

  getCourses: () => readData().courses,
  findCourseById: (id) => {
    const courses = readData().courses;
    return courses.find((c) => c.id === id);
  },
  createCourse: (courseData) => {
    const data = readData();
    const newCourse = {
      id: 'curso-' + Date.now(),
      createdAt: new Date().toISOString(),
      ...courseData
    };
    data.courses.unshift(newCourse);
    writeData(data);
    return newCourse;
  },
  updateCourse: (id, updatedFields) => {
    const data = readData();
    const index = data.courses.findIndex((c) => c.id === id);
    if (index === -1) return null;
    data.courses[index] = { ...data.courses[index], ...updatedFields };
    writeData(data);
    return data.courses[index];
  },
  deleteCourse: (id) => {
    const data = readData();
    data.courses = data.courses.filter((c) => c.id !== id);
    data.enrollments = data.enrollments.filter((e) => e.courseId !== id);
    writeData(data);
    return true;
  },

  getEnrollments: () => readData().enrollments,
  findEnrollment: (userId, courseId) => {
    const enrollments = readData().enrollments;
    return enrollments.find(
      (e) => e.userId === userId && e.courseId === courseId && e.status === 'ativa'
    );
  },
  createEnrollment: (userId, courseId) => {
    const data = readData();
    const existing = data.enrollments.find(
      (e) => e.userId === userId && e.courseId === courseId && e.status === 'ativa'
    );
    if (existing) return existing;

    const newEnrollment = {
      id: 'matr-' + Date.now() + '-' + Math.random().toString(36).substr(2, 5),
      userId,
      courseId,
      status: 'ativa',
      enrolledAt: new Date().toISOString()
    };
    data.enrollments.unshift(newEnrollment);
    writeData(data);
    return newEnrollment;
  },
  cancelEnrollment: (enrollmentId) => {
    const data = readData();
    const enrollment = data.enrollments.find((e) => e.id === enrollmentId);
    if (!enrollment) return null;
    enrollment.status = 'cancelada';
    writeData(data);
    return enrollment;
  }
};

app.use(express.json());
app.use(cookieParser());
app.use(express.static(path.join(__dirname, 'public')));

function authenticateToken(req, res, next) {
  const token = req.cookies.token;
  if (!token) {
    req.user = null;
    return next();
  }

  jwt.verify(token, JWT_SECRET, (err, decoded) => {
    if (err) {
      res.clearCookie('token');
      req.user = null;
      return next();
    }
    const user = db.findUserById(decoded.id);
    if (!user) {
      res.clearCookie('token');
      req.user = null;
      return next();
    }
    req.user = { id: user.id, name: user.name, email: user.email, role: user.role };
    next();
  });
}

function requireLogin(req, res, next) {
  if (!req.user) {
    return res.status(401).json({ error: 'Você precisa estar logado para realizar esta ação.' });
  }
  next();
}

function requireAdmin(req, res, next) {
  if (!req.user || req.user.role !== 'admin') {
    return res.status(403).json({ error: 'Acesso negado. Apenas o administrador tem permissão.' });
  }
  next();
}

app.use(authenticateToken);

app.post('/api/auth/register', (req, res) => {
  const { name, email, password } = req.body;
  if (!name || !email || !password) {
    return res.status(400).json({ error: 'Preencha todos os campos obrigatórios.' });
  }
  if (password.length < 6) {
    return res.status(400).json({ error: 'A senha deve ter pelo menos 6 caracteres.' });
  }

  const existing = db.findUserByEmail(email);
  if (existing) {
    return res.status(400).json({ error: 'Este e-mail já está cadastrado.' });
  }

  const salt = bcrypt.genSaltSync(10);
  const passwordHash = bcrypt.hashSync(password, salt);

  const newUser = db.createUser({
    name: name.trim(),
    email: email.trim().toLowerCase(),
    password: passwordHash
  });

  const token = jwt.sign({ id: newUser.id }, JWT_SECRET, { expiresIn: '7d' });
  res.cookie('token', token, {
    httpOnly: true,
    maxAge: 7 * 24 * 60 * 60 * 1000,
    sameSite: 'lax'
  });

  res.status(201).json({
    user: { id: newUser.id, name: newUser.name, email: newUser.email, role: newUser.role }
  });
});

app.post('/api/auth/login', (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) {
    return res.status(400).json({ error: 'Informe e-mail e senha.' });
  }

  const user = db.findUserByEmail(email);
  if (!user) {
    return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
  }

  const validPassword = bcrypt.compareSync(password, user.password);
  if (!validPassword) {
    return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
  }

  const token = jwt.sign({ id: user.id }, JWT_SECRET, { expiresIn: '7d' });
  res.cookie('token', token, {
    httpOnly: true,
    maxAge: 7 * 24 * 60 * 60 * 1000,
    sameSite: 'lax'
  });

  res.json({
    user: { id: user.id, name: user.name, email: user.email, role: user.role }
  });
});

app.post('/api/auth/logout', (req, res) => {
  res.clearCookie('token');
  res.json({ message: 'Logout realizado com sucesso.' });
});

app.get('/api/auth/me', (req, res) => {

});

app.get('/api/courses', (req, res) => {
  const courses = db.getCourses();
  const enrollments = db.getEnrollments();

  const coursesWithStatus = courses.map((course) => {
    let isEnrolled = false;
    if (req.user) {
      isEnrolled = enrollments.some(
        (e) => e.userId === req.user.id && e.courseId === course.id && e.status === 'ativa'
      );
    }
    const totalStudents = enrollments.filter(
      (e) => e.courseId === course.id && e.status === 'ativa'
    ).length;

    return {
      ...course,
      isEnrolled,
      totalStudents
    };
  });

  res.json(coursesWithStatus);
});

app.post('/api/courses', requireAdmin, (req, res) => {
  const { title, description, instructor, category, duration, content } = req.body;
  if (!title || !description || !instructor) {
    return res.status(400).json({ error: 'Título, descrição e instrutor são obrigatórios.' });
  }

  const newCourse = db.createCourse({
    title: title.trim(),
    description: description.trim(),
    instructor: instructor.trim(),
    category: (category || 'Geral').trim(),
    duration: (duration || 'Auto-instrucional').trim(),
    content: (content || 'Conteúdo das aulas disponível após matrícula.').trim()
  });

  res.status(201).json(newCourse);
});

app.put('/api/courses/:id', requireAdmin, (req, res) => {
  const courseId = req.params.id;
  const { title, description, instructor, category, duration, content } = req.body;

  const existing = db.findCourseById(courseId);
  if (!existing) {
    return res.status(404).json({ error: 'Curso não encontrado.' });
  }

  const updated = db.updateCourse(courseId, {
    title: title ? title.trim() : existing.title,
    description: description ? description.trim() : existing.description,
    instructor: instructor ? instructor.trim() : existing.instructor,
    category: category ? category.trim() : existing.category,
    duration: duration ? duration.trim() : existing.duration,
    content: content ? content.trim() : existing.content
  });

  res.json(updated);
});

app.delete('/api/courses/:id', requireAdmin, (req, res) => {
  const courseId = req.params.id;
  const existing = db.findCourseById(courseId);
  if (!existing) {
    return res.status(404).json({ error: 'Curso não encontrado.' });
  }

  db.deleteCourse(courseId);
  res.json({ message: 'Curso excluído com sucesso.' });
});

app.post('/api/enroll/:courseId', requireLogin, (req, res) => {
  const courseId = req.params.courseId;
  const course = db.findCourseById(courseId);
  if (!course) {
    return res.status(404).json({ error: 'Curso não encontrado.' });
  }

  const existing = db.findEnrollment(req.user.id, courseId);
  if (existing) {
    return res.status(400).json({ error: 'Você já está matriculado neste curso.' });
  }

  const enrollment = db.createEnrollment(req.user.id, courseId);
  res.status(201).json({ message: 'Matrícula realizada com sucesso!', enrollment });
});

app.get('/api/my-enrollments', requireLogin, (req, res) => {
  const allEnrollments = db.getEnrollments();
  const allCourses = db.getCourses();

  const userEnrollments = allEnrollments
    .filter((e) => e.userId === req.user.id && e.status === 'ativa')

          category: 'N/A',
          duration: 'N/A',

      };
    });

  res.json(userEnrollments);
});

app.get('/api/admin/enrollments', requireAdmin, (req, res) => {
  const enrollments = db.getEnrollments();
  const users = db.getUsers();
  const courses = db.getCourses();

  const report = enrollments.map((enr) => {
    const student = users.find((u) => u.id === enr.userId);
    const course = courses.find((c) => c.id === enr.courseId);

    return {
      id: enr.id,
      status: enr.status,
      enrolledAt: enr.enrolledAt,
      student: {
        id: student ? student.id : 'desconhecido',
        name: student ? student.name : 'Usuário Excluído',
        email: student ? student.email : 'N/A'
      },
      course: {
        id: course ? course.id : 'desconhecido',
        title: course ? course.title : 'Curso Removido'
      }
    };
  });

  res.json(report);
});

// Toast Notifications
function showToast(message, type = 'info') {
  const container = document.getElementById('toastContainer');
  const toast = document.createElement('div');
  toast.className = `toast toast-${type}`;
  toast.textContent = message;
  container.appendChild(toast);

  setTimeout(() => {
    toast.style.opacity = '0';
    toast.style.transition = 'opacity 0.3s ease';
    setTimeout(() => toast.remove(), 300);
  }, 4000);
}

// Controle de Autenticação
async function checkAuth() {
  try {
    const res = await fetch('/api/auth/me');
    const data = await res.json();
    currentUser = data.user;
    updateNavbar();
    loadCatalog();
  } catch (err) {
    currentUser = null;
    updateNavbar();
    loadCatalog();
  }
}

function updateNavbar() {
  const authArea = document.getElementById('authArea');
  const navMeusCursos = document.getElementById('navMeusCursos');
  const navAdmin = document.getElementById('navAdmin');

  if (currentUser) {
    const roleBadge = currentUser.role === 'admin' 
      ? '<span class="user-badge badge-admin">Admin</span>' 
      : '<span class="user-badge badge-student">Aluno</span>';

    authArea.innerHTML = `
      <div class="user-profile">
        <span class="user-name">${escapeHtml(currentUser.name)} ${roleBadge}</span>
        <button class="btn btn-secondary btn-sm" onclick="handleLogout()">Sair</button>
      </div>
    `;

    if (currentUser.role === 'admin') {
      navAdmin.classList.remove('hidden');
      navMeusCursos.classList.remove('hidden');
    } else {
      navAdmin.classList.add('hidden');
      navMeusCursos.classList.remove('hidden');
    }
  } else {
    authArea.innerHTML = `
      <button class="btn btn-secondary" onclick="openModal('modalLogin')">Entrar</button>
      <button class="btn btn-primary" onclick="openModal('modalRegister')">Criar Conta</button>
    `;
    navMeusCursos.classList.add('hidden');
    navAdmin.classList.add('hidden');
  }
}

async function handleLogin(e) {
  e.preventDefault();
  const email = document.getElementById('loginEmail').value;
  const password = document.getElementById('loginPassword').value;

  try {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao realizar login.', 'error');
      return;
    }

    currentUser = data.user;
    closeModal('modalLogin');
    document.getElementById('formLogin').reset();
    showToast(`Bem-vindo(a), ${currentUser.name}!`, 'success');
    updateNavbar();
    loadCatalog();

    if (currentUser.role === 'admin') {
      navigate('admin');
    } else {
      navigate('catalogo');
    }
  } catch (err) {
    showToast('Falha na comunicação com o servidor.', 'error');
  }
}

async function handleRegister(e) {
  e.preventDefault();
  const name = document.getElementById('regName').value;
  const email = document.getElementById('regEmail').value;
  const password = document.getElementById('regPassword').value;

  try {
    const res = await fetch('/api/auth/register', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, email, password })
    });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao criar conta.', 'error');
      return;
    }

    currentUser = data.user;
    closeModal('modalRegister');
    document.getElementById('formRegister').reset();
    showToast('Conta criada com sucesso!', 'success');
    updateNavbar();
    loadCatalog();
    navigate('catalogo');
  } catch (err) {
    showToast('Falha na comunicação com o servidor.', 'error');
  }
}

async function handleLogout() {
  try {
    await fetch('/api/auth/logout', { method: 'POST' });
    currentUser = null;
    updateNavbar();
    showToast('Você saiu da sua conta.', 'info');
    navigate('catalogo');
    loadCatalog();
  } catch (err) {
    showToast('Erro ao sair.', 'error');
  }
}

// Navegação entre abas principais
function navigate(viewName) {
  document.querySelectorAll('.view-section').forEach(sec => sec.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('active'));

  if (viewName === 'catalogo') {
    document.getElementById('sectionCatalogo').classList.add('active');
    document.getElementById('navCatalogo').classList.add('active');
    loadCatalog();
  } else if (viewName === 'meus-cursos') {
    if (!currentUser) {
      openModal('modalLogin');
      return;
    }
    document.getElementById('sectionMeusCursos').classList.add('active');
    document.getElementById('navMeusCursos').classList.add('active');
    loadMyCourses();
  } else if (viewName === 'admin') {
    if (!currentUser || currentUser.role !== 'admin') {
      showToast('Acesso exclusivo para administradores.', 'error');
      navigate('catalogo');
      return;
    }
    document.getElementById('sectionAdmin').classList.add('active');
    document.getElementById('navAdmin').classList.add('active');
    loadAdminData();
  }
}

// CATÁLOGO DE CURSOS
async function loadCatalog() {
  try {
    const res = await fetch('/api/courses');
    allCourses = await res.json();
    renderCatalog();

function renderCatalog() {
  const container = document.getElementById('catalogGrid');
  if (!allCourses || allCourses.length === 0) {
    container.innerHTML = `
      <div class="empty-state">
        <p>Nenhum curso disponível no momento.</p>
      </div>
    `;
    return;
  }

  container.innerHTML = allCourses.map(course => {
    let actionButton = '';

    if (course.isEnrolled) {
      actionButton = `
        <button class="btn btn-success btn-sm" onclick="openClassroom('${course.id}')">

      actionButton = `
        <button class="btn btn-primary btn-sm" onclick="enrollCourse('${course.id}')">

        </button>
      `;

    return `
      <div class="course-card">
        <div>
          <div class="course-tags">
            <span class="badge">${escapeHtml(course.category || 'Geral')}</span>
            <span class="badge badge-outline">${escapeHtml(course.duration || 'Auto-instrucional')}</span>

          <h3 class="course-title">${escapeHtml(course.title)}</h3>
          <p class="course-instructor">Instrutor: <b>${escapeHtml(course.instructor)}</b></p>
          <p class="course-description">${escapeHtml(course.description)}</p>
        </div>
        <div class="course-footer">
          <span class="course-stats">👥 ${course.totalStudents || 0} matriculados</span>
          ${actionButton}
        </div>
      </div>

  }).join('');

async function enrollCourse(courseId) {
  if (!currentUser) {
    showToast('Faça login ou crie uma conta para se matricular.', 'info');
    openModal('modalLogin');
    return;
  }

  try {
    const res = await fetch(`/api/enroll/${courseId}`, { method: 'POST' });
    const data = await res.json();

    if (!res.ok) {
      showToast(data.error || 'Erro ao realizar matrícula.', 'error');
      return;
    }

    showToast(data.message || 'Matrícula efetuada com sucesso!', 'success');
    loadCatalog();
  } catch (err) {
    showToast('Erro ao realizar matrícula.', 'error');
  }
}

function openClassroom(courseId) {
  let course = allCourses.find(c => c.id === courseId);
  if (!course) {
    const found = myEnrollments.find(e => e.course.id === courseId);
    if (found) course = found.course;

  document.getElementById('classroomTitle').textContent = course.title;
  document.getElementById('classroomCategory').textContent = course.category || 'Geral';
  document.getElementById('classroomDuration').textContent = course.duration || 'Flexível';

  document.getElementById('classroomDescription').textContent = course.description;

  openModal('modalClassroom');
}

// CRIAR / EDITAR CURSO (ADMIN)
function openCourseModal(courseId = null) {
  document.getElementById('formCourse').reset();
  const modalTitle = document.getElementById('modalCourseTitle');
  const editIdField = document.getElementById('courseEditId');

  if (courseId) {
    const course = allCourses.find(c => c.id === courseId);
    if (!course) return;
    modalTitle.textContent = 'Editar Curso';
    editIdField.value = course.id;
    document.getElementById('courseTitle').value = course.title;
    document.getElementById('courseCategory').value = course.category || '';
    document.getElementById('courseDuration').value = course.duration || '';
    document.getElementById('courseInstructor').value = course.instructor;
    document.getElementById('courseDescription').value = course.description;
    document.getElementById('courseContent').value = course.content || '';
  } else {
    modalTitle.textContent = 'Cadastrar Novo Curso';
    editIdField.value = '';
  }

  openModal('modalCourse');
}

async function handleSaveCourse(e) {
  e.preventDefault();
  const id = document.getElementById('courseEditId').value;
  const title = document.getElementById('courseTitle').value;
  const category = document.getElementById('courseCategory').value;
  const duration = document.getElementById('courseDuration').value;
  const instructor = document.getElementById('courseInstructor').value;
  const description = document.getElementById('courseDescription').value;
  const content = document.getElementById('courseContent').value;

  const bodyData = { title, category, duration, instructor, description, content };

  try {
    let res;
    if (id) {
      // Atualizar curso existente
      res = await fetch(`/api/courses/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(bodyData)
      });
    } else {
      // Criar novo curso
      res = await fetch('/api/courses', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(bodyData)
      });
    }

    const data = await res.json();
    if (!res.ok) {
      showToast(data.error || 'Erro ao salvar curso.', 'error');
      return;
    }

    showToast(id ? 'Curso atualizado com sucesso!' : 'Curso cadastrado com sucesso!', 'success');
    closeModal('modalCourse');
    loadAdminCourses();
  } catch (err) {
    showToast('Falha ao comunicar com o servidor.', 'error');
  }
}

// EXCLUIR CURSO (ADMIN)
async function handleDeleteCourse(courseId, courseTitle) {
  const confirmDelete = confirm(`Tem certeza que deseja remover o curso "${courseTitle}"?\nTodas as matrículas vinculadas a ele também serão removidas.`);
  if (!confirmDelete) return;

  try {
    const res = await fetch(`/api/courses/${courseId}`, { method: 'DELETE' });
    const data = await res.json();
    if (!res.ok) {
      showToast(data.error || 'Erro ao excluir o curso.', 'error');
      return;
    }

    showToast('Curso removido com sucesso!', 'success');
    loadAdminData();
  } catch (err) {
    showToast('Erro ao remover o curso.', 'error');
  }
}

// CANCELAR MATRÍCULA (ADMIN)
async function handleCancelEnrollment(enrollmentId, studentName, courseTitle) {
  const confirmCancel = confirm(`Deseja cancelar a matrícula de "${studentName}" no curso "${courseTitle}"?`);
  if (!confirmCancel) return;

  try {
    const res = await fetch(`/api/admin/enrollments/${enrollmentId}/cancel`, { method: 'POST' });
    const data = await res.json();
    if (!res.ok) {
      showToast(data.error || 'Erro ao cancelar matrícula.', 'error');
      return;
    }

    showToast('Matrícula cancelada com sucesso.', 'success');
    loadAdminEnrollments();
  } catch (err) {
    showToast('Erro ao cancelar matrícula.', 'error');
  }
}

// Prevenir injeção de HTML
function escapeHtml(text) {
  if (!text) return '';
  return String(text)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

# Plataforma de Cursos Online com Matrícula

Uma plataforma web completa, moderna e pronta para uso. Funciona tanto para alunos quanto para o administrador, sincronizando em tempo real com o servidor e salvando tudo de forma persistente.

---

## 🚀 Como Instalar e Rodar (Passo a Passo)

Você só precisa ter o **Node.js** instalado no seu computador ([baixar Node.js se não tiver](https://nodejs.org/)).

### 1. Abrir a pasta do projeto no Terminal ou Prompt de Comando
Abra o terminal na pasta onde descompactou os arquivos do projeto.

### 2. Instalar as dependências necessárias
Execute o seguinte comando e aperte **Enter**:
```bash
npm install
```

### 3. Iniciar o site
Execute o comando:
```bash
npm start
```

### 4. Acessar a plataforma
Abra o seu navegador de internet (Google Chrome, Edge, Safari, Firefox) e acesse:
👉 **http://localhost:3000**

---

## 📱 Recursos e Funcionamento

1. **Persistência Total:** Todas as informações (cursos criados, edições, alunos cadastrados e matrículas) ficam salvas em arquivo no próprio servidor (`data/database.json`). Mesmo se você reiniciar o computador ou fechar o navegador, nada se perde.
2. **Multi-dispositivo:** Se você acessar o endereço da máquina pelo celular (usando o IP do seu computador na rede local, por exemplo: `http://192.168.x.x:3000`), todas as ações feitas no celular aparecem imediatamente para quem está no computador.
3. **Catálogo de Cursos:** Alunos podem visualizar a carga horária, instrutor, categoria, número de matriculados e se matricular com 1 clique.
4. **Meus Cursos & Sala de Aula:** Alunos matriculados visualizam seus cursos e acessam a área com o conteúdo e material didático.
5. **Painel do Administrador:**
   - Cadastrar novos cursos com título, categoria, duração, instrutor, resumo e conteúdo.
   - Editar qualquer informação de um curso existente.
   - Excluir cursos da plataforma.
   - Visualizar a lista completa de todas as matrículas com nome do aluno, e-mail, data e status.
   - Cancelar matrículas ativas.
6. **Autenticação Segura:** Senhas criptografadas com `bcryptjs` e sessões protegidas por Cookies HTTP-only com `JWT`.
7. **Zero Serviços Externos:** Não depende de nenhum serviço pago, nuvem externa ou banco de dados externo. Tudo funciona 100% de forma autônoma.
