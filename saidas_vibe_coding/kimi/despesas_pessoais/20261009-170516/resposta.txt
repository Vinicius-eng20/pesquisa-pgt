# 💰 Controle de Despesas Pessoais

Sistema completo para gerenciamento de despesas pessoais com sincronização em tempo real.

## 👥 Tipos de Usuários

### 👤 Usuário Comum
- Registrar despesas com descrição, valor, categoria e data
- Organizar gastos por categorias
- Visualizar gastos mensais
- Acesso de qualquer dispositivo (computador/celular)

### 🔐 Administrador
- Criar e gerenciar categorias de gastos
- Ativar/desativar contas de usuários
- Visualizar total de gastos de todos os usuários
- Dashboard administrativo completo

---

## 🛠️ Tecnologias

- **Backend:** Node.js + Express
- **Banco de Dados:** SQLite (arquivo local)
- **Frontend:** HTML5, CSS3, JavaScript
- **Autenticação:** JWT (JSON Web Tokens)
- **Sessões:** Persistência no servidor

---

## 📦 Instalação e Execução

### 1️⃣ Instalar Node.js
Baixe e instale o Node.js: https://nodejs.org (versão 16 ou superior)

### 2️⃣ Instalar Dependências
```bash
cd despesa-pessoal
npm install
```

### 3️⃣ Executar o Servidor
```bash
npm start
```

### 4️⃣ Acessar o Sistema
Abra seu navegador e acesse: `http://localhost:3000`

```javascript
=== server.js ===
const express = require('express');
const cors = require('cors');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');
const db = require('./database');

const app = express();
const PORT = process.env.PORT || 3000;
const SECRET_KEY = 'sua_chave_secreta_super_segura_2024';

// Middlewares
app.use(cors());
app.use(express.json());
app.use(express.static('public'));

// Middleware de autenticação
function autenticar(req, res, next) {
    const token = req.headers.authorization?.split(' ')[1];
    
    if (!token) {
        return res.status(401).json({ erro: 'Token não fornecido' });
    }

    try {
        const decoded = jwt.verify(token, SECRET_KEY);
        req.usuarioId = decoded.id;
        req.isAdmin = decoded.isAdmin;
        next();
    } catch (erro) {
        return res.status(401).json({ erro: 'Token inválido' });
    }
}

// Middleware de administrador
function autenticarAdmin(req, res, next) {
    if (!req.isAdmin) {
        return res.status(403).json({ erro: 'Acesso negado' });
    }
    next();
}

// ===== ROTAS DE AUTENTICAÇÃO =====

// Registrar usuário
app.post('/api/registrar', async (req, res) => {
    const { nome, email, senha, isAdmin } = req.body;

    if (!nome || !email || !senha) {
        return res.status(400).json({ erro: 'Preencha todos os campos' });
    }

    try {
        const senhaHash = await bcrypt.hash(senha, 10);
        
        db.run(
            'INSERT INTO usuarios (nome, email, senha, isAdmin, ativo) VALUES (?, ?, ?, ?, 1)',
            [nome, email, senhaHash, isAdmin ? 1 : 0],
            function(erro) {
                if (erro) {
                    return res.status(400).json({ erro: 'Email já cadastrado' });
                }
                res.json({ mensagem: 'Usuário registrado com sucesso', id: this.lastID });
            }
        );
    } catch (erro) {
        res.status(500).json({ erro: 'Erro ao registrar usuário' });
    }
});

// Login
app.post('/api/login', (req, res) => {
    const { email, senha } = req.body;

    db.get('SELECT * FROM usuarios WHERE email = ?', [email], async (erro, usuario) => {
        if (erro || !usuario) {
            return res.status(401).json({ erro: 'Email ou senha incorretos' });
        }

        if (!usuario.ativo) {
            return res.status(403).json({ erro: 'Conta desativada' });
        }

        const senhaValida = await bcrypt.compare(senha, usuario.senha);
        if (!senhaValida) {
            return res.status(401).json({ erro: 'Email ou senha incorretos' });
        }

        const token = jwt.sign(
            { id: usuario.id, isAdmin: usuario.isAdmin },
            SECRET_KEY,
            { expiresIn: '30d' }
        );

        res.json({
            token,
            usuario: {
                id: usuario.id,
                nome: usuario.nome,
                email: usuario.email,
                isAdmin: usuario.isAdmin
            }
        });
    });
});

// ===== ROTAS DE DESPESAS =====

// Listar despesas do usuário
app.get('/api/despesas', autenticar, (req, res) => {
    const { mes, ano } = req.query;
    
    let sql = `
        SELECT d.*, c.nome as categoria_nome, c.cor as categoria_cor
        FROM despesas d
        JOIN categorias c ON d.categoria_id = c.id
        WHERE d.usuario_id = ?
    `;
    const params = [req.usuarioId];

    if (mes && ano) {
        sql += ` AND strftime('%m', d.data) = ? AND strftime('%Y', d.data) = ?`;
        params.push(mes.padStart(2, '0'), ano);
    }

    sql += ' ORDER BY d.data DESC';

    db.all(sql, params, (erro, despesas) => {
        if (erro) {
            return res.status(500).json({ erro: 'Erro ao buscar despesas' });
        }
        res.json(despesas);
    });
});

// Criar despesa
app.post('/api/despesas', autenticar, (req, res) => {
    const { descricao, valor, categoria_id, data } = req.body;

    if (!descricao || !valor || !categoria_id || !data) {
        return res.status(400).json({ erro: 'Preencha todos os campos' });
    }

    db.run(
        'INSERT INTO despesas (usuario_id, categoria_id, descricao, valor, data) VALUES (?, ?, ?, ?, ?)',
        [req.usuarioId, categoria_id, descricao, valor, data],
        function(erro) {
            if (erro) {
                return res.status(500).json({ erro: 'Erro ao criar despesa' });
            }
            res.json({ id: this.lastID, mensagem: 'Despesa registrada com sucesso' });
        }
    );
});

// Atualizar despesa
app.put('/api/despesas/:id', autenticar, (req, res) => {
    const { descricao, valor, categoria_id, data } = req.body;
    const { id } = req.params;

    db.run(
        'UPDATE despesas SET descricao = ?, valor = ?, categoria_id = ?, data = ? WHERE id = ? AND usuario_id = ?',
        [descricao, valor, categoria_id, data, id, req.usuarioId],
        function(erro) {
            if (erro || this.changes === 0) {
                return res.status(500).json({ erro: 'Erro ao atualizar despesa' });
            }
            res.json({ mensagem: 'Despesa atualizada com sucesso' });
        }
    );
});

// Deletar despesa
app.delete('/api/despesas/:id', autenticar, (req, res) => {
    db.run(
        'DELETE FROM despesas WHERE id = ? AND usuario_id = ?',
        [req.params.id, req.usuarioId],
        function(erro) {
            if (erro || this.changes === 0) {
                return res.status(500).json({ erro: 'Erro ao deletar despesa' });
            }
            res.json({ mensagem: 'Despesa deletada com sucesso' });
        }
    );
});

// ===== ROTAS DE CATEGORIAS =====

// Listar categorias
app.get('/api/categorias', autenticar, (req, res) => {
    db.all('SELECT * FROM categorias ORDER BY nome', (erro, categorias) => {
        if (erro) {
            return res.status(500).json({ erro: 'Erro ao buscar categorias' });
        }
        res.json(categorias);
    });
});

// Criar categoria (apenas admin)
app.post('/api/categorias', autenticar, autenticarAdmin, (req, res) => {
    const { nome, cor } = req.body;

    if (!nome) {
        return res.status(400).json({ erro: 'Nome é obrigatório' });
    }

    db.run(
        'INSERT INTO categorias (nome, cor) VALUES (?, ?)',
        [nome, cor || '#007bff'],
        function(erro) {
            if (erro) {
                return res.status(400).json({ erro: 'Categoria já existe' });
            }
            res.json({ id: this.lastID, mensagem: 'Categoria criada com sucesso' });
        }
    );
});

// ===== ROTAS ADMINISTRATIVAS =====

// Listar usuários (apenas admin)
app.get('/api/admin/usuarios', autenticar, autenticarAdmin, (req, res) => {
    db.all(
        'SELECT id, nome, email, isAdmin, ativo, created_at FROM usuarios ORDER BY nome',
        (erro, usuarios) => {
            if (erro) {
                return res.status(500).json({ erro: 'Erro ao buscar usuários' });
            }
            res.json(usuarios);
        }
    );
});

// Ativar/desativar usuário (apenas admin)
app.put('/api/admin/usuarios/:id/toggle', autenticar, autenticarAdmin, (req, res) => {
    db.run(
        'UPDATE usuarios SET ativo = NOT ativo WHERE id = ?',
        [req.params.id],
        function(erro) {
            if (erro || this.changes === 0) {
                return res.status(500).json({ erro: 'Erro ao atualizar usuário' });
            }
            res.json({ mensagem: 'Status do usuário atualizado' });
        }
    );
});

// Total de gastos de todos os usuários (apenas admin)
app.get('/api/admin/estatisticas', autenticar, autenticarAdmin, (req, res) => {
    const { mes, ano } = req.query;
    
    let sql = `
        SELECT 
            u.id,
            u.nome,
            u.email,
            COUNT(d.id) as total_despesas,
            COALESCE(SUM(d.valor), 0) as total_gasto
        FROM usuarios u
        LEFT JOIN despesas d ON u.id = d.usuario_id
    `;
    
    const params = [];
    if (mes && ano) {
        sql += ` AND strftime('%m', d.data) = ? AND strftime('%Y', d.data) = ?`;
        params.push(mes.padStart(2, '0'), ano);
    }
    
    sql += ' WHERE u.isAdmin = 0 GROUP BY u.id ORDER BY total_gasto DESC';

    db.all(sql, params, (erro, estatisticas) => {
        if (erro) {
            return res.status(500).json({ erro: 'Erro ao buscar estatísticas' });
        }
        res.json(estatisticas);
    });
});

// Iniciar servidor
app.listen(PORT, () => {
    console.log(`🚀 Servidor rodando em http://localhost:${PORT}`);
});
```

```javascript
=== database.js ===
const sqlite3 = require('sqlite3').verbose();
const path = require('path');

const dbPath = path.join(__dirname, 'database.sqlite');
const db = new sqlite3.Database(dbPath);

// Criar tabelas
db.serialize(() => {
    // Tabela de usuários
    db.run(`
        CREATE TABLE IF NOT EXISTS usuarios (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            email TEXT UNIQUE NOT NULL,
            senha TEXT NOT NULL,
            isAdmin INTEGER DEFAULT 0,
            ativo INTEGER DEFAULT 1,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
    `);

    // Tabela de categorias
    db.run(`
        CREATE TABLE IF NOT EXISTS categorias (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT UNIQUE NOT NULL,
            cor TEXT DEFAULT '#007bff',
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
    `);

    // Tabela de despesas
    db.run(`
        CREATE TABLE IF NOT EXISTS despesas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            usuario_id INTEGER NOT NULL,
            categoria_id INTEGER NOT NULL,
            descricao TEXT NOT NULL,
            valor REAL NOT NULL,
            data DATE NOT NULL,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
            FOREIGN KEY (categoria_id) REFERENCES categorias(id)
        )
    `);

    // Inserir categorias padrão
    const categoriasPadrao = [
        ['Alimentação', '#ff6b6b'],
        ['Transporte', '#4ecdc4'],
        ['Moradia', '#45b7d1'],
        ['Saúde', '#f9ca24'],
        ['Educação', '#6c5ce7'],
        ['Lazer', '#a29bfe'],
        ['Outros', '#95afc0']
    ];

    const stmt = db.prepare('INSERT OR IGNORE INTO categorias (nome, cor) VALUES (?, ?)');
    categoriasPadrao.forEach(cat => stmt.run(cat));
    stmt.finalize();

    console.log('✅ Banco de dados inicializado');
});

module.exports = db;
```

```javascript
document.getElementById('form-despesa').addEventListener('submit', async (e) => {

    const categoria_id = document.getElementById('desp-categoria').value;

            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${token}`
            },
            body: JSON.stringify(despesa)
        });
        
        if (res.ok) {
            alert(despesaEditando ? 'Despesa atualizada!' : 'Despesa adicionada!');
            document.getElementById('form-despesa').reset();
            despesaEditando = null;
            carregarDespesas();
        } else {
            const erro = await res.json();
            alert(erro.erro || 'Erro ao salvar despesa');
        }
    } catch (erro) {
        alert('Erro ao conectar com o servidor');
    }
});

async function carregarDespesas() {
    try {
        const res = await fetch('/api/despesas', {
            headers: { 'Authorization': `Bearer ${token}` }
        });
        despesas = await res.json();
        exibirDespesas(despesas);
    } catch (erro) {
        console.error('Erro ao carregar despesas:', erro);
    }
}

function exibirDespesas(lista) {
    const container = document.getElementById('lista-despesas');
    
    if (lista.length === 0) {
        container.innerHTML = '<div class="empty-state">Nenhuma despesa encontrada</div>';
        return;
    }
    
    container.innerHTML = lista.map(d => `
        <div class="despesa-item">
            <div class="despesa-info">
                <div class="despesa-descricao">${d.descricao}</div>
                <span class="despesa-categoria" style="background: ${d.categoria_cor}">
                    ${d.categoria_nome}
                </span>
                <small style="margin-left: 10px; color: #999;">${formatarData(d.data)}</small>
            </div>
            <div class="despesa-valor">R$ ${d.valor.toFixed(2)}</div>
            <div class="despesa-acoes">
                <button class="btn-edit" onclick="editarDespesa(${d.id})">Editar</button>
                <button class="btn-delete" onclick="deletarDespesa(${d.id})">Excluir</button>
            </div>
        </div>
    `).join('');
}

function filtrarDespesas() {
    const mesAno = document.getElementById('filtro-mes').value;
    if (!mesAno) {
        exibirDespesas(despesas);
        return;
    }
    
    const [ano, mes] = mesAno.split('-');
    const filtradas = despesas.filter(d => {
        const dataDespesa = new Date(d.data);
        return dataDespesa.getMonth() + 1 == mes && dataDespesa.getFullYear() == ano;
    });
    
    exibirDespesas(filtradas);
}

function limparFiltro() {
    document.getElementById('filtro-mes').value = '';
    exibirDespesas(despesas);
}

function editarDespesa(id) {
    const despesa = despesas.find(d => d.id === id);
    if (!despesa) return;
    
    document.getElementById('desp-descricao').value = despesa.descricao;
    document.getElementById('desp-valor').value = despesa.valor;
    document.getElementById('desp-categoria').value = despesa.categoria_id;
    document.getElementById('desp-data').value = despesa.data;
    
    despesaEditando = id;
    window.scrollTo({ top: 0, behavior: 'smooth' });
}

async function deletarDespesa(id) {
    if (!confirm('Deseja realmente excluir esta despesa?')) return;
    
    try {
        const res = await fetch(`/api/despesas/${id}`, {
            method: 'DELETE',
            headers: { 'Authorization': `Bearer ${token}` }
        });
        
        if (res.ok) {
            alert('Despesa excluída!');
            carregarDespesas();
        } else {
            alert('Erro ao excluir despesa');
        }
    } catch (erro) {
        alert('Erro ao conectar com o servidor');
    }
}

// === RELATÓRIO ===
async function gerarRelatorio() {
    const mesAno = document.getElementById('relatorio-mes').value;
    if (!mesAno) return;
    
    const [ano, mes] = mesAno.split('-');
    
    try {
        const res = await fetch(`/api/despesas?mes=${mes}&ano=${ano}`, {
            headers: { 'Authorization': `Bearer ${token}` }
        });
        const despesasMes = await res.json();
        
        // Agrupar por categoria
        const porCategoria = {};
        let total = 0;
        
        despesasMes.forEach(d => {
            if (!porCategoria[d.categoria_nome]) {
                porCategoria[d.categoria_nome] = {
                    total: 0,
                    cor: d.categoria_cor,
                    quantidade: 0
                };
            }
            porCategoria[d.categoria_nome].total += d.valor;
            porCategoria[d.categoria_nome].quantidade++;
            total += d.valor;
        });
        
        // Exibir relatório
        const content = document.getElementById('relatorio-content');
        let html = '';
        
        for (const [categoria, dados] of Object.entries(porCategoria)) {
            html += `
                <div class="relatorio-categoria" style="border-left-color: ${dados.cor}">
                    <div>
                        <strong>${categoria}</strong>
                        <small style="display: block; color: #999;">${dados.quantidade} despesa(s)</small>
                    </div>
                    <strong style="color: #ff6b6b;">R$ ${dados.total.toFixed(2)}</strong>
                </div>
            `;
        }
        
        html += `
            <div class="relatorio-total">
                <p>Total Gasto no Mês</p>
                <h2>R$ ${total.toFixed(2)}</h2>
            </div>
        `;
        
        content.innerHTML = html;
    } catch (erro) {
        console.error('Erro ao gerar relatório:', erro);
    }
}
```
