Para o sistema ser **“de verdade” na internet** (vários dispositivos acessando a mesma conta e os dados não sumirem), ele precisa de um **servidor com banco de dados** (e o projeto abaixo inclui isso). Sem servidor, não existe como sincronizar entre computador/celular usando “só um arquivo HTML”.

Abaixo está um projeto completo (Node.js + Express + SQLite) com:
- **Cliente**: ver cardápio, criar conta/entrar, fazer pedido, acompanhar seus pedidos.
- **Gerente**: CRUD do cardápio, mudar situação dos pedidos, ver todos os pedidos.
- **Persistência real**: banco SQLite em arquivo (não perde ao fechar o site).

# Sistema de Pedidos de Restaurante (Clientes + Gerente)

## Como rodar na sua máquina (para testar)
1. Instale o **Node.js (versão 18 ou superior)**.
2. Baixe este projeto e abra um terminal na pasta dele.
3. Instale as dependências:
   ```bash
   npm install
   ```
4. Rode o servidor:
   ```bash
   npm start
   ```
5. Abra no navegador:
   - http://localhost:3000

## Login do Gerente (primeiro acesso)
- E-mail: `gerente@restaurante.com`
- Senha: `123456`

> Troque essa senha depois (o projeto já salva usuários no banco; você pode criar outro usuário e mudar o papel para `manager` no banco se quiser).

## Como colocar “na internet de verdade” (opcional)
Você precisa hospedar o servidor em um provedor (ex.: um VPS). Depois de copiar os arquivos para o servidor:
- `npm install`
- `npm start`

O banco SQLite fica salvo no arquivo `data/restaurante.db` (persistente).

## Variáveis de ambiente (opcional, recomendado)
- `SESSION_SECRET` (obrigatório em produção)
- `PORT` (padrão 3000)
- `DB_FILE` (padrão `data/restaurante.db`)

const path = require("path");
const express = require("express");
const session = require("express-session");
const helmet = require("helmet");
const SQLiteStore = require("connect-sqlite3")(session);
const { db, init, seed } = require("./db");

const app = express();
const PORT = Number(process.env.PORT || 3000);

// Segurança básica
app.use(helmet());
app.use(express.json({ limit: "1mb" }));
app.use(express.urlencoded({ extended: false }));

// Sessões (login) persistidas no SQLite (não somem ao reiniciar)
app.use(
  session({
    store: new SQLiteStore({
      db: "sessions.sqlite",
      dir: path.join(__dirname, "..", "data"),
    }),
    secret: process.env.SESSION_SECRET || "troque-esta-chave-em-producao",
    resave: false,
    saveUninitialized: false,
    cookie: {
      httpOnly: true,
      sameSite: "lax",
      secure: false, // se for HTTPS, troque para true
      maxAge: 1000 * 60 * 60 * 24 * 7, // 7 dias
    },
  })
);

// Arquivos do front-end
app.use(express.static(path.join(__dirname, "..", "public")));

// Inicializa banco e dados básicos
init();
seed();

function requireAuth(req, res, next) {
  if (!req.session.user) return res.status(401).json({ error: "Não autenticado." });
  next();
}

function requireManager(req, res, next) {
  if (!req.session.user) return res.status(401).json({ error: "Não autenticado." });
  if (req.session.user.role !== "manager") return res.status(403).json({ error: "Acesso negado (somente gerente)." });
  next();
}

app.post("/api/register", (req, res) => {
  const { name, email, password } = req.body || {};
  if (!name || !email || !password) return res.status(400).json({ error: "Preencha nome, e-mail e senha." });

  const bcrypt = require("bcryptjs");
  const hash = bcrypt.hashSync(String(password), 10);

  try {
    const stmt = db.prepare("INSERT INTO users (name, email, password_hash, role) VALUES (?, ?, ?, 'customer')");
    const info = stmt.run(String(name), String(email).toLowerCase(), hash);

    req.session.user = { id: info.lastInsertRowid, name: String(name), email: String(email).toLowerCase(), role: "customer" };
    res.json({ ok: true });
  } catch (e) {
    if (String(e).includes("UNIQUE")) return res.status(409).json({ error: "Este e-mail já está cadastrado." });
    res.status(500).json({ error: "Erro ao cadastrar." });
  }
});

app.post("/api/login", (req, res) => {
  const { email, password } = req.body || {};
  if (!email || !password) return res.status(400).json({ error: "Preencha e-mail e senha." });

  const bcrypt = require("bcryptjs");
  const user = db.prepare("SELECT * FROM users WHERE email = ?").get(String(email).toLowerCase());
  if (!user) return res.status(401).json({ error: "E-mail ou senha inválidos." });

  const ok = bcrypt.compareSync(String(password), user.password_hash);
  if (!ok) return res.status(401).json({ error: "E-mail ou senha inválidos." });

  req.session.user = { id: user.id, name: user.name, email: user.email, role: user.role };
  res.json({ ok: true });
});

app.post("/api/logout", (req, res) => {
  req.session.destroy(() => res.json({ ok: true }));
});

app.get("/api/me", (req, res) => {
  res.json({ user: req.session.user || null });

app.get("/api/menu", (req, res) => {

app.post("/api/manager/menu", requireManager, (req, res) => {
  const { name, description, price } = req.body || {};
  if (!name || price === undefined) return res.status(400).json({ error: "Informe nome e preço." });

  const price_cents = Math.round(Number(price) * 100);
  if (!Number.isFinite(price_cents) || price_cents < 0) return res.status(400).json({ error: "Preço inválido." });

  const stmt = db.prepare("INSERT INTO menu_items (name, description, price_cents, is_active) VALUES (?, ?, ?, 1)");
  const info = stmt.run(String(name), String(description || ""), price_cents);
  res.json({ ok: true, id: info.lastInsertRowid });
});

app.put("/api/manager/menu/:id", requireManager, (req, res) => {
  const id = Number(req.params.id);
  const { name, description, price, is_active } = req.body || {};
  if (!id) return res.status(400).json({ error: "ID inválido." });

  const existing = db.prepare("SELECT * FROM menu_items WHERE id = ?").get(id);
  if (!existing) return res.status(404).json({ error: "Item não encontrado." });

  const price_cents = price === undefined ? existing.price_cents : Math.round(Number(price) * 100);
  if (!Number.isFinite(price_cents) || price_cents < 0) return res.status(400).json({ error: "Preço inválido." });

  db.prepare("UPDATE menu_items SET name = ?, description = ?, price_cents = ?, is_active = ? WHERE id = ?")
    .run(
      name === undefined ? existing.name : String(name),
      description === undefined ? existing.description : String(description || ""),
      price_cents,
      is_active === undefined ? existing.is_active : (is_active ? 1 : 0),
      id
    );

  res.json({ ok: true });
});

app.delete("/api/manager/menu/:id", requireManager, (req, res) => {
  const id = Number(req.params.id);
  if (!id) return res.status(400).json({ error: "ID inválido." });

  db.prepare("DELETE FROM menu_items WHERE id = ?").run(id);
  res.json({ ok: true });
});

/**
 * ORDERS (cliente cria; cliente vê os dele; gerente vê todos e muda status)
 */
app.post("/api/orders", requireAuth, (req, res) => {
  const user = req.session.user;
  const { items } = req.body || {};
  if (!Array.isArray(items) || items.length === 0) return res.status(400).json({ error: "Envie itens do pedido." });

  // valida itens e calcula total
  const getItem = db.prepare("SELECT id, name, price_cents, is_active FROM menu_items WHERE id = ?");
  let total = 0;

  const normalized = items.map((it) => {
    const menuId = Number(it.menu_item_id);
    const qty = Number(it.quantity);
    if (!menuId || !qty || qty <= 0) throw new Error("Item inválido no pedido.");
    const menu = getItem.get(menuId);
    if (!menu || !menu.is_active) throw new Error("Item indisponível no cardápio.");
    total += menu.price_cents * qty;
    return { menu_item_id: menuId, quantity: qty, unit_price_cents: menu.price_cents };
  });

  const tx = db.transaction(() => {
    const orderStmt = db.prepare("INSERT INTO orders (user_id, status, total_cents) VALUES (?, 'RECEBIDO', ?)");
    const orderInfo = orderStmt.run(user.id, total);

    const orderItemStmt = db.prepare(
      "INSERT INTO order_items (order_id, menu_item_id, quantity, unit_price_cents) VALUES (?, ?, ?, ?)"
    );

    for (const it of normalized) {
      orderItemStmt.run(orderInfo.lastInsertRowid, it.menu_item_id, it.quantity, it.unit_price_cents);
    }
    return orderInfo.lastInsertRowid;
  });

  try {
    const orderId = tx();
    res.json({ ok: true, order_id: orderId, total_cents: total });
  } catch (e) {
    res.status(400).json({ error: e.message || "Não foi possível criar o pedido." });
  }
});

app.get("/api/my-orders", requireAuth, (req, res) => {
  const user = req.session.user;

  const orders = db
    .prepare("SELECT id, status, total_cents, created_at FROM orders WHERE user_id = ? ORDER BY id DESC")
    .all(user.id);

  const itemsStmt = db.prepare(
    `SELECT oi.menu_item_id, mi.name, oi.quantity, oi.unit_price_cents
     FROM order_items oi
     JOIN menu_items mi ON mi.id = oi.menu_item_id
     WHERE oi.order_id = ?`
  );

  const withItems = orders.map((o) => ({
    ...o,
    items: itemsStmt.all(o.id),
  }));

  res.json({ orders: withItems });
});

app.get("/api/manager/orders", requireManager, (req, res) => {
  const orders = db
    .prepare(
      `SELECT o.id, o.status, o.total_cents, o.created_at, u.name AS customer_name, u.email AS customer_email
       FROM orders o
       JOIN users u ON u.id = o.user_id
       ORDER BY o.id DESC`
    )
    .all();

  const itemsStmt = db.prepare(
    `SELECT oi.menu_item_id, mi.name, oi.quantity, oi.unit_price_cents
     FROM order_items oi
     JOIN menu_items mi ON mi.id = oi.menu_item_id
     WHERE oi.order_id = ?`
  );

  const withItems = orders.map((o) => ({
    ...o,
    items: itemsStmt.all(o.id),
  }));

  res.json({ orders: withItems });
});

app.put("/api/manager/orders/:id/status", requireManager, (req, res) => {
  const id = Number(req.params.id);
  const { status } = req.body || {};
  const allowed = new Set(["RECEBIDO", "EM_PREPARO", "PRONTO", "ENTREGUE", "CANCELADO"]);

  if (!id) return res.status(400).json({ error: "ID inválido." });
  if (!allowed.has(String(status))) return res.status(400).json({ error: "Status inválido." });

  db.prepare("UPDATE orders SET status = ? WHERE id = ?").run(String(status), id);
  res.json({ ok: true });
});

// Fallback: sempre devolve a página principal
app.get("*", (req, res) => {
  res.sendFile(path.join(__dirname, "..", "public", "index.html"));
});

app.listen(PORT, () => {
  console.log(`Servidor rodando em http://localhost:${PORT}`);
});

```js
const path = require("path");
const fs = require("fs");
const Database = require("better-sqlite3");
const bcrypt = require("bcryptjs");

const dataDir = path.join(__dirname, "..", "data");
fs.mkdirSync(dataDir, { recursive: true });

const dbFile = process.env.DB_FILE || path.join(dataDir, "restaurante.db");
const db = new Database(dbFile);

function init() {
  db.pragma("journal_mode = WAL");

  db.exec(`
    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      email TEXT NOT NULL UNIQUE,
      password_hash TEXT NOT NULL,
      role TEXT NOT NULL CHECK (role IN ('customer','manager')),
      created_at TEXT NOT NULL DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS menu_items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      description TEXT NOT NULL DEFAULT '',
      price_cents INTEGER NOT NULL,
      is_active INTEGER NOT NULL DEFAULT 1,
      created_at TEXT NOT NULL DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS orders (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      user_id INTEGER NOT NULL,
      status TEXT NOT NULL,
      total_cents INTEGER NOT NULL,
      created_at TEXT NOT NULL DEFAULT (datetime('now')),
      FOREIGN KEY (user_id) REFERENCES users(id)
    );

    CREATE TABLE IF NOT EXISTS order_items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      order_id INTEGER NOT NULL,
      menu_item_id INTEGER NOT NULL,
      quantity INTEGER NOT NULL,
      unit_price_cents INTEGER NOT NULL,
      FOREIGN KEY (order_id) REFERENCES orders(id),
      FOREIGN KEY (menu_item_id) REFERENCES menu_items(id)
    );
  `);
}

function seed() {
  // Cria gerente padrão se não existir
  const managerEmail = "gerente@restaurante.com";
  const manager = db.prepare("SELECT id FROM users WHERE email = ?").get(managerEmail);
  if (!manager) {
    const hash = bcrypt.hashSync("123456", 10);
    db.prepare("INSERT INTO users (name, email, password_hash, role) VALUES (?, ?, ?, 'manager')")
      .run("Gerente", managerEmail, hash);
  }

  // Alguns itens iniciais do cardápio (só se estiver vazio)
  const count = db.prepare("SELECT COUNT(*) AS c FROM menu_items").get().c;
  if (count === 0) {
    const stmt = db.prepare("INSERT INTO menu_items (name, description, price_cents, is_active) VALUES (?, ?, ?, 1)");
    stmt.run("Prato do dia", "Arroz, feijão, salada e proteína", 2590);
    stmt.run("Massa alho e óleo", "Macarrão com alho e azeite", 2290);
    stmt.run("Suco natural", "Suco da fruta (300ml)", 890);
  }
}

module.exports = { db, init, seed };
```

let me = null;
let menu = [];
let cart = new Map(); // menu_item_id -> qty

const $ = (sel) => document.querySelector(sel);

function fmtMoney(cents){
  return (cents/100).toLocaleString("pt-BR", { style:"currency", currency:"BRL" });
}

async function api(path, opts={}){
  const res = await fetch(path, {
    headers: { "Content-Type": "application/json" },
    credentials: "same-origin",
    ...opts,
  });
  const data = await res.json().catch(() => ({}));
  if(!res.ok) throw new Error(data.error || "Erro na requisição.");
  return data;
}

function renderMenu(){
  const box = $("#menuList");
  box.innerHTML = "";
  const active = menu.filter(i => i.is_active === 1);

  if(active.length === 0){
    box.innerHTML = `<p class="muted">Nenhum item disponível no momento.</p>`;
    return;
  }

  for(const item of active){
    const div = document.createElement("div");
    div.className = "item";
    div.innerHTML = `
      <h4>${escapeHtml(item.name)}</h4>
      <div class="muted">${escapeHtml(item.description || "")}</div>
      <div class="row between" style="margin-top:10px">
        <span class="price">${fmtMoney(item.price_cents)}</span>
        <div class="row">
          <button class="btn" data-add="${item.id}">Adicionar</button>
        </div>
      </div>
    `;
    box.appendChild(div);
  }

  box.querySelectorAll("[data-add]").forEach(btn => {
    btn.addEventListener("click", () => {
      const id = Number(btn.getAttribute("data-add"));
      cart.set(id, (cart.get(id) || 0) + 1);
      renderCart();
    });
  });
}

function renderCart(){
  const box = $("#cartBox");
  const totalEl = $("#cartTotal");

  if(cart.size === 0){
    box.innerHTML = `<p class="muted">Adicione itens do cardápio.</p>`;
    totalEl.textContent = fmtMoney(0);
    return;
  }

  let total = 0;
  const rows = [];
  for(const [id, qty] of cart.entries()){
    const item = menu.find(i => i.id === id);
    if(!item) continue;
    const line = item.price_cents * qty;
    total += line;
    rows.push(`
      <div class="row between" style="padding:6px 0;border-bottom:1px solid var(--border)">
        <div>
          <div><strong>${escapeHtml(item.name)}</strong></div>
          <div class="muted small">${qty} × ${fmtMoney(item.price_cents)}</div>
        </div>
        <div class="row">
          <button class="btn" data-dec="${id}">-</button>
          <button class="btn" data-inc="${id}">+</button>
          <button class="btn danger" data-del="${id}">Remover</button>
        </div>
      </div>
    `);
  }

  box.innerHTML = rows.join("");
  totalEl.textContent = fmtMoney(total);

  box.querySelectorAll("[data-inc]").forEach(b => b.addEventListener("click", () => {
    const id = Number(b.getAttribute("data-inc"));
    cart.set(id, (cart.get(id) || 0) + 1);
    renderCart();
  }));
  box.querySelectorAll("[data-dec]").forEach(b => b.addEventListener("click", () => {
    const id = Number(b.getAttribute("data-dec"));
    const next = (cart.get(id) || 0) - 1;
    if(next <= 0) cart.delete(id); else cart.set(id, next);
    renderCart();
  }));
  box.querySelectorAll("[data-del]").forEach(b => b.addEventListener("click", () => {
    const id = Number(b.getAttribute("data-del"));
    cart.delete(id);
    renderCart();
  }));
}

async function placeOrder(){
  setOrderMessage("");
  if(cart.size === 0){
    setOrderMessage("Seu carrinho está vazio.");
    return;
  }

  const items = [...cart.entries()].map(([menu_item_id, quantity]) => ({ menu_item_id, quantity }));
  const r = await api("/api/orders", { method:"POST", body: JSON.stringify({ items }) });
  cart.clear();
  renderCart();
  setOrderMessage(`Pedido #${r.order_id} criado! Total: ${fmtMoney(r.total_cents)}`);
}

async function managerLoadMenu(){
  const { items } = await api("/api/menu");
  const box = $("#managerMenuList");
  box.innerHTML = "";

  if(items.length === 0){
    box.innerHTML = `<p class="muted">Nenhum item cadastrado.</p>`;
    return;
  }

  const table = document.createElement("table");
  table.innerHTML = `
    <thead>
      <tr><th>Item</th><th>Preço</th><th>Ativo</th><th></th></tr>
    </thead>
    <tbody></tbody>
  `;
  const tbody = table.querySelector("tbody");

  for(const it of items){
    const tr = document.createElement("tr");
    tr.innerHTML = `
      <td>
        <div><strong>${escapeHtml(it.name)}</strong></div>
        <div class="muted small">${escapeHtml(it.description || "")}</div>
      </td>
      <td>${fmtMoney(it.price_cents)}</td>
      <td>${it.is_active ? "Sim" : "Não"}</td>
      <td class="row">
        <button class="btn" data-edit="${it.id}">Editar</button>
        <button class="btn danger" data-remove="${it.id}">Remover</button>
      </td>
    `;
    tbody.appendChild(tr);
  }

  box.appendChild(table);

  box.querySelectorAll("[data-edit]").forEach(b => b.addEventListener("click", () => {
    const id = Number(b.getAttribute("data-edit"));
    const it = items.find(x => x.id === id);
    if(!it) return;

    const f = $("#menuItemForm");
    f.id.value = it.id;
    f.name.value = it.name;
    f.description.value = it.description;
    f.price.value = (it.price_cents/100).toFixed(2).replace(".", ",");
    f.is_active.checked = !!it.is_active;
    setMenuItemMessage("Editando item selecionado. Clique em Salvar.");
  }));

  box.querySelectorAll("[data-remove]").forEach(b => b.addEventListener("click", async () => {
    const id = Number(b.getAttribute("data-remove"));
    if(!confirm("Remover este item do cardápio?")) return;
    await api(`/api/manager/menu/${id}`, { method:"DELETE" });
    await managerLoadMenu();
    await loadMenu();
  }));
}

async function managerSaveMenuItem(ev){
  ev.preventDefault();
  setMenuItemMessage("");
  const f = ev.target;

  const payload = {
    name: f.name.value,
    description: f.description.value,
    price: String(f.price.value).replace(",", "."),
    is_active: f.is_active.checked
  };

  const id = Number(f.id.value || 0);
  if(id){
    await api(`/api/manager/menu/${id}`, { method:"PUT", body: JSON.stringify(payload) });
    setMenuItemMessage("Item atualizado.");
  }else{
    await api(`/api/manager/menu`, { method:"POST", body: JSON.stringify(payload) });
    setMenuItemMessage("Item cadastrado.");
  }

  f.reset();
  f.id.value = "";
  f.is_active.checked = true;

  await managerLoadMenu();
  await loadMenu();
}

async function managerLoadOrders(){
  const { orders } = await api("/api/manager/orders");
  const box = $("#allOrdersList");
  box.innerHTML = "";

  if(orders.length === 0){
    box.innerHTML = `<p class="muted">Nenhum pedido ainda.</p>`;
    return;
  }

  for(const o of orders){
    const div = document.createElement("div");
    div.className = "item";
    div.style.marginBottom = "10px";

    const itemsHtml = o.items.map(it => `
      <li>${escapeHtml(it.name)} — ${it.quantity} × ${fmtMoney(it.unit_price_cents)}</li>
    `).join("");

    div.innerHTML = `
      <div class="row between">
        <div>
          <div><strong>Pedido #${o.id}</strong> <span class="muted small">(${escapeHtml(o.created_at)})</span></div>
          <div class="muted small">Cliente: ${escapeHtml(o.customer_name)} — ${escapeHtml(o.customer_email)}</div>
        </div>
        ${statusBadge(o.status)}
      </div>
      <ul class="muted">${itemsHtml}</ul>
      <div class="row between">
        <strong>Total: ${fmtMoney(o.total_cents)}</strong>
        <div class="row">
          <select data-status="${o.id}">
            ${["RECEBIDO","EM_PREPARO","PRONTO","ENTREGUE","CANCELADO"].map(s =>
              `<option value="${s}" ${s===o.status?"selected":""}>${s}</option>`
            ).join("")}
          </select>
          <button class="btn primary" data-save="${o.id}">Salvar</button>
        </div>
      </div>
    `;
    box.appendChild(div);
  }

  box.querySelectorAll("[data-save]").forEach(btn => btn.addEventListener("click", async () => {
    const id = Number(btn.getAttribute("data-save"));
    const sel = box.querySelector(`select[data-status="${id}"]`);
    const status = sel.value;
    await api(`/api/manager/orders/${id}/status`, { method:"PUT", body: JSON.stringify({ status }) });
    await managerLoadOrders();
  }));
}

function escapeHtml(str){
  return String(str)
    .replaceAll("&","&amp;")
    .replaceAll("<","&lt;")
    .replaceAll(">","&gt;")
    .replaceAll('"',"&quot;")
    .replaceAll("'","&#039;");
}

function bindUI(){
  $("#tabLogin").addEventListener("click", () => {
    $("#tabLogin").classList.add("active");
    $("#tabRegister").classList.remove("active");
    $("#loginForm").classList.remove("hidden");
    $("#registerForm").classList.add("hidden");
    setAuthMessage("");
  });

  $("#tabRegister").addEventListener("click", () => {
    $("#tabRegister").classList.add("active");
    $("#tabLogin").classList.remove("active");
    $("#registerForm").classList.remove("hidden");
    $("#loginForm").classList.add("hidden");
    setAuthMessage("");
  });

  $("#loginForm").addEventListener("submit", async (ev) => {
    ev.preventDefault();
    setAuthMessage("");
    const f = ev.target;
    try{
      await api("/api/login", { method:"POST", body: JSON.stringify({
        email: f.email.value,
        password: f.password.value
      })});
      await loadMe();
      await loadMenu();
      await loadMyOrders();
      if(me.role === "manager"){
        await managerLoadMenu();
        await managerLoadOrders();
      }
    }catch(e){
      setAuthMessage(e.message);
    }
  });

  $("#registerForm").addEventListener("submit", async (ev) => {
    ev.preventDefault();
    setAuthMessage("");
    const f = ev.target;
    try{
      await api("/api/register", { method:"POST", body: JSON.stringify({
        name: f.name.value,
        email: f.email.value,
        password: f.password.value
      })});
      await loadMe();
      await loadMenu();
      await loadMyOrders();
    }catch(e){
      setAuthMessage(e.message);
    }
  });

  $("#btnLogout").addEventListener("click", async () => {
    await api("/api/logout", { method:"POST" });
    me = null;
    cart.clear();
    setAuthUI();
  });

  $("#btnGoMenu").addEventListener("click", async () => {
    if(!me) return;
    show("menuCard");
    await loadMenu();
  });

  $("#btnGoMyOrders").addEventListener("click", async () => {
    if(!me) return;
    show("myOrdersCard");
    await loadMyOrders();
  });

  $("#btnGoManager").addEventListener("click", async () => {
    if(!me || me.role !== "manager") return;
    show("managerCard");
    await managerLoadMenu();
    await managerLoadOrders();
  });

  $("#btnRefreshMenu").addEventListener("click", loadMenu);
  $("#btnPlaceOrder").addEventListener("click", async () => {
    try{ await placeOrder(); }
    catch(e){ setOrderMessage(e.message); }
  });

  $("#btnRefreshMyOrders").addEventListener("click", loadMyOrders);

  // Gerente
  $("#menuItemForm").addEventListener("submit", async (ev) => {
    try{ await managerSaveMenuItem(ev); }
    catch(e){ setMenuItemMessage(e.message); }
  });
  $("#btnClearForm").addEventListener("click", () => {
    const f = $("#menuItemForm");
    f.reset();
    f.id.value = "";
    f.is_active.checked = true;
    setMenuItemMessage("");
  });
  $("#btnReloadManagerMenu").addEventListener("click", managerLoadMenu);
  $("#btnReloadAllOrders").addEventListener("click", managerLoadOrders);
}
