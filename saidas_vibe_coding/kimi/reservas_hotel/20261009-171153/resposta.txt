const Database = require('better-sqlite3');
const path = require('path');
const bcrypt = require('bcrypt');

const dbPath = path.join(__dirname, 'data', 'hotel.db');
const db = new Database(dbPath);

// Criar diretório data se não existir
const fs = require('fs');
const dataDir = path.join(__dirname, 'data');
if (!fs.existsSync(dataDir)) {
  fs.mkdirSync(dataDir, { recursive: true });
}

function initializeDatabase() {
  // Tabela de usuários
  db.exec(`
    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      email TEXT UNIQUE NOT NULL,
      password TEXT NOT NULL,
      name TEXT NOT NULL,
      role TEXT NOT NULL CHECK(role IN ('guest', 'manager')),
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )
  `);

  // Tabela de quartos
  db.exec(`
    CREATE TABLE IF NOT EXISTS rooms (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      description TEXT,
      price_per_night DECIMAL(10,2) NOT NULL,
      capacity INTEGER NOT NULL,
      amenities TEXT,
      image_url TEXT,
      active INTEGER DEFAULT 1,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )
  `);

  // Tabela de reservas
  db.exec(`
    CREATE TABLE IF NOT EXISTS reservations (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      user_id INTEGER NOT NULL,
      room_id INTEGER NOT NULL,
      check_in DATE NOT NULL,
      check_out DATE NOT NULL,
      total_price DECIMAL(10,2) NOT NULL,
      status TEXT NOT NULL DEFAULT 'pending' CHECK(status IN ('pending', 'confirmed', 'cancelled', 'rejected')),
      notes TEXT,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (user_id) REFERENCES users(id),
      FOREIGN KEY (room_id) REFERENCES rooms(id)
    )
  `);

  // Índices para performance
  db.exec(`
    CREATE INDEX IF NOT EXISTS idx_reservations_dates ON reservations(check_in, check_out);
    CREATE INDEX IF NOT EXISTS idx_reservations_user ON reservations(user_id);
    CREATE INDEX IF NOT EXISTS idx_reservations_status ON reservations(status);
  `);

  console.log('Banco de dados inicializado com sucesso');
}

module.exports = { db, initializeDatabase };

=== middleware/auth.js ===
const jwt = require('jsonwebtoken');
const { db } = require('../database');

const JWT_SECRET = process.env.JWT_SECRET || 'seu-segredo-super-secreto-mude-em-producao';

function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];

  if (!token) {
    return res.status(401).json({ error: 'Acesso negado. Token não fornecido.' });
  }

  jwt.verify(token, JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ error: 'Token inválido ou expirado' });
    }
    req.user = user;
    next();
  });
}

function requireManager(req, res, next) {
  if (req.user.role !== 'manager') {
    return res.status(403).json({ error: 'Acesso restrito ao gerente' });
  }
  next();
}

function requireGuest(req, res, next) {
  if (req.user.role !== 'guest') {
    return res.status(403).json({ error: 'Acesso restrito a hóspedes' });
  }
  next();
}

module.exports = { authenticateToken, requireManager, requireGuest, JWT_SECRET };

const express = require('express');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
const { body, validationResult } = require('express-validator');
const { db } = require('../database');
const { JWT_SECRET } = require('../middleware/auth');

const router = express.Router();

// Registro de hóspede
router.post('/register', [
  body('email').isEmail().normalizeEmail(),
  body('password').isLength({ min: 6 }),
  body('name').trim().isLength({ min: 2 })
], async (req, res) => {
  try {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }

    const { email, password, name } = req.body;

    // Verificar se usuário já existe
    const existingUser = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
    if (existingUser) {
      return res.status(400).json({ error: 'Email já cadastrado' });
    }

    // Hash da senha
    const hashedPassword = await bcrypt.hash(password, 10);

    // Inserir usuário
    const stmt = db.prepare('INSERT INTO users (email, password, name, role) VALUES (?, ?, ?, ?)');
    const result = stmt.run(email, hashedPassword, name, 'guest');

    // Gerar token
    const token = jwt.sign(
      { id: result.lastInsertRowid, email, role: 'guest' },
      JWT_SECRET,
      { expiresIn: '7d' }
    );

    res.status(201).json({
      message: 'Cadastro realizado com sucesso',
      token,
      user: { id: result.lastInsertRowid, email, name, role: 'guest' }
    });
  } catch (error) {
    console.error(error);
    res.status(500).json({ error: 'Erro ao cadastrar usuário' });
  }
});

// Login
router.post('/login', [
  body('email').isEmail().normalizeEmail(),
  body('password').notEmpty()
], async (req, res) => {
  try {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }

    const { email, password } = req.body;

    // Buscar usuário
    const user = db.prepare('SELECT * FROM users WHERE email = ?').get(email);
    if (!user) {
      return res.status(401).json({ error: 'Email ou senha incorretos' });
    }

    // Verificar senha
    const validPassword = await bcrypt.compare(password, user.password);
    if (!validPassword) {
      return res.status(401).json({ error: 'Email ou senha incorretos' });
    }

    // Gerar token
    const token = jwt.sign(
      { id: user.id, email: user.email, role: user.role },
      JWT_SECRET,
      { expiresIn: '7d' }
    );

    res.json({
      message: 'Login realizado com sucesso',
      token,
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        role: user.role
      }
    });
  } catch (error) {
    console.error(error);
    res.status(500).json({ error: 'Erro ao fazer login' });
  }
});

    let params = [];

    // Se datas fornecidas, verificar disponibilidade
    if (checkIn && checkOut) {
      query = `
        SELECT r.* FROM rooms r
        WHERE r.active = 1
        AND r.id NOT IN (
          SELECT room_id FROM reservations
          WHERE status IN ('pending', 'confirmed')
          AND (
            (check_in <= ? AND check_out > ?)
            OR (check_in < ? AND check_out >= ?)
            OR (check_in >= ? AND check_out <= ?)
          )
        )
      `;
      params = [checkIn, checkIn, checkOut, checkOut, checkIn, checkOut];
    }

    const rooms = db.prepare(query).all(...params);
    res.json(rooms);

const express = require('express');
const { body, validationResult } = require('express-validator');
const { db } = require('../database');
const { authenticateToken, requireManager, requireGuest } = require('../middleware/auth');

const router = express.Router();

// Criar reserva (hóspede)
router.post('/', authenticateToken, requireGuest, [
  body('room_id').isInt(),
  body('check_in').isDate(),
  body('check_out').isDate(),
  body('notes').optional().trim()
], (req, res) => {
  try {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }

    const { room_id, check_in, check_out, notes } = req.body;
    const user_id = req.user.id;

    // Validar datas
    if (new Date(check_in) >= new Date(check_out)) {
      return res.status(400).json({ error: 'Data de check-out deve ser posterior ao check-in' });
    }

    if (new Date(check_in) < new Date()) {
      return res.status(400).json({ error: 'Data de check-in não pode ser no passado' });
    }

    // Verificar disponibilidade
    const conflicting = db.prepare(`
      SELECT COUNT(*) as count FROM reservations
      WHERE room_id = ? AND status IN ('pending', 'confirmed')
      AND (
        (check_in <= ? AND check_out > ?)
        OR (check_in < ? AND check_out >= ?)
        OR (check_in >= ? AND check_out <= ?)
      )
    `).get(room_id, check_in, check_in, check_out, check_out, check_in, check_out);

    if (conflicting.count > 0) {
      return res.status(400).json({ error: 'Quarto não disponível nas datas selecionadas' });
    }

    // Calcular preço total
    const room = db.prepare('SELECT price_per_night FROM rooms WHERE id = ?').get(room_id);
    if (!room) {
      return res.status(404).json({ error: 'Quarto não encontrado' });
    }

    const nights = Math.ceil((new Date(check_out) - new Date(check_in)) / (1000 * 60 * 60 * 24));
    const total_price = room.price_per_night * nights;

    // Criar reserva
    const stmt = db.prepare(`
      INSERT INTO reservations (user_id, room_id, check_in, check_out, total_price, notes)
      VALUES (?, ?, ?, ?, ?, ?)
    `);
    
    const result = stmt.run(user_id, room_id, check_in, check_out, total_price, notes);

    res.status(201).json({
      message: 'Reserva criada com sucesso. Aguardando confirmação do gerente.',
      reservationId: result.lastInsertRowid,
      total_price
    });
  } catch (error) {
    console.error(error);
    res.status(500).json({ error: 'Erro ao criar reserva' });
  }
});

// Listar reservas do usuário (hóspede)

// Estado global da aplicação
const App = {
    currentUser: null,
    token: localStorage.getItem('token'),
    apiUrl: '/api',

    init() {
        if (this.token) {
            this.validateToken();
        } else {
            this.showLogin();
        }
        this.setupNavigation();
    },

    async validateToken() {
        try {
            const response = await this.apiCall('/auth/validate', 'GET');
            if (response.user) {
                this.currentUser = response.user;
                this.showDashboard();
            } else {
                this.logout();
            }
        } catch (error) {
            this.logout();
        }
    },

    setupNavigation() {
        document.getElementById('nav-menu').addEventListener('click', (e) => {
            if (e.target.dataset.action) {
                e.preventDefault();
                this.handleNavigation(e.target.dataset.action);
            }
        });
    },

    handleNavigation(action) {
        switch(action) {
            case 'logout':
                this.logout();
                break;
            case 'rooms':
                this.showRooms();
                break;
            case 'my-reservations':
                this.showMyReservations();
                break;
            case 'all-reservations':
                this.showAllReservations();
                break;
            case 'manage-rooms':
                this.showManageRooms();
                break;
        }
    },

    renderNavbar() {
        const nav = document.getElementById('nav-menu');
        if (!this.currentUser) {
            nav.innerHTML = '';
            return;
        }

        let html = `<span class="user-info">Olá, ${this.currentUser.name}</span>`;
        
        if (this.currentUser.role === 'guest') {
            html += `
                <button data-action="rooms">Quartos</button>
                <button data-action="my-reservations">Minhas Reservas</button>
            `;
        } else if (this.currentUser.role === 'manager') {
            html += `
                <button data-action="manage-rooms">Gerenciar Quartos</button>
                <button data-action="all-reservations">Todas Reservas</button>
            `;
        }
        
        html += `<button data-action="logout">Sair</button>`;
        nav.innerHTML = html;
    },

    async apiCall(endpoint, method = 'GET', data = null) {
        const options = {
            method,
            headers: {
                'Content-Type': 'application/json',
            }
        };

        if (this.token) {
            options.headers['Authorization'] = `Bearer ${this.token}`;
        }

        if (data) {
            options.body = JSON.stringify(data);
        }

        const response = await fetch(this.apiUrl + endpoint, options);
        return await response.json();
    },

    showLogin() {
        document.getElementById('main-content').innerHTML = `
            <div class="form-container">
                <h2 class="text-center mb-20">Login</h2>
                <div id="alert-container"></div>
                <form id="login-form">
                    <div class="form-group">
                        <label>Email:</label>
                        <input type="email" id="email" required>
                    </div>
                    <div class="form-group">
                        <label>Senha:</label>
                        <input type="password" id="password" required>
                    </div>
                    <button type="submit" class="btn btn-primary">Entrar</button>
                </form>
                <p class="text-center mt-20">
                    Não tem conta? <a href="#" onclick="App.showRegister()">Cadastre-se</a>
                </p>
            </div>
        `;

        document.getElementById('login-form').addEventListener('submit', async (e) => {
            e.preventDefault();
            const email = document.getElementById('email').value;
            const password = document.getElementById('password').value;

            try {
                const response = await this.apiCall('/auth/login', 'POST', { email, password });
                
                if (response.token) {
                    this.token = response.token;
                    this.currentUser = response.user;
                    localStorage.setItem('token', this.token);
                    this.showDashboard();
                } else {
                    this.showAlert(response.error || 'Erro ao fazer login', 'error');
                }
            } catch (error) {
                this.showAlert('Erro ao conectar com o servidor', 'error');
            }
        });
    },

    showRegister() {

                </p>
            </div>

        document.getElementById('register-form').addEventListener('submit', async (e) => {
            e.preventDefault();
            const name = document.getElementById('name').value;
            const email = document.getElementById('email').value;
            const password = document.getElementById('password').value;

            try {

        const today = new Date().toISOString().split('T')[0];

    async loadRooms(checkIn = null, checkOut = null) {
        try {
            let url = '/rooms';
            if (checkIn && checkOut) {
                url += `?checkIn=${checkIn}&checkOut=${checkOut}`;
            }

            const rooms = await this.apiCall(url);
            const container = document.getElementById('rooms-container');

            if (rooms.length === 0) {
                container.innerHTML = '<p class="alert alert-error">Nenhum quarto disponível nas datas selecionadas.</p>';
                return;
            }

            container.innerHTML = `
                <div class="rooms-grid">
                    ${rooms.map(room => `
                        <div class="room-card">
                            <img src="${room.image_url || 'https://images.unsplash.com/photo-1611892440504-42a792e24d32?w=400'}" 
                                 alt="${room.name}" class="room-image">
                            <div class="room-info">
                                <h3 class="room-name">${room.name}</h3>
                                <p>${room.description || 'Quarto confortável e aconchegante.'}</p>
                                <p><strong>Capacidade:</strong> ${room.capacity} pessoas</p>
                                ${room.amenities ? `<p><strong>Comodidades:</strong> ${room.amenities}</p>` : ''}
                                <div class="room-price">R$ ${parseFloat(room.price_per_night).toFixed(2)} / noite</div>
                                ${checkIn && checkOut ? `
                                    <button class="btn btn-primary" onclick="App.showReservationForm(${room.id}, '${checkIn}', '${checkOut}', ${room.price_per_night})">
                                        Reservar
                                    </button>
                                ` : ''}
                            </div>
                        </div>
                    `).join('')}
                </div>
            `;
        } catch (error) {
            this.showAlert('Erro ao carregar quartos', 'error');
        }
    },

    showReservationForm(roomId, checkIn, checkOut, pricePerNight) {
        const nights = Math.ceil((new Date(checkOut) - new Date(checkIn)) / (1000 * 60 * 60 * 24));
        const totalPrice = nights * pricePerNight;

        const modal = document.createElement('div');
        modal.className = 'modal';
        modal.style.display = 'block';
        modal.innerHTML = `
            <div class="modal-content">
                <div class="modal-header">
                    <h3>Confirmar Reserva</h3>
                    <span class="close" onclick="this.closest('.modal').remove()">&times;</span>
                </div>
                <div id="modal-alert"></div>
                <form id="reservation-form">
                    <p><strong>Check-in:</strong> ${new Date(checkIn).toLocaleDateString('pt-BR')}</p>
                    <p><strong>Check-out:</strong> ${new Date(checkOut).toLocaleDateString('pt-BR')}</p>
                    <p><strong>Número de noites:</strong> ${nights}</p>
                    <p><strong>Valor total:</strong> R$ ${totalPrice.toFixed(2)}</p>
                    <div class="form-group">
                        <label>Observações (opcional):</label>
                        <textarea id="notes" placeholder="Alguma solicitação especial?"></textarea>
                    </div>
                    <button type="submit" class="btn btn-primary">Confirmar Reserva</button>
                </form>
            </div>
        `;

        document.body.appendChild(modal);

        document.getElementById('reservation-form').addEventListener('submit', async (e) => {
            e.preventDefault();
            const notes = document.getElementById('notes').value;

            try {
                const response = await this.apiCall('/reservations', 'POST', {
                    room_id: roomId,
                    check_in: checkIn,
                    check_out: checkOut,
                    notes
                });

                if (response.reservationId) {
                    modal.remove();
                    this.showAlert('Reserva criada com sucesso! Aguardando confirmação do gerente.', 'success');
                    this.showRooms();
                } else {
                    document.getElementById('modal-alert').innerHTML = 
                        `<div class="alert alert-error">${response.error || 'Erro ao criar reserva'}</div>`;
                }
            } catch (error) {
                document.getElementById('modal-alert').innerHTML = 
                    '<div class="alert alert-error">Erro ao conectar com o servidor</div>';
            }
        });
    },

            e.preventDefault();

            const data = {
                name: document.getElementById('room-name').value,
                description: document.getElementById('room-description').value,
                price_per_night: parseFloat(document.getElementById('room-price').value),
                capacity: parseInt(document.getElementById('room-capacity').value),
                amenities: document.getElementById('room-amenities').value,
                image_url: document.getElementById('room-image').value
            };

            try {
                const url = roomId ? `/rooms/${roomId}` : '/rooms';
                const method = roomId ? 'PUT' : 'POST';
                const response = await this.apiCall(url, method, data);

                if (response.message) {
                    modal.remove();
                    this.showAlert(response.message, 'success');
                    this.loadManageRooms();
                } else {
                    document.getElementById('modal-alert').innerHTML = 
                        `<div class="alert alert-error">${response.error || 'Erro ao salvar quarto'}</div>`;
                }
            } catch (error) {
                document.getElementById('modal-alert').innerHTML = 
                    '<div class="alert alert-error">Erro ao conectar com o servidor</div>';
            }
        });
    },

    async deleteRoom(id) {
        if (!confirm('Tem certeza que deseja remover este quarto?')) return;

        try {
            const response = await this.apiCall(`/rooms/${id}`, 'DELETE');
            this.showAlert(response.message || 'Quarto removido', 'success');
            this.loadManageRooms();
        } catch (error) {
            this.showAlert('Erro ao remover quarto', 'error');
        }
    },

    getStatusText(status) {
        const texts = {
            'pending': 'Pendente',
            'confirmed': 'Confirmada',
            'cancelled': 'Cancelada',
            'rejected': 'Recusada'
        };
        return texts[status] || status;
    },

    showAlert(message, type) {
        const container = document.getElementById('alert-container');
        container.innerHTML = `<div class="alert alert-${type}">${message}</div>`;
        setTimeout(() => container.innerHTML = '', 5000);
    },

    logout() {
        localStorage.removeItem('token');
        this.token = null;
        this.currentUser = null;
        this.showLogin();
    }

# Sistema de Reservas de Hotel

Sistema completo de reservas de hotel com área para hóspedes e gerente. Todos os dados ficam salvos em um banco de dados no servidor, permitindo acesso de qualquer dispositivo.

## Funcionalidades

### Para Hóspedes:
- Criar conta e fazer login
- Buscar quartos disponíveis por data
- Fazer reservas
- Ver suas reservas
- Cancelar reservas

### Para Gerente:
- Cadastrar, editar e remover quartos
- Ver todas as reservas
- Confirmar ou recusar reservas pendentes
- Gerenciar disponibilidade

## Instalação

### Requisitos:
- Node.js 18 ou superior ([baixe aqui](https://nodejs.org))

### Passo a passo:

1. **Instalar as dependências:**
```bash
npm install
```

2. **Criar usuário gerente:**
```bash
npm run init-admin
```
Isso criará um usuário gerente com:
- Email: `gerente@hotel.com`
- Senha: `admin123`

⚠️ **IMPORTANTE:** Anote essas credenciais e altere a senha após o primeiro acesso!

3. **Iniciar o servidor:**
```bash
npm start
```

4. **Acessar o site:**
Abra seu navegador e acesse: `http://localhost:3000`

## Uso

### Primeiro Acesso - Gerente:
1. Acesse `http://localhost:3000`
2. Faça login com:
   - Email: `gerente@hotel.com`
   - Senha: `admin123`
3. Cadastre os quartos do hotel
4. Gerencie as reservas

### Hóspedes:
1. Acesse `http://localhost:3000`
2. Clique em "Cadastre-se"
3. Crie sua conta
4. Busque quartos e faça reservas

Antes de colocar em produção:
