# Portal de Inscrição para Eventos

Sistema completo para gerenciamento de eventos com inscrições de participantes.

## Instalação

1. Certifique-se de ter o Node.js instalado (versão 14 ou superior)
   - Baixe em: https://nodejs.org/

2. Abra o terminal/prompt de comando na pasta do projeto

3. Instale as dependências:
```bash
npm install
```

## Executar o projeto

1. Inicie o servidor:
```bash
npm start
```

2. Abra seu navegador e acesse:
```
http://localhost:3000
```

## Contas de teste

### Organizador:
- Email: organizador@eventos.com
- Senha: organizador123

### Participante:
- Email: participante@eventos.com
- Senha: participante123

Você também pode criar novas contas pelo próprio site.

## Como usar

### Como Participante:
1. Faça login ou crie uma conta
2. Veja os eventos disponíveis na página inicial
3. Clique em "Inscrever-se" para participar de um evento
4. Acesse "Minhas Inscrições" para ver ou cancelar suas inscrições

### Como Organizador:
1. Faça login com uma conta de organizador
2. Use "Criar Evento" para adicionar novos eventos
3. Gerencie seus eventos (editar, cancelar, confirmar presença)
4. Veja todas as inscrições em "Gerenciar Inscrições"

## Observações

- Todos os dados ficam salvos no servidor (arquivo database.json)
- O site funciona em qualquer dispositivo (computador, tablet, celular)
- Os dados permanecem salvos mesmo após fechar o navegador

const express = require('express');
const session = require('express-session');
const bcrypt = require('bcrypt');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 3000;
const DB_FILE = path.join(__dirname, 'database.json');

// Middleware
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static('public'));
app.use(session({
  secret: 'evento-secreto-2024-portal-inscricoes',
  resave: false,
  saveUninitialized: false,
  cookie: { maxAge: 24 * 60 * 60 * 1000 } // 24 horas
}));

// Funções de banco de dados
function loadDatabase() {
  try {
    if (fs.existsSync(DB_FILE)) {
      const data = fs.readFileSync(DB_FILE, 'utf8');
      return JSON.parse(data);
    }
  } catch (error) {
    console.error('Erro ao carregar banco de dados:', error);
  }
  
  // Banco de dados inicial
  return {
    users: [],
    events: [],
    registrations: []
  };
}

function saveDatabase(db) {
  try {
    fs.writeFileSync(DB_FILE, JSON.stringify(db, null, 2));
    return true;
  } catch (error) {
    console.error('Erro ao salvar banco de dados:', error);
    return false;
  }
}

// Inicializar banco com usuários padrão
function initializeDatabase() {
  let db = loadDatabase();
  
  if (db.users.length === 0) {
    const organizadorHash = bcrypt.hashSync('organizador123', 10);
    const participanteHash = bcrypt.hashSync('participante123', 10);
    
    db.users.push({
      id: 1,
      name: 'Organizador Sistema',
      email: 'organizador@eventos.com',
      password: organizadorHash,
      role: 'organizador',
      createdAt: new Date().toISOString()
    });
    
    db.users.push({
      id: 2,
      name: 'Participante Teste',
      email: 'participante@eventos.com',
      password: participanteHash,
      role: 'participante',
      createdAt: new Date().toISOString()
    });
    
    saveDatabase(db);
  }
  
  return db;
}

// Middleware de autenticação
function isAuthenticated(req, res, next) {
  if (req.session.userId) {
    next();
  } else {
    res.status(401).json({ error: 'Não autorizado' });
  }
}

function isOrganizador(req, res, next) {
  if (req.session.userId && req.session.role === 'organizador') {
    next();
  } else {
    res.status(403).json({ error: 'Acesso negado. Apenas organizadores.' });
  }
}

// Rotas de autenticação
app.post('/api/register', async (req, res) => {
  const { name, email, password, role } = req.body;
  
  if (!name || !email || !password || !role) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios' });
  }
  
  if (role !== 'participante' && role !== 'organizador') {
    return res.status(400).json({ error: 'Tipo de usuário inválido' });
  }
  
  const db = loadDatabase();
  
  // Verificar se email já existe
  if (db.users.find(u => u.email === email)) {
    return res.status(400).json({ error: 'Email já cadastrado' });
  }
  
  // Criar novo usuário
  const hashedPassword = await bcrypt.hash(password, 10);
  const newUser = {
    id: db.users.length > 0 ? Math.max(...db.users.map(u => u.id)) + 1 : 1,
    name,
    email,
    password: hashedPassword,
    role,
    createdAt: new Date().toISOString()
  };
  
  db.users.push(newUser);
  saveDatabase(db);
  
  res.json({ message: 'Usuário criado com sucesso', userId: newUser.id });
});

app.post('/api/login', async (req, res) => {
  const { email, password } = req.body;
  
  if (!email || !password) {
    return res.status(400).json({ error: 'Email e senha são obrigatórios' });
  }
  
  const db = loadDatabase();
  const user = db.users.find(u => u.email === email);
  
  if (!user) {
    return res.status(401).json({ error: 'Email ou senha incorretos' });
  }
  
  const validPassword = await bcrypt.compare(password, user.password);
  
  if (!validPassword) {
    return res.status(401).json({ error: 'Email ou senha incorretos' });
  }
  
  req.session.userId = user.id;
  req.session.role = user.role;
  req.session.name = user.name;
  
  res.json({
    message: 'Login realizado com sucesso',
    user: {
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role
    }
  });
});

app.post('/api/logout', (req, res) => {
  req.session.destroy();
  res.json({ message: 'Logout realizado com sucesso' });
});

app.get('/api/me', isAuthenticated, (req, res) => {
  const db = loadDatabase();
  const user = db.users.find(u => u.id === req.session.userId);
  
  if (!user) {
    return res.status(404).json({ error: 'Usuário não encontrado' });
  }
  
  res.json({
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role
  });
});

  const event = db.events.find(e => e.id === parseInt(req.params.id));
  
  if (!event) {
    return res.status(404).json({ error: 'Evento não encontrado' });

  res.json(event);
});

app.post('/api/events', isOrganizador, (req, res) => {
  const { title, description, date, location, maxParticipants } = req.body;
  
  if (!title || !description || !date || !location || !maxParticipants) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios' });
  }
  
  const db = loadDatabase();
  
  const newEvent = {
    id: db.events.length > 0 ? Math.max(...db.events.map(e => e.id)) + 1 : 1,
    title,
    description,
    date,
    location,
    maxParticipants: parseInt(maxParticipants),
    organizerId: req.session.userId,
    status: 'ativo',
    createdAt: new Date().toISOString()
  };
  
  db.events.push(newEvent);
  saveDatabase(db);
  
  res.json({ message: 'Evento criado com sucesso', event: newEvent });
});

app.put('/api/events/:id', isOrganizador, (req, res) => {
  const { title, description, date, location, maxParticipants } = req.body;
  const db = loadDatabase();
  
  const eventIndex = db.events.findIndex(e => e.id === parseInt(req.params.id));
  
  if (eventIndex === -1) {
    return res.status(404).json({ error: 'Evento não encontrado' });
  }
  
  const event = db.events[eventIndex];
  
  if (event.organizerId !== req.session.userId) {
    return res.status(403).json({ error: 'Você não pode editar este evento' });
  }
  
  db.events[eventIndex] = {
    ...event,
    title: title || event.title,
    description: description || event.description,
    date: date || event.date,
    location: location || event.location,
    maxParticipants: maxParticipants ? parseInt(maxParticipants) : event.maxParticipants,
    updatedAt: new Date().toISOString()
  };
  
  saveDatabase(db);
  
  res.json({ message: 'Evento atualizado com sucesso', event: db.events[eventIndex] });
});

app.delete('/api/events/:id', isOrganizador, (req, res) => {
  const db = loadDatabase();
  const eventIndex = db.events.findIndex(e => e.id === parseInt(req.params.id));
  
  if (eventIndex === -1) {
    return res.status(404).json({ error: 'Evento não encontrado' });
  }
  
  const event = db.events[eventIndex];
  
  if (event.organizerId !== req.session.userId) {
    return res.status(403).json({ error: 'Você não pode cancelar este evento' });
  }
  
  db.events[eventIndex].status = 'cancelado';
  saveDatabase(db);
  
  res.json({ message: 'Evento cancelado com sucesso' });
});

// Rotas de inscrições
app.post('/api/registrations', isAuthenticated, (req, res) => {
  const { eventId } = req.body;
  
  if (!eventId) {
    return res.status(400).json({ error: 'ID do evento é obrigatório' });
  }
  
  const db = loadDatabase();
  const event = db.events.find(e => e.id === parseInt(eventId));
  
  if (!event) {
    return res.status(404).json({ error: 'Evento não encontrado' });
  }
  
  if (event.status !== 'ativo') {
    return res.status(400).json({ error: 'Este evento não está mais disponível' });
  }
  
  // Verificar se já está inscrito
  const existingRegistration = db.registrations.find(
    r => r.eventId === parseInt(eventId) && r.userId === req.session.userId && r.status !== 'cancelado'
  );
  
  if (existingRegistration) {
    return res.status(400).json({ error: 'Você já está inscrito neste evento' });
  }
  
  // Verificar vagas disponíveis
  const activeRegistrations = db.registrations.filter(
    r => r.eventId === parseInt(eventId) && r.status !== 'cancelado'
  );
  
  if (activeRegistrations.length >= event.maxParticipants) {
    return res.status(400).json({ error: 'Evento lotado' });
  }
  
  const newRegistration = {
    id: db.registrations.length > 0 ? Math.max(...db.registrations.map(r => r.id)) + 1 : 1,
    eventId: parseInt(eventId),
    userId: req.session.userId,
    status: 'inscrito',
    presence: false,
    createdAt: new Date().toISOString()
  };
  
  db.registrations.push(newRegistration);
  saveDatabase(db);
  
  res.json({ message: 'Inscrição realizada com sucesso', registration: newRegistration });
});

app.get('/api/registrations/my', isAuthenticated, (req, res) => {
  const db = loadDatabase();
  
  const userRegistrations = db.registrations
    .filter(r => r.userId === req.session.userId && r.status !== 'cancelado')
    .map(r => {
      const event = db.events.find(e => e.id === r.eventId);
      return {
        ...r,
        event
      };
    });
  
  res.json(userRegistrations);
});

app.get('/api/registrations/all', isOrganizador, (req, res) => {
  const db = loadDatabase();
  
  const allRegistrations = db.registrations
    .filter(r => r.status !== 'cancelado')
    .map(r => {
      const event = db.events.find(e => e.id === r.eventId);
      const user = db.users.find(u => u.id === r.userId);
      return {
        ...r,
        event,
        user: {
          id: user.id,
          name: user.name,
          email: user.email
        }
      };
    });
  
  res.json(allRegistrations);
});

app.delete('/api/registrations/:id', isAuthenticated, (req, res) => {
  const db = loadDatabase();
  const registrationIndex = db.registrations.findIndex(r => r.id === parseInt(req.params.id));
  
  if (registrationIndex === -1) {
    return res.status(404).json({ error: 'Inscrição não encontrada' });
  }
  
  const registration = db.registrations[registrationIndex];
  
  if (registration.userId !== req.session.userId) {
    return res.status(403).json({ error: 'Você não pode cancelar esta inscrição' });
  }
  
  db.registrations[registrationIndex].status = 'cancelado';
  saveDatabase(db);
  
  res.json({ message: 'Inscrição cancelada com sucesso' });
});

app.put('/api/registrations/:id/presence', isOrganizador, (req, res) => {
  const { presence } = req.body;
  const db = loadDatabase();
  
  const registrationIndex = db.registrations.findIndex(r => r.id === parseInt(req.params.id));
  
  if (registrationIndex === -1) {
    return res.status(404).json({ error: 'Inscrição não encontrada' });
  }
  
  const registration = db.registrations[registrationIndex];
  const event = db.events.find(e => e.id === registration.eventId);
  
  if (!event || event.organizerId !== req.session.userId) {
    return res.status(403).json({ error: 'Você não pode confirmar presença neste evento' });
  }
  
  db.registrations[registrationIndex].presence = presence === true;
  saveDatabase(db);
  
  res.json({ message: 'Presença atualizada com sucesso', registration: db.registrations[registrationIndex] });
});

// Estado da aplicação
let currentUser = null;
let currentView = 'events';
let allEvents = [];
let myRegistrations = [];
let allRegistrations = [];

// Inicialização
document.addEventListener('DOMContentLoaded', () => {
  checkAuth();
  setupEventListeners();
});

// Verificar autenticação
async function checkAuth() {
  try {
    const response = await fetch('/api/me');
    if (response.ok) {
      currentUser = await response.json();
      showMainScreen();
    } else {
      showAuthScreen();
    }
  } catch (error) {
    showAuthScreen();
  }
}

// Mostrar telas
function showAuthScreen() {
  document.getElementById('auth-screen').classList.add('active');
  document.getElementById('main-screen').classList.remove('active');
}

function showMainScreen() {
  document.getElementById('auth-screen').classList.remove('active');
  document.getElementById('main-screen').classList.add('active');
  
  document.getElementById('user-name').textContent = currentUser.name;
  
  if (currentUser.role === 'organizador') {
    document.body.classList.add('organizador');
  } else {
    document.body.classList.remove('organizador');
  }
  
  loadCurrentView();
}

// Event Listeners
function setupEventListeners() {
  // Tabs de autenticação
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const tab = btn.dataset.tab;
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      document.querySelectorAll('.auth-form').forEach(f => f.classList.remove('active'));
      btn.classList.add('active');
      document.getElementById(`${tab}-form`).classList.add('active');
    });
  });
  
  // Formulário de login
  document.getElementById('login-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    const email = document.getElementById('login-email').value;
    const password = document.getElementById('login-password').value;
    
    try {
      const response = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password })
      });
      
      const data = await response.json();
      
      if (response.ok) {
        currentUser = data.user;
        showToast('Login realizado com sucesso!', 'success');
        showMainScreen();
      } else {
        showToast(data.error || 'Erro ao fazer login', 'error');
      }
    } catch (error) {
      showToast('Erro ao fazer login', 'error');
    }
  });
  
  // Formulário de registro
  document.getElementById('register-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    const name = document.getElementById('register-name').value;
    const email = document.getElementById('register-email').value;
    const password = document.getElementById('register-password').value;
    const role = document.getElementById('register-role').value;
    
    try {
      const response = await fetch('/api/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name, email, password, role })
      });
      
      const data = await response.json();
      
      if (response.ok) {
        showToast('Conta criada com sucesso! Faça login.', 'success');
        document.querySelector('[data-tab="login"]').click();
        document.getElementById('login-email').value = email;
      } else {
        showToast(data.error || 'Erro ao criar conta', 'error');
      }
    } catch (error) {
      showToast('Erro ao criar conta', 'error');
    }
  });
  
  // Logout
  document.getElementById('logout-btn').addEventListener('click', async () => {
    try {
      await fetch('/api/logout', { method: 'POST' });
      currentUser = null;
      showToast('Logout realizado com sucesso!', 'success');
      showAuthScreen();
    } catch (error) {
      showToast('Erro ao fazer logout', 'error');
    }
  });
  
  // Menu lateral
  document.querySelectorAll('.menu-item').forEach(item => {
    item.addEventListener('click', () => {
      const view = item.dataset.view;
      document.querySelectorAll('.menu-item').forEach(m => m.classList.remove('active'));
      item.classList.add('active');
      currentView = view;
      loadCurrentView();
    });
  });
  
  // Formulário de criar evento
  document.getElementById('create-event-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    
    const eventData = {
      title: document.getElementById('event-title').value,
      description: document.getElementById('event-description').value,
      date: document.getElementById('event-date').value,
      location: document.getElementById('event-location').value,
      maxParticipants: document.getElementById('event-max').value
    };
    
    try {
      const response = await fetch('/api/events', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(eventData)
      });
      
      const data = await response.json();
      
      if (response.ok) {
        showToast('Evento criado com sucesso!', 'success');
        document.getElementById('create-event-form').reset();
        currentView = 'manage-events';
        document.querySelector('[data-view="manage-events"]').click();
      } else {
        showToast(data.error || 'Erro ao criar evento', 'error');
      }
    } catch (error) {
      showToast('Erro ao criar evento', 'error');
    }
  });
  
  // Formulário de editar evento
  document.getElementById('edit-event-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    
    const eventId = document.getElementById('edit-event-id').value;
    const eventData = {
      title: document.getElementById('edit-event-title').value,
      description: document.getElementById('edit-event-description').value,
      date: document.getElementById('edit-event-date').value,
      location: document.getElementById('edit-event-location').value,
      maxParticipants: document.getElementById('edit-event-max').value
    };
    
    try {
      const response = await fetch(`/api/events/${eventId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(eventData)
      });
      
      const data = await response.json();
      
      if (response.ok) {
        showToast('Evento atualizado com sucesso!', 'success');
        closeModal();
        loadCurrentView();
      } else {
        showToast(data.error || 'Erro ao atualizar evento', 'error');
      }
    } catch (error) {
      showToast('Erro ao atualizar evento', 'error');
    }
  });
  
  // Fechar modal
  document.querySelectorAll('.modal-close').forEach(btn => {
    btn.addEventListener('click', closeModal);
  });
  
  document.getElementById('edit-modal').addEventListener('click', (e) => {
    if (e.target.id === 'edit-modal') {
      closeModal();
    }
  });
}

// Carregar view atual
async function loadCurrentView() {
  document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
  document.getElementById(`${currentView}-view`).classList.add('active');
  
  switch (currentView) {
    case 'events':
      await loadEvents();
      break;
    case 'my-registrations':
      await loadMyRegistrations();
      break;
    case 'manage-events':
      await loadManageEvents();
      break;
    case 'all-registrations':
      await loadAllRegistrations();
      break;
  }
}

// Carregar eventos disponíveis
async function loadEvents() {
  try {
    const response = await fetch('/api/events');
    allEvents = await response.json();
    
    const container = document.getElementById('events-list');
    
    if (allEvents.length === 0) {
      container.innerHTML = `
        <div class="empty-state">
          <div class="empty-state-icon">📅</div>
          <h3>Nenhum evento disponível</h3>
          <p>Não há eventos cadastrados no momento.</p>
        </div>
      `;
      return;
    }
    
    // Carregar inscrições para verificar se já está inscrito
    const regResponse = await fetch('/api/registrations/my');
    const myRegs = await regResponse.json();
    const registeredEventIds = myRegs.map(r => r.eventId);
    
    container.innerHTML = allEvents.map(event => {
      const eventDate = new Date(event.date);
      const isRegistered = registeredEventIds.includes(event.id);
      
      // Contar inscrições ativas
      return `
        <div class="event-card">
          <h3>${event.title}</h3>
          <p>${event.description}</p>
          <div class="event-info">
            <div class="event-info-item">
              📅 ${eventDate.toLocaleDateString('pt-BR')} às ${eventDate.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}
            </div>
            <div class="event-info-item">
              📍 ${event.location}
            </div>
            <div class="event-info-item">
              👥 ${event.maxParticipants} vagas
            </div>
          </div>
          <div class="event-actions">
            ${isRegistered 
              ? '<button class="btn btn-secondary" disabled>Já inscrito</button>'
              : `<button class="btn btn-primary" onclick="registerForEvent(${event.id})">Inscrever-se</button>`
            }
          </div>
        </div>
      `;
    }).join('');
  } catch (error) {
    showToast('Erro ao carregar eventos', 'error');
  }
}

// Inscrever-se em evento
async function registerForEvent(eventId) {
  try {
    const response = await fetch('/api/registrations', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ eventId })
    });
    
    const data = await response.json();
    
    if (response.ok) {
      showToast('Inscrição realizada com sucesso!', 'success');
      loadEvents();
    } else {
      showToast(data.error || 'Erro ao fazer inscrição', 'error');
    }
  } catch (error) {
    showToast('Erro ao fazer inscrição', 'error');
  }
}

// Carregar minhas inscrições
async function loadMyRegistrations() {
  try {
    const response = await fetch('/api/registrations/my');
    myRegistrations = await response.json();
    
    const container = document.getElementById('my-registrations-list');
    
    if (myRegistrations.length === 0) {

          <h3>Nenhuma inscrição</h3>
          <p>Você ainda não se inscreveu em nenhum evento.</p>
        </div>
      `;
      return;
    }
    
    container.innerHTML = myRegistrations.map(reg => {
      const event = reg.event;
      const eventDate = new Date(event.date);
      
      return `
        <div class="registration-card">
          <div class="registration-header">
            <h3>${event.title}</h3>
            <span class="registration-badge ${reg.status}">${reg.status}</span>
          </div>
          <div class="registration-info">
            <div class="registration-info-item">
              <strong>Data e Hora</strong>
              ${eventDate.toLocaleDateString('pt-BR')} às ${eventDate.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}
            </div>
            <div class="registration-info-item">
              <strong>Local</strong>
              ${event.location}
            </div>
            <div class="registration-info-item">
              <strong>Presença</strong>
              ${reg.presence ? '✅ Confirmada' : '⏳ Pendente'}
            </div>
          </div>
          <p>${event.description}</p>
          <div class="registration-actions">
            <button class="btn btn-danger" onclick="cancelRegistration(${reg.id})">Cancelar Inscrição</button>
          </div>
        </div>
      `;
    }).join('');
  } catch (error) {
    showToast('Erro ao carregar inscrições', 'error');
  }
}

// Cancelar inscrição
async function cancelRegistration(registrationId) {
  if (!confirm('Tem certeza que deseja cancelar esta inscrição?')) {
    return;
  }
  
  try {
    const response = await fetch(`/api/registrations/${registrationId}`, {
      method: 'DELETE'
    });
    
    const data = await response.json();
    
    if (response.ok) {
      showToast('Inscrição cancelada com sucesso!', 'success');
      loadMyRegistrations();
    } else {
      showToast(data.error || 'Erro ao cancelar inscrição', 'error');
    }
  } catch (error) {
    showToast('Erro ao cancelar inscrição', 'error');
  }
}

    container.innerHTML = myEvents.map(event => {
      const eventDate = new Date(event.date);
      const eventRegs = allRegs.filter(r => r.eventId === event.id);
      const confirmedCount = eventRegs.filter(r => r.presence).length;
      
      return `
        <div class="event-card">
          <span class="event-status ${event.status}">${event.status}</span>
          <h3>${event.title}</h3>
          <p>${event.description}</p>
          <div class="event-info">
            <div class="event-info-item">
              📅 ${eventDate.toLocaleDateString('pt-BR')} às ${eventDate.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}
            </div>
            <div class="event-info-item">
              📍 ${event.location}
            </div>
            <div class="event-info-item">
              👥 ${eventRegs.length}/${event.maxParticipants} inscritos
            </div>
            <div class="event-info-item">
              ✅ ${confirmedCount} presenças confirmadas
            </div>
          </div>
          <div class="event-actions">
            ${event.status === 'ativo' ? `
              <button class="btn btn-primary" onclick="editEvent(${event.id})">Editar</button>
              <button class="btn btn-danger" onclick="cancelEvent(${event.id})">Cancelar Evento</button>
            ` : '<button class="btn btn-secondary" disabled>Evento Cancelado</button>'}
          </div>
        </div>
      `;
    }).join('');
  } catch (error) {
    showToast('Erro ao carregar eventos', 'error');
  }
}

// Editar evento
async function editEvent(eventId) {
  try {
    const response = await fetch(`/api/events/${eventId}`);
    const event = await response.json();
    
    document.getElementById('edit-event-id').value = event.id;
    document.getElementById('edit-event-title').value = event.title;
    document.getElementById('edit-event-description').value = event.description;
    
    // Formatar data para datetime-local
    const eventDate = new Date(event.date);
    const dateString = eventDate.toISOString().slice(0, 16);
    document.getElementById('edit-event-date').value = dateString;
    
    document.getElementById('edit-event-location').value = event.location;
    document.getElementById('edit-event-max').value = event.maxParticipants;
    
    document.getElementById('edit-modal').classList.add('active');
  } catch (error) {
    showToast('Erro ao carregar evento', 'error');
  }
}

// Cancelar evento
async function cancelEvent(eventId) {
  if (!confirm('Tem certeza que deseja cancelar este evento? Todos os inscritos serão notificados.')) {
    return;
  }
  
  try {
    const response = await fetch(`/api/events/${eventId}`, {
      method: 'DELETE'
    });
    
    const data = await response.json();
    
    if (response.ok) {
      showToast('Evento cancelado com sucesso!', 'success');
      loadManageEvents();
    } else {
      showToast(data.error || 'Erro ao cancelar evento', 'error');
    }
  } catch (error) {
    showToast('Erro ao cancelar evento', 'error');
  }
}

// Carregar todas as inscrições (organizador)
async function loadAllRegistrations() {
  try {
    const response = await fetch('/api/registrations/all');
    allRegistrations = await response.json();
    
    const container = document.getElementById('all-registrations-list');
    
    if (allRegistrations.length === 0) {
      container.innerHTML = `
        <div class="empty-state">
          <div class="empty-state-icon">👥</div>
          <h3>Nenhuma inscrição</h3>
          <p>Ainda não há inscrições em eventos.</p>
        </div>
      `;
      return;
    }
    
    container.innerHTML = allRegistrations.map(reg => {
      const event = reg.event;
      const user = reg.user;
      const eventDate = new Date(event.date);
      
      return `
        <div class="registration-card">
          <div class="registration-header">
            <div>
              <h3>${event.title}</h3>
              <p style="margin-top: 4px; color: #666;">Participante: ${user.name} (${user.email})</p>
            </div>
            <span class="registration-badge ${reg.presence ? 'presente' : 'inscrito'}">
              ${reg.presence ? 'Presente' : 'Inscrito'}
            </span>
          </div>
          <div class="registration-info">
            <div class="registration-info-item">
              <strong>Data e Hora</strong>
              ${eventDate.toLocaleDateString('pt-BR')} às ${eventDate.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}
            </div>
            <div class="registration-info-item">
              <strong>Local</strong>
              ${event.location}
            </div>
            <div class="registration-info-item">
              <strong>Inscrição</strong>
              ${new Date(reg.createdAt).toLocaleDateString('pt-BR')}
            </div>
          </div>
          ${event.organizerId === currentUser.id ? `
            <div class="registration-actions">
              <button class="btn ${reg.presence ? 'btn-secondary' : 'btn-success'}" 
                      onclick="togglePresence(${reg.id}, ${!reg.presence})">
                ${reg.presence ? 'Remover Presença' : 'Confirmar Presença'}
              </button>
            </div>
          ` : ''}
        </div>
      `;
    }).join('');
  } catch (error) {
    showToast('Erro ao carregar inscrições', 'error');
  }
}

// Confirmar/remover presença
async function togglePresence(registrationId, presence) {
  try {
    const response = await fetch(`/api/registrations/${registrationId}/presence`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ presence })
    });
    
    const data = await response.json();
    
    if (response.ok) {
      showToast(presence ? 'Presença confirmada!' : 'Presença removida!', 'success');
      loadAllRegistrations();
    } else {
      showToast(data.error || 'Erro ao atualizar presença', 'error');
    }
  } catch (error) {
    showToast('Erro ao atualizar presença', 'error');
  }
}

// Fechar modal
function closeModal() {
  document.getElementById('edit-modal').classList.remove('active');
}

// Toast de notificação
function showToast(message, type = 'info') {
  const toast = document.getElementById('toast');
  toast.textContent = message;
  toast.className = `toast ${type} show`;
  
  setTimeout(() => {
    toast.classList.remove('show');
  }, 3000);
}
