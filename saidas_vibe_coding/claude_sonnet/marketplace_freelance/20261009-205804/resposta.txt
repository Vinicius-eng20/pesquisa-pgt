# Marketplace de Serviços Freelance

Sistema completo de marketplace para contratação de serviços freelance.

## Requisitos

- Node.js 18+ instalado
- Navegador web moderno

## Instalação

```bash
npm install
```

## Executar o projeto

```bash
npm start
```

O site estará disponível em: http://localhost:3000

## Primeiro Acesso

### Conta Administrador (já criada):
- Email: admin@marketplace.com
- Senha: Admin123!

### Criar contas de usuários:
Clique em "Cadastrar" na página inicial e preencha o formulário.

## Funcionalidades

### Usuários (Freelancers/Clientes):
- Publicar serviços que oferece
- Contratar serviços de outros usuários
- Ver contratos em que participa
- Editar perfil

### Administrador:
- Aprovar ou remover serviços publicados
- Suspender contas de usuários
- Ver todos os serviços e contratos
- Painel completo de administração

## Notas

- Todos os dados são salvos no arquivo `database.json`
- O site funciona em qualquer dispositivo (responsivo)
- Não necessita conexão com serviços externos

const express = require('express');
const session = require('express-session');
const bcrypt = require('bcryptjs');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 3000;
const DB_FILE = path.join(__dirname, 'database.json');

// Middleware
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static('public'));
app.use(session({
  secret: 'marketplace-secret-key-2024',
  resave: false,
  saveUninitialized: false,
  cookie: { maxAge: 24 * 60 * 60 * 1000 }
}));

// Inicializar banco de dados
function initDatabase() {
  if (!fs.existsSync(DB_FILE)) {
    const initialData = {
      users: [
        {
          id: 1,
          name: 'Administrador',
          email: 'admin@marketplace.com',
          password: bcrypt.hashSync('Admin123!', 10),
          role: 'admin',
          suspended: false,
          createdAt: new Date().toISOString()
        }
      ],
      services: [],
      contracts: [],
      nextUserId: 2,
      nextServiceId: 1,
      nextContractId: 1
    };
    fs.writeFileSync(DB_FILE, JSON.stringify(initialData, null, 2));
  }
}

function readDB() {
  const data = fs.readFileSync(DB_FILE, 'utf8');
  return JSON.parse(data);
}

function writeDB(data) {
  fs.writeFileSync(DB_FILE, JSON.stringify(data, null, 2));
}

initDatabase();

// Middleware de autenticação
function requireAuth(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({ error: 'Não autenticado' });
  }
  next();
}

function requireAdmin(req, res, next) {
  if (!req.session.userId || req.session.userRole !== 'admin') {
    return res.status(403).json({ error: 'Acesso negado' });
  }
  next();
}

// Rotas de autenticação
app.post('/api/register', (req, res) => {
  const { name, email, password } = req.body;

  if (!name || !email || !password) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios' });
  }

  if (password.length < 6) {
    return res.status(400).json({ error: 'A senha deve ter no mínimo 6 caracteres' });
  }

  const db = readDB();
  
  if (db.users.find(u => u.email === email)) {
    return res.status(400).json({ error: 'Email já cadastrado' });
  }

  const newUser = {
    id: db.nextUserId++,
    name,
    email,
    password: bcrypt.hashSync(password, 10),
    role: 'user',
    suspended: false,
    createdAt: new Date().toISOString()
  };

  db.users.push(newUser);
  writeDB(db);

  req.session.userId = newUser.id;
  req.session.userName = newUser.name;
  req.session.userRole = newUser.role;

  res.json({ 
    success: true, 
    user: { 
      id: newUser.id, 
      name: newUser.name, 
      email: newUser.email, 
      role: newUser.role 
    } 
  });
});

app.post('/api/login', (req, res) => {
  const { email, password } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: 'Email e senha são obrigatórios' });
  }

  const db = readDB();
  const user = db.users.find(u => u.email === email);

  if (!user || !bcrypt.compareSync(password, user.password)) {
    return res.status(401).json({ error: 'Email ou senha incorretos' });
  }

  if (user.suspended) {
    return res.status(403).json({ error: 'Conta suspensa. Entre em contato com o administrador.' });
  }

  req.session.userId = user.id;
  req.session.userName = user.name;
  req.session.userRole = user.role;

  res.json({ 
    success: true, 
    user: { 
      id: user.id, 
      name: user.name, 
      email: user.email, 
      role: user.role 
    } 
  });
});

app.post('/api/logout', (req, res) => {
  req.session.destroy();
  res.json({ success: true });
});

app.get('/api/me', requireAuth, (req, res) => {
  const db = readDB();
  const user = db.users.find(u => u.id === req.session.userId);
  
  if (!user) {
    return res.status(404).json({ error: 'Usuário não encontrado' });
  }

  res.json({ 
    id: user.id, 
    name: user.name, 
    email: user.email, 
    role: user.role,
    suspended: user.suspended
  });
});

// Rotas de serviços
app.post('/api/services', requireAuth, (req, res) => {
  const { title, description, price, category } = req.body;

  if (!title || !description || !price || !category) {
    return res.status(400).json({ error: 'Todos os campos são obrigatórios' });
  }

  const db = readDB();
  const user = db.users.find(u => u.id === req.session.userId);

  if (user.suspended) {
    return res.status(403).json({ error: 'Conta suspensa' });
  }

  const newService = {
    id: db.nextServiceId++,
    userId: req.session.userId,
    userName: req.session.userName,
    title,
    description,
    price: parseFloat(price),
    category,
    status: 'pending',
    createdAt: new Date().toISOString()
  };

  db.services.push(newService);
  writeDB(db);

  res.json({ success: true, service: newService });
});

app.get('/api/services', requireAuth, (req, res) => {
  const db = readDB();
  let services = db.services;

  if (req.session.userRole !== 'admin') {
    services = services.filter(s => s.status === 'approved');
  }

  res.json(services);
});

app.get('/api/services/my', requireAuth, (req, res) => {
  const db = readDB();
  const myServices = db.services.filter(s => s.userId === req.session.userId);
  res.json(myServices);
});

app.delete('/api/services/:id', requireAuth, (req, res) => {
  const serviceId = parseInt(req.params.id);
  const db = readDB();
  
  const serviceIndex = db.services.findIndex(s => s.id === serviceId);
  
  if (serviceIndex === -1) {
    return res.status(404).json({ error: 'Serviço não encontrado' });
  }

  const service = db.services[serviceIndex];

  if (service.userId !== req.session.userId && req.session.userRole !== 'admin') {
    return res.status(403).json({ error: 'Sem permissão' });
  }

  db.services.splice(serviceIndex, 1);
  writeDB(db);

  res.json({ success: true });
});

app.patch('/api/services/:id/approve', requireAdmin, (req, res) => {
  const serviceId = parseInt(req.params.id);
  const db = readDB();
  
  const service = db.services.find(s => s.id === serviceId);
  
  if (!service) {
    return res.status(404).json({ error: 'Serviço não encontrado' });
  }

  service.status = 'approved';
  writeDB(db);

  res.json({ success: true, service });
});

app.delete('/api/admin/services/:id', requireAdmin, (req, res) => {
  const serviceId = parseInt(req.params.id);
  const db = readDB();
  
  const serviceIndex = db.services.findIndex(s => s.id === serviceId);
  
  if (serviceIndex === -1) {
    return res.status(404).json({ error: 'Serviço não encontrado' });
  }

  db.services.splice(serviceIndex, 1);
  writeDB(db);

  res.json({ success: true });
});

// Rotas de contratos
app.post('/api/contracts', requireAuth, (req, res) => {
  const { serviceId } = req.body;

  if (!serviceId) {
    return res.status(400).json({ error: 'ID do serviço é obrigatório' });
  }

  const db = readDB();
  const service = db.services.find(s => s.id === parseInt(serviceId));

  if (!service) {
    return res.status(404).json({ error: 'Serviço não encontrado' });
  }

  if (service.status !== 'approved') {
    return res.status(400).json({ error: 'Serviço não está aprovado' });
  }

  if (service.userId === req.session.userId) {
    return res.status(400).json({ error: 'Você não pode contratar seu próprio serviço' });
  }

  const user = db.users.find(u => u.id === req.session.userId);
  if (user.suspended) {
    return res.status(403).json({ error: 'Conta suspensa' });
  }

  const newContract = {
    id: db.nextContractId++,
    serviceId: service.id,
    serviceTitle: service.title,
    providerId: service.userId,
    providerName: service.userName,
    clientId: req.session.userId,
    clientName: req.session.userName,
    price: service.price,
    status: 'active',
    createdAt: new Date().toISOString()
  };

  db.contracts.push(newContract);
  writeDB(db);

  res.json({ success: true, contract: newContract });
});

app.get('/api/contracts', requireAuth, (req, res) => {
  const db = readDB();
  let contracts = db.contracts;

  if (req.session.userRole !== 'admin') {
    contracts = contracts.filter(c => 
      c.clientId === req.session.userId || c.providerId === req.session.userId
    );
  }

  res.json(contracts);
});

// Rotas de administração
app.get('/api/admin/users', requireAdmin, (req, res) => {
  const db = readDB();
  const users = db.users.map(u => ({
    id: u.id,
    name: u.name,
    email: u.email,
    role: u.role,
    suspended: u.suspended,
    createdAt: u.createdAt
  }));
  res.json(users);
});

app.patch('/api/admin/users/:id/suspend', requireAdmin, (req, res) => {
  const userId = parseInt(req.params.id);
  const db = readDB();
  
  const user = db.users.find(u => u.id === userId);
  
  if (!user) {
    return res.status(404).json({ error: 'Usuário não encontrado' });
  }

  if (user.role === 'admin') {
    return res.status(400).json({ error: 'Não é possível suspender um administrador' });
  }

  user.suspended = !user.suspended;
  writeDB(db);

  res.json({ success: true, user: { id: user.id, suspended: user.suspended } });
});

  buttons.forEach(btn => btn.classList.remove('active'));
  event.target.classList.add('active');

  }

  displayContracts(filtered);

async function hireService(serviceId) {
  if (!confirm('Deseja realmente contratar este serviço?')) {
    return;
  }

  try {
    const response = await fetch('/api/contracts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },

    const data = await response.json();

    if (response.ok) {

}

async function deleteService(serviceId) {
  if (!confirm('Deseja realmente excluir este serviço?')) {
    return;
  }

  try {
    const response = await fetch(`/api/services/${serviceId}`, {
      method: 'DELETE'
    });

    if (response.ok) {
      showModal('Sucesso', 'Serviço excluído com sucesso!');
      await loadMyServices();
    } else {
      const data = await response.json();
      showModal('Erro', data.error);
    }
  } catch (error) {
    showModal('Erro', 'Erro ao excluir serviço. Tente novamente.');
  }
}

          <h3>Nenhum serviço encontrado</h3>
          <p>Não há serviços cadastrados no sistema.</p>
        </div>
      `;
      return;
    }

    container.innerHTML = services.map(service => `
      <div class="service-card">
        <div class="service-header">
          <div>
            <div class="service-title">${service.title}</div>
            <div class="service-category">${getCategoryName(service.category)}</div>
          </div>
          <span class="service-status status-${service.status}">
            ${service.status === 'pending' ? 'Pendente' : 'Aprovado'}
          </span>
        </div>
        <div class="service-provider">Por: ${service.userName}</div>
        <div class="service-description">${service.description}</div>
        <div class="service-footer">
          <div class="service-price">R$ ${service.price.toFixed(2)}</div>
        </div>
        <div class="service-actions">
          ${service.status === 'pending' ? `
            <button class="btn btn-success btn-small" onclick="approveService(${service.id})">
              Aprovar
            </button>
          ` : ''}
          <button class="btn btn-danger btn-small" onclick="deleteServiceAdmin(${service.id})">
            Remover
          </button>
        </div>
      </div>
    `).join('');
  } catch (error) {
    console.error('Erro ao carregar todos os serviços:', error);
  }
}

async function approveService(serviceId) {
  try {
    const response = await fetch(`/api/services/${serviceId}/approve`, {
      method: 'PATCH'
    });

    if (response.ok) {
      showModal('Sucesso', 'Serviço aprovado com sucesso!');
      await loadPendingServices();
      await loadAllServices();
    } else {
      const data = await response.json();
      showModal('Erro', data.error);
    }
  } catch (error) {
    showModal('Erro', 'Erro ao aprovar serviço. Tente novamente.');
  }
}

async function deleteServiceAdmin(serviceId) {
  if (!confirm('Deseja realmente remover este serviço?')) {
    return;
  }

  try {
    const response = await fetch(`/api/admin/services/${serviceId}`, {
      method: 'DELETE'
    });

    if (response.ok) {
      showModal('Sucesso', 'Serviço removido com sucesso!');
      await loadPendingServices();
      await loadAllServices();
    } else {
      const data = await response.json();
      showModal('Erro', data.error);
    }
  } catch (error) {
    showModal('Erro', 'Erro ao remover serviço. Tente novamente.');
  }
}

async function toggleSuspendUser(userId, isSuspended) {
  const action = isSuspended ? 'reativar' : 'suspender';
  if (!confirm(`Deseja realmente ${action} esta conta?`)) {
    return;
  }

  try {
    const response = await fetch(`/api/admin/users/${userId}/suspend`, {
      method: 'PATCH'
    });

    if (response.ok) {
      showModal('Sucesso', `Conta ${isSuspended ? 'reativada' : 'suspensa'} com sucesso!`);
      await loadUsers();
    } else {
      const data = await response.json();
      showModal('Erro', data.error);
    }
  } catch (error) {
    showModal('Erro', 'Erro ao modificar conta. Tente novamente.');
  }
}

function showTab(tabName) {
  document.querySelectorAll('.tab-button').forEach(btn => btn.classList.remove('active'));
  document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));
  
  event?.target?.classList.add('active');
  document.getElementById(tabName + 'Tab').classList.add('active');
}

function showModal(title, message) {
  const modal = document.getElementById('modal');
  const modalBody = document.getElementById('modalBody');
  
  modalBody.innerHTML = `
    <h2>${title}</h2>
    <p style="margin-top: 15px; color: #666;">${message}</p>
    <button class="btn btn-primary" onclick="closeModal()" style="margin-top: 20px; width: 100%;">
      OK
    </button>
  `;
  
  modal.classList.add('show');
}

function closeModal() {
  document.getElementById('modal').classList.remove('show');
}

function getCategoryName(category) {
  const categories = {
    design: 'Design',
    programacao: 'Programação',
    marketing: 'Marketing',
    redacao: 'Redação',
    traducao: 'Tradução',
    consultoria: 'Consultoria',
    outros: 'Outros'
  };
  return categories[category] || category;
}
