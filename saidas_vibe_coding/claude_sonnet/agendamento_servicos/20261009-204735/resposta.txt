const express = require('express');
const session = require('express-session');
const bcrypt = require('bcryptjs');
const Database = require('better-sqlite3');
const path = require('path');

const app = express();
const db = new Database('agendamento.db');

// Configuração do banco de dados
db.exec(`
  CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL,
    tipo TEXT NOT NULL CHECK(tipo IN ('cliente', 'admin')),
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS servicos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    descricao TEXT,
    duracao INTEGER NOT NULL,
    preco REAL NOT NULL,
    ativo INTEGER DEFAULT 1,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS horarios_disponiveis (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    dia_semana INTEGER NOT NULL CHECK(dia_semana >= 0 AND dia_semana <= 6),
    hora_inicio TEXT NOT NULL,
    hora_fim TEXT NOT NULL,
    ativo INTEGER DEFAULT 1,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS agendamentos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    cliente_id INTEGER NOT NULL,
    servico_id INTEGER NOT NULL,
    data DATE NOT NULL,
    hora TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'pendente' CHECK(status IN ('pendente', 'confirmado', 'recusado', 'cancelado')),
    observacoes TEXT,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cliente_id) REFERENCES usuarios(id),
    FOREIGN KEY (servico_id) REFERENCES servicos(id)
  );
`);

// Middleware de autenticação
function requireAuth(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({ erro: 'Não autenticado' });
  }
  next();
}

function requireAdmin(req, res, next) {
  if (!req.session.userId || req.session.userTipo !== 'admin') {
    return res.status(403).json({ erro: 'Acesso negado' });
  }
  next();
}

// Rotas de autenticação
app.post('/api/registrar', (req, res) => {
  try {
    const { nome, email, senha } = req.body;
    
    if (!nome || !email || !senha) {
      return res.status(400).json({ erro: 'Todos os campos são obrigatórios' });
    }

    const usuarioExiste = db.prepare('SELECT id FROM usuarios WHERE email = ?').get(email);
    if (usuarioExiste) {
      return res.status(400).json({ erro: 'Email já cadastrado' });
    }

    const senhaHash = bcrypt.hashSync(senha, 10);
    const resultado = db.prepare('INSERT INTO usuarios (nome, email, senha, tipo) VALUES (?, ?, ?, ?)').run(
      nome,
      email,
      senhaHash,
      'cliente'
    );

    req.session.userId = resultado.lastInsertRowid;
    req.session.userNome = nome;
    req.session.userTipo = 'cliente';

    res.json({ sucesso: true, usuario: { id: resultado.lastInsertRowid, nome, tipo: 'cliente' } });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao registrar usuário' });
  }
});

app.post('/api/login', (req, res) => {
  try {
    const { email, senha } = req.body;
    
    const usuario = db.prepare('SELECT * FROM usuarios WHERE email = ?').get(email);
    
    if (!usuario || !bcrypt.compareSync(senha, usuario.senha)) {
      return res.status(401).json({ erro: 'Email ou senha inválidos' });
    }

    req.session.userId = usuario.id;
    req.session.userNome = usuario.nome;
    req.session.userTipo = usuario.tipo;

    res.json({ 
      sucesso: true, 
      usuario: { 
        id: usuario.id, 
        nome: usuario.nome, 
        tipo: usuario.tipo 
      } 
    });
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao fazer login' });
  }
});

// Rotas de agendamentos
app.get('/api/agendamentos', requireAuth, (req, res) => {
  try {
    let agendamentos;
    
    if (req.session.userTipo === 'admin') {
      agendamentos = db.prepare(`
        SELECT a.*, s.nome as servico_nome, u.nome as cliente_nome, u.email as cliente_email
        FROM agendamentos a
        JOIN servicos s ON a.servico_id = s.id
        JOIN usuarios u ON a.cliente_id = u.id
        ORDER BY a.data DESC, a.hora DESC
      `).all();
    } else {
      agendamentos = db.prepare(`
        SELECT a.*, s.nome as servico_nome, s.preco as servico_preco
        FROM agendamentos a
        JOIN servicos s ON a.servico_id = s.id
        WHERE a.cliente_id = ?
        ORDER BY a.data DESC, a.hora DESC
      `).all(req.session.userId);
    }

    res.json(agendamentos);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao buscar agendamentos' });
  }
});

app.get('/api/horarios-livres/:data/:servico_id', requireAuth, (req, res) => {
  try {
    const { data, servico_id } = req.params;
    const dataObj = new Date(data + 'T00:00:00');
    const diaSemana = dataObj.getDay();

    // Buscar horários disponíveis para o dia da semana
    const horariosDisponiveis = db.prepare(
      'SELECT * FROM horarios_disponiveis WHERE dia_semana = ? AND ativo = 1'
    ).all(diaSemana);

    if (horariosDisponiveis.length === 0) {
      return res.json([]);
    }

    // Buscar serviço para saber a duração
    const servico = db.prepare('SELECT duracao FROM servicos WHERE id = ?').get(servico_id);
    if (!servico) {
      return res.status(404).json({ erro: 'Serviço não encontrado' });
    }

    // Buscar agendamentos já feitos para esta data
    const agendamentosExistentes = db.prepare(
      'SELECT hora FROM agendamentos WHERE data = ? AND status IN ("pendente", "confirmado")'
    ).all(data);

    const horariosOcupados = new Set(agendamentosExistentes.map(a => a.hora));

    // Gerar slots de horários
    const horariosLivres = [];
    
    for (const horarioDisp of horariosDisponiveis) {
      const [horaInicio, minInicio] = horarioDisp.hora_inicio.split(':').map(Number);
      const [horaFim, minFim] = horarioDisp.hora_fim.split(':').map(Number);
      
      let horaAtual = horaInicio * 60 + minInicio;
      const horaFimMin = horaFim * 60 + minFim;
      
      while (horaAtual + servico.duracao <= horaFimMin) {
        const h = Math.floor(horaAtual / 60).toString().padStart(2, '0');
        const m = (horaAtual % 60).toString().padStart(2, '0');
        const horarioStr = `${h}:${m}`;
        
        if (!horariosOcupados.has(horarioStr)) {
          horariosLivres.push(horarioStr);
        }
        
        horaAtual += 30; // Intervalos de 30 minutos
      }
    }

    res.json(horariosLivres);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao buscar horários livres' });
  }
});

app.post('/api/agendamentos', requireAuth, (req, res) => {
  try {
    const { servico_id, data, hora, observacoes } = req.body;
    
    if (!servico_id || !data || !hora) {
      return res.status(400).json({ erro: 'Serviço, data e hora são obrigatórios' });
    }

    // Verificar se o horário ainda está disponível
    const agendamentoExistente = db.prepare(
      'SELECT id FROM agendamentos WHERE data = ? AND hora = ? AND status IN ("pendente", "confirmado")'
    ).get(data, hora);

    if (agendamentoExistente) {
      return res.status(400).json({ erro: 'Este horário já foi agendado' });
    }

    const resultado = db.prepare(
      'INSERT INTO agendamentos (cliente_id, servico_id, data, hora, observacoes) VALUES (?, ?, ?, ?, ?)'
    ).run(
      req.session.userId,
      servico_id,
      data,
      hora,
      observacoes || ''
    );

    const agendamento = db.prepare(`
      SELECT a.*, s.nome as servico_nome, s.preco as servico_preco
      FROM agendamentos a
      JOIN servicos s ON a.servico_id = s.id
      WHERE a.id = ?
    `).get(resultado.lastInsertRowid);

    res.json(agendamento);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao criar agendamento' });
  }
});

app.put('/api/agendamentos/:id/status', requireAuth, (req, res) => {
  try {
    const { status } = req.body;
    const agendamentoId = req.params.id;

    // Buscar agendamento
    const agendamento = db.prepare('SELECT * FROM agendamentos WHERE id = ?').get(agendamentoId);
    
    if (!agendamento) {
      return res.status(404).json({ erro: 'Agendamento não encontrado' });
    }

    // Verificar permissões
    if (status === 'cancelado' && agendamento.cliente_id !== req.session.userId) {
      return res.status(403).json({ erro: 'Você só pode cancelar seus próprios agendamentos' });
    }

    if ((status === 'confirmado' || status === 'recusado') && req.session.userTipo !== 'admin') {
      return res.status(403).json({ erro: 'Apenas administradores podem confirmar ou recusar agendamentos' });
    }

    db.prepare('UPDATE agendamentos SET status = ? WHERE id = ?').run(status, agendamentoId);

    const agendamentoAtualizado = db.prepare(`
      SELECT a.*, s.nome as servico_nome, u.nome as cliente_nome
      FROM agendamentos a
      JOIN servicos s ON a.servico_id = s.id
      JOIN usuarios u ON a.cliente_id = u.id
      WHERE a.id = ?
    `).get(agendamentoId);

    res.json(agendamentoAtualizado);
  } catch (erro) {
    res.status(500).json({ erro: 'Erro ao atualizar agendamento' });
  }
});

// Estado da aplicação
let usuarioAtual = null;

// Utilitários
function mostrarErro(mensagem, elementoId = 'auth-erro') {
  const elemento = document.getElementById(elementoId);
  if (elemento) {
    elemento.textContent = mensagem;
    elemento.classList.add('show');
    setTimeout(() => elemento.classList.remove('show'), 5000);
  }
}

function mostrarModal(mensagem) {
  const modal = document.getElementById('modal');
  const modalMessage = document.getElementById('modal-message');
  modalMessage.textContent = mensagem;
  modal.classList.add('show');
}

function fecharModal() {
  document.getElementById('modal').classList.remove('show');
}

function formatarData(data) {
  const d = new Date(data + 'T00:00:00');
  return d.toLocaleDateString('pt-BR');
}

function formatarPreco(preco) {
  return parseFloat(preco).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' });
}

function getDiaSemanaTexto(dia) {
  const dias = ['Domingo', 'Segunda-feira', 'Terça-feira', 'Quarta-feira', 'Quinta-feira', 'Sexta-feira', 'Sábado'];
  return dias[dia];
}

// API calls
async function fazerRequisicao(url, opcoes = {}) {
  try {
    const resposta = await fetch(url, {
      ...opcoes,
      headers: {
        'Content-Type': 'application/json',
        ...opcoes.headers,
      },
    });

    const dados = await resposta.json();

    if (!resposta.ok) {
      throw new Error(dados.erro || 'Erro na requisição');
    }

    return dados;
  } catch (erro) {
    console.error('Erro:', erro);
    throw erro;
  }
}

// Autenticação
async function verificarAutenticacao() {
  try {
    const dados = await fazerRequisicao('/api/usuario-atual');
    
    if (dados.autenticado) {
      usuarioAtual = dados.usuario;
      mostrarTelaCorreta();
    } else {
      mostrarTela('auth-screen');
    }
  } catch (erro) {
    mostrarTela('auth-screen');
  }
}

function mostrarTela(telaId) {
  document.querySelectorAll('.screen').forEach(tela => {
    tela.style.display = 'none';
  });
  document.getElementById(telaId).style.display = 'block';
}

function mostrarTelaCorreta() {
  if (!usuarioAtual) {
    mostrarTela('auth-screen');
    return;
  }

  if (usuarioAtual.tipo === 'admin') {
    document.getElementById('admin-nome').textContent = usuarioAtual.nome;
    mostrarTela('admin-screen');
    carregarDadosAdmin();
  } else {
    document.getElementById('user-nome').textContent = usuarioAtual.nome;
    mostrarTela('cliente-screen');
    carregarDadosCliente();
  }
}

// Eventos de autenticação
document.getElementById('login-form').addEventListener('submit', async (e) => {
  e.preventDefault();
  const formData = new FormData(e.target);
  const dados = Object.fromEntries(formData);

  try {
    const resposta = await fazerRequisicao('/api/login', {
      method: 'POST',
      body: JSON.stringify(dados),
    });

    usuarioAtual = resposta.usuario;
    mostrarTelaCorreta();
  } catch (erro) {
    mostrarErro(erro.message);
  }
});

document.addEventListener('click', (e) => {
  if (e.target.classList.contains('tab')) {
    const container = e.target.closest('.container') || e.target.closest('.auth-container');
    if (!container) return;

    const tabName = e.target.dataset.tab;
    
    container.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    e.target.classList.add('active');

    container.querySelectorAll('.tab-content').forEach(content => {
      content.style.display = 'none';
    });

    const targetContent = document.getElementById(tabName);
    if (targetContent) {
      targetContent.style.display = 'block';
    }
  }
});

async function carregarDadosCliente() {
  await carregarServicos();
  await carregarAgendamentosCliente();
  configurarDataMinima();
}

function configurarDataMinima() {
  const inputData = document.getElementById('input-data');
  const hoje = new Date();
  const amanha = new Date(hoje);
  amanha.setDate(amanha.getDate() + 1);
  inputData.min = amanha.toISOString().split('T')[0];
}

async function carregarServicos() {
  try {
    servicos = await fazerRequisicao('/api/servicos');
    renderizarServicos();
  } catch (erro) {
    console.error('Erro ao carregar serviços:', erro);
  }
}

function renderizarServicos() {
  const select = document.getElementById('select-servico');
  select.innerHTML = '<option value="">Selecione um serviço</option>';
  
  servicos.forEach(servico => {
    const option = document.createElement('option');
    option.value = servico.id;
    option.textContent = `${servico.nome} - ${formatarPreco(servico.preco)}`;
    select.appendChild(option);
  });
}

document.getElementById('select-servico').addEventListener('change', (e) => {
  const servicoId = e.target.value;
  const infoServico = document.getElementById('info-servico');
  
  if (!servicoId) {
    infoServico.innerHTML = '';
    return;
  }

  const servico = servicos.find(s => s.id == servicoId);
  if (servico) {
    infoServico.innerHTML = `
      <strong>${servico.nome}</strong><br>
      ${servico.descricao ? servico.descricao + '<br>' : ''}
      Duração: ${servico.duracao} minutos<br>
      Valor: ${formatarPreco(servico.preco)}
    `;
  }
});

document.getElementById('input-data').addEventListener('change', async (e) => {
  const data = e.target.value;
  const servicoId = document.getElementById('select-servico').value;
  const selectHorario = document.getElementById('select-horario');

  if (!data || !servicoId) {
    selectHorario.innerHTML = '<option value="">Selecione um serviço e uma data</option>';
    return;
  }

  try {
    const horariosLivres = await fazerRequisicao(`/api/horarios-livres/${data}/${servicoId}`);
    
    selectHorario.innerHTML = '<option value="">Selecione um horário</option>';
    
    if (horariosLivres.length === 0) {
      selectHorario.innerHTML = '<option value="">Nenhum horário disponível</option>';
      return;
    }

    horariosLivres.forEach(horario => {
      const option = document.createElement('option');
      option.value = horario;
      option.textContent = horario;
      selectHorario.appendChild(option);
    });
  } catch (erro) {
    console.error('Erro ao carregar horários:', erro);
    selectHorario.innerHTML = '<option value="">Erro ao carregar horários</option>';
  }
});

document.getElementById('form-novo-agendamento').addEventListener('submit', async (e) => {

async function carregarAgendamentosCliente() {
  try {
    agendamentos = await fazerRequisicao('/api/agendamentos');
    renderizarAgendamentosCliente();
  } catch (erro) {
    console.error('Erro ao carregar agendamentos:', erro);
  }
}

function renderizarAgendamentosCliente() {
  const container = document.getElementById('lista-agendamentos-cliente');
  
  if (agendamentos.length === 0) {
    container.innerHTML = '<div class="empty-state"><div class="empty-state-icon">📅</div><p>Você ainda não tem agendamentos</p></div>';
    return;
  }

  container.innerHTML = agendamentos.map(agendamento => `
    <div class="card">
      <div class="card-header">
        <div class="card-title">${agendamento.servico_nome}</div>
        <span class="status-badge status-${agendamento.status}">${agendamento.status.toUpperCase()}</span>
      </div>
      <div class="card-body">
        <div class="card-info">
          <div class="card-info-row">
            <span class="card-info-label">Data:</span>
            <span>${formatarData(agendamento.data)}</span>
          </div>
          <div class="card-info-row">
            <span class="card-info-label">Horário:</span>
            <span>${agendamento.hora}</span>
          </div>
          <div class="card-info-row">
            <span class="card-info-label">Valor:</span>
            <span>${formatarPreco(agendamento.servico_preco)}</span>
          </div>
          ${agendamento.observacoes ? `
            <div class="card-info-row">
              <span class="card-info-label">Observações:</span>
              <span>${agendamento.observacoes}</span>
            </div>
          ` : ''}
        </div>
      </div>
      ${agendamento.status === 'pendente' || agendamento.status === 'confirmado' ? `
        <div class="card-actions">
          <button class="btn btn-danger btn-small" onclick="cancelarAgendamento(${agendamento.id})">
            Cancelar Agendamento
          </button>
        </div>
      ` : ''}
    </div>
  `).join('');
}

async function cancelarAgendamento(id) {
  if (!confirm('Tem certeza que deseja cancelar este agendamento?')) {
    return;
  }

  try {
    await fazerRequisicao(`/api/agendamentos/${id}/status`, {
      method: 'PUT',
      body: JSON.stringify({ status: 'cancelado' }),
    });

    mostrarModal('Agendamento cancelado com sucesso');
    await carregarAgendamentosCliente();
  } catch (erro) {
    mostrarModal('Erro ao cancelar agendamento: ' + erro.message);
  }
}

document.getElementById('form-servico').addEventListener('submit', async (e) => {
  e.preventDefault();
  const formData = new FormData(e.target);
  const dados = Object.fromEntries(formData);
  
  const id = dados.id;
  delete dados.id;

  try {
    if (id) {
      dados.ativo = 1;
      await fazerRequisicao(`/api/servicos/${id}`, {
        method: 'PUT',
        body: JSON.stringify(dados),
      });
      mostrarModal('Serviço atualizado com sucesso');
    } else {
      await fazerRequisicao('/api/servicos', {
        method: 'POST',
        body: JSON.stringify(dados),
      });
      mostrarModal('Serviço criado com sucesso');
    }

    document.getElementById('form-servico-container').style.display = 'none';
    e.target.reset();
    await carregarServicosAdmin();
  } catch (erro) {
    mostrarModal('Erro ao salvar serviço: ' + erro.message);
  }
});

function editarServico(id) {
  const servico = servicos.find(s => s.id === id);
  if (!servico) return;

  const form = document.getElementById('form-servico');
  form.elements['id'].value = servico.id;
  form.elements['nome'].value = servico.nome;
  form.elements['descricao'].value = servico.descricao;
  form.elements['duracao'].value = servico.duracao;
  form.elements['preco'].value = servico.preco;

  document.getElementById('form-servico-container').style.display = 'block';
  form.scrollIntoView({ behavior: 'smooth' });
}

# Sistema de Agendamento de Serviços

Sistema completo de agendamento de serviços com autenticação, gerenciamento de serviços, horários e agendamentos.

## Funcionalidades

### Para Clientes:
- Criar conta e fazer login
- Ver serviços disponíveis com preços e descrições
- Ver horários disponíveis em tempo real
- Agendar serviços
- Ver histórico de agendamentos
- Cancelar agendamentos

### Para Administradores:
- Cadastrar e editar serviços (nome, descrição, duração, preço)
- Configurar horários de funcionamento por dia da semana
- Ver todos os agendamentos de todos os clientes
- Confirmar ou recusar agendamentos
- Dashboard completo de gerenciamento

## Instalação

### Requisitos
- Node.js (versão 14 ou superior)

### Passo a Passo

1. Abra o terminal/prompt de comando na pasta do projeto

2. Instale as dependências:
```bash
npm install
```

3. Inicie o servidor:
```bash
npm start
```

4. Abra o navegador e acesse:
```
http://localhost:3000
```

## Login Administrador Padrão

Ao iniciar o sistema pela primeira vez, um usuário administrador é criado automaticamente:

- **Email:** admin@sistema.com
- **Senha:** admin123

Use estas credenciais para fazer o primeiro acesso como administrador.

## Como Usar

### Primeira Vez

1. Acesse o sistema pelo navegador
2. Faça login como administrador (credenciais acima)
3. Cadastre os serviços oferecidos (ex: Corte de Cabelo, Manicure, etc)
4. Configure os horários de funcionamento (ex: Segunda a Sexta, 9h às 18h)
5. Agora os clientes podem se registrar e agendar!

### Como Cliente

1. Clique em "Registrar" e crie sua conta
2. Na aba "Novo Agendamento":
   - Escolha o serviço desejado
   - Selecione a data
   - Escolha um horário disponível
   - Adicione observações se necessário
   - Clique em "Agendar"
3. Na aba "Meus Agendamentos" você vê todos os seus agendamentos e pode cancelá-los

### Como Administrador

1. **Gerenciar Serviços:**
   - Clique em "Serviços"
   - Clique em "Novo Serviço" para adicionar
   - Preencha nome, descrição, duração (em minutos) e preço
   - Use "Editar" para modificar ou "Desativar" para remover

2. **Configurar Horários:**
   - Clique em "Horários"
   - Clique em "Novo Horário"
   - Escolha o dia da semana
   - Defina hora de início e fim
   - O sistema gera slots de 30 minutos automaticamente

3. **Gerenciar Agendamentos:**
   - Na aba "Agendamentos" você vê todos os pedidos
   - Clique em "Confirmar" para aprovar
   - Clique em "Recusar" para negar

## Tecnologias Utilizadas

- **Backend:** Node.js + Express
- **Banco de Dados:** SQLite (arquivo local, não precisa instalar nada)
- **Frontend:** HTML + CSS + JavaScript puro
- **Autenticação:** Sessões com bcrypt para senhas

## Dados Salvos

Todos os dados ficam salvos no arquivo `agendamento.db` que é criado automaticamente na pasta do projeto. Este arquivo contém:

- Usuários (clientes e administradores)
- Serviços cadastrados
- Horários de funcionamento
- Todos os agendamentos

Mesmo fechando o navegador ou reiniciando o servidor, os dados permanecem salvos.

- ✅ Funciona em qualquer computador ou celular
- ✅ Interface responsiva e fácil de usar
- ✅ Não precisa criar conta em outros sites
- ✅ Tudo funciona dentro do próprio sistema
- ✅ Dados salvos permanentemente

- ✅ Horários inteligentes (não permite conflitos)

O sistema roda na porta 3000. Se precisar mudar, pode definir a variável de ambiente PORT:

```bash
PORT=8080 npm start
```

## Problemas Comuns

**Erro ao instalar:** Certifique-se de que o Node.js está instalado. Baixe em nodejs.org

**Porta em uso:** Se a porta 3000 já estiver sendo usada, o sistema mostrará erro. Feche outros programas ou mude a porta.

**Banco de dados:** O arquivo agendamento.db é criado automaticamente. Não delete este arquivo ou perderá todos os dados.

## Suporte

O sistema é autossuficiente e não depende de serviços externos. Tudo funciona localmente no seu computador.
