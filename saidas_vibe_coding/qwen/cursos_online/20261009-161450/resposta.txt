const express = require('express');
const crypto = require('crypto');
const path = require('path');
const fs = require('fs');

const app = express();
const PORT = process.env.PORT || 3000;

const DATA_DIR = path.join(__dirname, 'data');
const DB_FILE = path.join(DATA_DIR, 'db.json');

app.use(express.json({ limit: '1mb' }));
app.use(express.static(path.join(__dirname, 'public')));

// ---------- Banco de dados (arquivo JSON local) ----------
function ensureDb() {
  if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });
  if (!fs.existsSync(DB_FILE)) {
    const initial = {
      users: [
        {
          id: 'admin',
          name: 'Administrador',
          email: 'admin@plataforma.com',
          passwordHash: hashPassword('admin123'),
          salt: '1',
          role: 'admin',
          createdAt: new Date().toISOString()
        }
      ],
      courses: [],
      enrollments: [],
      sessions: []
    };
    fs.writeFileSync(DB_FILE, JSON.stringify(initial, null, 2));
  }
}

function loadDb() {
  ensureDb();
  return JSON.parse(fs.readFileSync(DB_FILE, 'utf8'));
}

function saveDb(db) {
  ensureDb();
  fs.writeFileSync(DB_FILE, JSON.stringify(db, null, 2));
}

// ---------- Senhas ----------
function hashPassword(password, salt) {
  return crypto.scryptSync(password, salt, 64).toString('hex');
}

function makeSalt() {
  return crypto.randomBytes(16).toString('hex');
}

function verifyPassword(password, salt, hash) {
  const candidate = crypto.scryptSync(password, salt, 64).toString('hex');
  return crypto.timingSafeEqual(Buffer.from(candidate), Buffer.from(hash));
}

// ---------- Sessões (tokens) ----------
function createSession(userId) {
  const db = loadDb();
  const token = crypto.randomBytes(32).toString('hex');
  const expiry = Date.now() + 1000 * 60 * 60 * 24 * 7; // 7 dias
  db.sessions = db.sessions.filter(s => s.token !== token);
  db.sessions.push({ token, userId, expires: expiry });
  saveDb(db);
  return token;
}

function getSessionUserId(req) {
  const token = req.headers['x-session-token'];
  if (!token) return null;
  const db = loadDb();
  const session = db.sessions.find(s => s.token === token && s.expires > Date.now());
  if (!session) return null;
  return session.userId;
}

function destroySession(token) {
  if (!token) return;
  const db = loadDb();
  db.sessions = db.sessions.filter(s => s.token !== token);
  saveDb(db);
}

// ---------- Helpers ----------
const emailRe = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

function publicUser(u) {
  return { id: u.id, name: u.name, email: u.email, role: u.role };
}

function publicCourse(c) {
  const db = loadDb();
  const enrolledCount = db.enrollments.filter(e => e.courseId === c.id).length;
  return { ...c, enrolledCount };
}

function publicEnrollment(e) {
  const db = loadDb();
  const course = db.courses.find(c => c.id === e.courseId);
  const user = db.users.find(u => u.id === e.userId);
  return {
    id: e.id,
    courseId: e.courseId,
    userId: e.userId,
    courseName: course ? course.name : 'Curso removido',
    studentName: user ? user.name : 'Aluno removido',
    studentEmail: user ? user.email : '',
    enrolledAt: e.enrolledAt
  };
}

function genId() {
  return crypto.randomBytes(8).toString('hex');
}

app.post('/api/auth/register', (req, res) => {
  const { name, email, password } = req.body || {};
  if (!name || !email || !password) {
    return res.status(400).json({ error: 'Preencha nome, e-mail e senha.' });
  }
  if (!emailRe.test(email)) {
    return res.status(400).json({ error: 'E-mail inválido.' });
  }
  if (password.length < 4) {
    return res.status(400).json({ error: 'A senha precisa ter pelo menos 4 caracteres.' });
  }
  const db = loadDb();
  const norm = email.trim().toLowerCase();
  if (db.users.some(u => u.email === norm)) {
    return res.status(409).json({ error: 'Já existe uma conta com este e-mail.' });
  }
  const salt = makeSalt();
  const user = {
    id: genId(),
    name: name.trim(),
    email: norm,
    salt,
    passwordHash: hashPassword(password, salt),
    role: 'student',
    createdAt: new Date().toISOString()
  };
  db.users.push(user);
  saveDb(db);
  const token = createSession(user.id);
  res.json({ token, user: publicUser(user) });
});

app.post('/api/auth/login', (req, res) => {
  const { email, password } = req.body || {};
  if (!email || !password) {
    return res.status(400).json({ error: 'Informe e-mail e senha.' });
  }
  const db = loadDb();
  const norm = email.trim().toLowerCase();
  const user = db.users.find(u => u.email === norm);
  if (!user || !verifyPassword(password, user.salt, user.passwordHash)) {
    return res.status(401).json({ error: 'E-mail ou senha incorretos.' });
  }
  const token = createSession(user.id);
  res.json({ token, user: publicUser(user) });
});

app.post('/api/auth/logout', (req, res) => {

app.post('/api/enrollments', (req, res) => {
  const userId = getSessionUserId(req);
  if (!userId) return res.status(401).json({ error: 'Faça login.' });
  const db = loadDb();
  const user = db.users.find(u => u.id === userId);
  if (!user || user.role !== 'student') {
    return res.status(403).json({ error: 'Apenas alunos podem se matricular.' });
  }
  const courseId = (req.body || {}).courseId;
  const course = db.courses.find(c => c.id === courseId);
  if (!course) return res.status(404).json({ error: 'Curso não encontrado.' });
  if (db.enrollments.some(e => e.userId === userId && e.courseId === courseId)) {
    return res.status(409).json({ error: 'Você já está matriculado neste curso.' });
  }
  const enrollment = {
    id: genId(),
    userId,
    courseId,
    enrolledAt: new Date().toISOString()
  };
  db.enrollments.push(enrollment);
  saveDb(db);
  res.json({ enrollment: publicEnrollment(enrollment) });
});

const API = '/api';
let state = {
  token: localStorage.getItem('token') || null,
  user: null,
  mode: 'courses',
  myEnrolledCourseIds: []
};

// ---------- API ----------
async function api(path, options = {}) {
  const headers = { 'Content-Type': 'application/json' };
  if (state.token) headers['X-Session-Token'] = state.token;
  const res = await fetch(API + path, {
    method: options.method || 'GET',
    headers,
    body: options.body ? JSON.stringify(options.body) : undefined
  });
  let data = {};
  try { data = await res.json(); } catch (e) { /* sem corpo */ }
  if (!res.ok) {
    if (res.status === 401 && state.user) {
      logoutLocal();
      showAuth();
    }
    throw new Error(data.error || 'Erro ao comunicar com o servidor.');
  }
  return data;
}

// ---------- Helpers de UI ----------
function el(id) { return document.getElementById(id); }

function esc(s) {
  return String(s == null ? '' : s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

function setAuthMsg(text, isErr = true) {
  const m = el('authMsg');
  m.textContent = text || '';
  m.style.color = isErr ? 'var(--danger)' : 'var(--ok)';
}

// ---------- Navegação ----------
function buildNav() {
  const nav = el('nav');
  nav.innerHTML = '';
  const items = [
    { id: 'courses', label: 'Cursos' }
  ];
  if (state.user.role === 'student') {
    items.push({ id: 'mycourses', label: 'Meus cursos' });
  }
  if (state.user.role === 'admin') {
    items.push({ id: 'enrollments', label: 'Matrículas' });
  }
  items.forEach(it => {
    const b = document.createElement('button');
    b.textContent = it.label;
    b.dataset.mode = it.id;
    if (state.mode === it.id) b.classList.add('active');
    b.onclick = () => { state.mode = it.id; buildNav(); render(); };
    nav.appendChild(b);
  });
}

// ---------- Visões ----------
function showAuth() {
  el('topbar').style.display = 'none';
  el('homeView').style.display = 'none';
  el('authView').style.display = 'flex';
}

function showHome() {
  el('authView').style.display = 'none';
  el('topbar').style.display = 'flex';
  el('homeView').style.display = 'block';
  el('userName').textContent = state.user.name;
  el('newCourseBtn').style.display = state.user.role === 'admin' ? 'inline-block' : 'none';
  buildNav();
  render();
}

function render() {
  if (!state.user) return;
  const titles = {
    courses: ['Cursos disponíveis', 'Escolha um curso e se matricule.'],
    mycourses: ['Meus cursos', 'Cursos em que você está matriculado.'],
    enrollments: ['Todas as matrículas', 'Matrículas de todos os alunos.']
  };
  const t = titles[state.mode] || titles.courses;
  el('pageTitle').textContent = t[0];
  el('pageSub').textContent = t[1];

  if (state.mode === 'courses') return renderCourses();
  if (state.mode === 'mycourses') return renderMyCourses();
  if (state.mode === 'enrollments') return renderAllEnrollments();
}

async function loadMyEnrollments() {
  if (state.user.role !== 'student') { state.myEnrolledCourseIds = []; return; }
  try {
    const data = await api('/enrollments/my');
    state.myEnrolledCourseIds = data.enrollments.map(e => e.courseId);
  } catch (e) { state.myEnrolledCourseIds = []; }
}

// ---------- Cursos disponíveis ----------
async function renderCourses() {
  const box = el('homeContent');
  box.innerHTML = '<p class="empty">Carregando cursos…</p>';
  let courses = [];
  try {
    const data = await api('/courses');
    courses = data.courses;
  } catch (e) {
    box.innerHTML = '<div class="empty"><h3>Erro</h3><p>' + esc(e.message) + '</p></div>';
    return;
  }
  await loadMyEnrollments();

  if (!courses.length) {
    box.innerHTML = '<div class="empty"><h3>Nenhum curso ainda</h3><p>Em breve novos cursos estarão disponíveis.</p></div>';
    return;
  }

  box.innerHTML = '<div class="grid">' + courses.map(c => {
    const isAdmin = state.user.role === 'admin';
    const isEnrolled = state.myEnrolledCourseIds.includes(c.id);
    let foot = '';
    if (isAdmin) {
      foot = `<div class="card-actions">
        <button class="btn btn-ghost btn-sm" data-edit="${c.id}">Editar</button>
        <button class="btn btn-danger btn-sm" data-del="${c.id}">Remover</button>
      </div>`;
    } else if (isEnrolled) {
      foot = '<span class="enrolled-badge">✓ Matriculado</span>';
    } else {
      foot = `<button class="btn btn-enroll" data-enroll="${c.id}">Matricular-se</button>`;
    }
    return `<div class="course-card">
      <h3>${esc(c.name)}</h3>
      <div class="course-desc">${esc(c.description) || '<span style="opacity:.6">Sem descrição.</span>'}</div>
      <div class="course-meta">
        ${c.duration ? `<span>⏱ ${esc(c.duration)}</span>` : ''}
        ${c.price ? `<span>💰 ${esc(c.price)}</span>` : ''}
        <span>👥 ${c.enrolledCount} matriculado(s)</span>
      </div>
      <div class="course-foot">${foot}</div>
    </div>`;
  }).join('') + '</div>';

  bindCourseEvents(box);
}

function bindCourseEvents(box) {
  box.querySelectorAll('[data-enroll]').forEach(b => {
    b.onclick = async () => {
      b.disabled = true;
      try {
        await api('/enrollments', { method: 'POST', body: { courseId: b.dataset.enroll } });
        state.myEnrolledCourseIds.push(b.dataset.enroll);
        renderCourses();
      } catch (e) {
        alert(e.message);
        b.disabled = false;
      }
    };
  });
  box.querySelectorAll('[data-edit]').forEach(b => {
    b.onclick = () => openCourseModal(b.dataset.edit);
  });
  box.querySelectorAll('[data-del]').forEach(b => {
    b.onclick = async () => {
      if (!confirm('Remover este curso? Todas as matrículas dele também serão removidas.')) return;
      try {
        await api('/courses/' + b.dataset.del, { method: 'DELETE' });
        renderCourses();
      } catch (e) { alert(e.message); }
    };
  });
}

// ---------- Meus cursos ----------
async function renderMyCourses() {
  const box = el('homeContent');
  box.innerHTML = '<p class="empty">Carregando…</p>';

    const data = await api('/enrollments/my');

// ---------- Modal de curso ----------
function openCourseModal(courseId) {
  let course = null;
  if (courseId) {
    api('/courses').then(d => {
      course = d.courses.find(c => c.id === courseId);
      if (!course) return;
      el('courseModalTitle').textContent = 'Editar curso';
      el('courseId').value = course.id;
      el('courseName').value = course.name;
      el('courseDesc').value = course.description || '';
      el('courseDuration').value = course.duration || '';
      el('coursePrice').value = course.price || '';
      el('courseModal').style.display = 'flex';
    });
  } else {
    el('courseModalTitle').textContent = 'Novo curso';
    el('courseForm').reset();
    el('courseId').value = '';
    el('courseModal').style.display = 'flex';
  }
}
function closeCourseModal() { el('courseModal').style.display = 'none'; }

// ---------- Auth ----------
let authMode = 'login';

function setAuthMode(mode) {
  authMode = mode;
  document.querySelectorAll('.tab').forEach(t => t.classList.toggle('active', t.dataset.tab === mode));
  el('regNameWrap').style.display = mode === 'register' ? 'block' : 'none';
  el('authSubmit').textContent = mode === 'register' ? 'Criar conta' : 'Entrar';
  setAuthMsg('');
}

async function handleAuth(e) {
  e.preventDefault();
  const name = el('name').value.trim();
  const email = el('email').value.trim();
  const password = el('password').value;
  setAuthMsg('');
  const btn = el('authSubmit');
  btn.disabled = true;
  try {
    const payload = { email, password };
    if (authMode === 'register') payload.name = name;
    const data = await api('/auth/' + authMode, { method: 'POST', body: payload });
    state.token = data.token;
    state.user = data.user;
    localStorage.setItem('token', data.token);
    state.mode = state.user.role === 'admin' ? 'courses' : 'courses';
    showHome();
  } catch (err) {
    setAuthMsg(err.message);
  } finally {
    btn.disabled = false;
  }
}

function logoutLocal() {
  localStorage.removeItem('token');
  state.token = null;
  state.user = null;
}

async function handleLogout() {
  try { await api('/auth/logout', { method: 'POST' }); } catch (e) {}
  logoutLocal();
  showAuth();
}

// ---------- Inicialização ----------
async function init() {
  document.querySelectorAll('.tab').forEach(t => {
    t.onclick = () => setAuthMode(t.dataset.tab);
  });
  el('authForm').onsubmit = handleAuth;
  el('logoutBtn').onclick = handleLogout;
  el('newCourseBtn').onclick = () => openCourseModal(null);
  el('courseCancel').onclick = closeCourseModal;
  el('courseModal').onclick = (e) => { if (e.target === el('courseModal')) closeCourseModal(); };
  el('courseForm').onsubmit = async (e) => {
    e.preventDefault();
    const id = el('courseId').value;
    const body = {
      name: el('courseName').value,
      description: el('courseDesc').value,
      duration: el('courseDuration').value,
      price: el('coursePrice').value
    };
    try {
      if (id) await api('/courses/' + id, { method: 'PUT', body });
      else await api('/courses', { method: 'POST', body });
      closeCourseModal();
      renderCourses();
    } catch (err) { alert(err.message); }
  };

  // Tentar restaurar sessão
  if (state.token) {
    try {
      const data = await api('/auth/me');
      state.user = data.user;
      showHome();
      return;
    } catch (e) {
      logoutLocal();
    }
  }
  showAuth();
}

init();

# Plataforma de Cursos Online

Uma plataforma simples de cursos online com sistema de matrícula. Alunos podem ver cursos, se matricular e ver seus cursos. O administrador pode cadastrar, editar e remover cursos, ver todas as matrículas e cancelá-las.

Tudo fica salvo no próprio servidor (em um arquivo `data/db.json`). Por isso, se a mesma pessoa entrar do celular ou do computador, ela encontra as mesmas informações, e o que uma pessoa faz aparece para quem deve ver.

## Requisitos

- **Node.js 18** ou superior instalado no seu computador/servidor.
  - Para verificar: abra o terminal e digite `node -v`.
  - Se não tiver, baixe em: https://nodejs.org

## Como instalar e executar

1. Abra o terminal (ou Prompt de Comando) na pasta deste projeto.
2. Instale as dependências:

   ```bash
   npm install
   ```

3. Inicie o site:

   ```bash
   npm start
   ```

4. O site ficará disponível em:

   ```
   http://localhost:3000
   ```

## Contas

- **Administrador** (já criada):
  - E-mail: `admin@plataforma.com`
  - Senha: `admin123`

- **Alunos**: crie a conta pela aba "Criar conta" na tela de login.

## Funcionalidades

**Aluno:**
- Ver cursos disponíveis
- Matricular-se em um curso
- Ver os cursos em que está matriculado
- Cancelar sua própria matrícula

**Administrador:**
- Cadastrar, editar e remover cursos
- Ver todas as matrículas de todos os alunos
- Cancelar qualquer matrícula

## Como colocar na internet (opcional)

O servidor precisa rodar em um computador/servidor acessível pela internet. Você pode usar um serviço de hospedagem que execute Node.js (por exemplo, um VPS, Railway, Render ou Fly.io). Basta fazer upload deste projeto e rodar `npm install` e `npm start`.

> Observação: o site usa `http`. Para um ambiente profissional, recomende usar `https` (certificado). As informações ficam salvas em `data/db.json` — faça backup desse arquivo e da pasta `data`.
