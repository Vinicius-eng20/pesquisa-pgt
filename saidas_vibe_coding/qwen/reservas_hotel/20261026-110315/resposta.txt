def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def query(sql, args=()):
    conn = get_db()
    try:
        cur = conn.execute(sql, args)
        conn.commit()
        return cur
    finally:
        conn.close()


def query_all(sql, args=()):
    conn = get_db()
    try:
        cur = conn.execute(sql, args)
        rows = cur.fetchall()
        return [dict(r) for r in rows]
    finally:
        conn.close()

def overlaps(cid, cdoi):
    return not (cdoi <= cid or cid <= cdoi)


def is_taken(room_id, checkin, checkout, exclude_booking_id=None):
    sql = "SELECT id FROM bookings WHERE room_id=? AND status IN ('pending','confirmed')"
    args = [room_id]
    if exclude_booking_id is not None:
        sql += " AND id<>?"
        args.append(exclude_booking_id)
    sql += " AND ?<checkout AND ?>=checkin"
    rows = query_all(sql, args + [checkin, checkout])
    return len(rows) > 0


def conflicts_room(room_id, checkin, checkout):
    rows = query_all("SELECT * FROM rooms WHERE id=?", (room_id,))
    if not rows:
        return False
    return is_taken(room_id, checkin, checkout)

            "WHERE b.status='pending' GROUP BY b.room_id")
        for p in pending:
            for r in rooms:
                if r["id"] == p["room_id"]:
                    r["status"] = "pendente"
    return jsonify({"ok": True, "rooms": rooms})

@app.post("/api/bookings")
def create_booking():
    d = request.get_json() or {}
    user = d.get("user")
    if not user:
        return jsonify({"ok": False, "error": "Não autenticado."}), 401
    room_id = d.get("room_id")
    checkin, checkout = normalize_dates(d.get("checkin"), d.get("checkout"))
    if not checkin or not checkout:
        return jsonify({"ok": False, "error": "Escolha as datas de entrada e saída."}), 400
    if checkin >= checkout:
        return jsonify({"ok": False, "error": "A data de saída deve ser depois da entrada."}), 400
    rooms = query_all("SELECT * FROM rooms WHERE id=?", (room_id,))
    if not rooms:
        return jsonify({"ok": False, "error": "Quarto não encontrado."}), 404
    if is_taken(room_id, checkin, checkout):
        return jsonify({"ok": False,
                        "error": "Este quarto já está reservado para essas datas."}), 400
    try:
        cur = query("INSERT INTO bookings (room_id,user_id,checkin,checkout,status) "
                    "VALUES (?,?,?,'pending',?)",
                    (room_id, user["id"], checkin, checkout))
    except Exception:
        return jsonify({"ok": False, "error": "Não foi possível criar a reserva."}), 400
    row = query_all("SELECT b.*, r.name AS room_name, u.name AS guest_name "
                    "FROM bookings b JOIN rooms r ON r.id=b.room_id JOIN users u ON u.id=b.user_id "
                    "WHERE b.id=?", (cur.lastrowid,))[0]
    return jsonify({"ok": True, "booking": row})

function todayStr(){
  const d = new Date();
  const p = n => String(n).padStart(2,"0");
  return d.getFullYear()+"-"+p(d.getMonth()+1)+"-"+p(d.getDate());
}
function addDays(s, n){
  const d = new Date(s + "T00:00:00");
  d.setDate(d.getDate()+n);
  const p = x => String(x).padStart(2,"0");
  return d.getFullYear()+"-"+p(d.getMonth()+1)+"-"+p(d.getDate());
}
function fmtDate(s){
  if(!s) return "";
  const p = s.split("-");
  return p.length===3 ? (p[2]+"/"+p[1]+"/"+p[0]) : s;
}
function fmtPrice(x){
  return (Number(x)||0).toLocaleString("pt-BR",{style:"currency",currency:"BRL"});
}
function daysBetween(a,b){
  if(!a||!b) return 0;
  const ms = new Date(b+"T00:00:00") - new Date(a+"T00:00:00");
  return Math.round(ms/86400000);
}
const statusLabel = { pending:"Pendente", confirmed:"Confirmada", rejected:"Recusada", cancelled:"Cancelada" };
const statusChip  = { pending:"warn", confirmed:"ok", rejected:"bad", cancelled:"off" };

function toast(msg, kind){
  const box = document.getElementById("toasts") || (() => {
    const el = document.createElement("div"); el.id="toasts"; el.className="toasts";
    document.body.appendChild(el); return el;
  })();
  const t = document.createElement("div");
  t.className = "toast " + (kind||"");
  t.textContent = msg;
  box.appendChild(t);
  setTimeout(()=>{ t.style.opacity="0"; t.style.transition="opacity .3s"; setTimeout(()=>t.remove(),300); }, 2600);
}

async function api(path, opts={}){
  const res = await fetch(path, {
    method: opts.method || "GET",
    headers: {"Content-Type":"application/json"},
    body: JSON.stringify(opts.body || {})
  });
  const data = await res.json().catch(()=>({}));
  if(!res.ok) throw new Error(data.error || ("Erro " + res.status));
  return data;
}

function bindAuth(){
  const tl = document.getElementById("t-login"), tr = document.getElementById("t-reg");
  const msg = document.getElementById("auth-msg");
  const show = (txt, ok) => { msg.innerHTML = txt ? `<div class="form-msg ${ok?"ok":"err"}">${txt}</div>` : ""; };
  if(tl) tl.onclick = () => { authMode="login"; render(); };
  if(tr) tr.onclick = () => { authMode="reg"; render(); };
  const fl = document.getElementById("f-login");
  if(fl) fl.onsubmit = async e => {
    e.preventDefault();
    try{
      const fd = new FormData(fl);
      const data = await api("/api/login",{method:"POST",body:{username:fd.get("username"),password:fd.get("password")}});
      enter(data.user);
    }catch(err){ show(err.message); }
  };
  const fr = document.getElementById("f-reg");
  if(fr) fr.onsubmit = async e => {
    e.preventDefault();
    try{
      const fd = new FormData(fr);
      const data = await api("/api/register",{method:"POST",body:{
        username:fd.get("username"), name:fd.get("name"), password:fd.get("password")}});
      enter(data.user);
    }catch(err){ show(err.message); }
  };
}

async function bindGuestRooms(){
  const fin = document.getElementById("f-in");

  fin.min = todayStr();
  fin.onchange = async ()=>{ state.filter.checkin = fin.value;

async function loadGuestRooms(){
  const f = state.filter;
  const grid = document.getElementById("rooms-grid");
  const pill = document.getElementById("pill");
  let rooms;
  try{ rooms = (await api("/api/rooms",{method:"POST",body:{user:state.user,checkin:f.checkin,checkout:f.checkout}})).rooms; }
  catch(err){ grid.innerHTML = `<div class="empty"><div class="big">!</div>${err.message}</div>`; return; }
  const valid = !!(f.checkin && f.checkout && f.checkin < f.checkout);
  const n = rooms.filter(r=>r.available_selected).length;
  pill.textContent = valid ? `${n} quarto(s) para ${fmtDate(f.checkin)} – ${fmtDate(f.checkout)}` : "Mostre os quartos selecionando datas acima";
  if(!rooms.length){ grid.innerHTML = `<div class="empty"><div class="big">🏨</div>Nenhum quarto cadastrado ainda.</div>`; return; }
  grid.innerHTML = rooms.map(r=>{
    const off = !r.available;
    const taken = off || (valid && !r.available_selected);
    const sub = off ? "Indisponível"
      : (taken ? "Ocupado nestas datas"
      : (valid ? "Disponível para suas datas" : "Disponível"));
    return `
    <div class="room-card">
      <div class="room-thumb">
        <span class="cap">${r.capacity} pessoas</span>
        <span class="type">${r.type}</span>
      </div>
      <div class="room-body">
        <div class="room-name">${r.name}</div>
        <div class="room-desc">${r.description||"–"}</div>
        <span class="chip ${taken?"off":"ok"}" style="margin-bottom:10px">${sub}</span>
        <div class="room-foot">
          <div class="price">${fmtPrice(r.price)}<small>/noite</small></div>
          <button class="btn btn-teal btn-sm" data-book="${r.id}" ${taken?"disabled style='opacity:.45;cursor:not-allowed'":""}>Reservar</button>
        </div>
      </div>
    </div>`;
  }).join("");
  grid.querySelectorAll("[data-book]").forEach(b=>{
    b.onclick = ()=>{ const r = rooms.find(x=>x.id==b.dataset.book); openBookingModal(r); };
  });
}

function openBookingModal(room){
  const f = state.filter;
  let inD = f.checkin || todayStr(), outD = f.checkout || addDays(inD,1);
  const ov = document.createElement("div");
  ov.className="overlay";
  ov.innerHTML = `
    <div class="modal">
      <h3>Reservar ${room.name}</h3>
      <div class="sub">${room.type} · até ${room.capacity} pessoas · ${fmtPrice(room.price)}/noite</div>
      <div class="two-col">
        <div class="field"><label>Entrada</label><input type="date" id="m-in" value="${inD}" min="${todayStr()}"></div>
        <div class="field"><label>Saída</label><input type="date" id="m-out" value="${outD}"></div>
      </div>
      <div class="field"><label>Nome do hóspede</label><input id="m-guest" value="${state.user.name}"></div>
      <div id="m-msg"></div>
      <div class="modal-actions">
        <button class="btn btn-ghost" id="m-cancel">Cancelar</button>
        <button class="btn btn-teal" id="m-ok">Confirmar reserva</button>
      </div>
    </div>`;
  document.body.appendChild(ov);
  const mi = ov.querySelector("#m-in"), mo = ov.querySelector("#m-out");
  mo.min = addDays(mi.value,1);
  mi.onchange = ()=> mo.min = addDays(mi.value,1);
  const msg = ov.querySelector("#m-msg");
  const say = (t,ok)=> msg.innerHTML = t?`<div class="form-msg ${ok?"ok":"err"}">${t}</div>`:"";
  ov.querySelector("#m-cancel").onclick = ()=> ov.remove();
  ov.querySelector("#m-ok").onclick = async ()=>{
    const checkin = mi.value, checkout = mo.value;
    if(!checkin||!checkout) return say("Escolha as datas.",false);
    if(checkin>=checkout) return say("Saída deve ser após a entrada.",false);
    try{
      await api("/api/bookings",{method:"POST",body:{user:state.user,room_id:room.id,checkin,checkout}});
      ov.remove();
      toast("Reserva criada!", "ok");
      state.tab="minhas"; render();
    }catch(err){ say(err.message,false); }
  };
}

=== README.md ===
# Hotel Azul — Sistema de Reservas

Um site completo para reservas de hotel, com duas áreas:
- **Hóspede**: vê os quartos disponíveis nas datas escolhidas, faz e consulta/cancela reservas.
- **Gerente**: cadastra/edita/exclui quartos e confirma/recusa/exclui reservas, vendo tudo.

Os dados ficam salvos no próprio servidor (banco `hotel.db`), então:
- você entra de qualquer computador/celular e encontra a mesma informação;
- o que uma pessoa faz aparece para quem precisa ver;
- fechar e reabrir o site não apaga nada.

---

## 1. Instalar e rodar (no seu computador)

**Pré-requisito:** Python 3.8 ou mais novo. (Se não tiver, baixe em https://www.python.org — durante a instalação, marque a opção “Add Python to PATH”.)

Abra um terminal **na pasta deste projeto** e digite:

```
pip install flask
```

Depois inicie o servidor:

```
python app.py
```

- No **Windows** talvez precise de `pip3` no lugar de `pip`, e `python3` no lugar de `python`.
- O terminal ficará preso nesse comando — é normal (o servidor está rodando).

Abra no navegador:

```
http://localhost:5000
```

### Contas de teste
| Papel    | Usuário  | Senha    |
|----------|----------|----------|
| Gerente  | `gerente`| `123456` |
| Hóspede  | `hospede`| `123456` |

> Os 5 quartos de exemplo já vêm criados. Logue como **gerente** para vê-los.

Para parar o servidor: pressione `Ctrl + C` no terminal.

---

## 2. Como usar

**Hóspede**
1. Crie sua conta (ou use `hospede`/`123456`).
2. Em **Quartos**, escolha a data de entrada e de saída. Os quartos ocupados nessas datas aparecem como “Ocupado nestas datas”.
3. Clique em **Reservar** num quarto disponível. A reserva fica *Pendente* até o gerente confirmar.
4. Em **Minhas reservas**, veja, cancele ou acompanhe o status.

**Gerente**
1. Logue como `gerente`/`123456`.
2. Em **Reservas** (aba *Pendentes*), confirme ou recuse cada reserva. Também pode excluir.
3. Em **Quartos**, cadastre, edite, desative ou exclua quartos.

---

## 3. Publicar na internet para as pessoas (verdadeiro “funcionar na internet”)

No seu PC, `http://localhost:5000` só é visível para você. Para que os hóspedes e o gerente acessem de qualquer lugar, hospede o projeto num serviço **gratuito**. Recomendo dois:

### Opção A — Render (recomendado)
1. Crie uma conta grátis em https://render.com (com GitHub).
2. Crie um repositório vazio no GitHub e mande os arquivos deste projeto para ele (`git init; git add .; git commit -m "site"; git remote add origin ...; git push -u origin main`).
3. No Render, crie um **Web Service** apontando para esse repositório.
   - *Build Command*: `pip install -r requirements.txt`
   - *Start Command*: `python app.py`
   - Crie uma **variável de ambiente** `FLASK_APP=1` (apenas um flag para o banco).
4. O Render vai te dar um endereço `https://nome-alvo.onrender.com` — esse é o link público do seu hotel.

### Opção B — Railway
1. Crie conta em https://railway.app (também com GitHub).
2. “New Project” → “Deploy from GitHub repo” (o mesmo repositório).
3. Railway detecta Python; ele vai criar o banco por conta própria.
4. Faça “Generate Domain” para ter um link público.

### Opção C — Tunnelling rápido (para testar agora, sem conta)
Se só quer ver do celular **agora** (e estiver no mesmo Wi-Fi do seu PC):
```
pip install flask_ngrok
```
Depois, em `app.py`, após `init_db()` e antes do `if __name__`, adicione:
```python
if __name__ == "__main__":
    from flask_ngrok import run_with_ngrok
    run_with_ngrok(app)
```
Reinicie o `python app.py`. O terminal mostrará uma URL pública — acesse de qualquer lugar. *(Isso é para teste; para uso contínuo, use a Opção A ou B.)*

> Importante: para o “mesmo login de qualquer lugar” funcionar de verdade, todos precisam usar o **mesmo endereço público** e o **mesmo banco**. Os planos gratuitos podem “dormir” quando estão ociosos e, no Render, recomenda-se usar um banco externo (Postgres) para não perder dados — os dados ficam em `hotel.db` por padrão e são recriados se o banco não for persistido.
