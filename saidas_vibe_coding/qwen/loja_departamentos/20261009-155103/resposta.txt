O que você descreveu (dados salvos no próprio site, que funcionam de qualquer aparelho para várias pessoas ao mesmo tempo, sem criar conta em outros serviços) precisa de um **banco de dados** — é impossível salvar os dados *dentro* de um arquivo `.html` que fica no navegador de cada pessoa, porque cada computador teria uma cópia sua e nada se encontraria. Por isso criei um projeto de verdade: um **site com um banco de dados embutido no próprio servidor** (o arquivo de banco fica na pasta `data`, criado sozinho na primeira vez que você roda o servidor). Ele roda em um servidor qualquer com Node (ou até na sua própria máquina, se quiser testar).

Segue o projeto completo.

function hashPassword(pw) {
  return crypto.createHash('sha256').update(String(pw)).digest('hex');
}

let db;
function loadDB() {
  try {
    if (fs.existsSync(DB_FILE)) {
      db = JSON.parse(fs.readFileSync(DB_FILE, 'utf8'));
      if (!db.products) db.products = [];
      if (!db.orders) db.orders = [];
      if (!db.users) db.users = [];
      if (!db.nextId) db.nextId = 1;
    } else {
      fs.mkdirSync(DB_DIR, { recursive: true });
      db = defaultDB;
      saveDB();
    }
  } catch (e) {
    console.error('Erro ao carregar banco, usando padrão:', e.message);
    db = defaultDB;
  }
}
function saveDB() {
  fs.mkdirSync(DB_DIR, { recursive: true });
  fs.writeFileSync(DB_FILE, JSON.stringify(db, null, 2));
}
loadDB();

function newId() {
  const id = db.nextId++;
  saveDB();
  return id;
}

/* ---------------- Autenticação ---------------- */
function createToken(user) {
  return crypto.createHmac('sha256', 'mega-mall-secret').update(String(user.id)).digest('hex');
}
function getUserFromToken(token) {
  if (!token) return null;
  return db.users.find(u => createToken(u) === token) || null;
}
function requireAuth(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  const user = getUserFromToken(token);
  if (!user) return res.status(401).json({ error: 'Não autenticado' });
  req.user = user;
  next();
}
function requireAdmin(req, res, next) {
  if (req.user?.role !== 'admin') return res.status(403).json({ error: 'Acesso restrito ao lojista' });
  next();
}

app.post('/api/auth/register', (req, res) => {
  const { username, name, password } = req.body || {};
  if (!username || !name || !password) return res.status(400).json({ error: 'Preencha nome, usuário e senha.' });
  if (String(username).trim().length < 3) return res.status(400).json({ error: 'O usuário deve ter ao menos 3 caracteres.' });
  if (String(password).length < 4) return res.status(400).json({ error: 'A senha deve ter ao menos 4 caracteres.' });
  if (db.users.some(u => u.username.toLowerCase() === String(username).trim().toLowerCase()))
    return res.status(409).json({ error: 'Este usuário já existe.' });

  const id = newId();
  const user = { id, username: String(username).trim(), name: String(name).trim(), passwordHash: hashPassword(password), role: 'cliente' };
  db.users.push(user);
  saveDB();
  res.json({ token: createToken(user), user: publicUser(user) });
});

app.post('/api/auth/login', (req, res) => {
  const { username, password } = req.body || {};
  const user = db.users.find(u => u.username.toLowerCase() === String(username || '').trim().toLowerCase());
  if (!user || user.passwordHash !== hashPassword(password))
    return res.status(401).json({ error: 'Usuário ou senha incorretos.' });
  res.json({ token: createToken(user), user: publicUser(user) });
});

function publicUser(u) {

}

  res.json(product);

app.post('/api/orders', requireAuth, (req, res) => {
  const { items, customer } = req.body || {};
  if (!Array.isArray(items) || items.length === 0)
    return res.status(400).json({ error: 'O carrinho está vazio.' });

  const finalItems = [];
  let total = 0;
  for (const it of items) {
    const product = db.products.find(p => p.id === Number(it.productId));
    const qty = Math.max(1, Math.floor(Number(it.quantity) || 1));
    if (!product) return res.status(400).json({ error: 'Um dos produtos não existe mais.' });
    if (product.stock < qty)
      return res.status(400).json({ error: `Estoque insuficiente para "${product.name}".` });
    finalItems.push({
      productId: product.id,
      name: product.name,
      price: product.price,
      quantity: qty
    });
    total += product.price * qty;
    product.stock -= qty;
  }

  const id = newId();
  const order = {
    id,
    code: 'MM-' + String(id).padStart(5, '0'),
    userId: req.user.id,
    customer: {
      name: String(customer?.name || req.user.name || 'Cliente'),
      email: String(customer?.email || ''),
      phone: String(customer?.phone || ''),
      address: String(customer?.address || '')
    },
    payment: String(customer?.payment || 'cartao'),
    items: finalItems,
    total: Math.round(total * 100) / 100,
    status: 'pendente',
    createdAt: new Date().toISOString()
  };
  db.orders.unshift(order);
  saveDB();
  res.json(order);
});

app.patch('/api/orders/:id/status', requireAuth, requireAdmin, (req, res) => {
  const order = db.orders.find(o => o.id === Number(req.params.id));
  if (!order) return res.status(404).json({ error: 'Pedido não encontrado.' });
  const status = String(req.body?.status || '');
  if (!STATUSES.includes(status)) return res.status(400).json({ error: 'Situação inválida.' });

  const canceling = order.status !== 'cancelado' && status === 'cancelado';
  order.status = status;
  if (canceling) {
    for (const it of order.items) {
      const product = db.products.find(p => p.id === it.productId);
      if (product) product.stock += it.quantity;
    }
  }
  saveDB();
  res.json(order);
});

async function api(path, opts = {}) {
  const headers = { 'Content-Type': 'application/json', ...(opts.headers || {}) };
  if (state.token) headers.Authorization = 'Bearer ' + state.token;
  const res = await fetch('/api' + path, { ...opts, headers, body: opts.body ? JSON.stringify(opts.body) : undefined });
  let data = null;
  try { data = await res.json(); } catch (e) {}
  if (!res.ok) {
    if (res.status === 401 && path.startsWith('/api/auth') === false) {
      // sessão expirada
    }
    throw new Error((data && data.error) || 'Algo deu errado.');
  }
  return data;
}

/* =============== Carrinho =============== */
function saveCart() { localStorage.setItem('mm_cart', JSON.stringify(state.cart)); updateCartUI(); }
function cartCount() { return state.cart.reduce((a, i) => a + i.quantity, 0); }
function cartTotal() { return state.cart.reduce((a, i) => a + i.price * i.quantity, 0); }
function addToCart(p, qty = 1) {
  if (p.stock <= 0) return;
  const ex = state.cart.find(i => i.productId === p.id);
  if (ex) ex.quantity = Math.min(ex.quantity + qty, p.stock);
  else state.cart.push({ productId: p.id, name: p.name, price: p.price, quantity: Math.min(qty, p.stock) });
  saveCart();
  toast('Adicionado ao carrinho!', 'ok');
  openDrawer();
}
function setCartQty(id, qty) {
  const it = state.cart.find(i => i.productId === id);
  if (!it) return;
  const p = state.products.find(x => x.id === id);
  const max = p ? p.stock : 99;
  it.quantity = Math.max(1, Math.min(qty, max));
  if (it.quantity <= 0) state.cart = state.cart.filter(i => i.productId !== id);
  saveCart(); renderDrawer();
}
function removeCartItem(id) { state.cart = state.cart.filter(i => i.productId !== id); saveCart(); renderDrawer(); }

# MegaMall — Loja virtual de departamentos

Um shopping online completo: os **clientes** navegam, adicionam ao carrinho, finalizam pedidos e acompanham os seus pedidos; o **lojista** cadastra/edita/remove produtos, muda a situação de cada pedido e acompanha todas as vendas.

Tudo fica salvo em um **banco de dados dentro do próprio site** (arquivo `data/db.json`). As informações sobrevivem a reinícios e são compartilhadas entre todos: o que o cliente faz aparece no painel do lojista e vice-versa, em qualquer computador ou celular.

> **Por que precisa de um servidor?** Salvar dados *dentro* de um site estático (só um arquivo `.html`) não é possível: cada aparelho guardaria uma cópia própria e nada se encontraria. Por isso o MegaMall usa um servidor Node.js com um banco de dados embutido, criado automaticamente na primeira execução — **sem** criar conta em outros serviços e **sem** ligar a nada externo.

---

## O que precisa ter instalado

Apenas o **Node.js** (versão 16 ou mais recente).
- Baixe e instale em: https://nodejs.org
- Para confirmar, abra o terminal e digite:
  ```
  node -v
  ```
  Deve aparecer algo como `v18.x.x` ou `v20.x.x`.

---

## Como instalar e executar

1. Abra o terminal (ou "Prompt de comando") dentro da pasta do projeto.
2. Instale as dependências (só uma vez):
   ```
   npm install
   ```
3. Inicie a loja:
   ```
   npm start
   ```
4. Abra no navegador:
   ```
   http://localhost:3000
   ```

> Na primeira execução a pasta `data` é criada sozinha e já vem com produtos de exemplo e a conta do lojista.

---

## Contas

| Quem | Usuário | Senha |
|------|---------|-------|
| Lojista (já criada) | `admin` | `admin123` |
| Clientes | você cria | você escolhe |

- O lojista entra e cai direto no **Painel do Lojista** (produtos, pedidos e vendas).
- Qualquer visitante pode **Criar conta** na tela de login para comprar e acompanhar os próprios pedidos.
- Após logar, o botão **Entrar** vira o **seu nome** — clique nele para sair (logout).

---

## Como usar

**Cliente**
- Navegue pelas categorias ou use a busca.
- Toque em um produto para ver detalhes e **Adicionar ao carrinho** (🛒 no topo).
- No carrinho, ajuste quantidades e toque em **Finalizar Pedido**.
- Preencha os dados de entrega e confirme. (O pagamento é **simulado** — não é cobrado nada.)
- Em **Meus Pedidos** você acompanha a situação do seu pedido.

**Lojista (admin)**
- **📦 Produtos**: cadastrar, editar e excluir produtos (nome, categoria, preço, estoque e ícone).
- **🧾 Pedidos**: ver todos os pedidos de todos os clientes e mudar a situação (Pendente → Processando → Enviado → Entregue, ou Cancelar). Cancelar devolve o produto ao estoque.
- **📊 Vendas**: faturamento total, quantidade de pedidos e resumo por situação.

---

## Como colocar na internet (para acessar de qualquer lugar)

O site roda em qualquer servidor com Node.js — um VPS, um Raspberry Pi, um computador seu, etc.

**Opção 1 — Rodar em um servidor na nuvem** (ex.: um VPS barato):
```
# no servidor
npm install
npm start
```
Depois acesse pela IP/domínio do servidor: `http://SEU-IP:3000` (ou configure um domínio).

**Dica — deixar ligado o tempo todo e na porta padrão (80/443):** instale o [pm2](https://pm2.keymetrics.io/) e use um [reverse proxy](https://www.nginx.com/) (Nginx). Exemplo simples com pm2:
```
npm install -g pm2
pm2 start server.js --name megamall
pm2 save
pm2 startup
```
Para trocar a porta padrão, use: `PORT=8080 npm start`.

**Opção 2 — Testar localmente, mas acessar pelo celular na mesma rede:** rode `npm start` no computador e, no celular, acesse `http://IP-DO-COMPUTADOR:3000` (mesma rede Wi-Fi). Descubra o IP do computador com `ipconfig` (Windows) ou `ifconfig`/`ip a` (Mac/Linux).

## Onde ficam os dados

- Banco de dados: **`data/db.json`** (criado automaticamente). Faça backup deste arquivo para não perder nada.
- Para zerar a loja, apague a pasta `data` e reinicie (`npm start`) — ela é recriada com os produtos de exemplo.
- Cada navegador guarda apenas sua sessão local (o "estou logado"), para você não precisar digitar a senha a cada visita. Os dados de verdade estão no servidor.

## Solução de problemas

- **`npm` não é reconhecido** → o Node.js não está instalado ou o terminal precisa ser reaberto.
- **A porta 3000 já está ocupada** → use outra: `PORT=3001 npm start`.
- **Nada aparece ao recarregar** → confirme que o servidor está rodando (a mensagem "Loja no ar!" aparece no terminal).
- **Esqueci a senha do admin** → o admin padrão é sempre `admin` / `admin123` (o banco é o arquivo `data/db.json`; remova-o para recriar tudo do zero, se precisar).
