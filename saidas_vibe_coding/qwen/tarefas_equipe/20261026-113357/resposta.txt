import express from "express";
import expressSession from "express-session";
import bcrypt from "bcryptjs";
import { join, dirname } from "node:path";
import { fileURLToPath } from "node:url";
import { readFileSync, writeFileSync, existsSync } from "node:fs";

const __dirname = dirname(fileURLToPath(import.meta.url));
const DB_FILE = join(__dirname, "data.json");
const SALT_ROUNDS = 10;

const DEFAULT_STATUS = [
  "A fazer",
  "Em andamento",
  "Concluída"
];
const DEFAULT_PRIORIDADES = ["Baixa", "Média", "Alta"];

const app = express();
app.use(express.json());
app.use(express.static(join(__dirname, "public")));

app.use(expressSession({
  name: "sessao_gestao",
  secret: process.env.SESSAO_SECRET || "chave-secreta-nao-use-em-producao",
  resave: false,
  saveUninitialized: false,
  cookie: {
    httpOnly: true,
    maxAge: 1000 * 60 * 60 * 24 * 30
  }
}));

function uid() {
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10);
}

function getDb() {
  const empty = {
    users: [],
    tasks: [],
    status: DEFAULT_STATUS.slice(),
    prioridades: DEFAULT_PRIORIDADES.slice()
  };
  if (!existsSync(DB_FILE)) {
    writeFileSync(DB_FILE, JSON.stringify(empty, null, 2));
    return empty;
  }
  try {
    const parsed = JSON.parse(readFileSync(DB_FILE, "utf8"));
    return {
      users: parsed.users || [],
      tasks: parsed.tasks || [],
      status: parsed.status || DEFAULT_STATUS.slice(),
      prioridades: parsed.prioridades || DEFAULT_PRIORIDADES.slice()
    };
  } catch (err) {
    return empty;
  }
}

function saveDb(db) {
  writeFileSync(DB_FILE, JSON.stringify(db, null, 2));
}

function publicUser(user) {
  return {
    id: user.id,
    nome: user.nome,
    email: user.email,
    papel: user.papel
  };
}

function publicTask(task) {
  return {
    id: task.id,
    titulo: task.titulo,
    descricao: task.descricao,
    status: task.status,
    prioridade: task.prioridade,
    dataEntrega: task.dataEntrega,
    criadaPor: task.criadaPor,
    criadaPorNome: task.criadaPorNome,
    criadaEm: task.criadaEm,
    atualizadaEm: task.atualizadaEm,
    atribuidas: (task.atribuidas || []).map(a => ({
      userId: a.userId,
      nome: a.nome,
      done: !!a.done
    }))
  };
}

function isGerente(user) {
  return user && user.papel === "gerente";
}

function isAtribuida(user, task) {
  return (task.atribuidas || []).some(a => a.userId === user.id);
}

function auth(req, res, next) {
  const user = req.session.user;
  if (!user) {
    return res.status(401).json({ error: "Faça login para continuar." });
  }
  req.user = user;
  next();
}

function requireGerente(req, res, next) {
  if (!isGerente(req.user)) {
    return res.status(403).json({ error: "Só o gerente pode fazer isso." });
  }
  next();
}

// ===== AUTENTICACAO =====
app.post("/api/auth/registro", (req, res) => {
  const db = getDb();
  const { nome, email, senha } = req.body || {};
  if (!nome || !email || !senha) {
    return res.status(400).json({ error: "Preencha nome, e-mail e senha." });
  }
  if (senha.length < 4) {
    return res.status(400).json({ error: "A senha precisa ter pelo menos 4 caracteres." });
  }
  const emailNorm = String(email).trim().toLowerCase();
  if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(emailNorm)) {
    return res.status(400).json({ error: "E-mail inválido." });
  }
  if (db.users.some(u => u.email === emailNorm)) {
    return res.status(409).json({ error: "Já existe uma conta com este e-mail." });
  }

  const primeiro = db.users.length === 0;
  const papel = primeiro ? "gerente" : "membro";
  const salt = bcrypt.genSaltSync(SALT_ROUNDS);
  const user = {
    id: uid(),
    nome: String(nome).trim(),
    email: emailNorm,
    salt,
    hash: bcrypt.hashSync(String(senha), salt),
    papel
  };
  db.users.push(user);
  saveDb(db);

  req.session.user = {
    id: user.id,
    nome: user.nome,
    email: user.email,
    papel: user.papel
  };
  res.json({ usuario: publicUser(user) });
});

app.post("/api/auth/login", (req, res) => {
  const db = getDb();
  const { email, senha } = req.body || {};
  const emailNorm = String(email || "").trim().toLowerCase();
  const user = db.users.find(u => u.email === emailNorm);
  if (!user || !bcrypt.compareSync(String(senha || ""), user.hash)) {
    return res.status(401).json({ error: "E-mail ou senha incorretos." });
  }
  req.session.user = {
    id: user.id,
    nome: user.nome,
    email: user.email,
    papel: user.papel
  };
  res.json({ usuario: publicUser(user) });
});

app.post("/api/auth/logout", (req, res) => {
  req.session.destroy(err => {
    res.clearCookie("sessao_gestao");
    res.json({ ok: true });
  });
});

app.get("/api/auth/me", auth, (req, res) => {
  res.json({ usuario: publicUser(req.user) });
});

  res.json({ users: db.users.map(publicUser) });
});

// ===== TAREFAS: listar =====
app.get("/api/tasks", auth, (req, res) => {
  const db = getDb();
  let tasks = db.tasks;
  if (!isGerente(req.user)) {
    tasks = tasks.filter(t => isAtribuida(req.user, t));
  }
  tasks = tasks.slice().sort((a, b) => (b.criadaEm || 0) - (a.criadaEm || 0));
  res.json({ tasks: tasks.map(publicTask) });
});

// ===== TAREFAS: criar =====
app.post("/api/tasks", auth, (req, res) => {
  const db = getDb();
  const { titulo, descricao, prioridade, dataEntrega, atribuidas, paraSi } = req.body || {};
  if (!titulo || !String(titulo).trim()) {
    return res.status(400).json({ error: "A tarefa precisa de um título." });
  }
  const agora = Date.now();

  let lista;
  if (isGerente(req.user)) {
    lista = [];
    (atribuidas || []).forEach(id => {
      const u = db.users.find(x => x.id === id);
      if (u) lista.push({ userId: u.id, nome: u.nome, done: false });
    });
  } else {
    if (!paraSi) {
      return res.status(403).json({ error: "Você não pode atribuir tarefa para outros." });
    }
    lista = [{ userId: req.user.id, nome: req.user.nome, done: false }];
  }

  const task = {
    id: uid(),
    titulo: String(titulo).trim(),
    descricao: String(descricao || "").trim(),
    status: "A fazer",
    prioridade: prioridade || "Média",
    dataEntrega: dataEntrega || "",
    criadaPor: req.user.id,
    criadaPorNome: req.user.nome,
    criadaEm: agora,
    atualizadaEm: agora,
    atribuidas: lista
  };
  db.tasks.push(task);
  saveDb(db);
  res.json({ task: publicTask(task) });
});

// ===== TAREFAS: editar =====
app.put("/api/tasks/:id", auth, requireGerente, (req, res) => {
  const db = getDb();
  const task = db.tasks.find(t => t.id === req.params.id);
  if (!task) return res.status(404).json({ error: "Tarefa não encontrada." });

  const { titulo, descricao, prioridade, status, dataEntrega, atribuidas } = req.body || {};
  if (titulo !== undefined) task.titulo = String(titulo).trim();
  if (titulo && !String(titulo).trim()) {
    return res.status(400).json({ error: "O título não pode ficar vazio." });
  }
  if (descricao !== undefined) task.descricao = String(descricao).trim();
  if (prioridade !== undefined) task.prioridade = prioridade;
  if (status !== undefined) task.status = status;
  if (dataEntrega !== undefined) task.dataEntrega = dataEntrega;
  if (atribuidas !== undefined) {
    task.atribuidas = [];
    (atribuidas || []).forEach(id => {
      const u = db.users.find(x => x.id === id);
      if (u) task.atribuidas.push({ userId: u.id, nome: u.nome, done: false });
    });
  }
  task.atualizadaEm = Date.now();
  saveDb(db);
  res.json({ task: publicTask(task) });
});

// ===== TAREFAS: remover =====
app.delete("/api/tasks/:id", auth, requireGerente, (req, res) => {
  const db = getDb();
  const idx = db.tasks.findIndex(t => t.id === req.params.id);
  if (idx === -1) return res.status(404).json({ error: "Tarefa não encontrada." });
  db.tasks.splice(idx, 1);
  saveDb(db);
  res.json({ ok: true });
});

// ===== ATUALIZAR STATUS (membro) =====
app.put("/api/tasks/:id/status", auth, (req, res) => {
  const db = getDb();
  const task = db.tasks.find(t => t.id === req.params.id);
  if (!task) return res.status(404).json({ error: "Tarefa não encontrada." });
  if (!isGerente(req.user) && !isAtribuida(req.user, task)) {
    return res.status(403).json({ error: "Você não pode mudar o status desta tarefa." });
  }
  const { status } = req.body || {};
  if (!db.status.includes(status)) {
    return res.status(400).json({ error: "Status inválido." });
  }
  task.status = status;
  task.atualizadaEm = Date.now();
  saveDb(db);
  res.json({ task: publicTask(task) });
});

// ===== MARCAR COMO CONCLUIDA POR QUEM RECEBEU =====
app.put("/api/tasks/:id/done", auth, (req, res) => {
  const db = getDb();
  const task = db.tasks.find(t => t.id === req.params.id);
  if (!task) return res.status(404).json({ error: "Tarefa não encontrada." });
  if (!isGerente(req.user) && !isAtribuida(req.user, task)) {
    return res.status(403).json({ error: "Você não pode alterar esta tarefa." });
  }
  const { done } = req.body || {};
  const a = (task.atribuidas || []).find(x => x.userId === req.user.id);
  if (a) a.done = !!done;
  else if (isGerente(req.user) && !task.atribuidas.length) {
    task.status = done ? "Concluída" : "A fazer";
  }
  task.atualizadaEm = Date.now();
  saveDb(db);
  res.json({ task: publicTask(task) });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Gestão de Tarefas rodando em http://localhost:${PORT}`);
});

/* ---------- helpers ---------- */
function slug(s) {
  return String(s || "A fazer").replace(/[^a-z0-9]/gi, "");
}
function esc(s) {
  return String(s == null ? "" : s).replace(/[&<>"']/g, c => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
  }[c]));
}
function iniciais(nome) {
  return String(nome || "?").split(" ").filter(Boolean).slice(0, 2).map(p => p[0]).join("").toUpperCase();
}
function formatarData(iso) {
  if (!iso) return "";
  const d = new Date(iso + "T00:00:00");
  if (isNaN(d)) return iso;
  return d.toLocaleDateString("pt-BR");
}
function estaAtrasada(task) {
  if (!task.dataEntrega || task.status === "Concluída") return false;
  const hoje = new Date(); hoje.setHours(0,0,0,0);
  const d = new Date(task.dataEntrega + "T00:00:00");
  return d < hoje;
}
function ehGerente() { return state.usuario && state.usuario.papel === "gerente"; }
function estouAtribuido(task) {
  return (task.atribuidas || []).some(a => a.userId === state.usuario.id);
}

async function api(path, opts = {}) {
  const res = await fetch(path, {
    headers: { "Content-Type": "application/json" },
    ...opts,
    body: opts.body ? JSON.stringify(opts.body) : undefined
  });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.error || "Algo deu errado.");
  return data;
}

function tarefasFiltradas() {
  let t = state.tasks;
  if (state.filtro !== "todas") t = t.filter(x => x.status === state.filtro);
  if (state.busca) {
    t = t.filter(x =>
      x.titulo.toLowerCase().includes(state.busca) ||
      x.descricao.toLowerCase().includes(state.busca) ||
      (x.atribuidas || []).some(a => a.nome.toLowerCase().includes(state.busca))
    );
  }
  return t;
}

function cardHTML(task) {
  const atrasada = estaAtrasada(task);
  const podeGerenciar = ehGerente();
  const sou = estouAtribuido(task);
  const podeMudarStatus = podeGerenciar || sou;

  // avatares
  let avs = task.atribuidas.map(a => `<span class="avatar" title="${esc(a.nome)}">${esc(iniciais(a.nome))}</span>`).join("");
  if (!task.atribuidas.length) avs = `<span class="avatar more">–</span>`;

  const st = slug(task.status);
  const pri = slug(task.prioridade);

  // mini opcoes de status
  let statusHtml = "";
  if (podeMudarStatus) {
    statusHtml = state.config.status.map(s =>
      `<button class="mini st-mini ${s===task.status?'sel':''}" data-id="${task.id}" data-st="${esc(s)}">${esc(s)}</button>`
    ).join(" ");
  }

  let acoesHtml = `
    <div class="tarefa-acoes">
      <button class="mini" data-ver="${task.id}">Detalhes</button>`;
  if (podeGerenciar) {
    acoesHtml += `<button class="mini" data-editar="${task.id}">Editar</button>`;
    acoesHtml += `<button class="mini danger" data-del="${task.id}">Remover</button>`;
  }
  acoesHtml += `</div>`;

  let entregaHtml = "";
  if (task.dataEntrega) {
    entregaHtml = `<span class="${atrasada ? 'atrasado' : ''}">📅 ${esc(formatarData(task.dataEntrega))}${atrasada ? " (atrasado)" : ""}</span>`;
  }

  return `
    <div class="tarefa" data-id="${task.id}">
      <div class="tarefa-coluna">
        <div class="tarefa-top">
          <span class="badge st-${st}">${esc(task.status)}</span>
          <span class="badge pri-${pri}">${esc(task.prioridade)}</span>
          <span class="tarefa-titulo">${esc(task.titulo)}</span>
        </div>
        ${task.descricao ? `<div class="tarefa-desc">${esc(task.descricao)}</div>` : ""}
        <div class="tarefa-meta">
          <div class="avatares" title="${esc(task.atribuidas.map(a=>a.nome).join(', '))}">${avs}</div>
          <span>Por <b>${esc(task.criadaPorNome)}</b></span>
          ${entregaHtml}
        </div>
        ${statusHtml ? `<div class="status-opcoes" style="margin-top:10px">${statusHtml}</div>` : ""}
      </div>
      ${acoesHtml}
    </div>`;
}

# Sistema de Gestão de Tarefas da Equipe

Um site completo para gerenciar tarefas de uma equipe. O **gerente** cria e distribui tarefas para os membros, edita e remove qualquer tarefa e vê tudo. Os **membros** veem as tarefas que receberam, mudam o status e criam tarefas para si mesmos. Tudo fica salvo no próprio servidor do site (não no aparelho), então cada pessoa encontra as mesmas informações de qualquer computador ou celular, e o que uma pessoa faz aparece para quem precisa ver.

## Requisitos

- [Node.js](https://nodejs.org) versão **18** ou mais nova instalada.

## Como instalar e executar

1. Abra um terminal na pasta do projeto e rode:

```bash
npm install
```

2. Inicie o site:

```bash
npm start
```

3. Abra o navegador em:

```
http://localhost:3000
```

Para rodar em outra porta, use:

```bash
PORT=8080 npm start
```

## Como usar

- **Criar a conta do gerente:** na tela inicial, escolha **Criar conta**. A **primeira** conta criada é automaticamente o **gerente** da equipe.
- **Criar contas dos membros:** cada membro acessa o site de qualquer computador/celular e usa **Criar conta** com seu nome, e-mail e senha.
- **Entrar:** depois de criada, a pessoa escolhe **Entrar** e digita e-mail e senha.
- **Gerente:** pode criar tarefas e atribuí-las a um ou mais membros, editar e remover qualquer tarefa, mudar status e ver todas as tarefas de todos.
- **Membro:** vê as tarefas atribuídas a ele, muda o status (A fazer / Em andamento / Concluída) e cria tarefas apenas para si mesmo.

## Onde ficam os dados

Todos os dados (contas, tarefas, configurações) ficam salvos no arquivo `data.json`, criado automaticamente na pasta do projeto. Por isso as informações **não somem** ao fechar o site e são **compartilhadas** entre todos os usuários.

> **Importante:** para que as pessoas da equipe acessem de outros computadores/celulares, o servidor precisa estar disponível na internet (por exemplo, rodando em um computador com acesso de rede, ou publicado em uma hospedagem/serviço de deploy). Neste projeto não é preciso se cadastrar em nenhum outro site nem ligar com outros serviços — tudo fica dentro do próprio site.
